### 高精度
---
#### C++ 高精度整数
以 $10^9$ 为一位倒序存储，支持负数
```cpp
struct BigInt {
    static constexpr int BASE{1000000000};
    static constexpr int DIGITS{9};
    vector<int> a;
    int sign{1};
private:
    // ===== 1. 基础维护：所有功能依赖，不建议删除 =====
    void trim() {
        while (!a.empty() && a.back() == 0) a.pop_back();
        if (a.empty()) sign = 1;
    }
    static int abs_cmp(const BigInt &x, const BigInt &y) {
        if (x.a.size() != y.a.size()) return x.a.size() < y.a.size() ? -1 : 1;
        for (int i = int(x.a.size()) - 1; i >= 0; -- i) {
            if (x.a[i] != y.a[i]) return x.a[i] < y.a[i] ? -1 : 1;
        }
        return 0;
    }
    // ===== 2. 绝对值加减：BigInt 加减法依赖；大整数除法也依赖 =====
    // 若既不需要 BigInt +/- BigInt，也不需要 BigInt / BigInt，可连同公开加减接口一起删除。
    void add_abs(const BigInt &v) {
        int carry{0};
        for (int i = 0; i < int(v.a.size()) || carry; ++ i) {
            if (i == int(a.size())) a.push_back(0);
            i64 cur{a[i] + i64(carry) + (i < int(v.a.size()) ? v.a[i] : 0)};
            a[i] = int(cur % BASE);
            carry = int(cur / BASE);
        }
    }
    void sub_abs(const BigInt &v) {
        int carry{0}; // 要求 |*this| >= |v|
        for (int i = 0; i < int(v.a.size()) || carry; ++ i) {
            i64 cur{a[i] - i64(carry) - (i < int(v.a.size()) ? v.a[i] : 0)};
            carry = cur < 0;
            if (carry) cur += BASE;
            a[i] = int(cur);
        }
        trim();
    }
    // ===== 3. Karatsuba 辅助：只被高性能 BigInt * BigInt 使用 =====
    // 若改用下方 /** ... **/ 中的短版 operator*，本组可直接删除。
    using VLL = vector<i64>;
    static vector<int> convert_base(const vector<int> &v, int old_digits, int new_digits) {
        vector<i64> p(max(old_digits, new_digits) + 1, 1);
        for (int i = 1; i < int(p.size()); ++ i) p[i] = p[i - 1] * 10;
        vector<int> res;
        i64 cur{0};
        int cur_digits{0};
        for (int x : v) {
            cur += i64(x) * p[cur_digits];
            cur_digits += old_digits;
            while (cur_digits >= new_digits) {
                res.push_back(int(cur % p[new_digits]));
                cur /= p[new_digits];
                cur_digits -= new_digits;
            }
        }
        res.push_back(int(cur));
        while (!res.empty() && res.back() == 0) res.pop_back();
        return res;
    }
    static VLL karatsuba_multiply(const VLL &x, const VLL &y) {
        int n{int(x.size())};
        VLL res(2 * n);
        if (n <= 32) {
            for (int i = 0; i < n; ++ i)
                for (int j = 0; j < n; ++ j)
                    res[i + j] += x[i] * y[j];
            return res;
        }
        int k{n >> 1};
        VLL x0(x.begin(), x.begin() + k), x1(x.begin() + k, x.end());
        VLL y0(y.begin(), y.begin() + k), y1(y.begin() + k, y.end());
        VLL z0{karatsuba_multiply(x0, y0)}, z2{karatsuba_multiply(x1, y1)};
        for (int i = 0; i < k; ++ i) x1[i] += x0[i], y1[i] += y0[i];
        VLL z1{karatsuba_multiply(x1, y1)};
        for (int i = 0; i < int(z0.size()); ++ i) z1[i] -= z0[i];
        for (int i = 0; i < int(z2.size()); ++ i) z1[i] -= z2[i];
        for (int i = 0; i < int(z0.size()); ++ i) res[i] += z0[i];
        for (int i = 0; i < int(z1.size()); ++ i) res[i + k] += z1[i];
        for (int i = 0; i < int(z2.size()); ++ i) res[i + 2 * k] += z2[i];
        return res;
    }
public:
    // ===== 4. 构造、符号与输入输出：基础接口，通常保留 =====
    BigInt() = default;
    BigInt(i64 v) { *this = v; }
    BigInt(const string &s) { read(s); }
    bool is_zero() const { return a.empty(); }
    BigInt abs() const { BigInt res{*this}; res.sign = 1; return res; }
    BigInt &operator=(i64 v) {
        a.clear();
        sign = v < 0 ? -1 : 1;
        u64 x{u64(v)};
        if (v < 0) x = 0 - x;
        while (x) {
            a.push_back(int(x % BASE));
            x /= BASE;
        }
        return *this;
    }
    BigInt operator-() const {
        BigInt res{*this};
        if (!res.is_zero()) res.sign = -res.sign;
        return res;
    }
    void read(const string &s) {
        a.clear();
        sign = 1;
        int pos{0};
        if (!s.empty() && (s[0] == '+' || s[0] == '-')) {
            sign = s[0] == '-' ? -1 : 1;
            pos = 1;
        }
        for (int i = int(s.size()); i > pos; i -= DIGITS) {
            int x{0};
            for (int j = max(pos, i - DIGITS); j < i; ++ j) x = x * 10 + s[j] - '0';
            a.push_back(x);
        }
        trim();
    }
    friend istream &operator>>(istream &in, BigInt &v) {
        string s;
        in >> s;
        v.read(s);
        return in;
    }
    friend ostream &operator<<(ostream &out, const BigInt &v) {
        if (v.sign < 0) out << '-';
        if (v.a.empty()) return out << 0;
        out << v.a.back();
        for (int i = int(v.a.size()) - 2; i >= 0; -- i) {
            out << setw(DIGITS) << setfill('0') << v.a[i];
        }
        return out << setfill(' ');
    }
    // ===== 5. 比较：加减法和 BigInt 除法依赖，通常保留 =====
    friend bool operator<(const BigInt &x, const BigInt &y) {
        if (x.sign != y.sign) return x.sign < y.sign;
        int c{abs_cmp(x, y)};
        return x.sign == 1 ? c < 0 : c > 0;
    }
    friend bool operator>(const BigInt &x, const BigInt &y) { return y < x; }
    friend bool operator<=(const BigInt &x, const BigInt &y) { return !(y < x); }
    friend bool operator>=(const BigInt &x, const BigInt &y) { return !(x < y); }
    friend bool operator==(const BigInt &x, const BigInt &y) { return x.sign == y.sign && x.a == y.a; }
    friend bool operator!=(const BigInt &x, const BigInt &y) { return !(x == y); }
    // ===== 6. BigInt 加减：若不需要 +/- 且不需要 BigInt 除法，可连同第 2 组删除 =====
    BigInt &operator+=(const BigInt &v) {
        if (sign == v.sign) {
            add_abs(v);
        } else if (abs_cmp(*this, v) >= 0) {
            sub_abs(v);
        } else {
            BigInt t{v};
            t.sub_abs(*this);
            *this = move(t);
        }
        return *this;
    }
    BigInt &operator-=(const BigInt &v) { return *this += -v; }
    friend BigInt operator+(BigInt x, const BigInt &y) { return x += y; }
    friend BigInt operator-(BigInt x, const BigInt &y) { return x -= y; }
    // ===== 7. 与普通整数乘除：不用时可删；BigInt / BigInt 依赖 *= 和 /= =====
    BigInt &operator*=(i64 v) {
        if (v == 0 || is_zero()) return *this = 0;
        if (v < 0) sign = -sign;
        u64 x{u64(v)};
        if (v < 0) x = 0 - x;
        __uint128_t carry{0};
        for (int &d : a) {
            __uint128_t cur{__uint128_t(d) * x + carry};
            d = int(cur % BASE);
            carry = cur / BASE;
        }
        while (carry) {
            a.push_back(int(carry % BASE));
            carry /= BASE;
        }
        return *this;
    }
    BigInt &operator/=(i64 v) {
        assert(v != 0);
        if (v < 0) sign = -sign;
        u64 x{u64(v)};
        if (v < 0) x = 0 - x;
        __uint128_t rem{0};
        for (int i = int(a.size()) - 1; i >= 0; -- i) {
            __uint128_t cur{rem * BASE + a[i]};
            a[i] = int(cur / x);
            rem = cur % x;
        }
        trim();
        return *this;
    }
    i64 operator%(i64 v) const {
        assert(v != 0);
        u64 x{u64(v)};
        if (v < 0) x = 0 - x;
        __uint128_t rem{0};
        for (int i = int(a.size()) - 1; i >= 0; -- i) rem = (rem * BASE + a[i]) % x;
        i64 r{i64(rem)};
        return sign < 0 ? -r : r;
    }
    friend BigInt operator*(BigInt x, i64 v) { return x *= v; }
    friend BigInt operator*(i64 v, BigInt x) { return x *= v; }
    friend BigInt operator/(BigInt x, i64 v) { return x /= v; }
    // ===== 8. BigInt 乘法：完全不需要大整数乘大整数时，本组与第 3 组可直接删除 =====
    // 高性能版：Karatsuba；适合位数较多且乘法较重的题目。
    BigInt operator*(const BigInt &v) const {
        if (is_zero() || v.is_zero()) return 0;
        vector<int> a6{convert_base(a, DIGITS, 6)};
        vector<int> b6{convert_base(v.a, DIGITS, 6)};
        VLL x(a6.begin(), a6.end()), y(b6.begin(), b6.end());
        while (x.size() < y.size()) x.push_back(0);
        while (y.size() < x.size()) y.push_back(0);
        while (x.size() & (x.size() - 1)) x.push_back(0), y.push_back(0);
        VLL c{karatsuba_multiply(x, y)};
        BigInt res;
        res.sign = sign * v.sign;
        i64 carry{0};
        for (i64 z : c) {
            i64 cur{z + carry};
            res.a.push_back(int(cur % 1000000));
            carry = cur / 1000000;
        }
        while (carry) {
            res.a.push_back(int(carry % 1000000));
            carry /= 1000000;
        }
        res.a = convert_base(res.a, 6, DIGITS);
        res.trim();
        return res;
    }
    /**
    // 短版：朴素 O(nm) 乘法。
    // 若数据没有超长整数，通常已经足够；使用本版时删除上面的 operator*，
    // 并可直接删除第 3 组 convert_base / karatsuba_multiply / VLL。
    BigInt operator*(const BigInt &v) const {
        if (is_zero() || v.is_zero()) return 0;
        BigInt res;
        res.sign = sign * v.sign;
        res.a.assign(a.size() + v.a.size(), 0);
        for (int i = 0; i < int(a.size()); ++ i) {
            i64 carry{0};
            for (int j = 0; j < int(v.a.size()) || carry; ++ j) {
                i64 cur{res.a[i + j] + carry};
                if (j < int(v.a.size())) cur += i64(a[i]) * v.a[j];
                res.a[i + j] = int(cur % BASE);
                carry = cur / BASE;
            }
        }
        res.trim();
        return res;
    }
    **/
    BigInt &operator*=(const BigInt &v) { return *this = *this * v; }
    // ===== 9. BigInt 除法与取模：不用 / % 时整组删除 =====
    // 依赖：比较、第 2/6 组加减、第 7 组整数乘除。
    friend pair<BigInt, BigInt> divmod(const BigInt &x, const BigInt &y) {
        assert(!y.is_zero());
        int norm{BASE / (y.a.back() + 1)};
        BigInt a{x.abs() * norm}, b{y.abs() * norm}, q, r;
        q.a.resize(a.a.size());
        for (int i = int(a.a.size()) - 1; i >= 0; -- i) {
            r *= BASE;
            r += BigInt(a.a[i]);
            i64 s1{r.a.size() <= b.a.size() ? 0 : r.a[b.a.size()]};
            i64 s2{r.a.size() < b.a.size() ? 0 : r.a[b.a.size() - 1]};
            i64 d{(i64(BASE) * s1 + s2) / b.a.back()};
            r -= b * d;
            while (r < 0) r += b, -- d;
            q.a[i] = int(d);
        }
        q.sign = x.sign * y.sign;
        r.sign = x.sign;
        q.trim();
        r.trim();
        return {q, r / norm};
    }
    friend BigInt operator/(const BigInt &x, const BigInt &y) { return divmod(x, y).first; }
    friend BigInt operator%(const BigInt &x, const BigInt &y) { return divmod(x, y).second; }
    BigInt &operator/=(const BigInt &v) { return *this = *this / v; }
    BigInt &operator%=(const BigInt &v) { return *this = *this % v; }
    // ===== 10. 数论扩展：不需要 gcd / lcm 时以下代码可直接删除 =====
    // 依赖 BigInt 除法、取模和乘法。
    friend BigInt gcd(BigInt x, BigInt y) {
        x.sign = y.sign = 1;
        while (!y.is_zero()) x %= y, swap(x, y);
        return x;
    }
    friend BigInt lcm(const BigInt &x, const BigInt &y) {
        if (x.is_zero() || y.is_zero()) return 0;
        return (x / gcd(x, y) * y).abs();
    }
};
```
##### C++ 常见裁剪方式
###### 只需要高精度加减
保留：
- 第 `1`、`2`、`4`、`5`、`6` 组
- 输入输出只保留实际需要的接口
可以删除：
- 第 `3`、`7`、`8`、`9`、`10` 组
###### 只需要高精度乘普通整数
例如计算阶乘：
```cpp
BigInt ans{1};
for (int i = 2; i <= n; ++ i) ans *= i;
```
保留第 `1`、`4`、`7` 组即可；比较、加减、大整数乘除都可以删除
###### 只需要两个高精度整数相乘
保留第 `1`、`3`、`4`、`8` 组
若位数不大，建议改用 `operator*` 下方 `/** ... **/` 中的朴素版本，此时第 `3` 组也可以删除
##### 接口说明
###### 基础
- `BigInt()`：构造 `0`
- `BigInt(v)`：由 `i64` 构造
- `BigInt(s)`：由十进制字符串构造
- `read(s)`：读取有符号十进制字符串
- `is_zero()`：判断是否为 `0`
- `abs()`：返回绝对值
- `cin >> x / cout << x`：直接输入输出
##### 比较与加减
支持：
```cpp
a < b
a <= b
a > b
a >= b
a == b
a != b

a + b
a - b
a += b
a -= b
-a
```
##### 与普通整数运算
支持：
```cpp
a * k
k * a
a / k
a % k
a *= k
a /= k
```
其中 `k` 为 `i64`
###### 大整数乘除
支持：
```cpp
a * b
a / b
a % b
a *= b
a /= b
a %= b
```
同时需要商和余数时不要分别做 `/` 和 `%`，否则会执行两遍除法：
```cpp
auto [q, r] = divmod(a, b);
```
###### 数论
```cpp
BigInt g = gcd(a, b);
BigInt l = lcm(a, b);
```
结果均非负
##### 注意事项
- 内部以 `BASE = 10^9` 存储，不要直接把 `a[i]` 当十进制单个数位
- `0` 统一表示为 `a.empty()`，且 `sign = 1`
- `/ 0`、`% 0` 属于非法操作，代码中使用 `assert` 检查
- `BigInt / BigInt` 与 C++ 整数语义一致：商向 `0` 截断，余数与被除数同号
- 高性能 `operator*` 使用 Karatsuba；普通题若整数不长，使用其下方的朴素版本通常更值得，代码短很多
- 切换到朴素乘法后，应同时删除 `VLL`、`convert_base()`、`karatsuba_multiply()`
- 若只需要阶乘、组合递推等“高精度数 × 普通整数”操作，不要保留大整数乘法和除法
- 若题目允许 Python 且整体算法复杂度适合 Python，单纯为了高精度整数通常优先考虑 Python `int`
- Python 的 `//`、`%` 在负数时与 C++ 语义不同
- Python 的 `Decimal` 是**指定精度**十进制运算，不等价于所有运算都无限精确
- `Decimal` 应从字符串构造，避免先转换为二进制 `float`
##### 常见用法
###### 高精度阶乘
```cpp
BigInt ans{1};
for (int i = 2; i <= n; ++ i) ans *= i;
cout << ans << '\n';
```
这种题只需要“乘普通整数”，可以大量裁剪代码
###### 两个超长整数四则运算
```cpp
BigInt a, b;
cin >> a >> b;

cout << a + b << '\n';
cout << a - b << '\n';
cout << a * b << '\n';

if (!b.is_zero()) {
    auto [q, r] = divmod(a, b);
    cout << q << '\n';
    cout << r << '\n';
}
```
##### 复杂度
设 `n,m` 为两个高精度整数以 $10^9$ 为底后的块数
- 输入输出：$O(n)$
- 比较：$O(\max(n,m))$
- 加减：$O(\max(n,m))$
- 乘 / 除 / 模普通整数：$O(n)$
- Karatsuba 大整数乘法：$O(N^{\log_2 3})\approx O(N^{1.585})$
- `/** ... **/` 中朴素大整数乘法：$O(nm)$
- 大整数除法 `divmod()`：$O(nm)$
- `gcd()`：复杂度取决于欧几里得算法过程中各次大整数除法
- 空间复杂度通常为 $O(n+m)$
#### Python 原生高精度整数
Python 的 `int` 原生支持任意精度整数，不需要额外高精度类
##### 基础模板
```python
import sys
input = sys.stdin.readline

# 超长十进制整数输入 / 输出时关闭字符串转换位数限制
# 老版本 Python 没有该接口，因此先判断
if hasattr(sys, "set_int_max_str_digits"):
    sys.set_int_max_str_digits(0)
```
之后直接使用普通整数：
```python
a = int(input())
b = int(input())

c = a + b
d = a - b
e = a * b
q = a // b
r = a % b

x = pow(a, k)       # a^k
y = pow(a, k, mod)  # a^k mod mod，直接使用快速模幂
```
常用整数函数：
```python
from math import gcd, lcm, isqrt, factorial, comb

g = gcd(a, b)
l = lcm(a, b)
r = isqrt(a)        # floor(sqrt(a))，精确整数运算
f = factorial(n)
c = comb(n, k)
```
##### Python 负数除法与 C++ 不同
Python：
```python
-7 // 3 == -3
-7 % 3 == 2
```
因为 `//` 向负无穷方向取整
C++ 的整数除法向 `0` 截断：
```cpp
-7 / 3 == -2
-7 % 3 == -1
```
若 Python 必须模拟 C++ 的语义：
```python
q = abs(a) // abs(b)
if (a < 0) ^ (b < 0):
    q = -q
r = a - q * b
```

##### Python 高精度十进制浮点数
需要超过普通 `float` 精度时使用 `Decimal`：
```python
from decimal import Decimal, getcontext

getcontext().prec = 100

a = Decimal(input())
b = Decimal(input())

print(a + b)
print(a - b)
print(a * b)
print(a / b)
```
`prec` 表示运算结果保留的**有效数字位数**
输入十进制小数时优先：
```python
x = Decimal("0.1")
```
或：
```python
x = Decimal(input())
```
不要先经过 `float`：
```python
x = Decimal(0.1)    # 得到的是二进制 float 0.1 的精确值，不是十进制 0.1
```

##### Python 精确分数
若题目要求有理数始终保持精确，可以直接使用：
```python
from fractions import Fraction

a = Fraction(1, 3)
b = Fraction(2, 5)

print(a + b)
print(a * b)
print(a.numerator, a.denominator)
```
没有有理数需求时整段忽略
