### 高精度
---
#### C++ 高精度整数
```cpp
struct BigInt {
    static constexpr int BASE{ 1000000000 };
    static constexpr int DIGITS{ 9 };
    static constexpr int MUL_NAIVE{ 64 };
    static constexpr int DIV_NAIVE{ 64 };
    vector<int> a;
    int sign{ 1 };
private:
    using Vec = vector<int>;
    using u128 = __uint128_t;
    // ===== 1. 基础维护：所有功能依赖 =====
    static void trim_vec(Vec &v) { while (!v.empty() && v.back() == 0) v.pop_back(); }
    void trim() { trim_vec(a); if (a.empty()) sign = 1; }
    static int cmp_vec(const Vec &x, const Vec &y) {
        if (x.size() != y.size()) return x.size() < y.size() ? -1 : 1;
        for (int i = int(x.size()) - 1; i >= 0; -- i) if (x[i] != y[i]) return x[i] < y[i] ? -1 : 1;
        return 0;
    }
    static int abs_cmp(const BigInt &x, const BigInt &y) { return cmp_vec(x.a, y.a); }
    // ===== 2. 无符号向量加减与小整数乘除：快速乘除法内部依赖 =====
    static Vec add_vec(const Vec &x, const Vec &y) {
        int n = max(x.size(), y.size()), carry{ 0 };
        Vec z(n);
        for (int i = 0; i < n; ++ i) {
            i64 cur{ i64(carry) + (i < x.size() ? x[i] : 0) + (i < y.size() ? y[i] : 0) };
            if (cur >= BASE) cur -= BASE, carry = 1;
            else carry = 0;
            z[i] = int(cur);
        }
        if (carry) z.push_back(carry);
        return z;
    }
    // 要求 x >= y
    static Vec sub_vec(const Vec &x, const Vec &y) {
        Vec z{ x };
        int borrow{ 0 };
        for (int i = 0; i < z.size(); ++ i) {
            i64 cur{ i64(z[i]) - borrow - (i < y.size() ? y[i] : 0) };
            if (cur < 0) cur += BASE, borrow = 1;
            else borrow = 0;
            z[i] = int(cur);
        }
        trim_vec(z);
        return z;
    }
    static Vec mul_small_vec(const Vec &x, int v) {
        if (x.empty() || v == 0) return {};
        if (v == 1) return x;
        Vec z;
        z.reserve(x.size() + 1);
        i64 carry{ 0 };
        for (int d : x) {
            i64 cur{ i64(d) * v + carry };
            z.push_back(int(cur % BASE));
            carry = cur / BASE;
        }
        if (carry) z.push_back(int(carry));
        return z;
    }
    // v 满足 1 <= v < BASE，返回 { 商, 余数 }
    static pair<Vec, int> div_small_vec(const Vec &x, int v) {
        assert(1 <= v && v < BASE);
        Vec q(x.size());
        i64 rem{ 0 };
        for (int i = int(x.size()) - 1; i >= 0; -- i) {
            i64 cur{rem * BASE + x[i]};
            q[i] = int(cur / v);
            rem = cur % v;
        }
        trim_vec(q);
        return {move(q), int(rem)};
    }
    // ===== 3. 精确 NTT 乘法：BigInt * BigInt 与 Newton 除法依赖 =====
    // 小规模自动退化为朴素乘法；完全不需要 BigInt 乘除时，本组可整体删除
    template<int MOD, int G>
    struct NTT {
        static int power(int a, int b) {
            i64 r{ 1 };
            for (; b; b >>= 1, a = int(i64(a) * a % MOD)) if (b & 1) r = r * a % MOD;
            return int(r);
        }
        static void transform(vector<int> &a, bool inverse) {
            int n = a.size();
            for (int i = 1, j = 0; i < n; ++ i) {
                int bit{ n >> 1 };
                for (; j & bit; bit >>= 1) j ^= bit;
                j ^= bit;
                if (i < j) swap(a[i], a[j]);
            }
            for (int len = 2; len <= n; len <<= 1) {
                int wlen{ power(G, (MOD - 1) / len) };
                if (inverse) wlen = power(wlen, MOD - 2);
                for (int i = 0; i < n; i += len) {
                    i64 w{ 1 };
                    for (int j = 0; j < len / 2; ++ j) {
                        int u{ a[i + j] }, v{ int(a[i + j + len / 2] * w % MOD) };
                        int x{ u + v };
                        if (x >= MOD) x -= MOD;
                        int y{ u - v };
                        if (y < 0) y += MOD;
                        a[i + j] = x;
                        a[i + j + len / 2] = y;
                        w = w * wlen % MOD;
                    }
                }
            }
            if (inverse) {
                int inv_n{ power(n, MOD - 2) };
                for (int &x : a) x = int(i64(x) * inv_n % MOD);
            }
        }
        static vector<int> convolution(const vector<int> &x, const vector<int> &y, bool square) {
            if (x.empty() || y.empty()) return {};
            int need{ int(x.size() + y.size() - 1) }, n{ 1 };
            while (n < need) n <<= 1;
            assert((MOD - 1) % n == 0);
            vector<int> a(n);
            copy(x.begin(), x.end(), a.begin());
            transform(a, false);
            if (square) {
                for (int &v : a) v = int(i64(v) * v % MOD);
            } else {
                vector<int> b(n);
                copy(y.begin(), y.end(), b.begin());
                transform(b, false);
                for (int i = 0; i < n; ++ i) a[i] = int(i64(a[i]) * b[i] % MOD);
            }
            transform(a, true);
            a.resize(need);
            return a;
        }
    };
    static Vec convert_base(const Vec &v, int old_digits, int new_digits) {
        if (v.empty()) return {};
        array<i64, 10> p{};
        p[0] = 1;
        for (int i = 1; i <= max(old_digits, new_digits); ++ i) p[i] = p[i - 1] * 10;
        Vec res;
        res.reserve((i64(v.size()) * old_digits + new_digits - 1) / new_digits);
        i64 cur{ 0 };
        int cur_digits{ 0 };
        for (int x : v) {
            cur += i64(x) * p[cur_digits];
            cur_digits += old_digits;
            while (cur_digits >= new_digits) {
                res.push_back(int(cur % p[new_digits]));
                cur /= p[new_digits];
                cur_digits -= new_digits;
            }
        }
        if (cur) res.push_back(int(cur));
        trim_vec(res);
        return res;
    }
    static Vec mul_naive_vec(const Vec &x, const Vec &y) {
        if (x.empty() || y.empty()) return {};
        const Vec *a{ &x }, *b{ &y };
        if (a->size() < b->size()) swap(a, b);
        Vec z(a->size() + b->size());
        for (int i = 0; i < int(b->size()); ++ i) {
            u128 carry{0};
            for (int j = 0; j < int(a->size()) || carry; ++ j) {
                u128 cur{ u128(z[i + j]) + carry };
                if (j < int(a->size())) cur += u128((*b)[i]) * (*a)[j];
                z[i + j] = int(cur % BASE);
                carry = cur / BASE;
            }
        }
        trim_vec(z);
        return z;
    }
    static Vec mul_ntt_vec(const Vec &x, const Vec &y, bool square = false) {
        static constexpr int B{ 10000 }, BD{ 4 };
        static constexpr int M0{ 998244353 }, M1{ 469762049 };
        static constexpr i64 INV_M0_MOD_M1{ 208783132 };
        Vec a4{ convert_base(x, DIGITS, BD) };
        Vec b4{ square ? a4 : convert_base(y, DIGITS, BD) };
        auto c0{ NTT<M0, 3>::convolution(a4, b4, square) };
        auto c1{ NTT<M1, 3>::convolution(a4, b4, square) };
        Vec c4;
        c4.reserve(c0.size() + 8);
        u128 carry{ 0 };
        for (int i = 0; i < c0.size(); ++ i) {
            i64 d{ c1[i] - c0[i] % M1 };
            if (d < 0) d += M1;
            i64 t{ d * INV_M0_MOD_M1 % M1 };
            u128 cur{ u128(c0[i]) + u128(M0) * t + carry };
            c4.push_back(int(cur % B));
            carry = cur / B;
        }
        while (carry) {
            c4.push_back(int(carry % B));
            carry /= B;
        }
        return convert_base(c4, BD, DIGITS);
    }
    static Vec mul_vec(const Vec &x, const Vec &y) {
        if (x.empty() || y.empty()) return {};
        if (min(x.size(), y.size()) <= MUL_NAIVE) return mul_naive_vec(x, y);
        return mul_ntt_vec(x, y, &x == &y);
    }
    // ===== 4. 无符号长除法：仅供小除数或短商使用；Newton 的初值也依赖 =====
    static pair<Vec, Vec> divmod_naive_vec(const Vec &a, const Vec &b) {
        assert(!b.empty());
        if (a.empty()) return {{}, {}};
        if (b.size() == 1) {
            auto [q, r]{ div_small_vec(a, b[0]) };
            return { move(q), r ? Vec{r} : Vec{} };
        }
        if (cmp_vec(a, b) < 0) return {{}, a};
        int norm{ BASE / (b.back() + 1) };
        Vec x{ mul_small_vec(a, norm) }, y{ mul_small_vec(b, norm) };
        Vec q(x.size() - y.size() + 1);
        Vec r(x.end() - y.size(), x.end());
        trim_vec(r);
        for (int i = int(q.size()) - 1; i >= 0; -- i) {
            if (r.size() == y.size()) {
                if (cmp_vec(r, y) >= 0) q[i] = 1, r = sub_vec(r, y);
            } else if (r.size() > y.size()) {
                assert(r.size() == y.size() + 1);
                i64 rb{ i64(r.back()) * BASE + r[r.size() - 2] };
                i64 d{ min<i64>(BASE - 1, rb / y.back()) };
                Vec yd{ mul_small_vec(y, int(d)) };
                while (cmp_vec(r, yd) < 0) -- d, yd = sub_vec(yd, y);
                r = sub_vec(r, yd);
                while (cmp_vec(r, y) >= 0) ++ d, r = sub_vec(r, y);
                q[i] = int(d);
            }
            if (i) {
                r.insert(r.begin(), x[i - 1]);
                trim_vec(r);
            }
        }
        trim_vec(q);
        auto [rr, rem]{ div_small_vec(r, norm) };
        assert(rem == 0);
        return {move(q), move(rr)};
    }
    // ===== 5. Newton 倒数除法：大整数除法核心；依赖第 3/4 组 =====
    // 计算 floor(BASE^(a.size()+deg) / a)，要求 a.back() >= BASE / 2。
    static Vec calc_inv(const Vec &a, int deg) {
        assert(!a.empty() && a.back() >= BASE / 2);
        int k{ deg }, c = a.size();
        while (k > DIV_NAIVE) k = (k + 1) / 2;
        Vec z(c + k + 1);
        z.back() = 1;
        z = divmod_naive_vec(z, a).first;
        while (k < deg) {
            Vec s{mul_vec(z, z)};
            s.insert(s.begin(), 0);
            int take{ min(c, 2 * k + 1) };
            Vec t(2 * k + 1);
            copy(a.end() - take, a.end(), t.end() - take);
            Vec u{ mul_vec(s, t) };
            if (2 * k + 1 >= u.size()) u.clear();
            else u.erase(u.begin(), u.begin() + 2 * k + 1);
            Vec w(k + 1), w2{ add_vec(z, z) };
            w.insert(w.end(), w2.begin(), w2.end());
            z = sub_vec(w, u);
            if (!z.empty()) z.erase(z.begin());
            k *= 2;
        }
        int cut{ k - deg };
        if (cut > 0) z.erase(z.begin(), z.begin() + min(cut, int(z.size())));
        trim_vec(z);
        return z;
    }
    static pair<Vec, Vec> divmod_vec(const Vec &a, const Vec &b) {
        assert(!b.empty());
        if (a.empty()) return {{}, {}};
        if (cmp_vec(a, b) < 0) return {{}, a};
        if (b.size() <= DIV_NAIVE || int(a.size()) - int(b.size()) <= DIV_NAIVE) return divmod_naive_vec(a, b);
        int norm{ BASE / (b.back() + 1) };
        Vec x{mul_small_vec(a, norm)}, y{ mul_small_vec(b, norm) };
        int s = x.size(), t = y.size(), deg{ s - t + 2 };
        Vec z{ calc_inv(y, deg) };
        Vec q{ mul_vec(x, z) };
        int cut{ t + deg };
        if (cut >= q.size()) q.clear();
        else q.erase(q.begin(), q.begin() + cut);
        trim_vec(q);
        Vec yq{ mul_vec(y, q) };
        const Vec one{ 1 };
        while (cmp_vec(x, yq) < 0) q = sub_vec(q, one), yq = sub_vec(yq, y);
        Vec r{ sub_vec(x, yq) };
        while (cmp_vec(r, y) >= 0) q = add_vec(q, one), r = sub_vec(r, y);
        auto [rr, rem]{ div_small_vec(r, norm) };
        assert(rem == 0);
        trim_vec(q);
        return {move(q), move(rr)};
    }
    // ===== 6. 绝对值原地加减：公开 +/- 依赖 =====
    void add_abs(const BigInt &v) {
        int n = int(max(a.size(), v.a.size()));
        a.resize(n, 0);
        int carry{ 0 };
        for (int i = 0; i < n; ++ i) {
            i64 cur{ i64(a[i]) + carry + (i < v.a.size() ? v.a[i] : 0) };
            if (cur >= BASE) cur -= BASE, carry = 1;
            else carry = 0;
            a[i] = int(cur);
        }
        if (carry) a.push_back(carry);
    }
    void sub_abs(const BigInt &v) {
        int borrow{0};
        for (int i = 0; i < a.size(); ++ i) {
            i64 cur{ i64(a[i]) - borrow - (i < v.a.size() ? v.a[i] : 0) };
            if (cur < 0) cur += BASE, borrow = 1;
            else borrow = 0;
            a[i] = int(cur);
            if (i >= v.a.size() && !borrow) break;
        }
        trim();
    }
public:
    // ===== 7. 构造、符号与输入输出：基础接口，通常保留 =====
    BigInt() = default;
    BigInt(i64 v) { *this = v; }
    BigInt(const string &s) { read(s); }
    bool is_zero() const { return a.empty(); }
    BigInt abs() const { BigInt res{*this}; res.sign = 1; return res; }
    BigInt &operator=(i64 v) {
        a.clear();
        sign = v < 0 ? -1 : 1;
        u64 x{u64(v)};
        if (v < 0) x = 0 - x;
        while (x) a.push_back(int(x % BASE)), x /= BASE;
        return *this;
    }
    BigInt operator-() const {
        BigInt res{*this};
        if (!res.is_zero()) res.sign = -res.sign;
        return res;
    }
    void read(const string &s) {
        a.clear();
        sign = 1;
        int pos{ 0 };
        if (!s.empty() && (s[0] == '+' || s[0] == '-')) sign = s[0] == '-' ? -1 : 1, pos = 1;
        a.reserve((s.size() - pos + DIGITS - 1) / DIGITS);
        for (int i = s.size(); i > pos; i -= DIGITS) {
            int x{ 0 };
            for (int j = max(pos, i - DIGITS); j < i; ++ j) x = x * 10 + s[j] - '0';
            a.push_back(x);
        }
        trim();
    }
    string to_string() const {
        if (a.empty()) return "0";
        string s;
        s.reserve(a.size() * DIGITS + (sign < 0));
        if (sign < 0) s.push_back('-');
        s += std::to_string(a.back());
        char buf[DIGITS];
        for (int i = int(a.size()) - 2; i >= 0; -- i) {
            int x{ a[i] };
            for (int j = DIGITS - 1; j >= 0; -- j) buf[j] = char('0' + x % 10), x /= 10;
            s.append(buf, DIGITS);
        }
        return s;
    }
    friend istream &operator>>(istream &in, BigInt &v) { string s; if (in >> s) v.read(s); return in; }
    friend ostream &operator<<(ostream &out, const BigInt &v) { return out << v.to_string(); }
    // ===== 8. 比较：通常保留 =====
    friend bool operator<(const BigInt &x, const BigInt &y) {
        if (x.sign != y.sign) return x.sign < y.sign;
        int c{ abs_cmp(x, y) };
        return x.sign > 0 ? c < 0 : c > 0;
    }
    friend bool operator>(const BigInt &x, const BigInt &y) { return y < x; }
    friend bool operator<=(const BigInt &x, const BigInt &y) { return !(y < x); }
    friend bool operator>=(const BigInt &x, const BigInt &y) { return !(x < y); }
    friend bool operator==(const BigInt &x, const BigInt &y) { return x.sign == y.sign && x.a == y.a; }
    friend bool operator!=(const BigInt &x, const BigInt &y) { return !(x == y); }
    // ===== 9. BigInt 加减：不用 +/- 时可连同第 6 组删除 =====
    BigInt &operator+=(const BigInt &v) {
        if (sign == v.sign) add_abs(v);
        else if (abs_cmp(*this, v) >= 0) sub_abs(v);
        else {
            BigInt t{v};
            t.sub_abs(*this);
            *this = move(t);
        }
        return *this;
    }
    BigInt &operator-=(const BigInt &v) { return *this += -v; }
    friend BigInt operator+(BigInt x, const BigInt &y) { return x += y; }
    friend BigInt operator-(BigInt x, const BigInt &y) { return x -= y; }
    // ===== 10. 与普通整数乘除：不用时可直接删除 =====
    BigInt &operator*=(i64 v) {
        if (v == 0 || is_zero()) return *this = 0;
        if (v < 0) sign = -sign;
        u64 x{u64(v)};
        if (v < 0) x = 0 - x;
        if (x == 1) return *this;
        if (x < BASE) {
            i64 carry{0};
            for (int &d : a) {
                i64 cur{i64(d) * i64(x) + carry};
                d = int(cur % BASE);
                carry = cur / BASE;
            }
            if (carry) a.push_back(int(carry));
        } else {
            u128 carry{0};
            for (int &d : a) {
                u128 cur{u128(d) * x + carry};
                d = int(cur % BASE);
                carry = cur / BASE;
            }
            while (carry) a.push_back(int(carry % BASE)), carry /= BASE;
        }
        return *this;
    }
    BigInt &operator/=(i64 v) {
        assert(v != 0);
        if (is_zero()) return *this;
        if (v < 0) sign = -sign;
        u64 x{u64(v)};
        if (v < 0) x = 0 - x;
        if (x == 1) return *this;
        if (x < BASE) {
            i64 rem{ 0 };
            for (auto &val : views::reverse(a)) {
//            for (int i = int(a.size()) - 1; i >= 0; -- i) {
                i64 cur{ rem * BASE + val };
                val = int(cur / x);
                rem = cur % x;
            }
        } else {
            u128 rem{ 0 };
            for (auto &val : views::reverse(a)) {
//            for (int i = int(a.size()) - 1; i >= 0; -- i) {
                u128 cur{ rem * BASE + val };
                val = int(cur / x);
                rem = cur % x;
            }
        }
        trim();
        return *this;
    }
    i64 operator%(i64 v) const {
        assert(v != 0);
        u64 x{u64(v)};
        if (v < 0) x = 0 - x;
        if (x == 1 || is_zero()) return 0;
        u64 rem{ 0 };
        if (x < BASE) {
            for (const auto &val : views::reverse(a)) rem = (rem * BASE + val) % x;
//            for (int i = int(a.size()) - 1; i >= 0; -- i) rem = (rem * BASE + a[i]) % x;
        } else {
            u128 r{ 0 };
            for (const auto &val : views::reverse(a)) r = (r * BASE + val) % x;
//            for (int i = int(a.size()) - 1; i >= 0; -- i) r = (r * BASE + a[i]) % x;
            rem = u64(r);
        }
        i64 r{i64(rem)};
        return sign < 0 ? -r : r;
    }
    friend BigInt operator*(BigInt x, i64 v) { return x *= v; }
    friend BigInt operator*(i64 v, BigInt x) { return x *= v; }
    friend BigInt operator/(BigInt x, i64 v) { return x /= v; }
    // ===== 11. BigInt 乘法：小规模朴素，大规模双 NTT 精确卷积 =====
    BigInt operator*(const BigInt &v) const {
        if (is_zero() || v.is_zero()) return 0;
        BigInt res;
        res.sign = sign * v.sign;
        res.a = mul_vec(a, v.a);
        res.trim();
        return res;
    }
    BigInt &operator*=(const BigInt &v) { return *this = *this * v; }
    // ===== 12. BigInt 除法与取模：小规模长除法，大规模 Newton 倒数 =====
    // 同时需要商和余数时务必直接调用 divmod()，避免 / 与 % 各做一遍完整除法。
    friend pair<BigInt, BigInt> divmod(const BigInt &x, const BigInt &y) {
        assert(!y.is_zero());
        auto [qa, ra]{divmod_vec(x.a, y.a)};
        BigInt q, r;
        q.a = move(qa);
        r.a = move(ra);
        if (!q.a.empty()) q.sign = x.sign * y.sign;
        if (!r.a.empty()) r.sign = x.sign;
        return {move(q), move(r)};
    }
    friend BigInt operator/(const BigInt &x, const BigInt &y) { return divmod(x, y).first; }
    friend BigInt operator%(const BigInt &x, const BigInt &y) { return divmod(x, y).second; }
    BigInt &operator/=(const BigInt &v) { return *this = divmod(*this, v).first; }
    BigInt &operator%=(const BigInt &v) { return *this = divmod(*this, v).second; }
    // ===== 13. 数论扩展：不需要 gcd / lcm 时以下代码可直接删除 =====
    friend BigInt gcd(BigInt x, BigInt y) {
        x.sign = y.sign = 1;
        while (!y.is_zero()) {
            auto [q, r]{divmod(x, y)};
            x = move(y);
            y = move(r);
        }
        return x;
    }
    friend BigInt lcm(const BigInt &x, const BigInt &y) {
        if (x.is_zero() || y.is_zero()) return 0;
        return (x / gcd(x, y) * y).abs();
    }
};
```
##### 成员变量
- `a[i]`：绝对值按 $10^9$ 进制低位在前存储，零用空 `vector` 表示
- `sign`：符号，非零时为 `1 / -1`，零固定为 `1`
- `MUL_NAIVE`：乘法朴素 / NTT 的切换阈值，默认 `64`
- `DIV_NAIVE`：长除法 / Newton 除法的切换阈值，默认 `64`
##### 接口
- `BigInt(v)` / `BigInt(s)`：由 `i64` 或十进制字符串构造，$O(n)$
- `read(s)`：读取十进制字符串，$O(n)$
- `to_string()`：转十进制字符串，$O(n)$
- `is_zero()`：判断是否为零，$O(1)$
- `abs()`：返回绝对值副本，$O(n)$
- 比较运算：最坏 $O(n)$
- `+ / -`：高精加减，$O(n)$
- `* / / / %` 与 `i64`：$O(n)$
- `a * b`：小规模朴素乘法，大规模双 NTT 精确卷积
- `divmod(a, b)`：同时返回 `{a / b, a % b}`；小规模 / 短商走长除法，大规模走 Newton 倒数除法
- `a / b`：返回商
- `a % b`：返回余数
- `gcd(a, b)` / `lcm(a, b)`：高精最大公约数 / 最小公倍数
##### 注意事项
- 整数除法语义与 C++ 一致：商向 `0` 截断，余数与被除数同号
- 同时需要商和余数时使用 `divmod(a, b)`，不要分别写 `a / b` 与 `a % b`，否则会完整计算两遍除法
- NTT 使用 `998244353` 与 `469762049` 做 CRT，卷积内部转成 $10^4$ 进制；受第一模数限制，单次 NTT 长度最大为 $2^23$
- 当前实现覆盖 Library Checker `Big Integer` 的 $2	imes10^6$ 十进制位规模；若要处理远超千万位的单个整数，需要更换支持更长 2-adic 长度的 NTT 模数或改成其他卷积实现
- `MUL_NAIVE` / `DIV_NAIVE` 是常数阈值，不同评测机可以在 `32 ~ 128` 左右微调
##### 常见用法
###### 同时求商和余数
```cpp
auto [q, r] = divmod(a, b);
```
###### 只求商或余数
```cpp
BigInt q = a / b;
BigInt r = a % b;
```
###### 与普通整数运算
```cpp
BigInt x = a * 123456789LL;
x /= 998244353;
i64 r = x % 1000000007LL;
```
##### 复杂度
设 `n, m` 为 $10^9$ 进制 limb 数，$M(n)$ 为快速乘法复杂度

| 操作 | 时间复杂度 |
|---|---|
| 比较 / 加减 | $O(\max(n,m))$ |
| 与 `i64` 乘除模 | $O(n)$ |
| 小规模 BigInt 乘法 | $O(nm)$ |
| 大规模 BigInt 乘法 | $O(N\log N)$，`N` 为 NTT 长度 |
| 小除数 / 短商除法 | 近似 $O(n\min(m,n-m+1))$ |
| 大规模除法 | 约 $O(M(n)\log n)$ |
| `gcd` | 取决于欧几里得迭代次数与每次除法 |
空间复杂度：加减为 $O(n)$；快速乘除为 $O(N)$
#### BigInt（赛场短版）
```cpp
struct BigInt {
    static constexpr int BASE{ 1000000000 }, DIGITS{ 9 };
    vector<int> a;
    int sign{ 1 };
private:
    // ===== 1. 核心：所有模块依赖 =====
    void trim() { while (!a.empty() && a.back() == 0) a.pop_back(); if (a.empty()) sign = 1; }
    static int abs_cmp(const BigInt &x, const BigInt &y) {
        if (x.a.size() != y.a.size()) return x.a.size() < y.a.size() ? -1 : 1;
        for (int i = int(x.a.size()) - 1; i >= 0; -- i) if (x.a[i] != y.a[i]) return x.a[i] < y.a[i] ? -1 : 1;
        return 0;
    }
public:
    BigInt() = default;
    BigInt(i64 v) { *this = v; }
    BigInt(const string &s) { read(s); }
    bool is_zero() const { return a.empty(); }
    BigInt abs() const { BigInt r{*this}; r.sign = 1; return r; }
    BigInt &operator=(i64 v) {
        a.clear();
        sign = v < 0 ? -1 : 1;
        u64 x{ u64(v) };
        if (v < 0) x = 0 - x;
        while (x) a.push_back(x % BASE), x /= BASE;
        return *this;
    }
    BigInt operator-() const { BigInt r{ *this }; if (!r.is_zero()) r.sign = -r.sign; return r; }
    void read(const string &s) {
        a.clear(); sign = 1;
        int p{ 0 }; if (!s.empty() && (s[0] == '+' || s[0] == '-')) sign = s[0] == '-' ? -1 : 1, p = 1;
        for (int i = int(s.size()); i > p; i -= DIGITS) {
            int x{ 0 };
            for (int j = max(p, i - DIGITS); j < i; ++ j) x = x * 10 + s[j] - '0';
            a.push_back(x);
        }
        trim();
    }
    friend istream &operator>>(istream &in, BigInt &x) { string s; if (in >> s) x.read(s); return in; }
    friend ostream &operator<<(ostream &out, const BigInt &x) {
        if (x.sign < 0) out << '-';
        if (x.a.empty()) return out << 0;
        out << x.a.back();
        for (int i = int(x.a.size()) - 2; i >= 0; -- i) out << setw(DIGITS) << setfill('0') << x.a[i];
        return out << setfill(' ');
    }

    // ===== 2. 比较 =====
    friend bool operator<(const BigInt &x, const BigInt &y) {
        if (x.sign != y.sign) return x.sign < y.sign;
        int c{ abs_cmp(x, y) };
        return x.sign > 0 ? c < 0 : c > 0;
    }
    friend bool operator==(const BigInt &x, const BigInt &y) { return x.sign == y.sign && x.a == y.a; }
    friend bool operator!=(const BigInt &x, const BigInt &y) { return !(x == y); }
    friend bool operator>(const BigInt &x, const BigInt &y) { return y < x; }
    friend bool operator<=(const BigInt &x, const BigInt &y) { return !(y < x); }
    friend bool operator>=(const BigInt &x, const BigInt &y) { return !(x < y); }

private:
    // ===== 3. BigInt 加减 =====
    void add_abs(const BigInt &v) {
        a.resize(max(a.size(), v.a.size()));
        int carry{ 0 };
        for (int i = 0; i < a.size(); ++ i) {
            i64 cur{ i64(a[i]) + carry + (i < v.a.size() ? v.a[i] : 0) };
            a[i] = cur % BASE; carry = cur / BASE;
        }
        if (carry) a.push_back(carry);
    }
    void sub_abs(const BigInt &v) {
        int borrow{0};
        for (int i = 0; i < a.size(); ++ i) {
            i64 cur{ i64(a[i]) - borrow - (i < v.a.size() ? v.a[i] : 0) };
            if (cur < 0) cur += BASE, borrow = 1; else borrow = 0;
            a[i] = cur;
        }
        trim();
    }
public:
    BigInt &operator+=(const BigInt &v) {
        if (sign == v.sign) add_abs(v);
        else if (abs_cmp(*this, v) >= 0) sub_abs(v);
        else { BigInt t{v}; t.sub_abs(*this); *this = move(t); }
        return *this;
    }
    BigInt &operator-=(const BigInt &v) { return *this += -v; }
    friend BigInt operator+(BigInt x, const BigInt &y) { return x += y; }
    friend BigInt operator-(BigInt x, const BigInt &y) { return x -= y; }
    // ===== 4. 与 i64 乘 / 除 / 模：可按函数单独删除 =====
    BigInt &operator*=(i64 v) {
        if (!v || is_zero()) return *this = 0;
        if (v < 0) sign = -sign;
        u64 x{ u64(v) };
        if (v < 0) x = 0 - x;
        __uint128_t carry{ 0 };
        for (int &d : a) {
            __uint128_t cur{__uint128_t(d) * x + carry};
            d = cur % BASE; carry = cur / BASE;
        }
        while (carry) a.push_back(carry % BASE), carry /= BASE;
        return *this;
    }
    BigInt &operator/=(i64 v) {
        assert(v); if (is_zero()) return *this;
        if (v < 0) sign = -sign;
        u64 x{ u64(v) };
        if (v < 0) x = 0 - x;
        __uint128_t rem{ 0 };
        for (int i = int(a.size()) - 1; i >= 0; -- i) {
            __uint128_t cur{ rem * BASE + a[i] };
            a[i] = cur / x; rem = cur % x;
        }
        trim(); return *this;
    }
    i64 operator%(i64 v) const {
        assert(v); u64 x{ u64(v) }; if (v < 0) x = 0 - x;
        __uint128_t r{ 0 };
        for (int i = int(a.size()) - 1; i >= 0; -- i) r = (r * BASE + a[i]) % x;
        i64 z{i64(u64(r))};
        return sign < 0 ? -z : z;
    }
    friend BigInt operator*(BigInt x, i64 v) { return x *= v; }
    friend BigInt operator*(i64 v, BigInt x) { return x *= v; }
    friend BigInt operator/(BigInt x, i64 v) { return x /= v; }
    // ===== 5. BigInt 乘法：朴素 O(nm) =====
    friend BigInt operator*(const BigInt &x, const BigInt &y) {
        if (x.is_zero() || y.is_zero()) return 0;
        BigInt z;
        z.sign = x.sign * y.sign;
        z.a.assign(x.a.size() + y.a.size() + 1, 0);
        for (int i = 0; i < x.a.size(); ++ i) {
            __uint128_t carry{ 0 };
            for (int j = 0; j < y.a.size() || carry; ++ j) {
                __uint128_t cur{ __uint128_t(z.a[i + j]) + carry };
                if (j < y.a.size()) cur += __uint128_t(x.a[i]) * y.a[j];
                z.a[i + j] = cur % BASE; carry = cur / BASE;
            }
        }
        z.trim(); return z;
    }
    BigInt &operator*=(const BigInt &v) { return *this = *this * v; }
private:
    // ===== 6. BigInt 除法：Knuth 长除法；与 +/-、乘法模块无依赖 =====
    static pair<vector<int>, vector<int>> divmod_abs(const vector<int> &aa, const vector<int> &bb) {
        if (bb.size() == 1) {
            vector<int> q(aa.size()); i64 r{ 0 };
            for (int i = int(aa.size()) - 1; i >= 0; -- i) {
                i64 cur{r * BASE + aa[i]}; q[i] = cur / bb[0]; r = cur % bb[0];
            }
            while (!q.empty() && q.back() == 0) q.pop_back();
            return { move(q), r ? vector<int>{ int(r) } : vector<int>{} };
        }
        if (aa.size() < bb.size() || (aa.size() == bb.size() && lexicographical_compare(aa.rbegin(), aa.rend(), bb.rbegin(), bb.rend()))) return {{}, aa};
        int norm{ BASE / (bb.back() + 1) };
        auto mul = [&](const vector<int> &x) {
            vector<i64> z; z.reserve(x.size() + 1); i64 carry{0};
            for (int d : x) { i64 cur{i64(d) * norm + carry}; z.push_back(cur % BASE); carry = cur / BASE; }
            if (carry) z.push_back(carry);
            return z;
        };
        vector<i64> u{ mul(aa) }, v{ mul(bb) };
        u.push_back(0);
        int n = v.size(), m{ int(u.size()) - n - 1 };
        vector<int> q(m + 1);
        for (int j = m; j >= 0; -- j) {
            i64 num{ u[j + n] * BASE + u[j + n - 1] };
            i64 qt{ num / v[n - 1] }, rt{ num % v[n - 1] };
            if (qt == BASE) -- qt, rt += v[n - 1];
            while (n > 1 && rt < BASE && qt * v[n - 2] > BASE * rt + u[j + n - 2]) -- qt, rt += v[n - 1];
            i64 carry{ 0 };
            for (int i = 0; i < n; ++ i) {
                __int128 p{ __int128(qt) * v[i] + carry };
                i64 cur{ u[j + i] - i64(p % BASE) };
                carry = p / BASE;
                if (cur < 0) cur += BASE, ++ carry;
                u[j + i] = cur;
            }
            u[j + n] -= carry;
            if (u[j + n] < 0) {
                -- qt; i64 c{ 0 };
                for (int i = 0; i < n; ++ i) {
                    i64 cur{ u[j + i] + v[i] + c };
                    if (cur >= BASE) cur -= BASE, c = 1; else c = 0;
                    u[j + i] = cur;
                }
                u[j + n] += c;
            }
            q[j] = int(qt);
        }
        while (!q.empty() && q.back() == 0) q.pop_back();
        vector<int> r(n);
        for (int i = 0; i < n; ++ i) r[i] = u[i];
        i64 rem{ 0 };
        for (int i = n - 1; i >= 0; -- i) {
            i64 cur{ rem * BASE + r[i] }; r[i] = cur / norm; rem = cur % norm;
        }
        while (!r.empty() && r.back() == 0) r.pop_back();
        return {move(q), move(r)};
    }
public:
    // 同时需要商和余数时调用 divmod()，不要分别写 / 和 %
    friend pair<BigInt, BigInt> divmod(const BigInt &x, const BigInt &y) {
        assert(!y.is_zero());
        auto [qa, ra]{ divmod_abs(x.a, y.a) };
        BigInt q, r; q.a = move(qa); r.a = move(ra);
        if (!q.a.empty()) q.sign = x.sign * y.sign;
        if (!r.a.empty()) r.sign = x.sign;
        return {move(q), move(r)};
    }
    friend BigInt operator/(const BigInt &x, const BigInt &y) { return divmod(x, y).first; }
    friend BigInt operator%(const BigInt &x, const BigInt &y) { return divmod(x, y).second; }
    BigInt &operator/=(const BigInt &v) { return *this = divmod(*this, v).first; }
    BigInt &operator%=(const BigInt &v) { return *this = divmod(*this, v).second; }
};
```
##### 5B. BigInt 专用 FFT
模块 `5A`（上面结构中的朴素乘法）和本模块 `5B` **二选一**
这一版不依赖外部 FFT / NTT 板子，而是在 `BigInt` 内部放一套专用短 FFT：
- 固定使用 `complex<double>`
- `BASE = 1e9` 的一个 limb 恰好拆成三个 `FFT_BASE = 1e3` 系数
- 把两个实数组分别放进复数的实部和虚部，只做 **1 次正 FFT + 1 次逆 FFT**
- 不提供通用多项式卷积接口，只服务于高精乘法
```cpp
private:
    // ===== 5B. BigInt 专用 FFT 乘法：与模块 5A 二选一 =====
    using C = complex<double>;
    static constexpr double PI{ 3.1415926535897932384626433832795 };
    static constexpr int FFT_BASE{ 1000 };
    static void fft(vector<C> &a, bool inv) {
        int n = a.size();
        for (int i = 1, j = 0; i < n; ++ i) {
            int bit{ n >> 1 };
            for (; j & bit; bit >>= 1) j ^= bit;
            j ^= bit;
            if (i < j) swap(a[i], a[j]);
        }
        for (int len = 2; len <= n; len <<= 1) {
            double ang{ 2 * PI / len * (inv ? -1 : 1) };
            C wl{ cos(ang), sin(ang) };
            for (int i = 0; i < n; i += len) {
                C w{ 1 };
                for (int j = 0; j < len / 2; ++ j) {
                    C u{ a[i + j]}, v{a[i + j + len / 2] * w };
                    a[i + j] = u + v;
                    a[i + j + len / 2] = u - v;
                    w *= wl;
                }
            }
        }
        if (inv) for (C &x : a) x /= n;
    }
    static vector<int> fft_digits(const vector<int> &a) {
        vector<int> d;
        d.reserve(a.size() * 3);
        for (int x : a) {
            d.push_back(x % FFT_BASE); x /= FFT_BASE;
            d.push_back(x % FFT_BASE); x /= FFT_BASE;
            d.push_back(x);
        }
        while (!d.empty() && d.back() == 0) d.pop_back();
        return d;
    }

public:
    friend BigInt operator*(const BigInt &x, const BigInt &y) {
        if (x.is_zero() || y.is_zero()) return 0;
        vector<int> a{ fft_digits(x.a) }, b{fft_digits(y.a) };
        int need{ int(a.size() + b.size() - 1) }, n{ 1 };
        while (n < need) n <<= 1;
        // 实部放 a，虚部放 b；利用共轭对称性一次 FFT 同时得到两者频谱。
        vector<C> f(n);
        for (int i = 0; i < a.size(); ++ i) f[i].real(a[i]);
        for (int i = 0; i < b.size(); ++ i) f[i].imag(b[i]);
        fft(f, false);
        for (int i = 0; i <= n / 2; ++ i) {
            int j{ (n - i) & (n - 1) };
            C A{ (f[i] + conj(f[j])) * 0.5 };
            C B{ (f[i] - conj(f[j])) * C(0, -0.5) };
            C z{ A * B };
            f[i] = z;
            if (i != j) f[j] = conj(z);
        }
        fft(f, true);
        vector<int> d;
        d.reserve(need + 8);
        i64 carry{0};
        for (int i = 0; i < need; ++ i) {
            i64 cur{ llround(f[i].real()) + carry };
            d.push_back(cur % FFT_BASE);
            carry = cur / FFT_BASE;
        }
        while (carry) d.push_back(carry % FFT_BASE), carry /= FFT_BASE;
        while (!d.empty() && d.back() == 0) d.pop_back();
        BigInt z;
        z.sign = x.sign * y.sign;
        z.a.reserve((d.size() + 2) / 3);
        for (int i = 0; i < d.size(); i += 3) {
            int v{ d[i] };
            if (i + 1 < d.size()) v += d[i + 1] * FFT_BASE;
            if (i + 2 < d.size()) v += d[i + 2] * FFT_BASE * FFT_BASE;
            z.a.push_back(v);
        }
        z.trim();
        return z;
    }
    BigInt &operator*=(const BigInt &v) { return *this = *this * v; }
```
##### 裁剪索引

| 需求 | 保留模块 |
|---|---|
| 只存储 / 输入输出 | 1 |
| 比较 | 1 + 2 |

##### Python 负数除法与 C++ 不同


C++ 的整数除法向 `0` 截断：
```cpp

```

```python
q = abs(a) // abs(b)
if (a < 0) ^ (b < 0):
    q = -q
```

#### 已明确的数值与输入边界
BigInt 的 read/流输入要求有效十进制整数字符串（允许正负号与前导零）；不是带错误恢复的通用解析器。完整版精确 NTT 要求转换后的卷积长度不超过 2^23（由 998244353 限制）。短版 5B 使用 double FFT，存在浮点舍入风险，不能承诺任意长度精确；要求严格整数正确性时使用完整版精确 NTT 或短版 5A。
