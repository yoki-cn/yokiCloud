### Debug
#### 调试输出
```cpp
#ifndef ONLINE_JUDGE
struct Debug {
private:
    template<class T, class = void>
    struct IsIterable : false_type {};
    template<class T>
    struct IsIterable<T, void_t<
        decltype(begin(declval<const T &>())),
        decltype(end(declval<const T &>()))
    >> : true_type {};
    template<class T>
    struct IsPair : false_type {};
    template<class A, class B>
    struct IsPair<pair<A, B>> : true_type {};
    template<class T>
    struct IsTuple : false_type {};
    template<class... T>
    struct IsTuple<tuple<T...>> : true_type {};
    template<class T>
    static constexpr bool is_c_string =
        is_same_v<decay_t<T>, const char *> ||
        is_same_v<decay_t<T>, char *>;
    static void print_u128(u128 x) {
        char s[40];
        int n{ 0 };
        do { s[n ++] = char('0' + x % 10); x /= 10;
        } while (x > 0);
        while (n > 0) cerr << s[-- n];
    }
    template<class... T>
    static void print_list(const T &...x) {
        bool first{ true };
        ((cerr << (exchange(first, false) ? "" : ", "), print_one(x)), ...);
    }
    static string trim(const string &s) {
        size_t l{ 0 }, r{ s.size() };
        while (l < r && isspace((unsigned char)s[l])) ++ l;
        while (r > l && isspace((unsigned char)s[r - 1])) -- r;
        return s.substr(l, r - l);
    }
    static bool quoted(const string &s) {
        return s.size() >= 2 &&
            ((s.front() == '"' && s.back() == '"') ||
             (s.front() == '\'' && s.back() == '\''));
    }
    static vector<string> split_top_level(const string &s) {
        vector<string> res;
        string cur;
        int paren{ 0 }, brace{ 0 }, bracket{ 0 };
        char quote{ 0 };
        bool esc{ false };
        for (char c : s) {
            if (quote) {
                cur += c;
                if (esc) {
                    esc = false;
                } else if (c == '\\') {
                    esc = true;
                } else if (c == quote) {
                    quote = 0;
                }
                continue;
            }
            if (c == '\'' || c == '"') {
                quote = c;
                cur += c;
            } else if (c == '(') {
                ++ paren;
                cur += c;
            } else if (c == ')') {
                if (paren > 0) -- paren;
                cur += c;
            } else if (c == '{') {
                ++ brace;
                cur += c;
            } else if (c == '}') {
                if (brace > 0) -- brace;
                cur += c;
            } else if (c == '[') {
                ++ bracket;
                cur += c;
            } else if (c == ']') {
                if (bracket > 0) -- bracket;
                cur += c;
            } else if (c == ',' && paren == 0 && brace == 0 && bracket == 0) {
                res.push_back(trim(cur));
                cur.clear();
            } else {
                cur += c;
            }
        }
        if (!cur.empty() || res.empty()) res.push_back(trim(cur));
        return res;
    }
    static void print_named_impl(const vector<string> &, size_t) {
        cerr << '\n';
    }
    template<class T, class... U>
    static void print_named_impl(
        const vector<string> &names,
        size_t i,
        const T &x,
        const U &...rest
    ) {
        if (i >= names.size()) {
            print_one(x);
            ((cerr << ' ', print_one(rest)), ...);
            cerr << '\n';
            return;
        }
        if constexpr (is_c_string<T>) {
            if (quoted(names[i])) {
                print_one(x);
                cerr << ": ";
                print_named_impl(names, i + 1, rest...);
                return;
            }
        }
        cerr << names[i] << " = ";
        print_one(x);
        if constexpr (sizeof...(rest) == 0) {
            cerr << '\n';
        } else {
            cerr << ' ';
            print_named_impl(names, i + 1, rest...);
        }
    }
public:
    template<class T>
    static void print_one(const T &x) {
        using U = decay_t<T>;
        if constexpr (is_same_v<U, bool>) {
            cerr << (x ? "true" : "false");
        } else if constexpr (is_same_v<U, char>) {
            cerr << x;
        } else if constexpr (
            is_same_v<U, string> ||
            is_same_v<U, string_view>
        ) {
            cerr << x;
        } else if constexpr (is_c_string<T>) {
            if constexpr (is_array_v<remove_reference_t<T>>) {
                cerr << x;
            } else {
                cerr << (x ? x : "null");
            }
        } else if constexpr (is_same_v<U, i128>) {
            if (x < 0) {
                cerr << '-';
                print_u128(u128(-(x + 1)) + 1);
            } else {
                print_u128(u128(x));
            }
        } else if constexpr (is_same_v<U, u128>) {
            print_u128(x);
        } else if constexpr (IsPair<U>::value) {
            cerr << '(';
            print_one(x.first);
            cerr << ", ";
            print_one(x.second);
            cerr << ')';
        } else if constexpr (IsTuple<U>::value) {
            cerr << '(';
            apply([](const auto &...v) {
                print_list(v...);
            }, x);
            cerr << ')';
        } else if constexpr (IsIterable<T>::value) {
            cerr << '[';
            bool first{ true };
            for (const auto &v : x) {
                if (!first) cerr << ", ";
                first = false;
                print_one(v);
            }
            cerr << ']';
        } else {
            cerr << x;
        }
    }
    template<class... T>
    static void values(const T &...x) {
        bool first{ true };
        ((cerr << (exchange(first, false) ? "" : " "), print_one(x)), ...);
        cerr << '\n';
    }
    template<class... T>
    static void named(const string &names, const T &...x) {
        print_named_impl(split_top_level(names), 0, x...);
    }
};
#define dbg(...) (Debug::values(__VA_ARGS__))
#define dbgv(...) (Debug::named(#__VA_ARGS__, __VA_ARGS__))
#else
#define dbg(...) ((void)0)
#define dbgv(...) ((void)0)
#endif
```

struct RuntimeInfo {
    chrono::steady_clock::time_point start{ chrono::steady_clock::now() };
    ~RuntimeInfo() {
        double ms{ chrono::duration<double, milli>(chrono::steady_clock::now() - start).count() };
        auto flags{ cerr.flags() };
        auto precision{ cerr.precision() };
        cerr << fixed << setprecision(3) << "[Time] " << ms << " ms\n";
        cerr.flags(flags);
        cerr.precision(precision);
    }
};
RuntimeInfo runtime_info;
#endif
