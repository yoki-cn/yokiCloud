### 表达式解析与表达式树
---
支持十进制非负整数字面量、变量、括号，以及 `+ - * / % ^` 和一元正负号
- 中缀转后缀：支持语法检查、运算符优先级和结合性
- 后缀求值：支持变量赋值，检查除零、负指数和 `i64` 溢出
- 表达式树：通过后缀表达式建树，支持一元与二元运算符，以及恢复完全括号化的中缀表达式
```cpp
struct Expr {
    // ===== 基础辅助函数：被 postfix / eval / ExprTree 依赖 =====
    static bool op(const string &s) { return s == "+" || s == "-" || s == "*" || s == "/" || s == "%" || s == "^" || s == "u+" || s == "u-"; }
    static bool unary(const string &s) { return s == "u+" || s == "u-"; }
    static int pri(const string &s) {
        if (s == "^") return 4;
        if (unary(s)) return 3;
        if (s == "*" || s == "/" || s == "%") return 2;
        return 1;
    }
    static bool right(const string &s) { return s == "^" || unary(s); }

    // ===== 中缀转后缀：若只处理现成的后缀表达式，可删除本函数及 pri / right =====
    static vector<string> postfix(const string &s) {
        vector<string> out, st;
        bool need = true;
        for (int i = 0, n = s.size(); i < n; ) {
            unsigned char c = s[i];
            if (isspace(c)) { ++ i; continue; }
            if (isdigit(c) || isalpha(c) || c == '_') {
                if (!need) return {};
                int j = i;
                if (isdigit(c)) {
                    while (j < n && isdigit((unsigned char)s[j])) ++ j;
                } else {
                    while (j < n && (isalnum((unsigned char)s[j]) || s[j] == '_')) ++ j;
                }
                out.push_back(s.substr(i, j - i));
                i = j;
                need = false;
            } else if (c == '(') {
                if (!need) return {};
                st.push_back("(");
                ++ i;
            } else if (c == ')') {
                if (need) return {};
                while (!st.empty() && st.back() != "(") {
                    out.push_back(st.back());
                    st.pop_back();
                }
                if (st.empty()) return {};
                st.pop_back();
                ++ i;
                need = false;
            } else {
                string t(1, c);
                if (need && (c == '+' || c == '-')) {
                    st.push_back("u" + t); // 一元运算符直接入栈，允许 2^-3
                    ++ i;
                    continue;
                }
                if (need || !op(t)) return {};
                while (!st.empty() && st.back() != "(") {
                    int a = pri(t), b = pri(st.back());
                    if (!(b > a || (b == a && !right(t)))) break;
                    out.push_back(st.back());
                    st.pop_back();
                }
                st.push_back(t);
                ++ i;
                need = true;
            }
        }
        if (need) return {};
        while (!st.empty()) {
            if (st.back() == "(") return {};
            out.push_back(st.back());
            st.pop_back();
        }
        return out;
    }
    // ===== 后缀求值 =====
    static optional<i64> eval(const vector<string> &p, const unordered_map<string, i64> &val = {}) {
        vector<i64> st;
        auto fit = [](i128 x) { return x >= numeric_limits<i64>::min() && x <= numeric_limits<i64>::max(); };
        for (const string &s : p) {
            if (op(s)) {
                if (st.size() < (unary(s) ? 1U : 2U)) return nullopt;
                i128 x = st.back(); st.pop_back();
                if (unary(s)) {
                    if (s == "u-") x = -x;
                } else {
                    i128 b = x, a = st.back(); st.pop_back();
                    if (s == "+") x = a + b;
                    else if (s == "-") x = a - b;
                    else if (s == "*") x = a * b;
                    else if (s == "/" || s == "%") {
                        if (!b) return nullopt;
                        x = s == "/" ? a / b : a % b;
                    } else {
                        if (b < 0) return nullopt;
                        x = 1;
                        for (i64 e = (i64)b; e; e >>= 1) {
                            if (e & 1) {
                                x *= a;
                                if (!fit(x)) return nullopt;
                            }
                            if (e > 1) {
                                a *= a;
                                if (!fit(a)) return nullopt;
                            }
                        }
                    }
                }
                if (!fit(x)) return nullopt;
                st.push_back((i64)x);
            } else {
                i64 x;
                if (!s.empty() && isdigit((unsigned char)s[0])) {
                    auto [p, ec] = from_chars(s.data(), s.data() + s.size(), x);
                    if (ec != errc{} || p != s.data() + s.size()) return nullopt;
                } else {
                    auto it = val.find(s);
                    if (it == val.end()) return nullopt;
                    x = it->second;
                }
                st.push_back(x);
            }
        }
        if (st.size() != 1) return nullopt;
        return st.back();
    }
};
// ===== 表达式树 =====
struct ExprTree {
    struct Node {
        string s;
        int l{ -1 }, r{ -1 };
    };
    vector<Node> tr;
    int root{ -1 };
    bool build(const vector<string> &p) {
        tr.clear();
        root = -1;
        vector<int> st;
        for (const string &s : p) {
            int l = -1, r = -1;
            if (Expr::op(s)) {
                if (st.size() < (Expr::unary(s) ? 1U : 2U)) return false;
                r = st.back(); st.pop_back();
                if (Expr::unary(s)) swap(l, r);
                else { l = st.back(); st.pop_back(); }
            }
            st.push_back(tr.size());
            tr.push_back({s, l, r});
        }
        if (st.size() != 1) { tr.clear(); return false; }
        root = st.back();
        return true;
    }
    // ===== 恢复中缀表达式：以下函数可直接删除 =====
    void dump(int u, string &out) const {
        const auto &[s, l, r] = tr[u];
        if (l == -1) { out += s; return; }
        out += '(';
        if (r == -1) out += s[1];
        dump(l, out);
        if (r != -1) { out += s; dump(r, out); }
        out += ')';
    }
    string str() const {
        string out;
        if (root != -1) dump(root, out);
        return out;
    }
};
```
#### Expr 表达式解析与求值
- `Expr::op(s)`：判断 `s` 是否为支持的运算符，$O(1)$
- `Expr::unary(s)`：判断是否为一元正负号，$O(1)$
- `Expr::pri(s)`：返回合法运算符的优先级，$O(1)$
- `Expr::right(s)`：判断运算符是否右结合，$O(1)$
- `Expr::postfix(s)`：中缀表达式转后缀表达式，返回 `vector<string>`，语法不合法时返回空数组，$O(n)$
- `Expr::eval(p, val)`：求后缀表达式的整数值，返回 `optional<i64>`，失败时返回 `nullopt`，变量表 `val` 可省略
#### ExprTree 表达式树
- `ExprTree::build(p)`：根据后缀表达式建树，成功返回 `true`，失败返回 `false`，$O(n)$
- `ExprTree::root`：根节点编号，未成功建树时为 `-1`
- `ExprTree::tr[u].s`：节点保存的数字、变量或运算符
- `ExprTree::tr[u].l / r`：左右儿子，`-1` 表示不存在
- `ExprTree::str()`：恢复完全括号化的中缀表达式，$O(n)$
- `ExprTree::dump(u, out)`：将节点 `u` 对应子树的中缀表达式追加到 `out`，$O(k)$，其中 $k$ 为输出长度
节点编号为 `0base`，一元运算符只使用左儿子
#### 注意事项
#### 运算符优先级

| 优先级   | 运算符         | 结合性 |
| ----- | ----------- | --- |
| 4（最高） | `^`         | 右结合 |
| 3     | `u+`、`u-`   | 右结合 |
| 2     | `*`、`/`、`%` | 左结合 |
| 1（最低） | `+`、`-`     | 左结合 |
- `u+`、`u-` 是内部表示，输入中仍写普通的 `+`、`-`
- `-2^2` 解析为 `-(2^2)`，结果为 `-4`
- `(-2)^2` 解析为 `(-2)^2`，结果为 `4`
- `2^3^2` 解析为 `2^(3^2)`，结果为 `512`
- `2^-3` 可以正确解析，但整数求值不支持负指数
- `0^0` 按整数快速幂约定返回 `1`
#### 语法和数值范围
- 数字仅支持非负十进制整数字面量，变量名支持 `[A-Za-z_][A-Za-z0-9_]*`
- 不支持小数、隐式乘法、函数调用、赋值以及 `&&`、`||` 等其他运算符
- `postfix()` 只负责语法检查，不检查数值范围、变量是否存在以及除零
- `eval()` 对所有运算检查 `i64` 范围，不允许任何中间结果越界
- 单个数字字面量不能超过 `LLONG_MAX`，因此不能直接写 `-9223372036854775808`，但可以通过 `(-2)^63` 得到 `LLONG_MIN`
- `/` 和 `%` 遵循 C++ 有符号整数的截断除法规则
- `ExprTree::build()` 需要传入有效的后缀表达式，失败后不可访问节点
- `str()` 使用递归，极深的表达式树可能导致栈溢出
> [!tip] 如果题目使用浮点数、模运算或布尔表达式，可以保留 `postfix()` 和 `ExprTree`，只修改词法分析和求值规则，不必重写建树算法
#### 常见用法
#### 中缀表达式直接求值
```
auto p = Expr::postfix("-(12 + 3) * 4");
auto ans = Expr::eval(p);
if (ans) cout << *ans << '\n'; // -60
```
#### 输出后缀表达式
```cpp
auto p = Expr::postfix("a + b * c");
for (const string &s : p) cout << s << ' ';
// a b c * +
```
#### 带变量的表达式求值
```cpp
unordered_map<string, i64> val{
    {"a", 2}, {"b", 10}
};
auto ans = Expr::eval(Expr::postfix("a * (b - 3)"), val);
// ans = 14
```
#### 构建表达式树
```cpp
auto p = Expr::postfix("(a+b)*(c-d)");
ExprTree t;
if (t.build(p)) {
    cout << t.root << '\n';
    cout << t.str() << '\n'; // ((a+b)*(c-d))
}
```
#### 输出前缀表达式
建树后先序遍历即可得到前缀表达式，不需要额外实现转换算法
```cpp
vector<string> pre;
vector<int> st{t.root};
while (!st.empty()) {
    int u = st.back();
    st.pop_back();
    pre.push_back(t.tr[u].s);
    if (t.tr[u].r != -1) st.push_back(t.tr[u].r);
    if (t.tr[u].l != -1) st.push_back(t.tr[u].l);
}
// (a+b)*(c-d) -> * + a b - c d
```
#### 复杂度
设 $n$ 为表达式字符总数，$k$ 为后缀表达式的 Token 数，$h$ 为表达式树高度

| 操作          | 时间复杂度                 | 空间复杂度    |
| ----------- | --------------------- | -------- |
| `postfix()` | $O(n)$                | $O(n)$   |
| `eval()`    | $O(n+\sum \log(e+1))$ | $O(k)$   |
| `build()`   | $O(n)$                | $O(n)$   |
| `str()`     | $O(n)$                | $O(n+h)$ |
其中 $e$ 为各次乘方运算的非负指数，`eval()` 假设变量哈希查询均摊常数时间，且计算使用固定宽度整数，因此整体也可视为 $O(n)$
