### 2.5 FastIO
---
```cpp
using i128 = __int128_t;
using u128 = __uint128_t;

namespace FastIO {
struct Input {
private:
    static constexpr size_t S{ 1 << 20 };
    char buf[S];
    size_t p{ 0 }, n{ 0 };
    int get() {
        if (p == n) {
            n = fread(buf, 1, S, stdin);
            p = 0;
            if (n == 0) return EOF;
        }
        return (unsigned char)buf[p ++];
    }
    static bool digit(int c) { return '0' <= c && c <= '9'; }
    bool skip_blank(int &c) {
        c = get();
        while (c != EOF && c <= ' ') c = get();
        return c != EOF;
    }
    // ===== 整数读取核心：所有整数 read() 依赖，不建议删除 =====
    template<class U>
    bool read_abs(U &v, bool &neg) {
        int c;
        if (!skip_blank(c)) return false;
        neg = false;
        if (c == '+' || c == '-') {
            neg = c == '-';
            c = get();
        }
        if (!digit(c)) return false;
        v = 0;
        while (digit(c)) {
            v = v * 10 + U(c - '0');
            c = get();
        }
        return true;
    }
public:
    // ===== 标准整数：若完全不读整数，可连同 read_abs() 删除 =====
    template<class T,
        enable_if_t<is_integral_v<T> && !is_same_v<remove_cv_t<T>, bool>
            && !is_same_v<remove_cv_t<T>, i128>
            && !is_same_v<remove_cv_t<T>, u128>, int> = 0>
    bool read(T &x) {
        using U = make_unsigned_t<T>;
        U v;
        bool neg;
        if (!read_abs(v, neg)) return false;
        if constexpr (is_signed_v<T>) {
            x = neg ? (v ? -T(v - 1) - 1 : T{}) : T(v);
        } else {
            if (neg) return false;
            x = T(v);
        }
        return true;
    }
    // ===== 128 位整数：若不使用 i128/u128，以下两个函数可直接删除 =====
    bool read(i128 &x) {
        u128 v;
        bool neg;
        if (!read_abs(v, neg)) return false;
        x = neg ? (v ? -i128(v - 1) - 1 : i128{}) : i128(v);
        return true;
    }
    bool read(u128 &x) {
        bool neg;
        if (!read_abs(x, neg)) return false;
        return !neg;
    }
    // ===== 字符与字符串：若只读整数，以下代码可直接删除 =====
    bool read(char &c) {
        int x;
        if (!skip_blank(x)) return false;
        c = char(x);
        return true;
    }
    bool read(string &s) {
        int c;
        if (!skip_blank(c)) return false;
        s.clear();
        while (c != EOF && c > ' ') {
            s.push_back(char(c));
            c = get();
        }
        return true;
    }
    template<size_t N>
    bool read(char (&s)[N]) {
        static_assert(N > 0);
        int c;
        if (!skip_blank(c)) return false;
        size_t len{ 0 };
        bool ok{ true };
        while (c != EOF && c > ' ') {
            if (len + 1 < N) s[len ++] = char(c);
            else ok = false;
            c = get();
        }
        s[len] = '\0';
        return ok;
    }
    // ===== 容器读取：vector 必须提前 resize；若不需要可直接删除 =====
    template<class T, size_t N,
        enable_if_t<!is_same_v<remove_cv_t<T>, char>, int> = 0>
    bool read(T (&a)[N]) {
        for (T &x : a) if (!read(x)) return false;
        return true;
    }
    template<class T, size_t N>
    bool read(array<T, N> &a) {
        for (T &x : a) if (!read(x)) return false;
        return true;
    }
    template<class T, class A>
    bool read(vector<T, A> &a) {
        for (T &x : a) if (!read(x)) return false;
        return true;
    }
    // ===== 连续读入：若不需要 in.read(a, b, c)，可直接删除 =====
    template<class T, class U, class... V>
    bool read(T &x, U &y, V &...rest) { return read(x) && read(y, rest...); }
};
struct Output {
private:
    static constexpr size_t S{ 1 << 20 };
    char buf[S];
    size_t p{ 0 };
    template<class U>
    void write_unsigned(U x) {
        char s[40];
        int n{ 0 };
        do { s[n ++] = char('0' + x % 10); x /= 10;
        } while (x > 0);
        while (n > 0) put(s[-- n]);
    }
    void sep(bool &first) { if (!first) put(' '); first = false; }
    // ===== 单个标量输出：容器与多参数输出依赖，不建议删除 =====
    void write_scalar(char x) { put(x); }
    void write_scalar(string_view s) { for (char c : s) put(c); }
    void write_scalar(const char *s) { while (*s) put(*s ++); }
    template<size_t N>
    void write_scalar(const char (&s)[N]) {
        for (size_t i = 0; i < N && s[i] != '\0'; ++ i) put(s[i]);
    }
    template<class T,
        enable_if_t<is_integral_v<T> && !is_same_v<remove_cv_t<T>, bool>
            && !is_same_v<remove_cv_t<T>, i128>
            && !is_same_v<remove_cv_t<T>, u128>, int> = 0>
    void write_scalar(T x) {
        using U = make_unsigned_t<T>;
        U v;
        if constexpr (is_signed_v<T>) {
            if (x < 0) {
                put('-');
                v = U(-(x + 1));
                ++ v;
            } else v = U(x);
        } else v = U(x);
        write_unsigned(v);
    }
    // ===== 128 位整数：若不使用 i128/u128，以下两个函数可直接删除 =====
    void write_scalar(i128 x) {
        if (x < 0) {
            put('-');
            write_unsigned(u128(-(x + 1)) + 1);
        } else write_unsigned(u128(x));
    }
    void write_scalar(u128 x) { write_unsigned(x); }
    // ===== 参数展开核心：标量为一个 token，容器递归展开为若干 token =====
    template<class T>
    void write_tokens(bool &first, const T &x) { sep(first); write_scalar(x); }
    template<class T, size_t N,
        enable_if_t<!is_same_v<remove_cv_t<T>, char>, int> = 0>
    void write_tokens(bool &first, const T (&a)[N]) {
        for (const T &x : a) write_tokens(first, x);
    }
    template<class T, size_t N>
    void write_tokens(bool &first, const array<T, N> &a) {
        for (const T &x : a) write_tokens(first, x);
    }
    template<class T, class A>
    void write_tokens(bool &first, const vector<T, A> &a) {
        for (const T &x : a) write_tokens(first, x);
    }
public:
    ~Output() { flush(); }
    // ===== 基础输出：所有输出接口依赖，不建议删除 =====
    void flush() { if (p > 0) { fwrite(buf, 1, p, stdout); p = 0; } }
    void put(char c) { if (p == S) flush(); buf[p ++] = c; }
    // ===== 通用输出：标量与数组可任意混搭，token 之间自动加空格 =====
    template<class... T>
    void write(const T &...x) { bool first{ true }; (write_tokens(first, x), ...); }
    // ===== 行输出：与 write() 相同，但末尾自动补一个换行 =====
    template<class... T>
    void writeln(const T &...x) { write(x...); put('\n'); }
    // ===== 逐行输出：每个参数单独一行，数组内部仍以空格分隔 =====
    template<class... T>
    void writelines(const T &...x) { (writeln(x), ...); }
};
inline Input in;
inline Output out;
}
using namespace FastIO;
```
#### 注意事项
- 使用 `FastIO` 后也不要再通过其他输入函数读取 `stdin`，因为内部会提前读取一块数据
- `FastIO::Input` 和 `FastIO::Output` 各含约 $1\text{ MiB}$ 缓冲区，因此使用全局 `inline` 实例，避免占用较小的线程栈
- `Output` 正常析构时会自动刷新；交互题不能依赖析构，询问后必须调用 `flush()`
- 有符号最小值不能直接执行 `x = -x`；本板通过无符号类型处理，因此可以正确输出 `INT_MIN`、`LLONG_MIN` 和 `i128` 最小值
- `c <= ' '` 用于判断竞赛环境中的 ASCII 空白字符，比反复调用 `isspace()` 更轻；不用于需要处理特殊 locale 的文本程序
- `read(char&)` 会跳过空白；若题目需要逐字节读取空格或换行，应另行暴露内部 `get()`
- 浮点数通常直接使用 `iostream`；若输入量巨大且必须使用本板，可以先读为字符串再调用 `strtod/strtold`，没有必要把低频浮点解析长期塞进核心板子
