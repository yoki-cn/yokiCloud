### 组合数
```cpp
// 过程没有取模，因此使用取模类
template<class T = Z>
struct Comb {
    int n{0};
    vector<T> fac{T(1)}, ifac{T(1)}, inv{T(0)};
    explicit Comb(int n_max = 0) {
        inv.push_back(T(1));
        ensure(n_max);
    }
    void ensure(int m) {
        if (m <= n) return;
        if (m < 0 || m >= mod) throw invalid_argument("Comb requires 0 <= n < mod");
        int old{ n };
        n = m;
        fac.resize(n + 1);
        ifac.resize(n + 1);
        inv.resize(n + 1);
        for (int i = old + 1; i <= n; ++ i) {
            fac[i] = fac[i - 1] * T(i);
            if (i == 1) inv[i] = T(1);
            else inv[i] = -T(mod / i) * inv[mod % i];
            ifac[i] = ifac[i - 1] * inv[i];
        }
    }
    T Fac(int m) { ensure(m); return fac[m]; }
    T IFact(int m) { ensure(m); return ifac[m]; }
    T Inv(int m) { ensure(m); return inv[m]; }
    T C(int a, int b) {
        if (a < 0 || b < 0 || b > a) return T(0);
        ensure(a);
        return fac[a] * ifac[b] * ifac[a - b];
    }
    // 排列数 P(n,k) = n! / (n-k)!
    T P(int a, int b) {
        if (a < 0 || b < 0 || b > a) return T(0);
        ensure(a);
        return fac[a] * ifac[a - b];
    }
    // 可重复组合（把 K 个相同物品放进 N 类）：C(n+k-1, k)
    T H(int n_types, int k_take) {
        if (n_types < 0 || k_take < 0) return T(0);
        if (n_types == 0) return T(k_take == 0);
        int top{ n_types + k_take - 1 };
        if (top >= mod) throw invalid_argument("Comb::H exceeds supported range");
        return C((int)top, k_take);
    }
};
```
### Lucas
#### 复用 `Comb<T>`
要求全局 `mod` 为质数，并依赖前面的 `Comb<T>`
将下面接口直接放在 `Comb<T>` 的末尾：
```cpp
// ===== Lucas 定理：若不需要处理 n >= mod 的组合数，以下代码可直接删除 =====
T Lucas(i64 a, i64 b) {
    if (a < 0 || b < 0 || b > a) return T(0);
    T ans{1};
    while (b) {
        int x = a % mod, y = b % mod;
        if (y > x) return T(0);
        ans *= C(x, y);
        a /= mod;
        b /= mod;
    }
    return ans;
}
```
Lucas 定理：
$$
\binom{n}{k} \equiv \prod_{i\ge 0}\binom{n_i}{k_i} \pmod p,
$$
其中 `p = mod` 为质数，且
$$n=\sum_i n_ip^i,\quad k=\sum_i k_ip^i,\quad 0\le n_i,k_i<p$$
如果某一位满足 $k_i>n_i$，则
$$\binom nk\equiv0\pmod p$$
#### 接口说明
- `Lucas(a, b)`：计算 $\binom ab\bmod mod$，允许 `a >= mod`
- `a < 0`、`b < 0` 或 `b > a` 时返回 `0`
- 内部调用 `C(x, y)` 计算每一位的小组合数
由于当前 `Comb` 会按需扩容，因此 Lucas 查询只会把预处理范围扩大到实际出现过的最大 `a % mod`，不必预先计算整个 `0..mod-1`
#### 注意事项
- `mod` **必须是质数**
- Lucas 解决的是质数模下 `a >= mod` 时普通阶乘组合数失效的问题；合数模不能直接使用
- 若 `mod` 很大，例如 `1E9 + 7`，且某个 `a % mod` 也很大，则不能依靠 `Comb` 预处理到这一位置，否则时间和空间都会不可接受。
- Lucas 本身不会解决“模数很大且每一位也很大”的问题；此时应根据 `b` 的大小选择乘法式求组合数等方法
- 合数模的大组合数应使用扩展 Lucas / 质数幂组合数 + CRT，而不是普通Lucas
### Lucas (不预处理)
如果 `mod` 不固定，或者不希望为 Lucas 数位建立阶乘表，可以使用下面的独立版本
```cpp
i64 qpow_mod(i64 a, i64 b, i64 p) {
    i64 ans{1};
    while (b) {
        if (b & 1) ans = (i128)ans * a % p;
        a = (i128)a * a % p;
        b >>= 1;
    }
    return ans;
}
i64 C_mod_prime(i64 n, i64 k, i64 p) {
    if (n < 0 || k < 0 || k > n) return 0;
    k = min(k, n - k);
    i64 a{1}, b{1};
    for (i64 i = 1; i <= k; ++ i) {
        a = (i128)a * (n - k + i) % p;
        b = (i128)b * i % p;
    }
    return (i128)a * qpow_mod(b, p - 2, p) % p;
}
i64 lucas(i64 n, i64 k, i64 p) {
    if (n < 0 || k < 0 || k > n) return 0;
    i64 ans{1};
    while (k) {
        i64 a = n % p, b = k % p;
        if (b > a) return 0;
        ans = (i128)ans * C_mod_prime(a, b, p) % p;
        n /= p;
        k /= p;
    }
    return ans;
}
```
#### 使用场景
```cpp
i64 ans = lucas(n, k, p);
```
其中 `p` 可以在运行时给出，但必须为质数
这个版本不需要 $O(p)$ 空间，适合：
- `p` 每组数据不同；
- 查询次数较少；
- 每一位的 $\min(k_i,n_i-k_i)$ 较小
若同一个小质数 `p` 下有大量查询，优先使用 `Comb + Lucas` 版本
#### 复杂度
##### `Comb + Lucas`
设一次查询的 Lucas 数位数为
$$d=O(\log_{mod} n)$$
若阶乘表已经覆盖所有需要的数位：

|操作|时间复杂度|
|---|--:|
|`Lucas(n, k)`|$O(\log_{mod} n)$|
|每个数位的 `C()`|$O(1)$|
若使用前面的动态 `Comb`，设历史所有 Lucas 查询中实际访问过的最大数位为 `M`，则所有扩容的累计复杂度为：
$$O(M)$$
空间复杂度为：
$$O(M)$$
因此无需强制预处理整个 `0..mod-1`
##### 不预处理版本
设
$$n_i=n\bmod p,\quad k_i=k\bmod p$$
单次查询复杂度为
$$O\left( \sum_i\min(k_i,n_i-k_i) + \log_p n\cdot\log p \right)$$
空间复杂度为 $O(1)$
