### MInt
```cpp
template<class Policy>  
struct MInt {  
    u32 x{};  
    constexpr MInt() = default;  
    constexpr MInt(i64 v) : x(norm(v)) {}  
    // ===== 基础接口与四则运算：核心功能，不建议删除 =====  
    static constexpr u32 mod() { return Policy::mod(); }  
    static constexpr u32 norm(i64 v) {  
        i64 m{ mod() };  
        if (v >= m || v <= -m) v %= m;  
        if (v < 0) v += m;  
        return u32(v);  
    }  
    constexpr u32 val() const { return x; }  
    explicit constexpr operator u32() const { return x; }  
    constexpr MInt &operator+=(MInt b) {  
        u32 m{ mod() };  
        if (x >= m - b.x) x -= m - b.x;  
        else x += b.x;  
        return *this;  
    }  
    constexpr MInt &operator-=(MInt b) {  
        u32 m{ mod() };  
        if (x < b.x) x += m - b.x;  
        else x -= b.x;  
        return *this;  
    }  
    constexpr MInt &operator*=(MInt b) {  
        x = u64(x) * b.x % mod();  
        return *this;  
    }  
    friend constexpr MInt operator+(MInt a, MInt b) { return a += b; }  
    friend constexpr MInt operator-(MInt a, MInt b) { return a -= b; }  
    friend constexpr MInt operator*(MInt a, MInt b) { return a *= b; }  
    constexpr MInt operator-() const { return x ? mod() - x : 0; }  
    // ===== 幂、逆元与除法：若只需要加减乘，以下代码可直接删除 =====  
    constexpr MInt pow(i64 b) const {  
        bool minus{ b < 0 };  
        u64 e{ minus ? u64(-(b + 1)) + 1 : u64(b) };  
        MInt a = *this, res(1);  
        for (; e; e >>= 1, a *= a) if (e & 1) res *= a;  
        return minus ? res.inv() : res;  
    }  
    constexpr MInt inv() const {  
        assert(x != 0);  
        return pow(i64(mod()) - 2);  
    }  
    constexpr MInt &operator/=(MInt b) { return *this *= b.inv(); }  
    friend constexpr MInt operator/(MInt a, MInt b) { return a /= b; }  
    // ===== 快捷运算符：若只使用 pow() / inv()，以下代码可直接删除 =====  
    constexpr MInt &operator^=(i64 b) { return *this = pow(b); }  
    friend constexpr MInt operator^(MInt a, i64 b) { return a.pow(b); }  
    friend constexpr MInt operator~(MInt a) { return a.inv(); }  
    // ===== 比较与 I/O：不需要时以下代码可直接删除 =====  
    friend constexpr bool operator==(MInt a, MInt b) { return a.x == b.x; }  
    friend constexpr bool operator!=(MInt a, MInt b) { return a.x != b.x; }  
    friend constexpr bool operator<(MInt a, MInt b) { return a.x < b.x; }  
    friend istream &operator>>(istream &in, MInt &a) {  
        i64 v;  
        in >> v;  
        a = v;  
        return in;  
    }
    friend ostream &operator<<(ostream &out, MInt a) { return out << a.val(); }
    // ===== 动态模数专属：只使用静态模数时可直接删除 =====
    static void set_mod(u32 value) { Policy::set_mod(value); }
};

template<u32 P>
struct StaticMod { static constexpr u32 mod() { return P; } };
using Z = MInt<StaticMod<mod>>;
  
// ===== 动态模数：只使用静态模数时，以下代码可整段删除 =====  
template<int Id>
struct DynamicMod {  
    inline static u32 modv{};  
    static u32 mod() { return modv; }  
    static void set_mod(u32 value) { modv = value; }  
};  
template<int Id = 0>  
using DZ = MInt<DynamicMod<Id>>;
```
#### 注意事项
- 模数必须满足 `mod() > 1`；静态版需保证模板参数合法，动态版必须在首次构造非零值或进行模运算前调用 `DZ::set_mod(p)`
- 动态模数修改后，之前创建的 `DZ` 对象不会重新取模；不要在仍使用旧对象时修改模数。多测模数不同时，应先 `set_mod()`，再创建本组所有模整数
- `inv()`、`/`、`~` 和负指数 `pow(-k)` 均要求**模数为质数**；同时求逆的元素必须非零
- `pow(b)` 支持负指数；`b < 0` 时按 $a^b=(a^{-b})^{-1}$ 计算，因此底数必须可逆。代码可正确处理 `LLONG_MIN`
- 内部始终维护 `0 <= x < mod()`；整数构造和输入会自动规范化负数及超出模数范围的值
- 构造函数与输入仅直接支持 `i64` 范围内的整数；若输入值可能超过 `i64`，需要单独处理字符串取模
- 模数类型为 `u32`，支持不超过 `UINT32_MAX` 的模数
- `val()` 和显式转换到 `u32` 返回的是 `[0, mod())` 内的标准余数，而不是原始输入值
- `operator<` 比较的是标准余数 `x` 的大小，仅用于排序等场景，不表示模意义下存在自然的大小关系
