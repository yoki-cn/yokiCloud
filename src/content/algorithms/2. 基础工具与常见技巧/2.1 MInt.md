### MInt
---
```cpp
constexpr int mod{ 998244353 };

template<uint32_t P>
struct MInt {
    using u32 = uint32_t;
    using u64 = uint64_t;
    static_assert(P > 1);
    u32 x{};
    constexpr MInt() = default;
    constexpr MInt(i64 v) : x(norm(v)) {}
    // ===== 基础接口与四则运算 =====
    static constexpr u32 mod() { return P; }
    static constexpr u32 norm(i64 v) {
        if (v >= P || v <= -i64(P)) v %= P;
        if (v < 0) v += P;
        return u32(v);
    }
    constexpr u32 val() const { return x; }
    explicit constexpr operator u32() const { return x; }
    constexpr MInt &operator+=(MInt b) {
        if (x >= P - b.x) x -= P - b.x;
        else x += b.x;
        return *this;
    }
    constexpr MInt &operator-=(MInt b) {
        if (x < b.x) x += P - b.x;
        else x -= b.x;
        return *this;
    }
    constexpr MInt &operator*=(MInt b) {
        x = u64(x) * b.x % P;
        return *this;
    }
    friend constexpr MInt operator+(MInt a, MInt b) { return a += b; }
    friend constexpr MInt operator-(MInt a, MInt b) { return a -= b; }
    friend constexpr MInt operator*(MInt a, MInt b) { return a *= b; }
    constexpr MInt operator-() const { return x ? P - x : 0; }
    // ===== 幂、逆元与除法：若只需要加减乘，以下代码可直接删除 =====
    constexpr MInt pow(i64 b) const {
        bool minus{ b < 0 };
        u64 e{ minus ? u64(-(b + 1)) + 1 : u64(b) };
        MInt a = *this, res(1);
        for (; e; e >>= 1, a *= a) if (e & 1) res *= a;
        return minus ? res.inv() : res;
    }
    constexpr MInt inv() const {
        assert(x != 0);
        return pow(i64(P) - 2);
    }
    constexpr MInt &operator/=(MInt b) { return *this *= b.inv(); }
    friend constexpr MInt operator/(MInt a, MInt b) { return a /= b; }
    // ===== 快捷运算符：若只使用 pow() / inv()，以下代码可直接删除 =====
    constexpr MInt &operator^=(i64 b) { return *this = pow(b); }
    friend constexpr MInt operator^(MInt a, i64 b) { return a.pow(b); }
    friend constexpr MInt operator~(MInt a) { return a.inv(); }
    // ===== 比较与 I/O：不需要时以下代码可直接删除 =====
    friend constexpr bool operator==(MInt a, MInt b) { return a.x == b.x; }
    friend constexpr bool operator!=(MInt a, MInt b) { return a.x != b.x; }
    friend constexpr bool operator<(MInt a, MInt b) { return a.x < b.x; }
    friend istream &operator>>(istream &in, MInt &a) {
        i64 v;
        if (in >> v) a = v;
        return in;
    }
    friend ostream &operator<<(ostream &out, MInt a) { return out << a.val(); }
};
using Z = MInt<mod>;
```
#### 接口说明
- `MInt<P>(v)`：构造模 `P` 意义下的整数，并规范化到 `[0, P)`
- `mod()`：返回模数 `P`
- `val()`：返回 `[0, P)` 内的标准余数
- `+`、`-`、`*`、`/`：模意义下四则运算
- `pow(b)`：计算 $a^b$；支持负指数
- `inv()`：计算乘法逆元
- `^ b`：等价于 `pow(b)`
- `~a`：等价于 `a.inv()`
#### 注意事项
- 模数 `P` 必须满足 `P > 1`，代码通过 `static_assert` 在编译期检查
- `inv()`、`/`、`~` 和负指数 `pow(-k)` 使用费马小定理，因此要求 **`P` 为质数**；求逆元素必须非零
- `pow(b)` 支持负指数；`b < 0` 时按 $a^b=(a^{-b})^{-1}$ 计算，代码可以正确处理 `LLONG_MIN`
- 内部始终维护 `0 <= x < P`；整数构造和输入会自动规范化负数及超出模数范围的值
- 构造函数与输入直接支持 `i64` 范围内的整数；若输入值可能超过 `i64`，需要使用字符串自行取模
- 模数类型为 `u32`，因此要求 `P <= UINT32_MAX`
- 乘法使用 `u64(x) * b.x`，对任意 `u32` 模数均不会在乘法阶段溢出 `u64`
- `operator<` 比较标准余数 `x` 的整数大小，仅用于排序等场景，不表示模意义下存在自然大小关系
