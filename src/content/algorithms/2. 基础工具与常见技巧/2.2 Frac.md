### Frac
---
```cpp
struct Frac {
    using u64 = uint64_t;
    using i128 = __int128_t;
    using ld = long double;
    i64 num{ 0 }, den{ 1 };
    Frac() = default;
    Frac(i64 num_) : num(num_) {}
    Frac(i64 num_, i64 den_) : num(num_), den(den_) { normalize(); }
private:
    // ===== 基础辅助：约分、四则运算依赖，不建议删除 =====
    static u64 mag(i64 x) { return x < 0 ? u64(-(x + 1)) + 1 : u64(x); }
    static u64 gcd_mag(i64 a, i64 b) {
        u64 x{ mag(a) }, y{ mag(b) };
        while (y) {
            x %= y;
            swap(x, y);
        }
        return x;
    }
    static i64 gcd_abs(i64 a, i64 b) { return i64(gcd_mag(a, b)); }
    static bool fits_i64(i128 x) {
        return i128(numeric_limits<i64>::min()) <= x && x <= i128(numeric_limits<i64>::max());
    }
    // 求 gcd(|n|, d)，d > 0
    static i64 gcd_num(i128 n, i64 d) {
        if (fits_i64(n)) return gcd_abs(i64(n), d);
        return gcd_abs(i64(n % d), d);
    }
    // 同时约去两个 i64 的 gcd；额外处理 LLONG_MIN / LLONG_MIN
    static pair<i64, i64> cancel(i64 a, i64 b) {
        u64 g{ gcd_mag(a, b) };
        if (g <= u64(numeric_limits<i64>::max())) {
            i64 d{ i64(g) };
            return { a / d, b / d };
        }
        return { -1, -1 };
    }
    void set_reduced(i128 n, i128 d) {
        if (d < 0) {
            n = -n;
            d = -d;
        }
        assert(d > 0 && fits_i64(n) && fits_i64(d));
        num = i64(n);
        den = i64(d);
    }
    // ===== 加减核心：operator+= / operator-= 共用 =====
    Frac &add_assign(const Frac &b, int s) {
        if (b.num == 0) return *this;
        if (num == 0) {
            set_reduced(i128(s) * b.num, b.den);
            return *this;
        }
        if (den == 1) {
            set_reduced(i128(num) * b.den + i128(s) * b.num, b.den);
            return *this;
        }
        if (b.den == 1) {
            set_reduced(i128(num) + i128(s) * b.num * den, den);
            return *this;
        }
        if (den == b.den) {
            i128 n{ i128(num) + i128(s) * b.num };
            i64 g{ gcd_num(n, den) };
            set_reduced(n / g, den / g);
            return *this;
        }
        i64 g{ gcd_abs(den, b.den) };
        i64 a1{ den / g }, b1{ b.den / g };
        i128 n{ i128(num) * b1 + i128(s) * b.num * a1 };
        i64 h{ gcd_num(n, g) };
        set_reduced(n / h, i128(a1) * (b.den / h));
        return *this;
    }
public:
    // ===== 规范化：直接修改 num / den 后调用；正常四则运算无需调用 =====
    void normalize() {
        assert(den != 0);
        if (num == 0) {
            den = 1;
            return;
        }
        u64 g{ gcd_mag(num, den) };
        i128 n{ num }, d{ den };
        if (d < 0) {
            n = -n;
            d = -d;
        }
        set_reduced(n / i128(g), d / i128(g));
    }
    // ===== 四则运算：始终保持最简分数 =====
    Frac &operator+=(const Frac &b) { return add_assign(b, 1); }
    Frac &operator-=(const Frac &b) { return add_assign(b, -1); }
    Frac &operator*=(const Frac &b) {
        if (num == 0 || b.num == 0) {
            num = 0;
            den = 1;
            return *this;
        }
        if (den == 1 && b.den == 1) {
            set_reduced(i128(num) * b.num, 1);
            return *this;
        }
        if (den == 1) {
            i64 g{ gcd_abs(num, b.den) };
            set_reduced(i128(num / g) * b.num, b.den / g);
            return *this;
        }
        if (b.den == 1) {
            i64 g{ gcd_abs(b.num, den) };
            set_reduced(i128(num) * (b.num / g), den / g);
            return *this;
        }
        i64 g1{ gcd_abs(num, b.den) };
        i64 g2{ gcd_abs(b.num, den) };
        set_reduced(
            i128(num / g1) * (b.num / g2),
            i128(den / g2) * (b.den / g1)
        );
        return *this;
    }
    Frac &operator/=(const Frac &b) {
        assert(b.num != 0);
        if (num == 0 || (b.num == 1 && b.den == 1)) return *this;
        auto [a1, b1]{ cancel(num, b.num) };
        if (b.den == 1) {
            set_reduced(a1, i128(den) * b1);
            return *this;
        }
        if (den == 1) {
            set_reduced(i128(a1) * b.den, b1);
            return *this;
        }
        i64 g{ gcd_abs(den, b.den) };
        set_reduced(
            i128(a1) * (b.den / g),
            i128(den / g) * b1
        );
        return *this;
    }
    friend Frac operator+(Frac a, const Frac &b) { return a += b; }
    friend Frac operator-(Frac a, const Frac &b) { return a -= b; }
    friend Frac operator*(Frac a, const Frac &b) { return a *= b; }
    friend Frac operator/(Frac a, const Frac &b) { return a /= b; }
    Frac operator-() const {
        Frac res;
        res.set_reduced(-i128(num), den);
        return res;
    }
    friend bool operator==(const Frac &a, const Frac &b) {
        return a.num == b.num && a.den == b.den;
    }
    friend bool operator!=(const Frac &a, const Frac &b) { return !(a == b); }
    friend bool operator<(const Frac &a, const Frac &b) {
        return i128(a.num) * b.den < i128(b.num) * a.den;
    }
    friend bool operator>(const Frac &a, const Frac &b) { return b < a; }
    friend bool operator<=(const Frac &a, const Frac &b) { return !(b < a); }
    friend bool operator>=(const Frac &a, const Frac &b) { return !(a < b); }
    // ===== 浮点转换与输出：不需要时可直接删除 =====
    explicit operator ld() const { return ld(num) / ld(den); }
    friend ostream &operator<<(ostream &out, const Frac &x) {
        out << x.num;
        if (x.den != 1) out << '/' << x.den;
        return out;
    }
};
```
#### 必须使用 `i128` 存储时
不能只把成员类型机械改成 `i128`。至少需要恢复安全的非交叉乘比较：
```cpp
using i128 = __int128_t;
using u128 = __uint128_t;
static u128 mag(i128 x) {
    return x < 0 ? u128(-(x + 1)) + 1 : u128(x);
}
// 比较非负分数 a / b 与 c / d，不使用交叉乘
static int cmp_pos(u128 a, u128 b, u128 c, u128 d) {
    bool rev{ false };
    while (true) {
        u128 q1{ a / b }, q2{ c / d };
        if (q1 != q2) {
            int res{ q1 < q2 ? -1 : 1 };
            return rev ? -res : res;
        }
        a %= b;
        c %= d;
        if (!a || !c) {
            int res{ a == c ? 0 : (a ? 1 : -1) };
            return rev ? -res : res;
        }
        swap(a, b);
        swap(c, d);
        rev = !rev;
    }
}
```
