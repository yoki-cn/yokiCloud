### Matrix
---
```cpp
template<typename T = i64>
struct Matrix {
    int n, m;
    vector<vector<T>> mat;
    Matrix(int n, int m) : n(n), m(m), mat(n, vector<T>(m, T{})) {}
    static Matrix E(int n) {
        Matrix res(n, n);
        for (int i = 0; i < n; ++ i) res[i][i] = T{1};
        return res;
    }
    vector<T>& operator[](int i) { return mat[i]; }
    const vector<T>& operator[](int i) const { return mat[i]; }
    Matrix operator+(const Matrix& o) const {
        Matrix res(n, m);
        for (int i = 0; i < n; ++ i) {
            for (int j = 0; j < m; ++ j) res[i][j] = mat[i][j] + o[i][j];
        }
        return res;
    }
    Matrix operator*(const Matrix& o) const {
        Matrix res(n, o.m);
        for (int i = 0; i < n; ++ i) {
            for (int k = 0; k < m; ++ k) {
                if (mat[i][k] == T{0}) continue;
                for (int j = 0; j < o.m; ++ j) {
                    res[i][j] = res[i][j] + mat[i][k] * o[k][j];
                }
            }
        }
        return res;
    }
};
template<class T>
Matrix<T> matrix_power(Matrix<T> a, u64 e) {
    Matrix<T> res = Matrix<T>::E(a.n);
    for (; e; e >>= 1, a = a * a)
        if (e & 1) res = res * a;
    return res;
}
```
### 定长 Matrix
---
```cpp
template<int N, int M = N, typename T = i64>
struct StaticMatrix {
    array<array<T, M>, N> mat;
    StaticMatrix() { for (auto& row : mat) row.fill(T{0}); }
    static StaticMatrix E() {
        StaticMatrix res;
        for (int i = 0; i < min(N, M); ++ i) res[i][i] = T{1};
        return res;
    }
    array<T, M>& operator[](int i) { return mat[i]; }
    const array<T, M>& operator[](int i) const { return mat[i]; }
    StaticMatrix operator+(const StaticMatrix& o) const {
        StaticMatrix res;
        for (int i = 0; i < N; ++ i) {
            for (int j = 0; j < M; ++ j) res[i][j] = mat[i][j] + o[i][j];
        }
        return res;
    }
    template<int K>
    StaticMatrix<N, K, T> operator*(const StaticMatrix<M, K, T>& o) const {
        StaticMatrix<N, K, T> res;
        for (int i = 0; i < N; ++ i) {
            for (int k = 0; k < M; ++ k) {
                if (mat[i][k] == T{0}) continue;
                for (int j = 0; j < K; ++ j) {
                    res[i][j] = res[i][j] + mat[i][k] * o[k][j];
                }
            }
        }
        return res;
    }
};
using Mat2 = StaticMatrix<2>;
template<int N, class T>
StaticMatrix<N, N, T> power(StaticMatrix<N, N, T> a, u64 e) {
    auto res = StaticMatrix<N, N, T>::E();
    for (; e; e >>= 1, a = a * a)
        if (e & 1) res = res * a;
    return res;
}
```

### 热带 / 常用半环矩阵速查

| 普通矩阵           | Min-Plus                | Max-Plus                | Min-Max                     | Max-Min                     |
| -------------- | ----------------------- | ----------------------- | --------------------------- | --------------------------- |
| `T ans = 0`    | `T ans = INF`           | `T ans = -INF`          | `T ans = INF`               | `T ans = -INF`              |
| `ans += x * y` | `ans = min(ans, x + y)` | `ans = max(ans, x + y)` | `ans = min(ans, max(x, y))` | `ans = max(ans, min(x, y))` |
| 零矩阵全 `0`       | 全 `INF`                 | 全 `-INF`                | 全 `INF`                     | 全 `-INF`                    |
| 单位阵对角 `1`      | 对角 `0`                  | 对角 `0`                  | 对角 `-INF`                   | 对角 `INF`                    |
| 单位阵非对角 `0`     | 非对角 `INF`               | 非对角 `-INF`              | 非对角 `INF`                   | 非对角 `-INF`                  |
#### Min-Plus
##### 图论含义
若
```cpp
A[u][v] = edge_weight(u, v);
```
那么
$$
(A^k)_{uv}
$$
表示：
> 从 $u$ 到 $v$，**恰好经过 $k$ 条边**的最短路长度

因此适合：
* 固定边数最短路
* 最小转移代价
* DP + 矩阵快速幂
---
#### Max-Plus
##### 常见意义
$$
(A^k)_{uv}
$$
表示：
> 从 \(u\) 到 \(v\)，恰好进行 \(k\) 次转移能够得到的最大收益

常用于：
* 固定步数最大权路径
* 最大收益 DP
* 调度
* Max-Plus 线性系统
---
#### Min-Max
##### 图论意义
路径的代价不是边权之和，而是
$$
\max_{e\in P}w_e
$$
然后在所有路径中取最小：
$$
\min_P\max_{e\in P}w_e
$$
也就是典型的 **minimax path / 最小瓶颈路**

---
#### Max-Min
##### 图论意义
一条路径的容量为：
$$
\min_{e\in P}w_e
$$
然后选容量最大的路径：
$$
\max_P\min_{e\in P}w_e
$$
即：
* widest path
* 最大瓶颈路径
* 最大运输容量
---
#### Boolean 半环
若 `A[u][v] = true` 表示有边，则
$$
(A^k)_{uv}
$$
表示：
> 是否存在从 $u$ 到 $v$ 恰好走 $k$ 条边的路径
