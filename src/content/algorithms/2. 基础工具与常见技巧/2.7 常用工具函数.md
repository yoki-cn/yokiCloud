### 平方根
```cpp
u64 isqrt(u64 n) {
    u64 x{ u64(sqrt(ld(n))) };
    while (x > 0 && x > n / x) -- x;
    while (x + 1 <= n / (x + 1)) ++ x;
    return x;
}
```
### 立方根
```cpp
u64 icbrt(u64 n) {
    u64 x{ u64(cbrt(ld(n))) };
    auto le = [&](u64 v) { return v == 0 || v <= n / v / v; };
    while (!le(x)) -- x;
    while (le(x + 1)) ++ x;
    return x;
}
```
### 底为 2 的整数对数
```cpp
// 要求 x > 0，返回 floor(log2(x))
int floor_log2(u64 x) { return 63 - __builtin_clzll(x); }
// 要求 x > 0，返回 ceil(log2(x))
int ceil_log2(u64 x) { return floor_log2(x) + ((x & (x - 1)) != 0); }
```
### 任意整数底数的对数
```cpp
// 要求 base >= 2, x >= 1，返回最大的 k，使 base^k <= x
int floor_log(u64 x, u64 base) {
    int k{ 0 };
    while (x >= base) {
        x /= base;
        ++ k;
    }
    return k;
}
// 要求 base >= 2, x >= 1，返回最小的 k，使 base^k >= x
int ceil_log(u64 x, u64 base) {
    int k{ 0 };
    u128 p{ 1 };
    while (p < x) {
        p *= base;
        ++ k;
    }
    return k;
}
```
### 有符号整数 `floor_div` / `ceil_div`
```cpp
template<class T>
T floor_div(T a, T b) {
    T q{ a / b }, r{ a % b };
    if (r != 0 && ((r > 0) != (b > 0))) -- q;
    return q;
}
template<class T>
T ceil_div(T a, T b) {
    T q{ a / b }, r{ a % b };
    if (r != 0 && ((r > 0) == (b > 0))) ++ q;
    return q;
}
```
### `chmin` / `chmax` / `sign`
```cpp
template<class T, class U>
bool chmin(T &a, const U &b) {
    if (b < a) {
        a = b;
        return true;
    }
    return false;
}
template<class T, class U>
bool chmax(T &a, const U &b) {
    if (a < b) {
        a = b;
        return true;
    }
    return false;
}
template<class T>
int sign(const T &x) { return (T{} < x) - (x < T{}); }
```
