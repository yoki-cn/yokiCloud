$$
\frac1{1-x}=\sum_{n\ge0}x^n,\quad\frac1{(1-x)^r}=\sum_{n\ge0}\binom{n+r-1}{r-1}x^n,
$$


$$
e^x=\sum_{n\ge0}\frac{x^n}{n!},\quad\log(1+x)=\sum_{n\ge1}\frac{(-1)^{n-1}}n x^n,
$$


$$
\frac1{1-x-x^2}=\sum_{n\ge0}F_{n+1}x^n,\quad\sum_{n\ge0}H_nx^n=\frac{-\log(1-x)}{1-x},
$$


$$
\sum_{n\ge0}n x^n=\frac{x}{(1-x)^2},\quad \sum_{n\ge0}n^2x^n=\frac{x(1+x)}{(1-x)^3}.
$$


$F_0=0,F_1=1$，$H_0=0$。若 $A(x)$ 为形式幂级数，则


$$
[x^n]\frac{A(x)}{1-x}=\sum_{k=0}^n a_k,\quad [x^n]\frac{A(x)}{(1-x)^r}=\sum_{k=0}^n\binom{n-k+r-1}{r-1}a_k.
$$
