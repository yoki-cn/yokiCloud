$$
\sum_{k=0}^m(-1)^k\binom nk=(-1)^m\binom{n-1}m\quad(0\le m<n).
$$


$$
\sum_{k=0}^m\binom nk=2^n-\sum_{k=m+1}^n\binom nk,\qquad \sum_{k=0}^m k\binom nk=n\sum_{k=0}^{m-1}\binom{n-1}k.
$$


$$
\sum_{k=0}^{n}\binom nk\frac1{k+1}=\frac{2^{n+1}-1}{n+1},\qquad\sum_{k=0}^{n}(-1)^k\binom nk\frac1{k+1}=\frac1{n+1}.
$$


$$
\sum_{k=0}^{n}\binom nk\frac1{k+r}=\int_0^1 x^{r-1}(1+x)^n\,dx\quad(r>0).
$$
