$$
\sin^2x+\cos^2x=1,\quad\sin(x\pm y)=\sin x\cos y\pm\cos x\sin y,
$$


$$
\cos(x\pm y)=\cos x\cos y\mp\sin x\sin y,\quad\tan(x+y)=\frac{\tan x+\tan y}{1-\tan x\tan y},
$$


$$
\sin2x=2\sin x\cos x,\quad\cos2x=1-2\sin^2x=2\cos^2x-1,
$$


$$
\sin x\sin y=\frac{\cos(x-y)-\cos(x+y)}2,\quad\cos x\cos y=\frac{\cos(x-y)+\cos(x+y)}2,
$$


$$
\sin x+\sin y=2\sin\frac{x+y}2\cos\frac{x-y}2,\quad\cos x+\cos y=2\cos\frac{x+y}2\cos\frac{x-y}2.
$$



$$
e^{ix}=\cos x+i\sin x,\quad\cos nx+i\sin nx=(\cos x+i\sin x)^n,
$$


$$
\sum_{k=0}^{n-1}e^{ikx}=e^{i(n-1)x/2}\frac{\sin(nx/2)}{\sin(x/2)}\quad(\sin(x/2)\ne0),
$$


$$
\sum_{k=1}^n\sin kx=\frac{\sin(nx/2)\sin((n+1)x/2)}{\sin(x/2)},\quad\sum_{k=1}^n\cos kx=\frac{\sin(nx/2)\cos((n+1)x/2)}{\sin(x/2)}.
$$
