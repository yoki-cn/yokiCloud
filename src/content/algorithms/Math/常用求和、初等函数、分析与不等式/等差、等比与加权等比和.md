$$
\sum_{k=1}^nk=\frac{n(n+1)}2,\quad\sum_{k=1}^nk^2=\frac{n(n+1)(2n+1)}6,\quad\sum_{k=1}^nk^3=\left(\frac{n(n+1)}2\right)^2.
$$



$$
\sum_{k=0}^{n-1}r^k=\frac{1-r^n}{1-r}\quad(r\ne1),\qquad\sum_{k=0}^{n-1}kr^k=\frac{r-nr^n+(n-1)r^{n+1}}{(1-r)^2}\quad(r\ne1),
$$



$$
\sum_{k=0}^{n-1}k^2r^k=\frac{r(1+r)-n^2r^n+(2n^2-2n-1)r^{n+1}-(n-1)^2r^{n+2}}{(1-r)^3}\quad(r\ne1).
$$


$|r|<1$ 时有无限级数：


$$
\sum_{k\ge0}r^k=\frac1{1-r},\quad\sum_{k\ge0}kr^k=\frac r{(1-r)^2},\quad\sum_{k\ge0}k^2r^k=\frac{r(1+r)}{(1-r)^3},
$$


$$
\sum_{k\ge0}k^3r^k=\frac{r(1+4r+r^2)}{(1-r)^4},\qquad\sum_{k\ge0}\binom{k+s-1}s r^k=\frac{r}{(1-r)^{s+1}}\quad(s\ge1).
$$
