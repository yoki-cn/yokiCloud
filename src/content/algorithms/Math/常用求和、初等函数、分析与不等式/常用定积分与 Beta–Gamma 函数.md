$$
\Gamma(s)=\int_0^\infty t^{s-1}e^{-t}\,dt\quad(s>0),\quad\Gamma(s+1)=s\Gamma(s),\quad\Gamma(n+1)=n!\quad(n\in\mathbb Z_{\ge0}),
$$


$$
B(a,b)=\int_0^1t^{a-1}(1-t)^{b-1}\,dt=\frac{\Gamma(a)\Gamma(b)}{\Gamma(a+b)}\quad(a,b>0),
$$


$$
\Gamma(1/2)=\sqrt\pi,\quad\int_{-\infty}^\infty e^{-ax^2}\,dx=\sqrt{\frac\pi a}\quad(a>0),
$$


$$
\int_0^{\pi/2}\sin^n x\,dx=\int_0^{\pi/2}\cos^n x\,dx=\frac{\sqrt\pi\,\Gamma((n+1)/2)}{2\Gamma((n+2)/2)}\quad(n>-1),
$$


$$
\int_0^1x^m(1-x)^n\,dx=\frac{m!n!}{(m+n+1)!}\quad(m,n\in\mathbb Z_{\ge0}).
$$
