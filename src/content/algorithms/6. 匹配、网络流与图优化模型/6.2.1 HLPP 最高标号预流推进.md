### HLPP（最高标号预流推进）
---
**解决问题：** 求有向图最大流，使用 Highest Label Push-Relabel，并加入 Gap 与 Global Relabel 启发式，适合作为 Dinic 在较难网络上的性能互补版本
```cpp
template<class Cap = i64>
struct HLPP {
    static_assert(numeric_limits<Cap>::is_integer && numeric_limits<Cap>::is_signed);
    struct Edge {
        int to, rev;
        Cap cap;
    };
    int n;
    vector<vector<Edge>> g;
    explicit HLPP(int n_) : n(n_), g(n) {}
    void add_edge(int u, int v, Cap cap) {
        if (u == v) return;
        Edge a{ v, int(g[v].size()), cap }, b{ u, int(g[u].size()), 0 };
        g[u].push_back(a);
        g[v].push_back(b);
    }
    Cap max_flow(int s, int t) {
        if (s == t) return 0;
        vector<Cap> excess(n);
        vector<int> h(n, n), cur(n), cnt(2 * n + 2);
        vector<char> active(n);
        vector<vector<int>> bucket(2 * n + 2);
        int highest{};
        // 从汇点沿反向残量边 BFS，重建精确距离标号与活跃桶
        auto global_relabel = [&]() {
            ranges::fill(h, n);
            ranges::fill(cur, 0);
            ranges::fill(cnt, 0);
            ranges::fill(active, false);
            for (auto &v : bucket) v.clear();
            queue<int> q;
            h[t] = 0;
            q.push(t);
            while (!q.empty()) {
                int u{ q.front() };
                q.pop();
                for (const Edge &e : g[u]) {
                    const Edge &r{ g[e.to][e.rev] };
                    if (r.cap > 0 && h[e.to] == n) h[e.to] = h[u] + 1, q.push(e.to);
                }
            }
            h[s] = n;
            for (int u = 0; u < n; ++ u) ++ cnt[h[u]];
            highest = 0;
            for (int u = 0; u < n; ++ u) if (u != s && u != t && excess[u] > 0 && h[u] < n) {
                active[u] = true;
                bucket[h[u]].push_back(u);
                highest = max(highest, h[u]);
            }
        };
        auto activate = [&](int u) {
            if (u == s || u == t || excess[u] == 0 || h[u] >= n || active[u]) return;
            active[u] = true;
            bucket[h[u]].push_back(u);
            highest = max(highest, h[u]);
        };
        // 源点饱和推出预流，源点 excess 无需维护
        h[s] = n;
        for (Edge &e : g[s]) if (e.cap > 0) {
            Cap f{ e.cap };
            e.cap = 0;
            g[e.to][e.rev].cap += f;
            excess[e.to] += f;
        }
        global_relabel();
        int work{};
        while (highest >= 0) {
            while (highest >= 0 && bucket[highest].empty()) -- highest;
            if (highest < 0) break;
            int u{ bucket[highest].back() };
            bucket[highest].pop_back();
            active[u] = false;
            if (excess[u] == 0 || h[u] != highest || h[u] >= n) continue;
            while (excess[u] > 0) {
                if (cur[u] == int(g[u].size())) {
                    int old{ h[u] }, nh{ 2 * n };
                    for (const Edge &e : g[u]) if (e.cap > 0) nh = min(nh, h[e.to] + 1);
                    -- cnt[old];
                    h[u] = min(nh, 2 * n);
                    ++ cnt[h[u]];
                    cur[u] = 0;
                    // Gap：高度 old 断层后，old 以上且仍低于 n 的点均无法到达汇点
                    if (old < n && cnt[old] == 0) {
                        for (int v = 0; v < n; ++ v) if (v != s && v != t && old < h[v] && h[v] < n) {
                            -- cnt[h[v]];
                            h[v] = n;
                            ++ cnt[n];
                            cur[v] = 0;
                        }
                    }
                    if (h[u] >= n) break;
                    highest = max(highest, h[u]);
                    continue;
                }
                Edge &e{ g[u][cur[u]] };
                if (e.cap > 0 && h[u] == h[e.to] + 1) {
                    Cap f{ min(excess[u], e.cap) };
                    e.cap -= f;
                    g[e.to][e.rev].cap += f;
                    excess[u] -= f;
                    excess[e.to] += f;
                    activate(e.to);
                } else ++ cur[u];
            }
            activate(u);
            if (++ work % max(1, 4 * n) == 0) global_relabel();
        }
        return excess[t];
    }
};
```
#### 接口说明
##### 接口
- `HLPP<Cap>(n)`：建立 `n` 点有向残量网络，$O(n)$
- `add_edge(u, v, cap)`：加入容量为 `cap` 的有向边，自环自动忽略，均摊 $O(1)$
- `max_flow(s, t)`：使用最高标号预流推进求最大流值
##### `add_edge()` 条件
- `0 <= u, v < n`
- `cap >= 0`
- `Cap` 必须为有符号整数类型
#### 算法说明
HLPP 不维护从始至终合法的普通流，而是允许中间点暂时拥有正的 `excess`，不断从**当前高度最大的活跃点**向外推流
- `h[u]`：高度标号，只允许沿 `h[u] = h[v] + 1` 的残量边推流
- `excess[u]`：流入减流出的超额流量，`excess[u] > 0` 的普通点为活跃点
- `relabel`：当前点无法继续推流时，将高度提高到可推边所需的最小高度
- Highest Label：优先处理高度最大的活跃点，减少无效往返
- Gap：若某个高度层为空，则更高且仍可达汇点范围内的点实际上都无法到达汇点
- Global Relabel：从汇点沿反向残量边 BFS，重新计算到汇点的精确残量距离
#### 注意事项
- 本版定位是 **Dinic 的性能互补版本**，普通题默认仍优先使用 Dinic
- `max_flow()` 会原地修改残量网络，默认同一对象只求解一次
- 本实现允许最终仍有超额流滞留在无法到达汇点的点上，因此残量状态是**最大预流**而不保证是普通可行流
- 因此不要使用本板恢复逐边实际流量，也不要直接从最终残量图恢复最小割
- 需要边流量、最小割、限流增广、修改容量、上下界流等接口时使用 Dinic
- 源点初始化时不维护 `excess[s]`，避免源点所有出边容量之和超过 `Cap` 时产生无意义的负溢出
- 仍需保证单条容量、各点 `excess` 和最终最大流不会超过 `Cap`
- 无向容量边需要显式加入两个方向，自动建立的反向边仅为残量边
- 自环不会影响 $s-t$ 最大流值，因此直接忽略，也避免自环时 `rev` 的特殊处理
- 平行边可以直接加入，不影响正确性
- Global Relabel 的触发周期 `4 * n` 是启发式参数，可以按实际数据调整
#### 常见用法
```cpp
HLPP<i64> flow(n);
for (auto [u, v, cap] : edges) flow.add_edge(u, v, cap);
i64 ans{ flow.max_flow(s, t) };
```
> [!tip]
> 稀疏普通网络、需要恢复方案时优先 Dinic；只求最大流值，且 Dinic 在稠密图或特殊构造上较慢时再尝试 HLPP
#### 复杂度
通用最坏时间复杂度：$O(n^2m)$
空间复杂度：$O(n+m)$
