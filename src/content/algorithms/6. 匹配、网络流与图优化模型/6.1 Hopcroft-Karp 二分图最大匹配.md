### 二分图最大匹配（Hopcroft--Karp）
---
```cpp
struct HopcroftKarp {
    int nl, nr, shortest{ -1 }, matching{ 0 };
    vector<vector<int>> g;
    vector<int> ml, mr, dis, ptr;
    HopcroftKarp(int nl_, int nr_) : nl(nl_), nr(nr_), g(nl), ml(nl, -1), mr(nr, -1), dis(nl), ptr(nl) {}
    void add_edge(int l, int r) { g[l].push_back(r); }
    // ===== Hopcroft--Karp 核心 =====
    // 每轮只保留最短增广路所在的分层
    bool bfs() {
        queue<int> q;
        ranges::fill(dis, -1);
        for (int u = 0; u < nl; ++ u) if (ml[u] == -1) dis[u] = 0, q.push(u);
        shortest = -1;
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            if (shortest != -1 && dis[u] + 1 > shortest) continue;
            for (int v : g[u]) {
                int w = mr[v];
                if (w == -1) {
                    if (shortest == -1) shortest = dis[u] + 1;
                } else if (dis[w] == -1 && (shortest == -1 || dis[u] + 1 < shortest)) {
                    dis[w] = dis[u] + 1;
                    q.push(w);
                }
            }
        }
        return shortest != -1;
    }
    bool dfs(int u) {
        for (int &i = ptr[u]; i < g[u].size(); ++ i) {
            int v = g[u][i], w = mr[v];
            if (w == -1) {
                if (dis[u] + 1 != shortest) continue;
            } else if (dis[w] != dis[u] + 1 || !dfs(w)) {
                continue;
            }
            ml[u] = v;
            mr[v] = u;
            return true;
        }
        dis[u] = -1;
        return false;
    }
    // 可以在已有匹配上继续增广，因此支持增量加边 / 加点后再次调用
    int max_matching() {
        while (bfs()) {
            ranges::fill(ptr, 0);
            for (int u = 0; u < nl; ++ u) if (ml[u] == -1 && dfs(u)) ++ matching;
        }
        return matching;
    }
    // ===== 匹配方案 =====
    vector<pair<int, int>> matching_edges() const {
        vector<pair<int, int>> ans;
        ans.reserve(matching);
        for (int u = 0; u < nl; ++ u) if (ml[u] != -1) ans.emplace_back(u, ml[u]);
        return ans;
    }
    // ===== 动态建图 =====
    int add_left() {
        g.push_back({});
        ml.push_back(-1);
        dis.push_back(0);
        ptr.push_back(0);
        return nl ++;
    }
    int add_right() {
        mr.push_back(-1);
        return nr ++;
    }
    // ===== 删边 / 重算 =====
    // 删除一条 l-r 边；存在重边时只删除一条
    // 若删除的是最后一条承载当前匹配的 l-r 边，会自动解除该匹配
    bool erase_edge(int l, int r) {
        auto it = find(g[l].begin(), g[l].end(), r);
        if (it == g[l].end()) return false;
        g[l].erase(it);
        if (ml[l] == r && find(g[l].begin(), g[l].end(), r) == g[l].end()) {
            ml[l] = mr[r] = -1;
            -- matching;
        }
        return true;
    }
    // 保留图，只清空当前匹配
    void clear_matching() {
        ranges::fill(ml, -1);
        ranges::fill(mr, -1);
        matching = 0;
    }
    // ===== König 定理扩展：点覆盖 / 独立集 =====
    // 从所有未匹配左点出发，沿“非匹配边 L->R、匹配边 R->L”搜索
    pair<vector<char>, vector<char>> alternating_reachable() const {
        vector<char> zl(nl), zr(nr);
        queue<int> q;
        for (int u = 0; u < nl; ++ u) if (ml[u] == -1) zl[u] = true, q.push(u);
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            for (int v : g[u]) {
                if (ml[u] == v || zr[v]) continue;
                zr[v] = true;
                if (mr[v] != -1 && !zl[mr[v]]) {
                    zl[mr[v]] = true;
                    q.push(mr[v]);
                }
            }
        }
        return {zl, zr};
    }
    // 最小点覆盖 = (L \ ZL) ∪ ZR
    pair<vector<int>, vector<int>> min_vertex_cover() const {
        auto [zl, zr] = alternating_reachable();
        vector<int> left, right;
        for (int u = 0; u < nl; ++ u) if (!zl[u]) left.push_back(u);
        for (int v = 0; v < nr; ++ v) if (zr[v]) right.push_back(v);
        return {left, right};
    }
    // 最大独立集 = ZL ∪ (R \ ZR)
    pair<vector<int>, vector<int>> max_independent_set() const {
        auto [zl, zr] = alternating_reachable();
        vector<int> left, right;
        for (int u = 0; u < nl; ++ u) if (zl[u]) left.push_back(u);
        for (int v = 0; v < nr; ++ v) if (!zr[v]) right.push_back(v);
        return {left, right};
    }
    // ===== 最小边覆盖 =====
    // 要求当前匹配已经最大，且整个二分图不存在孤立点
    vector<pair<int, int>> min_edge_cover() const {
        vector<pair<int, int>> ans;
        vector<int> any(nr, -1);
        for (int u = 0; u < nl; ++ u) {
            for (int v : g[u]) if (any[v] == -1) any[v] = u;
        }
        for (int u = 0; u < nl; ++ u) if (ml[u] != -1) ans.emplace_back(u, ml[u]);
        for (int u = 0; u < nl; ++ u) if (ml[u] == -1) ans.emplace_back(u, g[u][0]);
        for (int v = 0; v < nr; ++ v) if (mr[v] == -1) ans.emplace_back(any[v], v);
        return ans;
    }
};
```
#### 成员变量
- `g[u]`：左部点 `u` 相邻的所有右部点
- `ml[u]`：左部点 `u` 当前匹配的右部点，未匹配为 `-1`
- `mr[v]`：右部点 `v` 当前匹配的左部点，未匹配为 `-1`
- `matching`：当前匹配边数
- `nl`、`nr`：当前左右部点数
#### 接口
- `HopcroftKarp(nl,nr)`：建立左右部大小分别为 `nl`、`nr` 的二分图，$O(n_l+n_r)$
- `add_edge(l,r)`：加入左部 `l` 到右部 `r` 的边，均摊 $O(1)$
- `max_matching()`：从当前匹配继续增广到最大匹配，返回当前总匹配数，$O(m\sqrt{n_l+n_r})$
- `matching_edges()`：返回当前匹配中的所有 `{左点,右点}`，$O(n_l)$
- `add_left()`：新增一个左部点并返回编号，均摊 $O(1)$
- `add_right()`：新增一个右部点并返回编号，均摊 $O(1)$
- `erase_edge(l,r)`：删除一条 `l-r` 边；若最后一条该重边正被匹配则同时解除匹配，$O(\deg(l))$
- `clear_matching()`：保留图并清空当前匹配，$O(n_l+n_r)$
- `alternating_reachable()`：在最大匹配对应交错图中返回 `{ZL,ZR}` 的可达标记，$O(n+m)$
- `min_vertex_cover()`：恢复一个最小点覆盖，返回 `{左部点集合,右部点集合}`，$O(n+m)$
- `max_independent_set()`：恢复一个最大独立集，返回 `{左部点集合,右部点集合}`，$O(n+m)$
- `min_edge_cover()`：恢复一个最小边覆盖，返回边集合，$O(n+m)$
#### 注意事项
- 左右两侧使用独立的 `0base` 编号：左部为 `0..nl-1`，右部为 `0..nr-1`
- `ml` / `mr` 本身就是一种匹配方案；`matching_edges()` 只是将它整理成边集合
- 最大匹配通常不唯一，`ml` / `mr` 保存的是算法找到的其中一种方案
- 可以在已有最大匹配后继续 `add_edge()`、`add_left()`、`add_right()`，再调用 `max_matching()` 增量增广，不需要清空原匹配
- `erase_edge()` 后建议重新调用 `max_matching()`；删除匹配边后原匹配可能已经不再最大
- `erase_edge()` 使用 `vector::erase`，只适合少量动态修改；若存在大量删边操作，不建议使用本板子模拟全动态匹配
- 改边可以使用 `erase_edge(old_l,old_r)`、`add_edge(new_l,new_r)` 后重新调用 `max_matching()`
- `min_vertex_cover()`、`max_independent_set()`、`alternating_reachable()` 必须在当前图已经得到最大匹配后调用
- 最小点覆盖大小等于最大匹配大小
- 最大独立集大小为 `nl + nr - matching`
- `min_edge_cover()` 必须在最大匹配后调用，并要求所有点都不是孤立点；其大小为 `nl + nr - matching`
- 左部存在完全匹配当且仅当 `matching == nl`
- 右部存在完全匹配当且仅当 `matching == nr`
- 整张二分图存在完美匹配需要 `nl == nr && matching == nl`
- 重边不影响最大匹配正确性，但会增加常数
- 点具有容量的多重匹配更适合转最大流；带权匹配使用 KM 或费用流
- 该模板只处理二分图；一般图最大匹配使用带花树
#### 常见用法
##### 最大匹配并恢复一种方案
```cpp
HopcroftKarp hk(nl, nr);
for (auto [l, r] : edges) hk.add_edge(l, r);
cout << hk.max_matching() << '\n';
for (auto [l, r] : hk.matching_edges()) {
    cout << l << ' ' << r << '\n';
}
```
也可以直接使用 `ml`：
```cpp
hk.max_matching();
for (int l = 0; l < nl; ++ l) {
    if (hk.ml[l] != -1) {
        int r = hk.ml[l];
        // l 与 r 匹配
    }
}
```
##### 判断并恢复左部完全匹配
```cpp
if (hk.max_matching() == hk.nl) {
    for (int l = 0; l < hk.nl; ++ l) {
        int r = hk.ml[l];
        // 每个左点 l 都分配给唯一右点 r
    }
}
```
##### 增量加入边或点
已经求过最大匹配后，不需要重新建图：
```cpp
hk.max_matching();
hk.add_edge(l, r);
int u = hk.add_left();
int v = hk.add_right();
hk.add_edge(u, v);
int matching = hk.max_matching();
```
##### 删除或修改边
```cpp
hk.max_matching();
hk.erase_edge(u, v);
hk.max_matching();
```
修改一条边：
```cpp
hk.erase_edge(u1, v1);
hk.add_edge(u2, v2);
hk.max_matching();
```
##### 最小点覆盖
```cpp
hk.max_matching();
auto [left, right] = hk.min_vertex_cover();
for (int u : left) {
    // 选择左部点 u
}
for (int v : right) {
    // 选择右部点 v
}
```
覆盖中的点数恒为：
```cpp
int cover_size = left.size() + right.size(); // == hk.matching
```
##### 最大独立集
```cpp
hk.max_matching();
auto [left, right] = hk.max_independent_set();
```
其中任意两个被选点之间都不存在二分图边，且：
```cpp
int independent_size = left.size() + right.size();
// == hk.nl + hk.nr - hk.matching
```
##### 最小边覆盖
要求图中没有孤立点：
```cpp
hk.max_matching();
auto cover = hk.min_edge_cover();
for (auto [u, v] : cover) {
    // 选择边 u-v
}
```
边数为：
```cpp
int edge_cover_size = hk.nl + hk.nr - hk.matching;
```
##### DAG 最小路径覆盖并恢复方案
对于 DAG，将每个原点 `u` 拆成左部 `u` 和右部 `u`；原图存在边 `u -> v` 时加入二分图边 `u_L -> v_R`
```cpp
HopcroftKarp hk(n, n);
for (auto [u, v] : edges) hk.add_edge(u, v);
int matching = hk.max_matching();
int path_count = n - matching;
```
恢复每条路径：
```cpp
vector<int> pre(n, -1), nxt(n, -1);
for (auto [u, v] : hk.matching_edges()) {
    nxt[u] = v;
    pre[v] = u;
}
vector<vector<int>> paths;
for (int s = 0; s < n; ++ s) if (pre[s] == -1) {
    vector<int> path;
    for (int u = s; u != -1; u = nxt[u]) {
        path.push_back(u);
    }
    paths.push_back(move(path));
}
```
此时 `paths.size() == n - matching`
##### 网格骨牌 / 配对方案
将网格按黑白染色划分为左右部，相邻可配对格子连边：
```cpp
int matching = hk.max_matching();
if (matching * 2 == cells) {
    for (auto [black, white] : hk.matching_edges()) {
        // black 与 white 构成一块骨牌
    }
}
```
#### 复杂度
设 $n=n_l+n_r$，边数为 $m$

| 操作 | 时间复杂度 |
| --- | --- |
| `add_edge()` | 均摊 $O(1)$ |
| `add_left()` / `add_right()` | 均摊 $O(1)$ |
| `erase_edge()` | $O(\deg(l))$ |
| `clear_matching()` | $O(n)$ |
| `max_matching()` | $O(m\sqrt n)$ |
| `matching_edges()` | $O(n_l)$ |
| `alternating_reachable()` | $O(n+m)$ |
| `min_vertex_cover()` | $O(n+m)$ |
| `max_independent_set()` | $O(n+m)$ |
| `min_edge_cover()` | $O(n+m)$ |
空间复杂度：$O(n+m)$
多次增量修改并反复调用 `max_matching()` 时，总复杂度按各次调用累加，不应直接视为一次 $O(m\sqrt n)$
