### Dinic 最大流
---
```cpp
template<class Cap = i64>
struct Dinic {
    static_assert(numeric_limits<Cap>::is_integer && numeric_limits<Cap>::is_signed);
    struct Edge {
        int from, to;
        Cap cap, init;
    };
    int n;
    vector<Edge> e;
    vector<vector<int>> g;
    vector<int> level, it;
    explicit Dinic(int n_, int m_hint = 0) : n(n_), g(n), level(n), it(n) {
        e.reserve(2 * m_hint);
    }
    // ===== 基础建边与 Dinic =====
    int add_edge(int u, int v, Cap cap, Cap rcap = 0) {
        int id = e.size() / 2;
        g[u].push_back(e.size());
        e.push_back({u, v, cap, cap});
        g[v].push_back(e.size());
        e.push_back({v, u, rcap, rcap});
        return id;
    }
    bool bfs(int s, int t, Cap scale = 1) {
        ranges::fill(level, -1);
        queue<int> q;
        level[s] = 0;
        q.push(s);
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            for (int id : g[u]) {
                const Edge &edge = e[id];
                if (edge.cap < scale || level[edge.to] != -1) continue;
                level[edge.to] = level[u] + 1;
                q.push(edge.to);
            }
        }
        return level[t] != -1;
    }
    Cap dfs(int u, int t, Cap f, Cap scale = 1) {
        if (u == t) return f;
        Cap sent = 0;
        for (int &i = it[u]; i < g[u].size() && sent < f; ++ i) {
            int id = g[u][i];
            Edge &edge = e[id];
            if (edge.cap < scale || level[edge.to] != level[u] + 1) continue;
            Cap got = dfs(edge.to, t, min(f - sent, edge.cap), scale);
            if (got == 0) continue;
            edge.cap -= got;
            e[id ^ 1].cap += got;
            sent += got;
        }
        if (sent < f) level[u] = -1;
        return sent;
    }
    Cap max_flow(int s, int t, Cap limit = numeric_limits<Cap>::max()) {
        if (s == t || limit == 0) return 0;
        Cap flow = 0;
        while (flow < limit && bfs(s, t)) {
            ranges::fill(it, 0);
            flow += dfs(s, t, limit - flow);
        }
        return flow;
    }
    // ===== 容量缩放：bfs / dfs 的 scale 保留默认值即可 =====
    Cap max_flow_scaling(int s, int t, Cap limit = numeric_limits<Cap>::max()) {
        if (s == t || limit == 0) return 0;
        Cap mx = 0;
        for (const Edge &edge : e) mx = max(mx, edge.cap);
        Cap scale = 1;
        while (scale <= mx / 2) scale *= 2;
        Cap flow = 0;
        for (; scale > 0 && flow < limit; scale /= 2) {
            while (flow < limit && bfs(s, t, scale)) {
                ranges::fill(it, 0);
                flow += dfs(s, t, limit - flow, scale);
            }
        }
        return flow;
    }
    // ===== 动态容量：用于逐步开放边或修改容量 =====
    void increase_cap(int id, Cap x) {
        assert(x >= 0);
        e[2 * id].cap += x;
        e[2 * id].init += x;
    }
    // 保持当前净流量不变；若新容量无法容纳当前流量则修改失败
    bool set_cap(int id, Cap cap) {
        assert(cap >= 0);
        Edge &edge = e[2 * id];
        Cap f = edge.init - edge.cap;
        if (cap < f) return false;
        edge.cap = cap - f;
        edge.init = cap;
        return true;
    }
    void reset_flow() { for (Edge &edge : e) edge.cap = edge.init; }
    // ===== 方案恢复 =====
    tuple<int, int, Cap, Cap> get_edge(int id) const {
        const Edge &edge = e[2 * id];
        return {edge.from, edge.to, edge.init, edge.init - edge.cap};
    }
    Cap edge_flow(int id) const { return e[2 * id].init - e[2 * id].cap; }
    vector<char> min_cut(int s) const {
        vector<char> vis(n);
        vector<int> stk{s};
        vis[s] = true;
        while (!stk.empty()) {
            int u = stk.back();
            stk.pop_back();
            for (int id : g[u]) {
                const Edge &edge = e[id];
                if (edge.cap > 0 && !vis[edge.to]) {
                    vis[edge.to] = true;
                    stk.push_back(edge.to);
                }
            }
        }
        return vis;
    }
    // ===== 上下界流辅助 =====
    void disable_edge(int id) { e[2 * id].cap = e[2 * id + 1].cap = 0; }
};
```
#### 成员变量
- `e[2*id]` / `e[2*id+1]`：第 `id` 条逻辑边的正向 / 反向残量边
- `Edge::cap`：当前残量容量
- `Edge::init`：该方向当前设定的初始容量
- `level[u]`：当前分层图中点 `u` 的层数
- `it[u]`：当前弧位置
#### 接口
- `Dinic(n,m_hint)`：建立 `n` 点残量网络；可选预留约 `m_hint` 条逻辑边，$O(n)$
- `add_edge(u,v,cap)`：加入容量为 `cap` 的有向边，返回逻辑边编号，均摊 $O(1)$
- `add_edge(u,v,cap,rcap)`：正反方向初始容量分别为 `cap`、`rcap`
- `max_flow(s,t)`：在**当前残量网络**上继续增广，返回本次新增最大流
- `max_flow(s,t,limit)`：本次最多再发送 `limit` 单位流
- `max_flow_scaling(s,t,limit)`：容量缩放版最大流
- `increase_cap(id,x)`：将第 `id` 条边的正向容量增加 `x`，保留当前流，$O(1)$
- `set_cap(id,cap)`：保持当前流不变，将正向容量修改为 `cap`；若无法容纳当前净流则返回 `false`，$O(1)$
- `reset_flow()`：清空所有已有流量，恢复为当前设定容量，$O(m)$
- `get_edge(id)`：返回 `{from,to,正向容量,当前净正向流量}`，$O(1)$
- `edge_flow(id)`：返回当前净正向流量，$O(1)$
- `min_cut(s)`：返回当前残量网络从 `s` 可达的点集合，$O(n+m)$
- `disable_edge(id)`：直接清空该逻辑边正反两条残量边，仅用于特殊建模，$O(1)$
#### 注意事项
- 点编号为 `0base`，容量使用**有符号整数类型**
- `max_flow()` 修改残量网络，并且返回的是**本次新增流量**，不是历史总流量
- 因此 `ans += flow.max_flow(s, t);` 可以在同一个残量网络上反复继续增广
- 添加新边或增加边容量不会破坏已有流的可行性，因此可以直接继续调用 `max_flow()`
- `set_cap()` 可以降低容量，但要求新容量仍能容纳该边当前净流量
- 若需要把容量降低到当前流量以下，不能只局部修改该边；最简单可靠的方法是 `reset_flow()` 后修改容量并重新求流
- 如果原网络已经达到最大流，而只**降低**若干边容量，并且 `set_cap()` 均成功，则旧流仍然可行，因此仍然是新网络的最大流
- `add_edge(u,v,c,c)` 表示两个方向共享残量关系的无向容量边；两条互相独立的有向边应分别调用两次 `add_edge`
- `get_edge()` / `edge_flow()` 返回净正向流量；若 `rcap > 0`，流量可能为负，范围为 `[-rcap,cap]`
- `min_cut()` 只有在没有因为 `limit` 提前停止、确实已经得到最大流后，才对应最小割
- `disable_edge()` 不是一般意义上的动态删边；若该边已经承载流量，直接清空正反残量会破坏普通流量语义
- `disable_edge()` 主要用于上下界流等建模中移除辅助边；普通删边优先 `reset_flow()` 后 `set_cap(id,0)`
- `reset_flow()` 会恢复所有边，包括之前被 `disable_edge()` 清空的边
- 多源多汇使用超级源汇；超级边容量取可证明的流量上界，不要无脑使用可能溢出的 `INF`
- 总流量、容量之和以及中间结果都必须能放进 `Cap`
> 普通题默认使用 `max_flow()`。容量跨度特别大时可以尝试 `max_flow_scaling()`；若 Dinic 的分层表现依然很差，再考虑 HLPP
#### 常见用法
##### 最大流与最小割
```cpp
Dinic<i64> flow(n, m);
for (auto [u, v, cap] : edges) {
    flow.add_edge(u, v, cap);
}
i64 ans = flow.max_flow(s, t);
auto side = flow.min_cut(s);
```
##### 保存输入边流量
```cpp
Dinic<i64> flow(n, m);
vector<int> id(m);
for (int i = 0; i < m; ++ i) {
    id[i] = flow.add_edge(u[i], v[i], cap[i]);
}
flow.max_flow(s, t);
for (int i = 0; i < m; ++ i) {
    auto [u, v, cap, used] = flow.get_edge(id[i]);
}
```
##### 查询边的流量与恢复方案
`add_edge()` 返回逻辑边编号，可在求流前保存需要恢复的边：
```cpp
int id = flow.add_edge(u, v, cap);
i64 ans = flow.max_flow(s, t);
i64 f = flow.edge_flow(id); // 边 u -> v 的当前净流量
```
##### 限制本次只发送一定流量
`limit` 表示**这一次调用**最多增加多少流：
```cpp
i64 sent = flow.max_flow(s, t, k);
```
如果需要分阶段发送：
```cpp
i64 ans = 0;
ans += flow.max_flow(s, t, 10);
// 根据当前流量做一些处理
ans += flow.max_flow(s, t, 20);
// 再继续发送至多 20
```
每次都基于上一次留下的残量网络继续
##### 一单位一单位增广
某些需要观察每个流量阶段的题目可以：
```cpp
i64 ans = 0;
while (true) {
    i64 add = flow.max_flow(s, t, 1);
    if (add == 0) break;
    ans += add;
    // 此时最大流恰好又增加了 1
}
```
这种写法会反复执行 Dinic，若不需要中间状态，不应代替一次完整的 `max_flow()`
##### 逐步加入 / 松弛边
已经求出当前图最大流后，可以继续加入新的边：
```cpp
i64 ans = flow.max_flow(s, t);
for (auto [u, v, cap] : new_edges) {
    flow.add_edge(u, v, cap);
}
ans += flow.max_flow(s, t);
```
原有流仍然是一组合法流，因此只需要在新残量网络上继续增广
例如边按照某个阈值逐步开放：
```cpp
Dinic<i64> flow(n);
i64 cur = 0;
for (int i = 0; i < groups.size(); ++ i) {
    for (auto [u, v, cap] : groups[i]) {
        flow.add_edge(u, v, cap);
    }
    cur += flow.max_flow(s, t);
    if (cur >= need) {
        // 第 i 组边开放后第一次达到 need
        break;
    }
}
```
常见于：
- 按边权从小到大逐步允许使用边
- 二分图中逐步允许新的匹配关系
- 随时间开放道路 / 通道
- 按等级逐步松弛建模限制
##### 预先建边，再逐步开放容量
如果边的结构一开始就已知，可以先用容量 `0` 建边：
```cpp
int id = flow.add_edge(u, v, 0);
i64 ans = flow.max_flow(s, t);
flow.increase_cap(id, 5);
ans += flow.max_flow(s, t);
flow.increase_cap(id, 10);
ans += flow.max_flow(s, t);
```
容量依次为：
```text
0 -> 5 -> 15
```
每次增加容量后都可以直接利用旧残量网络继续增广
这通常比每次从头重新建图更合适
##### 同时逐步增加多条边容量
```cpp
i64 ans = flow.max_flow(s, t);
for (auto [id, delta] : changes) {
    flow.increase_cap(id, delta);
}
ans += flow.max_flow(s, t);
```
如果修改全部属于：
```text
加边
增加容量
```
则已有流永远保持可行，可以一直复用残量网络
##### 直接修改某条边容量
假设某条边当前：
```text
容量 = 10
流量 = 6
```
那么修改为 `8` 是安全的：
```cpp
bool ok = flow.set_cap(id, 8);
```
此时：
```text
容量 = 8
流量 = 6
剩余容量 = 2
```
也可以修改为更大：
```cpp
flow.set_cap(id, 20);
i64 add = flow.max_flow(s, t);
```
因为出现了新的剩余容量，所以可以继续增广
##### 安全降低容量
```cpp
i64 used = flow.edge_flow(id);
if (new_cap >= used) {
    flow.set_cap(id, new_cap);
}
```
若网络此前已经是最大流，并且只是降低容量：
```cpp
flow.set_cap(id, new_cap);
```
只要修改成功，当前最大流值不会变化，也不需要重新求最大流
原因是当前流仍然可行，而容量减小不可能产生比原最大流更大的答案
##### 容量降低到当前流量以下
例如：
```text
当前容量 = 10
当前流量 = 8
新容量 = 5
```
则：
```cpp
flow.set_cap(id, 5);
```
会返回 `false`
因为原来的 `8` 单位流已经无法通过容量为 `5` 的边
若不需要维护动态流，最稳妥的做法是重新开始：
```cpp
flow.reset_flow();
bool ok = flow.set_cap(id, 5);
assert(ok);
i64 ans = flow.max_flow(s, t);
```
若有很多容量需要同时修改：
```cpp
flow.reset_flow();
for (auto [id, cap] : changes) {
    bool ok = flow.set_cap(id, cap);
    assert(ok);
}
i64 ans = flow.max_flow(s, t);
```
重置后所有边流量均为 `0`，因此任意非负容量都可以安全设置
##### 多轮参数修改
如果参数变化只有增加：
```cpp
i64 ans = flow.max_flow(s, t);
for (int round = 0; round < q; ++ round) {
    for (auto [id, delta] : increase[round]) {
        flow.increase_cap(id, delta);
    }
    ans += flow.max_flow(s, t);
    // ans 为这一轮修改后的新最大流
}
```
这是利用残量网络最重要的动态用法之一
##### 每轮从零重新求
若每轮容量可能任意增加或降低，并且降低可能越过当前流量：
```cpp
for (int round = 0; round < q; ++ round) {
    flow.reset_flow();
    for (auto [id, cap] : new_cap[round]) {
        flow.set_cap(id, cap);
    }
    i64 ans = flow.max_flow(s, t);
}
```
##### 暂时关闭一条普通边
若希望真正按普通网络语义把一条边容量改成 `0`：
```cpp
flow.reset_flow();
flow.set_cap(id, 0);
i64 ans = flow.max_flow(s, t);
```
不要直接：
```cpp
flow.disable_edge(id);
```
`disable_edge()` 是给上下界流等特殊残量图建模使用的
##### 容量缩放
```cpp
Dinic<i64> flow(n, m);
for (auto [u, v, cap] : edges) {
    flow.add_edge(u, v, cap);
}
i64 ans = flow.max_flow_scaling(s, t);
```
普通题优先使用 `max_flow()`；容量跨度非常大且普通 Dinic 表现不好时再尝试
##### 二分图容量匹配 / b-Matching
左部点 `u` 最多匹配 `capL[u]` 次，右部点 `v` 最多匹配 `capR[v]` 次，每条允许匹配边 `(u, v)` 至多使用一次
建立网络：
- `S -> L_u`：容量 `capL[u]`
- `L_u -> R_v`：容量 `1`
- `R_v -> T`：容量 `capR[v]`
最大流即最大匹配数
```cpp
int S = nl + nr;
int T = S + 1;
Dinic<i64> flow(T + 1, nl + int(es.size()) + nr);
for (int u = 0; u < nl; ++ u) {
    flow.add_edge(S, u, capL[u]);
}
vector<tuple<int, int, int>> edges;
for (auto [u, v] : es) {
    int id = flow.add_edge(u, nl + v, 1);
    edges.emplace_back(u, v, id);
}
for (int v = 0; v < nr; ++ v) {
    flow.add_edge(nl + v, T, capR[v]);
}
i64 ans = flow.max_flow(S, T);
```
这里不需要拆左右部点，因为它们的容量已经分别由：
```
S -> 左部点
右部点 -> T
```
限制
##### 恢复容量匹配方案
保存所有 `L -> R` 边的编号，最大流结束后检查其流量：
```cpp
vector<vector<int>> ml(nl), mr(nr);
for (auto [u, v, id] : edges) {
    if (flow.edge_flow(id) == 1) {
        ml[u].push_back(v);
        mr[v].push_back(u);
    }
}
```
其中：
- `ml[u]`：左部点 `u` 最终匹配的所有右部点
- `mr[v]`：右部点 `v` 最终匹配的所有左部点
若左右两侧容量均为 `1`，就退化为普通二分图最大匹配

> [!tip]  
> 若匹配边 `(u, v)` 还带费用或收益，容量模型不变，  
> 将 `L_u -> R_v` 改成容量 `1`、对应费用的边，并使用最小费用流
##### 一般图的点容量限制
若一般流网络中点 `u` 最多允许 `cap[u]` 单位流经过，将一个点拆成：
```
u_in -> u_out
```
并令这条边容量为 `cap[u]`
原来的边：
```
u -> v
```
改为：
```
u_out -> v_in
```
例如原边容量均为 `1`：
```cpp
auto in = [&](int u) { return 2 * u; };
auto out = [&](int u) { return 2 * u + 1; };
Dinic<i64> flow(2 * n);
for (int u = 0; u < n; ++ u) {
    flow.add_edge(in(u), out(u), cap[u]);
}
for (auto [u, v] : es) {
    flow.add_edge(out(u), in(v), 1);
}
```
若源点、汇点同样受点容量限制：
```cpp
i64 ans = flow.max_flow(in(s), out(t));
```
若源点或汇点不受点容量限制，可以将对应拆点边容量设为足够大的 `INF`，或根据建图直接从 `out(s)` / 流入 `in(t)`
##### 只恢复有流量的原图边
若原问题的方案就是“哪些原图边最终被使用”，建图时保存这些边：
```cpp
struct Info {
    int u, v, id;
};
vector<Info> edges;
for (auto [u, v, cap] : es) {
    int id = flow.add_edge(u, v, cap);
    edges.push_back({u, v, id});
}
flow.max_flow(s, t);
for (auto [u, v, id] : edges) {
    i64 f = flow.edge_flow(id);
    if (f > 0) {
        // 原图边 u -> v 使用了 f 单位流
    }
}
```
不应直接根据残量 `e[2 * id].cap == 0` 判断边是否被使用：  
容量大于 `1` 时，一条边即使还有剩余容量，也可能已经承载了流量。  
统一使用 `edge_flow(id)` 更稳妥
#### 复杂度
设点数为 $n$、逻辑边数为 $m$

| 操作                             | 时间复杂度         |
| ------------------------------ | ------------- |
| `add_edge()`                   | 均摊 $O(1)$     |
| `increase_cap()` / `set_cap()` | $O(1)$        |
| `reset_flow()`                 | $O(m)$        |
| `get_edge()` / `edge_flow()`   | $O(1)$        |
| `min_cut()`                    | $O(n+m)$      |
| `max_flow()`                   | 一般图 $O(n^2m)$ |
| 单位网络 `max_flow()`              | $O(m\sqrt n)$ |
空间复杂度：$O(n+m)$
`max_flow_scaling()` 至多经过 $O(\log U)$ 个容量尺度，其中 $U$ 为当前最大残量容量；可保守视为在 Dinic 上额外增加一个 $O(\log U)$ 的尺度因子，主要作为实践优化使用
