### 6.1.3 Weighted Blossom 一般图最大权匹配
---
```cpp
template<class T = i64>
struct WeightedBlossom {
    static_assert(numeric_limits<T>::is_integer && numeric_limits<T>::is_signed);
    struct Edge { int u{}, v{}; T w{}; };
    int n, m{}, id{};
    vector<tuple<int, int, T>> edges;
    vector<vector<Edge>> e;
    vector<vector<int>> b, flower;
    vector<T> lab;
    vector<int> lk, sl, st, f, s, ed, q;
    T slack(const Edge &x) const { return lab[x.u] + lab[x.v] - x.w * 2; }
    void update_slack(int u, int v) { if (!sl[v] || slack(e[u][v]) < slack(e[sl[v]][v])) sl[v] = u; }
    void set_slack(int v) {
        sl[v] = 0;
        for (int u = 1; u <= n; ++ u) if (e[u][v].w > 0 && st[u] != v && s[st[u]] == 0) update_slack(u, v);
    }
    void push(int u) {
        if (u <= n) q.push_back(u);
        else for (int v : flower[u]) push(v);
    }
    void set_st(int u, int v) {
        st[u] = v;
        if (u > n) for (int x : flower[u]) set_st(x, v);
    }
    int get_pos(int u, int v) {
        int p{ int(find(flower[u].begin(), flower[u].end(), v) - flower[u].begin()) };
        if (p & 1) {
            reverse(flower[u].begin() + 1, flower[u].end());
            return int(flower[u].size()) - p;
        }
        return p;
    }
    // ===== 匹配与缩花：核心功能，不建议删除 =====
    void set_match(int u, int v) {
        lk[u] = e[u][v].v;
        if (u <= n) return;
        Edge w{ e[u][v] };
        int x{ b[u][w.u] }, y{ get_pos(u, x) };
        for (int i = 0; i < y; ++ i) set_match(flower[u][i], flower[u][i ^ 1]);
        set_match(x, v);
        rotate(flower[u].begin(), flower[u].begin() + y, flower[u].end());
    }
    void augment(int u, int v) {
        int w{ st[lk[u]] };
        set_match(u, v);
        if (!w) return;
        set_match(w, st[f[w]]);
        augment(st[f[w]], w);
    }
    int lca(int u, int v) {
        for (++ id; u || v; swap(u, v)) {
            if (!u) continue;
            if (ed[u] == id) return u;
            ed[u] = id;
            u = st[lk[u]];
            if (u) u = st[f[u]];
        }
        return 0;
    }
    void add_blossom(int u, int a, int v) {
        int x{ n + 1 }, i, j;
        while (x <= m && st[x]) ++ x;
        if (x > m) ++ m;
        lab[x] = T{};
        s[x] = st[x] = 0;
        lk[x] = lk[a];
        flower[x].clear();
        flower[x].push_back(a);
        for (i = u; i != a; i = st[f[j]]) {
            flower[x].push_back(i);
            flower[x].push_back(j = st[lk[i]]);
            push(j);
        }
        reverse(flower[x].begin() + 1, flower[x].end());
        for (i = v; i != a; i = st[f[j]]) {
            flower[x].push_back(i);
            flower[x].push_back(j = st[lk[i]]);
            push(j);
        }
        set_st(x, x);
        for (i = 1; i <= m; ++ i) e[x][i].w = e[i][x].w = T{};
        fill(b[x].begin() + 1, b[x].begin() + n + 1, 0);
        for (int u0 : flower[x]) {
            for (int v0 = 1; v0 <= m; ++ v0) if (!e[x][v0].w || slack(e[u0][v0]) < slack(e[x][v0])) {
                e[x][v0] = e[u0][v0];
                e[v0][x] = e[v0][u0];
            }
            for (int v0 = 1; v0 <= n; ++ v0) if (b[u0][v0]) b[x][v0] = u0;
        }
        set_slack(x);
    }
    void expand_blossom(int u) {
        for (int x : flower[u]) set_st(x, x);
        int a{ b[u][e[u][f[u]].u] }, r{ get_pos(u, a) };
        for (int i = 0; i < r; i += 2) {
            int x{ flower[u][i] }, y{ flower[u][i + 1] };
            f[x] = e[y][x].u;
            s[x] = 1;
            s[y] = sl[x] = 0;
            set_slack(y);
            push(y);
        }
        s[a] = 1;
        f[a] = f[u];
        for (int i = r + 1; i < int(flower[u].size()); ++ i) {
            s[flower[u][i]] = -1;
            set_slack(flower[u][i]);
        }
        st[u] = 0;
    }
    bool on_edge(const Edge &x) {
        int u{ st[x.u] }, v{ st[x.v] };
        if (s[v] == -1) {
            f[v] = x.u;
            s[v] = 1;
            int a{ st[lk[v]] };
            sl[v] = sl[a] = s[a] = 0;
            push(a);
        } else if (s[v] == 0) {
            int a{ lca(u, v) };
            if (!a) {
                augment(u, v);
                augment(v, u);
                return true;
            }
            add_blossom(u, a, v);
        }
        return false;
    }
    // ===== 原始对偶增广：维护标号与 slack =====
    bool bfs() {
        fill(s.begin() + 1, s.begin() + m + 1, -1);
        fill(sl.begin() + 1, sl.begin() + m + 1, 0);
        q.clear();
        size_t h{};
        for (int i = 1; i <= m; ++ i) if (st[i] == i && !lk[i]) {
            f[i] = s[i] = 0;
            push(i);
        }
        if (q.empty()) return false;
        while (true) {
            while (h < q.size()) {
                int u{ q[h ++] };
                if (s[st[u]] == 1) continue;
                for (int v = 1; v <= n; ++ v) if (e[u][v].w > 0 && st[u] != st[v]) {
                    if (slack(e[u][v]) != 0) update_slack(u, st[v]);
                    else if (on_edge(e[u][v])) return true;
                }
            }
            T x{ numeric_limits<T>::max() / 4 };
            for (int i = n + 1; i <= m; ++ i) if (st[i] == i && s[i] == 1) x = min(x, lab[i] >> 1);
            for (int i = 1; i <= m; ++ i) if (st[i] == i && sl[i] && s[i] != 1) {
                x = min(x, slack(e[sl[i]][i]) >> (s[i] + 1));
            }
            for (int i = 1; i <= n; ++ i) if (s[st[i]] != -1) {
                lab[i] += T(s[st[i]] * 2 - 1) * x;
                if (lab[i] <= 0) return false;
            }
            for (int i = n + 1; i <= m; ++ i) if (st[i] == i && s[st[i]] != -1) {
                lab[i] += T(2 - s[st[i]] * 4) * x;
            }
            q.clear();
            h = 0;
            for (int i = 1; i <= m; ++ i) if (st[i] == i && sl[i] && st[sl[i]] != i && slack(e[sl[i]][i]) == 0) {
                if (on_edge(e[sl[i]][i])) return true;
            }
            for (int i = n + 1; i <= m; ++ i) if (st[i] == i && s[i] == 1 && lab[i] == 0) expand_blossom(i);
        }
    }

public:
    explicit WeightedBlossom(int n_) : n(n_) {}
    void add_edge(int u, int v, T w) { if (u != v) edges.emplace_back(u + 1, v + 1, w); }
    pair<T, vector<int>> max_weight_matching() {
        int cap{ 2 * n + 2 };
        e.assign(cap, vector<Edge>(cap));
        b.assign(cap, vector<int>(n + 1));
        flower.assign(cap, {});
        lab.assign(cap, T{});
        lk.assign(cap, 0);
        sl.assign(cap, 0);
        st.assign(cap, 0);
        f.assign(cap, 0);
        s.assign(cap, 0);
        ed.assign(cap, 0);
        q.clear();
        m = n;
        id = 0;
        for (int i = 1; i <= n; ++ i) st[i] = i;
        for (int i = 1; i <= n; ++ i) for (int j = 1; j <= n; ++ j) e[i][j] = { i, j, T{} };
        T wm{};
        for (auto [u, v, w] : edges) if (w > e[u][v].w) {
            e[u][v].w = e[v][u].w = w;
            wm = max(wm, w);
        }
        for (int i = 1; i <= n; ++ i) b[i][i] = i;
        fill(lab.begin() + 1, lab.begin() + n + 1, wm);
        while (bfs()) {}
        T value{};
        vector<int> match(n, -1);
        for (int i = 1; i <= n; ++ i) if (lk[i]) {
            if (i < lk[i]) value += e[i][lk[i]].w;
            match[i - 1] = lk[i] - 1;
        }
        return { value, match };
    }
};
```
#### 接口
- `WeightedBlossom<T>(n)`：建立 `n` 点一般无向图
- `add_edge(u, v, w)`：加入权值为 `w` 的无向边，自环忽略
- `max_weight_matching()`：返回 `{最大权值和, match}`，其中 `match[u]` 为 `u` 的匹配点，未匹配为 `-1`
- 平行边只保留同一对端点中的最大权值
- 外部点编号为 `0base`
#### 注意事项
- 求的是**允许部分点不匹配时的最大权值匹配**，不是最大基数匹配，也不强制完美匹配
- 内部只处理正权边，权值为 `0` 或负数的边不会被选；在允许空匹配时这不影响最大权值，但不会为了相同权值额外增大匹配边数
- 若需要“先最大化匹配边数，再最大化权值”，可给每条合法边增加一个足够大的统一常数
- 若已知 `|w| <= W`，可取 `C > nW`，将边权改为 `w + C`，此时先最大化匹配边数，同基数下再最大化原权值
- 若要求最大权完美匹配，可使用上述变换并检查所有点是否均被匹配；若存在未匹配点则不存在完美匹配
- 最小权完美匹配可将权值变为 `C - cost`，其中 `C` 需同时保证基数优先且中间值不溢出
- `T` 必须是有符号整数类型，且需要容纳 `2w`、标号、slack 和答案和；权值较大时使用 `i128`
- 重复调用 `max_weight_matching()` 会从保存的边重新求解，因此可以继续 `add_edge()` 后再次调用
- 一般二分图最大权指派优先使用 Hungarian / KM，带权 Blossom 只用于一般图
- 该实现使用稠密矩阵维护缩花间的最优边，空间为 $O(n^2)$
#### 常见用法
##### 最大权匹配
```cpp
WeightedBlossom<i64> bm(n);
for (auto [u, v, w] : edges) bm.add_edge(u, v, w);
auto [ans, match] = bm.max_weight_matching();
for (int u = 0; u < n; ++ u) if (match[u] != -1 && u < match[u]) {
    int v{ match[u] };
    // 匹配边 (u, v)
}
```
##### 最大权完美匹配
若 `|w| <= W`，先加入足够大的基数奖励
```cpp
i128 C = i128(n) * W + 1;
WeightedBlossom<i128> bm(n);
for (auto [u, v, w] : edges) bm.add_edge(u, v, i128(w) + C);
auto [value, match] = bm.max_weight_matching();
bool ok{ true };
for (int u = 0; u < n; ++ u) {
    if (match[u] == -1) ok = false;
}
if (ok) value -= C * (n / 2);
```
##### 最大基数优先，再最大化权值
同样使用 `w + C`
```cpp
i128 C = i128(n) * W + 1;
WeightedBlossom<i128> bm(n);
for (auto [u, v, w] : edges) bm.add_edge(u, v, i128(w) + C);
auto [value, match] = bm.max_weight_matching();
int cnt{};
for (int u = 0; u < n; ++ u) cnt += match[u] != -1;
cnt /= 2;
i128 original_value = value - C * cnt;
```
#### 复杂度
时间复杂度：$O(n^3)$
空间复杂度：$O(n^2)$
