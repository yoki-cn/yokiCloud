### 最小费用流
---
```cpp
template<class Cap = i64, class Cost = i64>
struct MinCostFlow {
    static_assert(numeric_limits<Cap>::is_integer && numeric_limits<Cap>::is_signed);
    static_assert(numeric_limits<Cost>::is_integer && numeric_limits<Cost>::is_signed);
    struct Edge {
        int from, to;
        Cap cap, init;
        Cost cost;
    };
    int n, sc{ 8 };
    bool upd{};
    vector<Edge> e;
    vector<vector<int>> g;
    explicit MinCostFlow(int n_) : n(n_), g(n) {}
    // ===== 最大流 + 整数费用缩放 =====
    int add_edge(int u, int v, Cap cap, Cost cost) {
        int id{ int(e.size()) / 2 };
        g[u].push_back(e.size());
        e.push_back({ u, v, cap, cap, cost });
        g[v].push_back(e.size());
        e.push_back({ v, u, 0, 0, -cost });
        return id;
    }
    pair<Cap, Cost> flow(int s, int t, Cap limit = numeric_limits<Cap>::max()) {
        if (s == t || limit == 0) return { 0, 0 };
        using T = conditional_t<(sizeof(Cost) < sizeof(i128)), i128, Cost>;
        vector<int> dep(n), cur(n);
        Cap f{};
        T c{};
        for (int i = 0; i < int(e.size()); i += 2) c -= T(e[i].init - e[i].cap) * e[i].cost;
        auto dfs = [&](auto &&self, int u, Cap lim) -> Cap {
            if (u == t) return lim;
            for (int &i = cur[u]; i < int(g[u].size()); ++ i) {
                int id{ g[u][i] };
                Edge &x{ e[id] };
                if (x.cap == 0 || dep[x.to] != dep[u] + 1) continue;
                Cap z{ self(self, x.to, min(lim, x.cap)) };
                if (z == 0) continue;
                x.cap -= z;
                e[id ^ 1].cap += z;
                return z;
            }
            return 0;
        };
        while (f < limit) {
            ranges::fill(dep, -1);
            queue<int> q;
            dep[s] = 0;
            q.push(s);
            while (!q.empty()) {
                int u{ q.front() };
                q.pop();
                for (int id : g[u]) if (e[id].cap > 0 && dep[e[id].to] == -1) {
                    dep[e[id].to] = dep[u] + 1;
                    q.push(e[id].to);
                }
            }
            if (dep[t] == -1) break;
            ranges::fill(cur, 0);
            while (f < limit) {
                Cap z{ dfs(dfs, s, limit - f) };
                if (z == 0) break;
                f += z;
            }
        }
        cancel_negative_cycles();
        for (int i = 0; i < int(e.size()); i += 2) c += T(e[i].init - e[i].cap) * e[i].cost;
        assert(c >= numeric_limits<Cost>::min() && c <= numeric_limits<Cost>::max());
        return { f, Cost(c) };
    }
    // ===== 方案恢复：若不需要逐边流量，以下函数可直接删除 =====
    tuple<int, int, Cap, Cap, Cost> get_edge(int id) const {
        const Edge &x{ e[2 * id] };
        return { x.from, x.to, x.init, x.init - x.cap, x.cost };
    }
    // ===== 流量重设与费用优化 =====
    void change_edge(int id, Cap cap, Cap used) {
        Edge &x{ e[2 * id] }, &r{ e[2 * id + 1] };
        x.init = cap;
        x.cap = cap - used;
        r.init = 0;
        r.cap = used;
    }
    // 保持各点净流量，用整数费用缩放消除残量负环
    Cost cancel_negative_cycles() {
        using T = conditional_t<(sizeof(Cost) < sizeof(i128)), i128, Cost>;
        using F = conditional_t<(sizeof(Cap) < sizeof(i128)), i128, Cap>;
        assert(sc >= 2);
        vector<T> p(n);
        vector<F> b(n);
        vector<int> cur(n);
        T eps{}, ans{};
        for (int i = 0; i < int(e.size()); i += 2) {
            ans -= T(e[i].init - e[i].cap) * e[i].cost;
            T c{ T(e[i].cost) * (n + 1) };
            eps = max(eps, c < 0 ? -c : c);
        }
        auto rc = [&](int id) { return T(e[id].cost) * (n + 1) + p[e[id].from] - p[e[id].to]; };
        auto push = [&](int id, Cap z) {
            e[id].cap -= z;
            e[id ^ 1].cap += z;
            if (e[id].from != e[id].to) b[e[id].from] -= z, b[e[id].to] += z;
        };
        while (eps > 0) {
            eps = max(T(1), eps / sc);
            for (int id = 0; id < int(e.size()); ++ id) if (e[id].cap > 0 && rc(id) < 0) push(id, e[id].cap);
            // 可选价格更新：反向多源最短路，每轮执行一次
            if (upd) {
                constexpr T INF{ numeric_limits<T>::max() };
                vector<T> d(n, INF);
                priority_queue<pair<T, int>, vector<pair<T, int>>, greater<pair<T, int>>> q;
                for (int u = 0; u < n; ++ u) if (b[u] < 0) d[u] = 0, q.push({ 0, u });
                T mx{};
                while (!q.empty()) {
                    auto [du, u] = q.top();
                    q.pop();
                    if (du != d[u]) continue;
                    mx = du;
                    for (int id : g[u]) if (e[id ^ 1].cap > 0) {
                        int v{ e[id].to };
                        T nd{ du + rc(id ^ 1) + eps };
                        if (nd < d[v]) d[v] = nd, q.push({ nd, v });
                    }
                }
                for (int u = 0; u < n; ++ u) p[u] -= min(d[u], mx);
            }
            ranges::fill(cur, 0);
            queue<int> q;
            for (int u = 0; u < n; ++ u) if (b[u] > 0) q.push(u);
            while (!q.empty()) {
                int u{ q.front() };
                q.pop();
                while (b[u] > 0) {
                    if (cur[u] == int(g[u].size())) {
                        T d{ numeric_limits<T>::max() };
                        for (int id : g[u]) if (e[id].cap > 0 && e[id].to != u) d = min(d, rc(id));
                        assert(d != numeric_limits<T>::max());
                        p[u] -= d + eps;
                        cur[u] = 0;
                    }
                    int id{ g[u][cur[u]] }, v{ e[id].to };
                    if (e[id].cap == 0 || rc(id) >= 0) { ++ cur[u]; continue; }
                    Cap z{ Cap(min(b[u], F(e[id].cap))) };
                    F old{ b[v] };
                    push(id, z);
                    if (v != u && old <= 0 && b[v] > 0) q.push(v);
                }
            }
            if (eps == 1) break;
        }
        for (int i = 0; i < int(e.size()); i += 2) ans += T(e[i].init - e[i].cap) * e[i].cost;
        assert(ans >= numeric_limits<Cost>::min() && ans <= numeric_limits<Cost>::max());
        return Cost(ans);
    }
};
```
#### 成员变量
- `e[2 * id] / e[2 * id + 1]`：第 `id` 条逻辑边的正向 / 反向残量边
- `Edge::cap`：当前残量容量
- `Edge::init`：原边容量，反向残量边为 `0`
- `Edge::cost`：单位流量费用
#### 接口
- `MinCostFlow(n)`：建立 `n` 点费用残量网络，$O(n)$
- `add_edge(u,v,cap,cost)`：加入容量 `cap`、单位费用 `cost` 的有向边，返回逻辑边编号，均摊 $O(1)$
- `flow(s,t)`：在当前残量状态上继续增广至最大流，返回 `{本次新增流量, 本次新增最小费用}`
- `flow(s,t,limit)`：最多继续增广 `limit` 单位流
- `get_edge(id)`：返回 `{from,to,capacity,flow,cost}`，$O(1)$
- `change_edge(id,cap,used)`：将逻辑边重设为总容量 `cap`、当前流量 `used`，$O(1)$
- `cancel_negative_cycles()`：保持每个点净流量不变并消除残量负环，返回费用改变量，返回值非正
#### 注意事项
- 点编号为 `0base`，要求 `0 <= s,t,u,v < n`
- `add_edge()` 要求 `cap >= 0`，且 `cost` 与 `-cost` 都能由 `Cost` 表示
- `flow()` 要求 `limit >= 0`
- `flow(s,t,limit)` 表示**至多发送**，要求恰好 `limit` 时必须检查返回流量
- 允许负费用边和任意位置的负费用环；先用 Dinic 确定新增流量，再用整数费用缩放优化整个残量网络
- `flow()` 返回的费用是本次调用前后总费用之差，包含独立负环的优化；不必预先调用 `cancel_negative_cycles()`
- `s == t` 或 `limit == 0` 时原样返回 `{0,0}`，不优化费用；只优化当前流请调用 `cancel_negative_cycles()`
- `flow()` 会修改残量网络，再次调用表示在当前流基础上继续增广，返回值仍只统计本次增量
- 最大费用流可将所有费用取负，最后将答案费用取反
- `change_edge()` 要求 `0 <= used <= cap`，且调用者必须自行保证修改后整张网络仍满足所需流量守恒
- `cancel_negative_cycles()` 要求当前流满足需要保持的点平衡 / 净流量条件，消圈不会改变各点净流量
- `Cap` 容纳容量和返回流量；内部点盈余至少用 `i128`，必须容纳所有入边容量之和
- 内部费用类型 `T` 至少为 `i128`，必须容纳 `(n+1)*cost`、势能、约化费用、逐边费用乘积及累加中间值；不是任意 `i128` 输入都安全
- `Cost` 必须容纳返回费用差及费用优化阶段的费用差；总费用较大或优化前后差值较大时使用 `i128`
- Dinic 的递归深度可能达到 `n`，注意运行环境栈容量
#### 整数缩放的正确性
每轮先饱和负约化费用边，再推送正盈余并调整势能，最终恢复每个点调用前的净流量。所有残量边的约化费用至少为 `-eps`。
费用先乘 `n+1`；当 `eps == 1` 时，长度至多 `n` 的简单环费用不小于 `-n`，但它必须是 `n+1` 的整数倍，因此不可能为负。没有残量负环即为当前净流量下的最小费用。这是整数精确算法，不使用浮点近似。
费用缩放与 epsilon 最优性的依据见 [Goldberg、Tardos、Tarjan：Network Flow Algorithms](https://www.cs.cornell.edu/~eva/Network.Flow.Algorithms.pdf)。
#### 常见用法
##### 恰好发送指定流量
```cpp
MinCostFlow<i64, i64> mcf(n);
for (auto [u, v, cap, cost] : edges) mcf.add_edge(u, v, cap, cost);
auto [flow, cost] = mcf.flow(s, t, need);
if (flow != need) cout << "Impossible\n";
else cout << cost << '\n';
```
##### 恢复边流量
```cpp
int id{ mcf.add_edge(u, v, cap, cost) };
mcf.flow(s, t);
auto [from, to, capacity, used, unit_cost] = mcf.get_edge(id);
```
##### 存在负费用环
```cpp
Cost cost{};
cost += mcf.cancel_negative_cycles();
auto [f, c] = mcf.flow(s, t, need);
cost += c;
```
上面保留了先优化初始环流的写法，也可以直接调用 `flow()`。正常求流后再次优化返回 `0`：
```cpp
cost += mcf.cancel_negative_cycles();
```
#### 复杂度
设网络有 $n$ 个点、$m$ 条逻辑边，$C$ 为最大单位费用绝对值。

| 操作 | 时间复杂度 |
|---|---:|
| Dinic 求流量 | $O(n^2m)$ |
| `cancel_negative_cycles()` | $O(n^2m\log(2+nC))$，整数费用缩放 |
| `flow()` | Dinic 与费用缩放之和 |
| `get_edge()` / `change_edge()` | $O(1)$ |

空间复杂度：$O(n+m)$。这份实现优先保持代码紧凑，价格更新默认关闭；小流量简单网络上未必比逐次最短路快。

#### 常用方法：可选启发式
`upd` 默认 `false`，设为 `true` 后，每轮饱和负边后执行一次价格更新：从亏流点反向跑多源 Dijkstra，批量调整势能，减少逐点重标号。单轮额外开销 $O((n+m)\log(n+m))$、空间 $O(n+m)$；短路径、小图不一定更快，建议用目标数据比较后决定。
`sc` 默认 `8`，要求至少 `2`，常用 `4/8/16`。较大值减少缩放轮数，但可能增加单轮推送和重标号，不保证更快。两项只影响运行过程，不改变整数最优性或接口语义。
```cpp
MinCostFlow<i64, i128> mcf(3);
mcf.upd = true; // 开启每轮价格更新
mcf.sc = 4;     // 可省略，默认 8
mcf.add_edge(0, 1, 2, -3);
mcf.add_edge(1, 2, 2, 1);
auto [f, c] = mcf.flow(0, 2); // f=2，c=-4
assert(f == 2 && c == -4);
```
只优化循环流时同样先设置 `upd/sc`，再调用 `cancel_negative_cycles()`；上下界包装见 6.4.1。

价格更新使用边长 `rc+eps`。调用位置在饱和负边后，残量 `rc>=0`，可用 Dijkstra；距离满足 `d[u]<=rc(u,v)+eps+d[v]`，令 `p[u]-=d[u]` 后仍有 `rc>=-eps`。不可达点统一减去最大有限距离，保持跨越可达集合边的同一不等式；没有亏流点时不调整。路径距离、中间加法须可表示且有限距离严格小于内部 `T::max()`。不要把这段直接搬到可能有 `rc<-eps` 的位置。
启发式背景可参见 [OR-Tools 的费用流价格更新开关与说明](https://github.com/google/or-tools/blob/stable/ortools/graph/min_cost_flow.h)；这里使用紧凑的每轮一次版本。
