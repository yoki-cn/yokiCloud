### 一般图最大匹配（Edmonds Blossom）
---
```cpp
struct Blossom {
    int n;
    vector<vector<int>> g;
    vector<int> match, p, base, q;
    vector<char> used, blossom;
    explicit Blossom(int n_) : n(n_), g(n), match(n, -1), p(n), base(n), q(n), used(n), blossom(n) {}
    void add_edge(int u, int v) { if (u != v) g[u].push_back(v), g[v].push_back(u); }
    // 求交错森林中两个点所在花的公共基点
    int lca(int a, int b) {
        vector<char> seen(n);
        while (true) {
            a = base[a];
            seen[a] = true;
            if (match[a] == -1) break;
            a = p[match[a]];
        }
        while (true) {
            b = base[b];
            if (seen[b]) return b;
            b = p[match[b]];
        }
    }
    void mark_path(int v, int b, int child) {
        while (base[v] != b) {
            blossom[base[v]] = blossom[base[match[v]]] = true;
            p[v] = child;
            child = match[v];
            v = p[match[v]];
        }
    }
    // 从未匹配根出发 BFS；遇到交错奇环时缩花
    int find_path(int root) {
        ranges::fill(used, false);
        ranges::fill(p, -1);
        iota(base.begin(), base.end(), 0);
        int ql{ 0 }, qr{ 0 };
        q[qr ++] = root;
        used[root] = true;
        while (ql < qr) {
            int v{ q[ql ++] };
            for (int u : g[v]) {
                if (base[v] == base[u] || match[v] == u) continue;
                if (u == root || (match[u] != -1 && p[match[u]] != -1)) {
                    int b{ lca(v, u) };
                    ranges::fill(blossom, false);
                    mark_path(v, b, u);
                    mark_path(u, b, v);
                    for (int i = 0; i < n; ++ i) if (blossom[base[i]]) {
                        base[i] = b;
                        if (!used[i]) used[i] = true, q[qr ++] = i;
                    }
                } else if (p[u] == -1) {
                    p[u] = v;
                    if (match[u] == -1) return u;
                    u = match[u];
                    used[u] = true;
                    q[qr ++] = u;
                }
            }
        }
        return -1;
    }
    int max_matching() {
        for (int root = 0; root < n; ++ root) if (match[root] == -1) {
            int v{ find_path(root) };
            if (v == -1) continue;
            while (v != -1) {
                int pv{ p[v] }, nv{ pv == -1 ? -1 : match[pv] };
                match[v] = pv;
                if (pv != -1) match[pv] = v;
                v = nv;
            }
        }
        int ans{};
        for (int v : match) ans += v != -1;
        return ans / 2;
    }
};
```
#### 成员变量
- `g[u]`：点 `u` 的邻接表
- `match[u]`：点 `u` 的匹配点，未匹配为 `-1`
- `base[u]`：当前缩花后 `u` 所属花的基点
- `p[u]`：当前交错森林中的前驱
#### 接口
- `Blossom(n)`：建立 `n` 个点的无向图，$O(n)$
- `add_edge(u, v)`：加入无向边，自环自动忽略，均摊 $O(1)$
- `max_matching()`：从当前匹配继续增广至最大基数匹配，返回当前匹配边数，$O(n^3)$
- 求解后 `match[u] = v` 当且仅当 `match[v] = u`，未匹配点为 `-1`
#### 注意事项
- 点编号为 `0base`，调用接口需保证 `0 <= u, v < n`
- 解决**一般无向图最大基数匹配**，即最大化匹配边数，不处理边权
- 核心区别于二分图匹配：交错森林中遇到奇环时将其缩成一个花，再继续寻找增广路；常见竞赛实现复杂度为 $O(n^3)$
- 平行边不影响正确性但会增加常数，自环直接忽略
- 可以先调用 `max_matching()`，继续 `add_edge()` 后再次调用；原匹配会作为合法初始匹配继续增广
- 重复调用 `max_matching()` 返回的是**当前最大匹配总边数**，不是本次新增的匹配数
- 不要直接修改 `match`，除非能保证它始终表示一个合法匹配
- 二分图优先使用 Hopcroft--Karp，代码更简单且复杂度更优
- 一般图最大权匹配需要带权 Blossom，是另一套明显更复杂的算法；参考模板体系也将无权 Blossom 与一般图最大权匹配分开整理
- 随机贪心或 `shuffle` 匹配只能用于启发式加速，不能代替 Blossom 保证答案正确
#### 常见用法
```cpp
Blossom bm(n);
for (auto [u, v] : edges) bm.add_edge(u, v);

int matching{ bm.max_matching() };
for (int u = 0; u < n; ++ u) if (bm.match[u] != -1 && u < bm.match[u]) {
    int v{ bm.match[u] };
    // 匹配边 (u, v)
}
```
##### 判断是否存在完美匹配
```cpp
bool perfect{ n % 2 == 0 && bm.max_matching() * 2 == n };
```
##### 动态加入边后继续增广
```cpp
int ans{ bm.max_matching() };

bm.add_edge(u, v);
ans = bm.max_matching();
```
无需清空已有 `match`
#### 复杂度
时间复杂度：$O(n^3)$
空间复杂度：$O(n+m)$
