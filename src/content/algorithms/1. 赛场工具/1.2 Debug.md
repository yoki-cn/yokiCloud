### Debug
---
#### 调试输出
---
```cpp
using i128 = __int128_t;
using u128 = __uint128_t;

#ifdef LOCAL
struct Debug {
    // ===== 运行上下文 =====
    static inline int tc{ -1 };
    static inline uint64_t seed{};
    static inline bool has_seed{};
    static void set_seed(uint64_t x) { seed = x; has_seed = true; }
private:
    template<class T, class = void> struct IsRange : false_type {};
    template<class T> struct IsRange<T, void_t<
        decltype(begin(declval<const T &>())),
        decltype(end(declval<const T &>()))>> : true_type {};
    template<class T> struct IsPair : false_type {};
    template<class A, class B> struct IsPair<pair<A, B>> : true_type {};
    template<class T> struct IsTuple : false_type {};
    template<class... T> struct IsTuple<tuple<T...>> : true_type {};
    template<class T> static constexpr bool is_cstr =
        is_same_v<decay_t<T>, const char *> || is_same_v<decay_t<T>, char *>;
    static void print128(u128 x) {
        char s[40];
        int n{};
        do { s[n ++] = char('0' + x % 10); x /= 10; } while (x);
        while (n) cerr << s[-- n];
    }
    template<class... T> static void list(const T &...x) {
        bool fst = true;
        ((cerr << (exchange(fst, false) ? "" : ", "), show(x)), ...);
    }
    static string trim(const string &s) {
        size_t l{}, r = s.size();
        while (l < r && isspace((unsigned char)s[l])) ++ l;
        while (r > l && isspace((unsigned char)s[r - 1])) -- r;
        return s.substr(l, r - l);
    }
    static bool quoted(const string &s) {
        return s.size() >= 2 &&
            ((s.front() == '"' && s.back() == '"') ||
             (s.front() == '\'' && s.back() == '\''));
    }
    static vector<string> split(const string &s) {
        vector<string> a;
        string cur;
        int p{}, b{}, k{};
        char q{};
        bool esc{};
        for (char c : s) {
            if (q) {
                cur += c;
                if (esc) esc = false;
                else if (c == '\\') esc = true;
                else if (c == q) q = 0;
            } else if (c == '\'' || c == '"') q = c, cur += c;
            else if (c == '(') ++ p, cur += c;
            else if (c == ')') -- p, cur += c;
            else if (c == '{') ++ b, cur += c;
            else if (c == '}') -- b, cur += c;
            else if (c == '[') ++ k, cur += c;
            else if (c == ']') -- k, cur += c;
            else if (c == ',' && !p && !b && !k) {
                a.push_back(trim(cur));
                cur.clear();
            } else cur += c;
        }
        if (!cur.empty() || a.empty()) a.push_back(trim(cur));
        return a;
    }
    static void named_impl(const vector<string> &, size_t) { cerr << '\n'; }
    template<class T, class... U>
    static void named_impl(const vector<string> &a, size_t i, const T &x, const U &...rest) {
        if (i >= a.size()) {
            show(x);
            ((cerr << ' ', show(rest)), ...);
            cerr << '\n';
            return;
        }
        if constexpr (is_cstr<T>) {
            if (quoted(a[i])) {
                show(x);
                cerr << ": ";
                named_impl(a, i + 1, rest...);
                return;
            }
        }
        cerr << a[i] << " = ";
        show(x);
        if constexpr (sizeof...(rest) == 0) cerr << '\n';
        else {
            cerr << ' ';
            named_impl(a, i + 1, rest...);
        }
    }
    // ===== 位置与运行上下文：不需要 tc/seed 时删除对应两行 =====
    static void head(const char *tag, const char *file, int line, const char *fn) {
        cerr << '[' << tag << ' ' << file << ':' << line << ' ' << fn << ']';
        if (tc >= 0) cerr << " tc=" << tc;
        if (has_seed) cerr << " seed=" << seed;
        cerr << ' ';
    }
public:
    // ===== 基础输出 =====
    template<class T> static void show(const T &x) {
        using U = decay_t<T>;
        if constexpr (is_same_v<U, bool>) cerr << (x ? "true" : "false");
        else if constexpr (is_same_v<U, char> || is_same_v<U, string> ||
                           is_same_v<U, string_view>) cerr << x;
        else if constexpr (is_cstr<T>) {
            if constexpr (is_array_v<remove_reference_t<T>>) cerr << x;
            else cerr << (x ? x : "null");
        } else if constexpr (is_same_v<U, i128>) {
            if (x < 0) cerr << '-', print128(u128(-(x + 1)) + 1);
            else print128(u128(x));
        } else if constexpr (is_same_v<U, u128>) print128(x);
        else if constexpr (IsPair<U>::value) {
            cerr << '('; show(x.first); cerr << ", "; show(x.second); cerr << ')';
        } else if constexpr (IsTuple<U>::value) {
            cerr << '(';
            apply([](const auto &...v) { list(v...); }, x);
            cerr << ')';
        } else if constexpr (IsRange<T>::value) {
            cerr << '[';
            bool fst = true;
            for (const auto &v : x) {
                if (!fst) cerr << ", ";
                fst = false;
                show(v);
            }
            cerr << ']';
        } else cerr << x;
    }
    template<class... T>
    static void values(const char *file, int line, const char *fn, const T &...x) {
        head("DBG", file, line, fn);
        bool fst = true;
        ((cerr << (exchange(fst, false) ? "" : " "), show(x)), ...);
        cerr << '\n';
    }
    template<class... T>
    static void named(const char *file, int line, const char *fn, const string &s, const T &...x) {
        head("DBG", file, line, fn);
        named_impl(split(s), 0, x...);
    }
    // ===== 失败检查：不需要时可连同 dcheck / dcheckv 宏删除 =====
    [[noreturn]] static void fail(const char *e, const char *file, int line, const char *fn) {
        head("CHECK", file, line, fn);
        cerr << "failed: " << e << '\n';
        cerr.flush();
        abort();
    }
    template<class... T>
    [[noreturn]] static void fail(const char *e, const char *file, int line,
                                  const char *fn, const string &s, const T &...x) {
        head("CHECK", file, line, fn);
        cerr << "failed: " << e << '\n';
        head("VARS", file, line, fn);
        named_impl(split(s), 0, x...);
        cerr.flush();
        abort();
    }
};
#define dbg(...) (Debug::values(__FILE__, __LINE__, __func__, __VA_ARGS__))
#define dbgv(...) (Debug::named(__FILE__, __LINE__, __func__, #__VA_ARGS__, __VA_ARGS__))
#define dcheck(e) do { if (!(e)) Debug::fail(#e, __FILE__, __LINE__, __func__); } while (false)
#define dcheckv(e, ...) do { if (!(e)) Debug::fail(#e, __FILE__, __LINE__, __func__, #__VA_ARGS__, __VA_ARGS__); } while (false)
#else
#define dbg(...) ((void)0)
#define dbgv(...) ((void)0)
#define dcheck(e) ((void)0)
#define dcheckv(e, ...) ((void)0)
#endif
```
- `dbg(x...)`：输出若干变量的值，自动附带文件、行号和函数名
- `dbgv(x...)`：输出变量名与值，保留原来的字符串标签用法
- `Debug::tc`：当前测试编号，默认 `-1` 表示不显示
- `Debug::set_seed(x)`：设置随机种子，之后的调试信息自动附带 `seed`
- `dcheck(cond)`：条件不成立时打印位置与表达式，随后调用 `abort()`
- `dcheckv(cond, x...)`：与 `dcheck` 相同，额外输出指定变量的名称和值
##### 自动输出代码位置
```cpp
int n = 10;
vector<int> a{1, 2, 3};

dbg(n, a);
dbgv("Before DFS", n, a);
```
输出示意：
```
[DBG main.cpp:42 solve] 10 [1, 2, 3]
[DBG main.cpp:43 solve] Before DFS: n = 10 a = [1, 2, 3]
```
##### 检查可能导致 RE 的条件
```cpp
dcheck(0 <= p && p < int(a.size()));
dcheckv(0 <= p && p < int(a.size()), p, a.size(), a);
```
第二个检查失败时，会输出类似：
```
[CHECK main.cpp:51 solve] failed: 0 <= p && p < int(a.size())
[VARS main.cpp:51 solve] p = 7 a.size() = 3 a = [1, 2, 3]
```
适合在数组下标、除数、递归入口、图节点编号等高风险位置使用
##### 注意事项
- `LOCAL` 未定义时，`dbg`、`dbgv`、`dcheck`、`dcheckv` 均不会求值参数，因此不要在这些宏里执行必需的修改操作
- `dcheck` 和 `dcheckv` 是调试断言，不应代替正式程序中的合法性判断
- `abort()`、段错误等异常终止通常不会执行 `RuntimeInfo` 和 `ScopeTimer` 的析构函数，因此不保证输出最终时间
- `dbgv` 的变量名拆分支持常见括号、数组及初始化列表，但直接含有模板参数逗号的复杂表达式，必要时用括号包裹
- `Debug::tc` 和 `seed` 只记录主动设置的上下文，无法自动知道任意崩溃时全部局部变量
#### 运行时间
---
```cpp
#ifdef LOCAL
struct RuntimeInfo {
    using Clock = chrono::steady_clock;
    Clock::time_point st{ Clock::now() };
    i64 cnt{}, id{};
    double sum{}, mx{};
    static double ms(Clock::time_point t) {
        return chrono::duration<double, milli>(Clock::now() - t).count();
    }
    // ===== 多测计时：不需要时可删除 run() 与 cnt/id/sum/mx =====
    template<class F> void run(F &&f) {
        auto t = Clock::now();
        forward<F>(f)();
        double d = ms(t);
        if (!cnt || d > mx) mx = d, id = cnt + 1;
        sum += d;
        ++ cnt;
    }
    ~RuntimeInfo() {
        auto flags = cerr.flags();
        auto prec = cerr.precision();
        cerr << fixed << setprecision(3) << "[Time] total=" << ms(st) << " ms";
        if (cnt) {
            cerr << " solve=" << sum << " ms avg=" << sum / cnt
                 << " ms max=" << mx << " ms (#" << id << ")";
        }
        cerr << '\n';
        cerr.flags(flags);
        cerr.precision(prec);
    }
};
inline RuntimeInfo runtime_info;

// ===== 局部计时 =====
struct ScopeTimer {
    const char *s;
    chrono::steady_clock::time_point st{ chrono::steady_clock::now() };
    explicit ScopeTimer(const char *s_ = "scope") : s(s_) {}
    ~ScopeTimer() {
        auto flags = cerr.flags();
        auto prec = cerr.precision();
        cerr << fixed << setprecision(3) << "[Time] " << s << '='
             << chrono::duration<double, milli>(chrono::steady_clock::now() - st).count() << " ms\n";
        cerr.flags(flags);
        cerr.precision(prec);
    }
};
#endif
```
如果删除 `run()` 和相关统计成员，还需同步删除析构函数里的 `if (cnt)` 统计输出段；如果完全不需要 `RuntimeInfo`，则可连同 `ScopeTimer` 一起删除
- `runtime_info`：全局计时对象，正常结束时自动输出总耗时
- `runtime_info.run(f)`：执行函数 `f`，记录该次耗时
- `ScopeTimer(s)`：创建局部计时对象，离开作用域时输出耗时
- `cnt`：已计时次数
- `sum`：所有 `run()` 的累计耗时
- `mx`：最慢一次 `run()` 的耗时
- `id`：最慢一次的编号，从 `1` 开始
##### 多测计时与 RE 上下文
```cpp
int t{ 1 };
// cin >> t;
for (int i = 1; i <= t; ++ i) {
#ifdef LOCAL
    runtime_info.run(solve);
#else
    solve();
#endif
}
```
`Start case` 在每组开始时立即输出，即使之后发生段错误，也能通过最后一条记录判断程序执行到了哪组测试
如果已经确认程序正确、不需要调试输出，可以仅保留计时
```cpp
#ifdef LOCAL
runtime_info.run(solve);
#else
solve();
#endif
```
##### 局部计时
```cpp
#ifdef LOCAL
{
    ScopeTimer tm{"build"};
    build();
}
#else
build();
#endif
```
总计时与局部计时相互独立，可以同时保留
#### RE 定位与现代 C++ 扩展
---
##### C++20 `std::source_location`
如果只需要一个能直接打印调用位置的独立函数，可以使用 C++20 标准库，无需宏
```cpp
#if defined(LOCAL) && __cplusplus >= 202002L
#include <source_location>
void here(const std::source_location &loc = std::source_location::current()) {
    cerr << "[HERE] " << loc.file_name() << ':' << loc.line() << ' ' << loc.function_name() << '\n';
}
#endif
```

```cpp
#ifdef LOCAL
here();
#endif
```
该函数的默认参数会获取调用位置，而非函数定义位置
##### C++23 `std::stacktrace`
支持该特性的标准库可以在程序仍能正常执行到调试函数时获取调用栈
```cpp
// 放在头文件区域
#if __has_include(<stacktrace>)
#include <stacktrace>

#ifdef __cpp_lib_stacktrace
cerr << std::stacktrace::current() << '\n';
#endif
```
可以放进 `Debug::fail()`，在 `abort()` 前打印调用栈
部分编译器或标准库尚不完整支持 `std::stacktrace`，某些版本还需要额外链接库，因此不建议纳入默认 Debug 核心
#### Linux 峰值内存统计
---
如果还希望查看本地测试的内存峰值：
```cpp
#if defined(LOCAL) && defined(__linux__)
#include <sys/resource.h>
void peak_mem() {
    rusage ru{};
    if (getrusage(RUSAGE_SELF, &ru) == 0) {
        cerr << "[Mem] peak=" << ru.ru_maxrss / 1024.0 << " MiB\n";
    }
}
#endif

#if defined(LOCAL) && defined(__linux__)
peak_mem();
#endif
```
可以放到 `RuntimeInfo` 析构函数中，Linux 的 `ru_maxrss` 单位为 KiB，输出换算为 MiB
##### 注意事项
- `run()` 计时包含 `solve()` 内的输入输出，但不包含函数外的全局预处理
- 对真正的 RE，优先使用 Sanitizer；需要进一步查看递归调用栈和函数参数时使用 GDB
#### Debug 编译、运行与调试命令手册
---
##### 环境与命令约定
###### 不同环境的区别

|环境|编译器|程序运行方式|推荐用途|
|---|---|---|---|
|Linux（Bash）|GCC / Clang|`./main`|完整调试、Sanitizer、性能及内存分析|
|Windows CMD|MinGW-w64 GCC|`main.exe`|Windows 原生编译、调试|
|Windows PowerShell|MinGW-w64 GCC|`.\main.exe`|Windows 原生编译、调试|
|Windows Git Bash|MinGW-w64 GCC|`./main.exe`|使用 Bash 风格命令运行 Windows 程序|
|Windows WSL|Linux GCC / Clang|`./main`|复现 Linux 评测环境|
> [!important] 编译参数由编译器决定，运行命令由终端决定，系统功能由操作系统决定
> 
> 以下默认使用 GCC。Windows 需要安装 MinGW-w64 / MSYS2 等 GCC 工具链，不能直接把这些参数用于 MSVC 的 `cl.exe`
> 
> Git Bash 不等于 Linux，WSL 内使用 Linux GCC 编译产生的是 Linux 程序，而不是 Windows 原生 `.exe`

##### 常用命令速查
###### Linux / WSL
```
# 普通编译
g++ main.cpp -std=gnu++20 -O2 -o main

# 编译警告
g++ main.cpp -std=gnu++20 -O2 -Wall -Wextra -Wshadow -o main

# 调试版本
g++ main.cpp -std=gnu++20 -Og -g3 -DLOCAL -o main

# Sanitizer 检查
g++ main.cpp -std=gnu++20 -O1 -g -DLOCAL -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all -o main

# STL 检查
g++ main.cpp -std=gnu++20 -O0 -g -D_GLIBCXX_DEBUG -DLOCAL -o main

# 运行并重定向
./main < in.txt > out.txt 2> err.txt

# 对比输出
diff -u ans.txt out.txt

# 查看时间、峰值内存
/usr/bin/time -v ./main < in.txt > out.txt

# GDB 定位崩溃
gdb -q -batch -ex 'run < in.txt' -ex 'bt full' -ex 'info args' -ex 'info locals' --args ./main
```
###### Windows CMD
```
REM 普通编译
g++ main.cpp -std=gnu++20 -O2 -o main.exe

REM 编译警告
g++ main.cpp -std=gnu++20 -O2 -Wall -Wextra -Wshadow -o main.exe

REM 调试版本
g++ main.cpp -std=gnu++20 -Og -g3 -DLOCAL -o main.exe

REM STL 检查（MinGW-w64 GCC + libstdc++）
g++ main.cpp -std=gnu++20 -O0 -g -D_GLIBCXX_DEBUG -DLOCAL -o main.exe

REM 运行并重定向
main.exe < in.txt > out.txt 2> err.txt

REM 比对输出
fc.exe /N ans.txt out.txt

REM GDB 定位崩溃（需 Windows 版 GDB）
gdb -q -batch -ex "run < in.txt" -ex "bt full" -ex "info args" -ex "info locals" --args main.exe
```
###### Windows PowerShell
编译命令与 CMD 基本相同，运行时采用 `.\main.exe`
```
# 普通编译
g++ main.cpp -std=gnu++20 -O2 -o main.exe

# 直接运行
.\main.exe

# 使用 CMD 完成输入输出重定向
cmd /c ".\main.exe < in.txt > out.txt 2> err.txt"

# 对比输出
fc.exe /N ans.txt out.txt

# GDB
gdb -q -batch -ex "run < in.txt" -ex "bt full" -ex "info args" -ex "info locals" --args .\main.exe
```
PowerShell 不提供 Bash/CMD 风格的 `< in.txt` 输入重定向，且较旧的 PowerShell 版本对原生程序输出重定向可能进行文本处理，因此竞赛程序的精确 I/O 测试优先通过 `cmd /c` 执行

> [!warning] Windows 原生 GCC 的 Sanitizer 支持取决于工具链及运行库版本，不能保证 `-fsanitize=address,undefined` 在每个 MinGW 环境下都可用
> 
> 如果需要稳定使用完整的 ASan + UBSan，推荐 Linux / WSL；Clang 在 Windows 上也有 ASan 支持，但仍需相应的运行库
##### GCC 编译参数详解
###### 基础参数
通用命令格式：
```
g++ main.cpp -std=gnu++20 -O2 -DLOCAL -g -o main
```

|参数|功能|限制与说明|
|---|---|---|
|`main.cpp`|待编译的源文件|多文件可填写多个文件名|
|`-o main`|指定输出文件|Windows 建议写 `main.exe`|
|`-std=gnu++17`|GNU C++17|支持 `__int128` 等 GNU 扩展|
|`-std=gnu++20`|GNU C++20|支持 `ranges`、`source_location` 等|
|`-std=gnu++23`|GNU C++23|新功能还取决于标准库版本|
|`-std=c++20`|标准 C++20 模式|GNU 扩展可能仍可用，但在严格检查下可能产生诊断|
|`-DLOCAL`|定义 `LOCAL` 宏|启用你现有的 Debug 代码|
|`-DONLINE_JUDGE`|定义 `ONLINE_JUDGE` 宏|仅适用于使用该宏判断环境的代码|
|`-DNDEBUG`|禁用标准 `assert()`|不影响自定义 `dcheck()`|
|`-g`|生成调试符号|主要影响文件体积，不是运行时检查|
|`-g3`|生成更多调试信息|包含宏定义等额外信息|
|`-ggdb3`|针对 GDB 生成详细调试信息|使用 GDB 时可选|
`-g3` 与 `-ggdb3` 选择其一即可，后者更偏向 GDB 的调试需求
###### 优化等级

|参数|作用|推荐场景|
|---|---|---|
|`-O0`|关闭大部分优化|GDB 单步调试|
|`-Og`|兼顾基本优化与可调试性|日常 Debug|
|`-O1`|基础优化|Sanitizer|
|`-O2`|常规优化|正式性能测试|
|`-O3`|更激进的优化|卡常对比测试|
|`-Ofast`|允许部分非严格标准优化|确认语义安全后尝试|
这些优化等级通常只选择一个，若重复指定，以最后生效的优化等级为准
推荐搭配：
```
# GDB
-O0 -g3

# 日常调试
-Og -g3

# Sanitizer
-O1 -g

# 性能测试
-O2
```
`-Ofast` 可能改变浮点数等运算的语义，不应作为 Debug 选项
###### 编译警告

|参数|检查内容|注意事项|
|---|---|---|
|`-Wall`|常见潜在错误|并非开启所有警告|
|`-Wextra`|更多额外警告|可能产生未使用参数等提示|
|`-Wshadow`|变量名遮蔽|竞赛中的局部重名可能产生较多警告|
|`-Wshadow=compatible-local`|较有针对性的局部变量遮蔽检查|可替代更严格的 `-Wshadow`|
|`-Wconversion`|可能改变数值的隐式类型转换|对 `size_t` 到 `int` 等场景较敏感|
|`-Wsign-conversion`|有符号与无符号类型转换|提示数量可能较多|
|`-Wfloat-equal`|浮点数直接相等比较|部分精确比较是有意设计的|
|`-Werror`|将警告视为编译错误|调试时可用，日常不必默认开启|
|`-fmax-errors=5`|最多显示五个编译错误|不限制普通警告的总数|
`-Wshadow` 和 `-Wconversion` 的检查范围可以按 GCC 文档进一步缩小，适合逐项排查警告
推荐组合：
```
-Wall -Wextra -Wshadow
```
需要严格检查转换时：
```
-Wall -Wextra -Wshadow -Wconversion -Wsign-conversion
```
##### 运行时错误检查
###### Sanitizer 参数

|参数|作用|可能的限制|
|---|---|---|
|`-fsanitize=address`|检查部分内存越界、释放后访问等|增加内存和运行开销，需要运行库支持|
|`-fsanitize=undefined`|检查多种未定义行为|不覆盖所有逻辑错误|
|`-fsanitize=address,undefined`|同时启用 ASan 与 UBSan|Linux 优先使用|
|`-fno-sanitize-recover=all`|检测到支持的错误后停止|适合快速定位首次错误|
|`-fno-omit-frame-pointer`|保留栈帧指针|改善部分调用栈信息|
|`-fno-optimize-sibling-calls`|避免部分尾调用优化|使部分调用栈更完整|
|`-g`|输出调试符号|便于获得源码位置|
GCC 官方建议使用较低优化等级及帧指针辅助获取可读的 Sanitizer 调用栈
Linux / WSL 完整写法：
```bash
g++ main.cpp -std=gnu++20 -O1 -g -DLOCAL -Wall -Wextra -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all -o main
./main < in.txt > out.txt 2> err.txt
```
也可以只检查特定错误，例如仅检查未定义行为：
```
g++ main.cpp -std=gnu++20 -O1 -g -fsanitize=undefined -o main
```
Windows 原生工具链应先确认相应 Sanitizer 是否提供运行库，Clang 的 Windows ASan 支持也存在平台和功能上的限制
###### STL 检查参数

|参数|作用|限制|
|---|---|---|
|`-D_GLIBCXX_ASSERTIONS`|启用 libstdc++ 的额外前置条件断言|检查范围有限|
|`-D_GLIBCXX_DEBUG`|开启更完整的容器、迭代器调试|可能改变部分容器布局、增加耗时和内存|
轻量检查：
```
g++ main.cpp -std=gnu++20 -O2 -D_GLIBCXX_ASSERTIONS -DLOCAL -o main
```
完整 STL 调试：
```
g++ main.cpp -std=gnu++20 -O0 -g -D_GLIBCXX_DEBUG -DLOCAL -o main
```
以上同样适用于使用 GNU libstdc++ 的 Windows MinGW-w64 GCC
`_GLIBCXX_DEBUG` 会改变部分标准库模板的布局，混合编译多个文件或链接预编译库时需要注意 ABI 兼容性

> [!tip] 建议分别构建 Sanitizer 版与 STL Debug 版，而不是默认把两者全叠加
> 
> 两者可以在受支持的配置中组合，但性能和内存开销更高，出现问题时也不容易判断是哪一类检查导致的
##### GDB 调试命令
###### 启动参数

|参数|功能|

|`-batch`|批处理模式，执行完命令自动退出|
|`-ex "命令"`|在启动时执行一条 GDB 命令，可重复|
|`-x file`|从文件读取 GDB 命令|
|`--args`|指定被调试程序及其命令行参数，后续内容交给程序|
|`-return-child-result`|在适用情况下传回被调试程序的退出码|
这些启动参数可以按需要组合
###### GDB 内部命令

|命令|功能|
|---|---|
|`run` / `r`|启动程序|
|`run < in.txt`|使用文件作为程序输入|
|`bt`|查看调用栈|
|`bt full`|查看调用栈及局部变量|
|`frame 0`|选择第 0 层栈帧，即当前出错位置|

|`next` / `n`|单步执行，不进入函数|
|`step` / `s`|单步执行，进入函数|
|`continue` / `c`|继续运行|
|`list` / `l`|查看附近源码|
|`quit` / `q`|退出 GDB|
`bt full` 通常比单独执行 `info locals` 更适合定位深层递归中的错误，因为它可以查看多层调用栈

###### 运行、重定向与结果检查
###### 重定向符号

| 符号            | 功能                 |
| ------------- | ------------------ |
| `< in.txt`    | 从文件读取标准输入          |
| `> out.txt`   | 将标准输出写入文件，覆盖原内容    |
| `>> out.txt`  | 将标准输出追加到文件末尾       |
| `2> err.txt`  | 将标准错误输出写入文件        |
| `2>> err.txt` | 追加标准错误输出           |
| `2>&1`        | 将标准错误输出合并到当前标准输出目标 |
| `\|`          | 将左侧程序输出连接到右侧程序输入   |
注意 `2>&1` 与其他重定向符号的先后顺序可能影响结果
Linux、CMD 和 Git Bash 支持这些常见重定向方式，PowerShell 对输出流的处理规则有所不同，尤其需要注意输入重定向与编码
###### 对比答案
Linux / WSL：
```
./main < in.txt > out.txt
diff -u ans.txt out.txt
```
Windows CMD：
```
main.exe < in.txt > out.txt
fc.exe /N ans.txt out.txt
```
Windows PowerShell：
```
cmd /c ".\main.exe < in.txt > out.txt"
fc.exe /N ans.txt out.txt
```
常用比对参数：

| 参数          | 作用         | 适用环境         |
| ----------- | ---------- | ------------ |
| `diff -u`   | 统一格式显示差异   | Linux / Bash |
| `diff -w`   | 忽略空白数量差异   | Linux / Bash |
| `fc.exe /N` | 显示不同文本行及行号 | Windows      |
| `fc.exe /B` | 逐字节比较文件    | Windows      |
| `fc.exe /W` | 压缩空白差异     | Windows      |
`fc.exe` 返回 `0` 表示相同、`1` 表示不同、`2` 表示发生比较错误
以上均为普通文本或字节比较，并不能替代浮点误差判定、答案不唯一等题目的 SPJ

# Linux / Bash：紧跟程序运行之后

```

```
REM Windows CMD

```

```
# Windows PowerShell

退出码非零通常表示程序异常结束或自行报告失败，但具体数值会受到程序、操作系统和工具链影响
###### 时间、内存与运行限制
**Linux / WSL**
```
# 简单计时
time ./main < in.txt > out.txt

# GNU time：详细资源统计
/usr/bin/time -v ./main < in.txt > out.txt

# 把时间与内存报告单独保存
/usr/bin/time -v -o perf.txt ./main < in.txt > out.txt 2> err.txt

# 限制运行时间为 2 秒
timeout 2s ./main < in.txt

# 查看当前栈限制
ulimit -s

# 将当前 Shell 的栈软限制尝试设为 64 MiB
ulimit -s 65536
```

|参数或命令|作用|限制|
|---|---|---|
|`time`|查看整体执行耗时|不一定提供内存信息|
|`/usr/bin/time -v`|查看时间、CPU 及峰值内存等|需要 GNU time|
|`-o perf.txt`|将 GNU time 报告写入文件|不影响程序的标准输出|
|`timeout 2s`|最多运行约 2 秒|需要 GNU coreutils|
|`ulimit -s`|查询或设置栈空间限制|仅影响当前 Shell 及后续子进程|
|`ulimit -c unlimited`|允许尽可能大的 core dump|仍受系统 core dump 配置限制|
GNU time 的最大常驻内存指标通常以 KiB 表示，而 `timeout` 超时后默认返回码为 `124`，特殊选项或信号情况除外
`ulimit -s` 不能改变评测机的栈限制，并且修改栈限制可能影响 ASan 的运行环境，不能将其作为通用修复手段
**Windows PowerShell**
```
# 运行并测量总耗时
(Measure-Command {
    cmd /c ".\main.exe < in.txt > out.txt 2> err.txt"
}).TotalMilliseconds
```
`Measure-Command` 返回一个 `TimeSpan` 对象，可以读取 `TotalMilliseconds`、`TotalSeconds` 等属性
它测量的是整个命令块执行时间，包括启动 `cmd.exe` 和程序的开销，因此只适合比较较长的运行过程，不适合微小函数的精细测速
Windows CMD 的 `timeout /t 2` 只是暂停命令处理器 2 秒，并不是限制目标程序执行 2 秒，与 Linux 的 `timeout 2s ./main` 不同
Windows 原生环境没有与 Linux `/usr/bin/time -v` 完全等价的内置命令。峰值内存可以考虑任务管理器、性能分析工具或专门的 Windows API 实现；想使用本章 Linux 工具时，优先通过 WSL
##### 参数组合规则
###### 推荐组合

|使用场景|推荐组合|
|---|---|
|正式编译|`-std=gnu++20 -O2`|
|基础警告|`-O2 -Wall -Wextra -Wshadow`|
|日常 Debug|`-Og -g3 -DLOCAL -Wall -Wextra -Wshadow`|
|GDB 定位 RE|`-O0 -g3 -DLOCAL -fno-omit-frame-pointer`|
|Sanitizer|`-O1 -g -DLOCAL -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all`|
|STL 轻量检查|`-O2 -D_GLIBCXX_ASSERTIONS`|
|STL 完整检查|`-O0 -g -D_GLIBCXX_DEBUG`|
|性能对比|分别使用 `-O2`、`-O3` 编译并测试|
###### 不推荐的组合

|组合|原因|
|---|---|
|`-O0 -O3`|同类优化等级重复，按最后生效的设置处理|
|`-Ofast` + 浮点正确性调试|可能改变浮点语义|
|`-DNDEBUG` + 依赖 `assert()` 的调试|会禁用断言|
|`-D_GLIBCXX_DEBUG` + 正式性能测试|改变容器行为与开销|
|`-fsanitize=address` + `-static`|ASan 静态链接存在支持限制|
|`-march=native` + 未知评测机 CPU|可能生成目标 CPU 不支持的指令|
|Sanitizer + 严格时空限制模拟|检查工具会增加运行时间及内存|
##### 按需拼接示例
例如想在 Linux 上定位 RE，同时开启警告和变量检查：
```bash
g++ main.cpp -std=gnu++20 -O1 -g3 -DLOCAL -Wall -Wextra -Wshadow -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all -o main
./main < in.txt > out.txt 2> err.txt
```
想在 Windows 上检查 STL 越界，并开启你的 `dbg()` / `dbgv()`：
```
g++ main.cpp -std=gnu++20 -O0 -g3 -DLOCAL -D_GLIBCXX_DEBUG -Wall -Wextra -Wshadow -o main.exe
main.exe < in.txt > out.txt 2> err.txt
```
想在 Linux 上统计真实运行耗时和峰值内存：
```bash
g++ main.cpp -std=gnu++20 -O2 -o main
/usr/bin/time -v -o perf.txt ./main < in.txt > out.txt
```
这时不要使用 Sanitizer 或 STL Debug，因为它们会改变程序的时间与内存表现
###### 注意事项
- `-DLOCAL` 对应你的 `#ifdef LOCAL`，如果忘记添加，调试输出和计时功能就不会启用
- `-g` 本身不会检查越界，`-O0` 本身也不会检查越界
- Sanitizer、STL Debug、GDB 各自解决不同的问题，不能相互完全替代
- Windows 上的 `g++` 不一定是同一个发行版，支持的参数和运行库可能不同
- 尽量保留和评测机相同的 C++ 标准、编译器及标准库版本
- 对于跨平台竞赛项目，最稳妥的是分别保留 Windows 原生调试命令与 Linux / WSL 调试命令，而不是强行合并成一条
最终，这份手册可以遵循一个简单的拼接顺序：
```
编译器 + 源文件 + C++ 标准 + 优化等级
       + 宏定义 + 警告选项 + 调试信息
       + 检查工具 + 输出文件
```
