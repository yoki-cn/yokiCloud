### IO
---
#### FastIO
---
使用 `fread` / `fwrite` 缓冲输入输出，支持普通整数、`i128`、`u128`、字符、字符串及常见容器
- `read()` 支持连续读取多个参数、数组和已分配空间的容器
- `write()` 自动在不同元素之间添加空格
- `writeln()` 自动换行，`writelines()` 将每个参数分别输出一行
- `get()`、`line()` 分别支持原始字符和整行读取
- 不支持浮点数读写，相关方法放在后面的标准库速查中
```cpp
using i128 = __int128_t;
using u128 = __uint128_t;

namespace FastIO {
struct Input {
private:
    static constexpr size_t S{ 1 << 20 };
    char buf[S];
    size_t p{}, n{};
    static bool digit(int c) { return '0' <= c && c <= '9'; }
    bool skip(int &c) {
        c = get();
        while (c != EOF && c <= ' ') c = get();
        return c != EOF;
    }
    // ===== 整数读取核心 =====
    template<class U>
    bool read_abs(U &v, bool &neg) {
        int c;
        if (!skip(c)) return false;
        neg = false;
        if (c == '-' || c == '+') {
            neg = c == '-';
            c = get();
        }
        if (!digit(c)) return false;
        v = 0;
        bool ok = true;
        constexpr U lim = ~U(0);
        while (digit(c)) {
            U d = U(c - '0');
            if (v > lim / 10 || (v == lim / 10 && d > lim % 10)) ok = false;
            else if (ok) v = v * 10 + d;
            c = get();
        }
        return ok;
    }
public:
    // ===== 原始字符读取 =====
    int get() {
        if (p == n) {
            n = fread(buf, 1, S, stdin);
            p = 0;
            if (!n) return EOF;
        }
        return (unsigned char)buf[p ++];
    }
    // ===== 标准整数 =====
    template<class T,
        enable_if_t<is_integral_v<T> && !is_same_v<T, bool>
            && !is_same_v<T, i128> && !is_same_v<T, u128>, int> = 0>
    bool read(T &x) {
        using U = make_unsigned_t<T>;
        U v;
        bool neg;
        if (!read_abs(v, neg)) return false;
        if constexpr (is_signed_v<T>) {
            U lim = U(numeric_limits<T>::max()) + U(neg);
            if (v > lim) return false;
            x = neg ? (v ? -T(v - 1) - 1 : T{}) : T(v);
        } else {
            if (neg) return false;
            x = T(v);
        }
        return true;
    }
    // ===== 128 位整数 =====
    bool read(i128 &x) {
        u128 v;
        bool neg;
        if (!read_abs(v, neg)) return false;
        if (v > (u128(1) << 127) - 1 + u128(neg)) return false;
        x = neg ? (v ? -i128(v - 1) - 1 : i128{}) : i128(v);
        return true;
    }
    bool read(u128 &x) {
        u128 v;
        bool neg;
        if (!read_abs(v, neg) || neg) return false;
        x = v;
        return true;
    }
    // ===== 字符与字符串 =====
    bool read(char &c) {
        int x;
        if (!skip(x)) return false;
        c = char(x);
        return true;
    }
    bool read(string &s) {
        int c;
        if (!skip(c)) return false;
        s.clear();
        while (c != EOF && c > ' ') {
            s.push_back(char(c));
            c = get();
        }
        return true;
    }
    template<size_t N>
    bool read(char (&s)[N]) {
        static_assert(N > 0);
        int c;
        if (!skip(c)) return false;
        size_t len{};
        bool ok = true;
        while (c != EOF && c > ' ') {
            if (len + 1 < N) s[len ++] = char(c);
            else ok = false;
            c = get();
        }
        s[len] = '\0';
        return ok;
    }
    // ===== 整行读取 =====
    bool line(string &s) {
        int c = get();
        if (c == EOF) return false;
        s.clear();
        while (c != EOF && c != '\n') {
            s += char(c);
            c = get();
        }
        if (!s.empty() && s.back() == '\r') s.pop_back();
        return true;
    }
    // ===== 容器读取 =====
    template<class T, size_t N,
        enable_if_t<!is_same_v<T, char>, int> = 0>
    bool read(T (&a)[N]) {
        for (T &x : a) if (!read(x)) return false;
        return true;
    }
    template<class T, size_t N>
    bool read(array<T, N> &a) {
        for (T &x : a) if (!read(x)) return false;
        return true;
    }
    template<class T, class A>
    bool read(vector<T, A> &a) {
        for (T &x : a) if (!read(x)) return false;
        return true;
    }
    // ===== 连续读取 =====
    template<class T, class U, class... V>
    bool read(T &x, U &y, V &...v) { return read(x) && read(y, v...); }
};
struct Output {
private:
    static constexpr size_t S{ 1 << 20 };
    char buf[S];
    size_t p{};
    template<class U>
    void write_unsigned(U x) {
        char s[40];
        int n{};
        do {
            s[n ++] = char('0' + x % 10);
            x /= 10;
        } while (x);
        while (n) put(s[-- n]);
    }
    void sep(bool &first) {
        if (!first) put(' ');
        first = false;
    }
    // ===== 标量输出 =====
    void scalar(char c) { put(c); }
    void scalar(bool x) { put(x ? '1' : '0'); }
    void scalar(string_view s) { for (char c : s) put(c); }
    void scalar(const char *s) { while (*s) put(*s ++); }
    template<class T,
        enable_if_t<is_integral_v<T> && !is_same_v<T, bool>
            && !is_same_v<T, i128> && !is_same_v<T, u128>, int> = 0>
    void scalar(T x) {
        using U = make_unsigned_t<T>;
        U v;
        if constexpr (is_signed_v<T>) {
            if (x < 0) {
                put('-');
                v = U(-(x + 1)) + 1;
            } else v = U(x);
        } else v = U(x);
        write_unsigned(v);
    }
    // ===== 128 位整数输出 =====
    void scalar(i128 x) {
        if (x < 0) {
            put('-');
            write_unsigned(u128(-(x + 1)) + 1);
        } else write_unsigned(u128(x));
    }
    void scalar(u128 x) { write_unsigned(x); }
    // ===== 容器展开 =====
    template<class T>
    void tokens(bool &first, const T &x) {
        sep(first);
        scalar(x);
    }
    template<class T, size_t N,
        enable_if_t<!is_same_v<T, char>, int> = 0>
    void tokens(bool &first, const T (&a)[N]) {
        for (const T &x : a) tokens(first, x);
    }
    template<class T, size_t N>
    void tokens(bool &first, const array<T, N> &a) {
        for (const T &x : a) tokens(first, x);
    }
    template<class T, class A>
    void tokens(bool &first, const vector<T, A> &a) {
        for (const T &x : a) tokens(first, x);
    }
public:
    ~Output() { flush(); }
    // ===== 基础输出 =====
    void flush() {
        if (p) {
            fwrite(buf, 1, p, stdout);
            p = 0;
        }
    }
    void put(char c) {
        if (p == S) flush();
        buf[p ++] = c;
    }
    // ===== 通用输出 =====
    template<class... T>
    void write(const T &...x) {
        if constexpr (sizeof...(T)) {
            bool first = true;
            (tokens(first, x), ...);
        }
    }
    // ===== 扩展输出 =====
    template<class... T>
    void writeln(const T &...x) { write(x...); put('\n'); }
    template<class... T>
    void writelines(const T &...x) { (writeln(x), ...); }
};
inline Input in;
inline Output out;
}
using namespace FastIO;
```
#### Input
- `in.get()`：读取一个原始字节，包含空格、换行，EOF 返回 `-1`，均摊 $O(1)$
- `in.read(x)`：读取整数、非空白字符或字符串，成功返回 `true`，失败返回 `false`
- `in.read(a, b, c)`：依次读取多个参数，任一失败则停止
- `in.read(a)`：当 `a` 是 C 数组、`array`、`vector` 时，逐元素读取
- `in.line(s)`：读取一整行，不包含行末换行符，支持空行，遇到 EOF 且没有可读取字符时返回 `false`
#### Output
- `out.put(c)`：输出一个原始字符
- `out.flush()`：将缓冲区内的数据写入 `stdout`
- `out.write(x...)`：输出多个标量或容器，元素之间自动添加空格，不自动换行
- `out.writeln(x...)`：与 `write()` 相同，末尾添加换行
- `out.writelines(x...)`：每个参数输出一行
- `out.writeln()`：输出一个空行  
#### 容器约定
- `vector` 必须事先 `resize()`，不支持直接读取未分配大小的容器
- 支持嵌套的 `vector`、`array` 和普通数组
- 不支持 `vector<bool>` 的代理引用读写
- `char[N]` 作为以 `'\0'` 结尾的字符串处理，而不是逐元素处理
#### 注意事项
- `FastIO` 输入会提前缓存数据，不能与 `cin`、`scanf`、`getchar` 混合读取同一个 `stdin`
- 使用 `FastIO` 输出时，也不应与 `cout`、`printf` 混合输出，以免缓冲顺序与预期不一致
- `read()` 假定整数 token 只含数字及可选的前置正负号，且以空白或 EOF 分隔，不是通用文本语法分析器
- 整数读取溢出、非法符号、EOF 时返回 `false`，单个标量读取失败不会修改目标变量，容器多元素读取失败时，之前的元素可能已经修改
- `read(char[N])` 超长时会截断存储、消费整个 token 并返回 `false`
- `get()` 返回 `int`，不能用 `char` 保存 EOF
- `line()` 在 `\r\n` 行尾时自动去掉末尾 `\r`
- 使用 `read()` 后调用 `line()` 时，整数或 token 读取已消费一个结束字符，注意是否还有同一行的剩余内容
- 正常程序退出时 `Output` 析构函数自动刷新缓冲区，交互题不建议使用该 `fread` 方案
- 本实现的输出不进行特殊的 `boolalpha`、浮点格式化、进制格式化
#### 常见用法
##### 标准 I/O（iostream）
```cpp
ios::sync_with_stdio(false);
cin.tie(nullptr);
```
- `sync_with_stdio(false)`：关闭 C++ 标准流与 C 标准流之间的同步
- `cin.tie(nullptr)`：取消输入前自动刷新 `cout`
- `'\n'`：只添加换行，通常优先于 `endl`
- `endl`：换行并刷新输出缓冲区
##### 字符、字符串与整行输入
```cpp
char c;
cin >> c;          // 跳过前导空白
cin.get(c);        // 读取一个原始字符，包括空格和换行

string s, line;
cin >> s;          // 读取空白分隔的字符串
getline(cin, line); // 读取整行，不含换行符

char buf[100];
cin >> setw(100) >> buf; // 最多读取 99 个字符
```
读取整数后接着使用 `getline()` 时，需要先处理残留换行符
```cpp
int n;
cin >> n;
cin.ignore(numeric_limits<streamsize>::max(), '\n');
string s;
getline(cin, s);
```
也可以写成：
```cpp
getline(cin >> ws, s);
```
但 `ws` 会跳过所有前导空白，也可能跳过空行或行首空格，不能用于需要保留这些信息的场景
##### 未知数量整数输入
读到 EOF 为止
```cpp
vector<int> a;
for (int x; cin >> x;) a.push_back(x);
```
按行读取，一行内的整数个数未知
```cpp
string s;
getline(cin, s);
stringstream ss(s);
vector<int> a;
for (int x; ss >> x;) a.push_back(x);
```
使用 `FastIO` 时：
```cpp
int x;
while (in.read(x)) {
    // ...
}
```
##### `scanf` / `printf` 格式化输入输出
```cpp
int x;
i64 y;
double a;
long double b;
char c, s[100];

scanf("%d%lld%lf%Lf", &x, &y, &a, &b);
scanf(" %c", &c);      // 跳过前导空白
scanf("%99s", s);      // 限制字符串长度

printf("%d %lld\n", x, y);
printf("%.12f\n", a);
printf("%.12Lf\n", b);
```
常见格式说明符：

|类型|`scanf`|`printf`|
|---|---|---|
|`int`|`%d`|`%d`|
|`unsigned int`|`%u`|`%u`|
|`i64`|`%lld`|`%lld`|
|`u64`|`%llu`|`%llu`|
|`float`|`%f`|`%f`|
|`double`|`%lf`|`%f`|
|`long double`|`%Lf`|`%Lf`|
|`char`|`%c`|`%c`|
|C 字符串|`%s`|`%s`|
`scanf` 读取 C 字符串应指定宽度，至少预留一个字符存放 `'\0'`
##### `stringstream` 常用操作
###### 字符串与整数互转
```cpp
stringstream ss;
ss << 123 << ' ' << -45;
string s = ss.str(); // "123 -45"

stringstream tt(s);
int x, y;
tt >> x >> y;
```
###### 按照指定分隔符拆分
```cpp
stringstream ss("aa,bb,,cc");
string s;
vector<string> a;
while (getline(ss, s, ',')) a.push_back(s);
// {"aa", "bb", "", "cc"}
```
不涉及格式化时，整数与字符串转换优先考虑 `to_string()`、`stoi()`、`stoll()` 或 C++17 的 `from_chars()`、`to_chars()`
##### 浮点数与格式化输出
```cpp
double x = 1.23456789;
cout << fixed << setprecision(3) << x << '\n'; // 1.235
cout << scientific << setprecision(6) << x << '\n';
cout << defaultfloat << x << '\n';

cout << setw(6) << setfill('0') << 42 << '\n'; // 000042
cout << boolalpha << true << '\n';             // true
cout << noboolalpha << true << '\n';           // 1
```
- `fixed`：`setprecision(k)` 表示小数点后保留 `k` 位
- `scientific`：科学计数法，`setprecision(k)` 表示小数点后 `k` 位
- `defaultfloat`：恢复默认浮点格式，此时精度表示有效数字位数
- `setw(k)`：只影响下一次格式化插入
- `setfill(c)`、`setprecision(k)`、`fixed` 等设置会持续生效，必要时需恢复
##### C++20 `std::format`
需要 `<format>`，适用于格式化生成字符串
```cpp
// C++20，需编译器标准库支持 <format>
string s = std::format("{} {:.3f}", 42, 1.23456);
// s = "42 1.235"
string t = std::format("{:08d}", 42);
// t = "00000042"
```
`std::format()` 返回 `string`，并不直接输出
##### C++23 `std::print` / `std::println`
需要 `<print>`，可以直接格式化输出，无需构造字符串流
```cpp
// C++23，需标准库支持 <print>
std::print("{} {}", 10, 20);
std::println(" {}", 30); // 10 20 30
std::println("{:.6f}", 1.23456789);
```
`std::print` 不自动换行，`std::println` 自动添加换行，均支持格式说明符
> [!warning] `std::format` 属于 C++20，`std::print` / `std::println` 属于 C++23，但 `-std=gnu++20` 或 `-std=gnu++23` 并不保证评测机标准库已实现相关接口
> 
##### FastIO 常用调用
```cpp
int n, m;
i64 x;
string s;
in.read(n, m, x, s);

vector<int> a(n);
in.read(a);

out.writeln(n, m, x);
out.writeln(a);
out.writelines(n, m, x);

// 原始字符与整行
int c = in.get();
string line;
in.line(line);

// 立即刷新
out.flush();
```
输出容器时会递归展开其中的元素
```cpp
vector<vector<int>> a{{1, 2}, {3, 4}};
out.writeln(a); // 1 2 3 4
```
##### 文件重定向与本地调试
使用条件编译进行文件重定向
```cpp
int main() {
#ifdef LOCAL
    freopen("in.txt", "r", stdin);
    freopen("out.txt", "w", stdout);
#endif
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    solve();
    return 0;
}
```
GNU C++ 编译：
```cpp
g++ main.cpp -std=gnu++17 -O2 -DLOCAL -o main
./main
```
Shell 重定向：
```cpp
./main < in.txt > out.txt
./main < in.txt > out.txt 2> err.txt
time ./main < in.txt > out.txt
```
Windows CMD：
```cpp
main.exe < in.txt > out.txt
main.exe < in.txt > out.txt 2> err.txt
```
- `stdin`、`stdout`、`stderr` 分别为标准输入、标准输出、标准错误输出
- `2> err.txt` 将标准错误输出重定向到文件
- `freopen()` 应在任何实际读写之前调用
- 本地调试优先使用 Shell 重定向，无需修改源代码
#### 复杂度

|操作|时间复杂度|空间复杂度|
|---|---|---|
|`in.get()` / `out.put()`|均摊 $O(1)$|$O(1)$|
|`in.read(x)` 整数|$O(d)$|$O(1)$|
|`in.read(s)` 字符串|$O(n)$|$O(n)$|
|`in.line(s)`|$O(n)$|$O(n)$|
|`in.read(a)` 容器|$O(\sum d_i)$|$O(1)$（不计容器）|
|`out.write(x...)`|$O(L)$|$O(1)$（不计缓冲区）|
|`out.flush()`|$O(B)$|$O(1)$|
|`getline()`|$O(n)$|$O(n)$|
|`stringstream` 解析|$O(n)$|$O(n)$|
其中：
- $d$ 为一个整数的十进制位数
- $n$ 为字符串长度
- $L$ 为输出总字符数
- $B$ 为待刷新缓冲区中的字节数
- FastIO 的固定缓冲区总空间约为 $2\text{ Mib}$
