### 常用工具函数
---
```cpp
// ===== 基础运算 =====
template<class T>
T power(T a, i64 b) {
    assert(b >= 0);
    T res{ 1 };
    while (b) {
        if (b & 1) res *= a;
        b >>= 1;
        if (b) a *= a;
    }
    return res;
}
template<class T>
T sqr(T x) { return x * x; }
// ===== 整数平方根与立方根 =====
u64 isqrt(u64 n) {
    u64 x{ u64(sqrt(ld(n))) };
    while (x > 0 && x > n / x) -- x;
    while (x + 1 <= n / (x + 1)) ++ x;
    return x;
}
u64 icbrt(u64 n) {
    u64 x{ u64(cbrt(ld(n))) };
    auto le = [&](u64 v) { return v == 0 || v <= n / v / v; };
    while (!le(x)) -- x;
    while (le(x + 1)) ++ x;
    return x;
}
// ===== 整数根扩展：不需要可整段删除 =====
u64 ceil_sqrt(u64 n) {
    u64 x{ isqrt(n) };
    return x + (x * x != n);
}
u64 ceil_cbrt(u64 n) {
    u64 x{ icbrt(n) };
    return x + (x * x * x != n);
}
bool is_square(u64 n) {
    u64 x{ isqrt(n) };
    return x * x == n;
}
bool is_cube(u64 n) {
    u64 x{ icbrt(n) };
    return x * x * x == n;
}
// ===== 整数对数 =====
// 要求 x > 0
int floor_log2(u64 x) {
    assert(x > 0);
    return bit_width(x) - 1;
}
int ceil_log2(u64 x) {
    assert(x > 0);
    return bit_width(x - 1);
}
// 要求 base >= 2, x >= 1
int floor_log(u64 x, u64 base) {
    assert(base >= 2 && x >= 1);
    int k{ 0 };
    while (x >= base) {
        x /= base;
        ++ k;
    }
    return k;
}
int ceil_log(u64 x, u64 base) {
    assert(base >= 2 && x >= 1);
    return x == 1 ? 0 : floor_log(x - 1, base) + 1;
}
// ===== 有符号整除与取模 =====
// 要求 b != 0，且 a / b 可由 T 表示
template<class T>
T floor_div(T a, T b) {
    static_assert(is_integral_v<T> && is_signed_v<T>);
    assert(b != 0 && !(a == numeric_limits<T>::min() && b == T(-1)));
    T q{ a / b }, r{ a % b };
    if (r != 0 && ((r > 0) != (b > 0))) -- q;
    return q;
}
template<class T>
T ceil_div(T a, T b) {
    static_assert(is_integral_v<T> && is_signed_v<T>);
    assert(b != 0 && !(a == numeric_limits<T>::min() && b == T(-1)));
    T q{ a / b }, r{ a % b };
    if (r != 0 && ((r > 0) == (b > 0))) ++ q;
    return q;
}
template<class T>
T floor_mod(T a, T b) {
    static_assert(is_integral_v<T> && is_signed_v<T>);
    assert(b != 0 && !(a == numeric_limits<T>::min() && b == T(-1)));
    T r{ a % b };
    if (r != 0 && ((r > 0) != (b > 0))) r += b;
    return r;
}
// ===== 比较与更新 =====
template<class T, class U>
bool chmin(T &a, const U &b) {
    if (b < a) {
        a = b;
        return true;
    }
    return false;
}
template<class T, class U>
bool chmax(T &a, const U &b) {
    if (a < b) {
        a = b;
        return true;
    }
    return false;
}
template<class T>
int sign(const T &x) {
    return (T{} < x) - (x < T{});
}
// ===== 二进制辅助：不需要可整段删除 =====
// 返回 >= x 的最小 2 的幂；溢出返回 0
u64 ceil_pow2(u64 x) {
    if (x > (1ULL << 63)) return 0;
    return bit_ceil(x);
}
u64 lowbit(u64 x) {
    return x & -x;
}
// ===== 安全绝对差：不需要可直接删除 =====
// 支持完整 i64 范围，结果为 u64
u64 abs_diff(i64 a, i64 b) {
    return a < b ? u64(b) - u64(a) : u64(a) - u64(b);
}
// ===== 整数幂比较与通用 k 次根：不需要可整段删除 =====
// 返回 a^b <= lim，避免乘法溢出，要求 b >= 0
bool power_le(u64 a, int b, u64 lim) {
    assert(b >= 0);
    u64 p{ 1 };
    for (int i = 0; i < b; ++ i) {
        if (a && p > lim / a) return false;
        p *= a;
    }
    return p <= lim;
}
// 要求 1 <= k <= 64，返回 floor(n^(1/k))
u64 kth_root(u64 n, int k) {
    assert(1 <= k && k <= 64);
    if (k == 1 || n <= 1) return n;
    u64 l{ 1 }, r{ 1ULL << ((64 + k - 1) / k) };
    while (r - l > 1) {
        u64 mid{ l + (r - l) / 2 };
        if (power_le(mid, k, n)) l = mid;
        else r = mid;
    }
    return l;
}
```

#### 接口说明

| 函数                         | 功能                 |
| -------------------------- | ------------------ |
| `power(a,b)`               | 泛型非负整数快速幂          |
| `sqr(x)`                   | 求平方                |
| `isqrt` / `icbrt`          | 整数根下取整             |
| `ceil_sqrt` / `ceil_cbrt`  | 整数根上取整             |
| `is_square` / `is_cube`    | 判断完全平方 / 立方        |
| `floor_log2` / `ceil_log2` | 底数为 2 的整数对数        |
| `floor_log` / `ceil_log`   | 任意整数底数的整数对数        |
| `floor_div` / `ceil_div`   | 数学意义的向下 / 向上取整除法   |
| `floor_mod`                | 与下取整除法一致的余数        |
| `chmin` / `chmax`          | 更新极值，返回是否更新        |
| `sign`                     | 返回 \(-1,0,1\)      |
| `ceil_pow2`                | 向上取整到 2 的幂         |
| `lowbit`                   | 最低有效 1 对应的值        |
| `abs_diff`                 | 有符号 64 位整数安全绝对差    |
| `power_le`                 | 安全判断 $a^b\leq lim$ |
| `kth_root`                 | 精确整数 $k$ 次根下取整     |
