### 异或线性基
---
`T` 为无符号整数（默认 `u64`），`B` 是有效二进制位数。
#### 普通异或线性基
---
```cpp
template<class T = u64, int B = numeric_limits<T>::digits>
struct LinearBasis {
    static_assert(is_unsigned_v<T>);
    static_assert(1 <= B && B <= 64 && B <= numeric_limits<T>::digits);
    using value_type = T;
    array<T, B> p{};
    int siz{ 0 };
    // ===== 基础接口：插入、判定、最值与秩 =====
    bool insert(T x) {
        if (siz == B) return false;
        for (int i = B - 1; i >= 0; -- i) if (x >> i & 1) {
            if (!p[i]) {
                p[i] = x;
                ++ siz;
                return true;
            }
            x ^= p[i];
        }
        return false;
    }
    bool contains(T x) const {
        for (int i = B - 1; i >= 0; -- i) if (x >> i & 1) {
            if (!p[i]) return false;
            x ^= p[i];
        }
        return true;
    }
    T max_xor(T x = 0) const {
        for (int i = B - 1; i >= 0; -- i) if ((x ^ p[i]) > x) x ^= p[i];
        return x;
    }
    T min_xor(T x) const {
        for (int i = B - 1; i >= 0; -- i) if ((x ^ p[i]) < x) x ^= p[i];
        return x;
    }
    T min_positive() const {
        for (int i = 0; i < B; ++ i) if (p[i]) return p[i];
        return 0;
    }
    int rank() const { return siz; }
    void clear() {
        p.fill(0);
        siz = 0;
    }
    // ===== 规范化与第 k 小 =====
    vector<T> basis() const {
        auto q{ p };
        for (int i = 0; i < B; ++ i) if (q[i]) {
            for (int j = i - 1; j >= 0; -- j) {
                if (q[i] >> j & 1) q[i] ^= q[j];
            }
        }
        vector<T> res;
        res.reserve(siz);
        for (int i = 0; i < B; ++ i) if (q[i]) res.push_back(q[i]);
        return res;
    }
    optional<T> kth(u64 k) const {
        if (siz < 64 && k >= (1ULL << siz)) return nullopt;
        auto a{ basis() };
        T res{};
        for (int i = 0; i < siz; ++ i) if (k >> i & 1) res ^= a[i];
        return res;
    }
    // ===== 张成空间运算：合并 / 求交 =====
    void merge(const LinearBasis &o) {
        for (int i = B - 1; i >= 0; -- i) if (o.p[i]) insert(o.p[i]);
    }
    LinearBasis intersection(const LinearBasis &o) const {
        // q 表示两个空间的并空间基，tag 表示 q 中对应向量的当前空间分量
        auto q{ p };
        auto tag{ p };
        LinearBasis res;
        for (int i = B - 1; i >= 0; -- i) if (o.p[i]) {
            T x{ o.p[i] }, t{};
            bool independent{ false };
            for (int j = B - 1; j >= 0; -- j) if (x >> j & 1) {
                if (!q[j]) {
                    q[j] = x;
                    tag[j] = t;
                    independent = true;
                    break;
                }
                x ^= q[j];
                t ^= tag[j];
            }
            // 出现并空间中的线性关系时，t 同时属于两个原空间
            if (!independent) res.insert(t);
        }
        return res;
    }
};
```
#### 前缀区间线性基（静态区间 / 末尾追加）
---
查询区间均为 **`[l,r)`**。维护每个前缀的最大右端位置主元。
```cpp
template<class T = u64, int B = numeric_limits<T>::digits>
struct PrefixLinearBasis {
    static_assert(is_unsigned_v<T>);
    static_assert(1 <= B && B <= 64 && B <= numeric_limits<T>::digits);
    struct Version {
        array<T, B> value{};
        array<int, B> pos;
        Version() { pos.fill(-1); }
    };
    vector<Version> pre{ 1 };
    PrefixLinearBasis() = default;
    explicit PrefixLinearBasis(const vector<T> &a) {
        pre.reserve(a.size() + 1);
        for (T x : a) push(x);
    }
    int size() const { return int(pre.size()) - 1; }
    void push(T x) {
        int position{ size() };
        Version cur{ pre.back() };
        for (int i = B - 1; i >= 0; -- i) if (x >> i & 1) {
            if (!cur.value[i]) {
                cur.value[i] = x;
                cur.pos[i] = position;
                break;
            }
            // 同一主元优先保留位置更靠后的向量
            if (cur.pos[i] < position) {
                swap(cur.value[i], x);
                swap(cur.pos[i], position);
            }
            x ^= cur.value[i];
        }
        pre.push_back(cur);
    }
    int range_rank(int l, int r) const {
        const Version &v{ pre[r] };
        int res{};
        for (int i = 0; i < B; ++ i) res += v.pos[i] >= l;
        return res;
    }
    bool range_contains(int l, int r, T x) const {
        const Version &v{ pre[r] };
        for (int i = B - 1; i >= 0; -- i) if (x >> i & 1) {
            if (v.pos[i] < l) return false;
            x ^= v.value[i];
        }
        return true;
    }
    T range_max(int l, int r, T x = 0) const {
        const Version &v{ pre[r] };
        for (int i = B - 1; i >= 0; -- i) {
            if (v.pos[i] >= l && (x ^ v.value[i]) > x) x ^= v.value[i];
        }
        return x;
    }
    T range_min(int l, int r, T x) const {
        const Version &v{ pre[r] };
        for (int i = B - 1; i >= 0; -- i) {
            if (v.pos[i] >= l && (x ^ v.value[i]) < x) x ^= v.value[i];
        }
        return x;
    }
};
```
##### 线段树结点信息（单点修改、区间查询）
---
直接接入已有 `SegTree<Info>`
```cpp
template<class Basis = LinearBasis<>>
struct BasisInfo {
    using T = typename Basis::value_type;
    Basis basis;
    BasisInfo() = default;
    explicit BasisInfo(T x) { basis.insert(x); }
    friend BasisInfo operator+(BasisInfo a, const BasisInfo &b) {
        a.basis.merge(b.basis);
        return a;
    }
};
```
##### 可撤销异或线性基（线段树分治 / DFS 回溯）
一次插入只会新增一个主元，历史中记录主元位置，`-1` 表示本次秩没有变化。
```cpp
template<class T = u64, int B = numeric_limits<T>::digits>
struct RollbackLinearBasis {
    static_assert(is_unsigned_v<T>);
    static_assert(1 <= B && B <= 64 && B <= numeric_limits<T>::digits);
    array<T, B> p{};
    int siz{ 0 };
    vector<int> hist;
    // ===== 修改与回滚 =====
    int snapshot() const { return int(hist.size()); }
    bool insert(T x) {
        if (siz == B) {
            hist.push_back(-1);
            return false;
        }
        for (int i = B - 1; i >= 0; -- i) if (x >> i & 1) {
            if (!p[i]) {
                p[i] = x;
                ++ siz;
                hist.push_back(i);
                return true;
            }
            x ^= p[i];
        }
        hist.push_back(-1);
        return false;
    }
    void rollback(int snap) {
        while (int(hist.size()) > snap) {
            int i{ hist.back() };
            hist.pop_back();
            if (i != -1) {
                p[i] = 0;
                -- siz;
            }
        }
    }
    void clear() {
        p.fill(0);
        siz = 0;
        hist.clear();
    }
    // ===== 基础查询 =====
    int rank() const { return siz; }
    bool contains(T x) const {
        for (int i = B - 1; i >= 0; -- i) if (x >> i & 1) {
            if (!p[i]) return false;
            x ^= p[i];
        }
        return true;
    }
    T max_xor(T x = 0) const {
        for (int i = B - 1; i >= 0; -- i) if ((x ^ p[i]) > x) x ^= p[i];
        return x;
    }
    T min_xor(T x) const {
        for (int i = B - 1; i >= 0; -- i) if ((x ^ p[i]) < x) x ^= p[i];
        return x;
    }
};
```
##### 带来源的线性基（最多 64 个输入元素）
---
除了“是否可表示”，还可恢复**一组实现指定 XOR 的下标集合**，以及一组非空 XOR 为 0 的线性关系。这一块与普通 `LinearBasis` 独立，按需复制。
```cpp
template<class T = u64, int B = numeric_limits<T>::digits>
struct WitnessLinearBasis {
    static_assert(is_unsigned_v<T>);
    static_assert(1 <= B && B <= 64 && B <= numeric_limits<T>::digits);
    array<T, B> p{};
    array<u64, B> from{};
    int n{0}, siz{0};
    // 成功插入返回 nullopt；相关时返回非空零异或下标掩码
    optional<u64> insert(T x) {
        assert(n < 64);
        u64 mask = 1ULL << n ++;
        for (int i = B - 1; i >= 0; -- i) if (x >> i & 1) {
            if (!p[i]) {
                p[i] = x;
                from[i] = mask;
                ++ siz;
                return nullopt;
            }
            x ^= p[i];
            mask ^= from[i];
        }
        return mask;
    }
    // nullopt = 不可表示；其他情况返回一组下标掩码（0 可由空集表示）
    optional<u64> represent(T x) const {
        u64 mask = 0;
        for (int i = B - 1; i >= 0; -- i) if (x >> i & 1) {
            if (!p[i]) return nullopt;
            x ^= p[i];
            mask ^= from[i];
        }
        return mask;
    }
    int rank() const { return siz; }
};
```
#### 接口说明
##### 普通 `LinearBasis<T,B>`
- `insert(x)`：插入，若提升秩返回 `true`；$O(B)$。
- `contains(x)`：是否可由某个子集异或得到；$O(B)$。
- `max_xor(x=0)` / `min_xor(x)`：$x\oplus y$ 的最大 / 最小值，$y$ 为任意可表示值；$O(B)$。
- `min_positive()`：最小**正**可表示值，秩为零时返回 `0`；$O(B)$。
- `rank()` / `clear()`：秩 $O(1)$、清空 $O(B)$。
- `basis()`：按最低到最高主元排序的规范基，**不修改原基**；$O(B^2)$。
- `kth(k)`：第 `k` 小**不同**可表示 XOR，`0base` 且包括 `0`；越界 `nullopt`；$O(B^2)$。
- `merge(other)`：张成空间的和 $V+W$，而非集合并去重；$O(B^2)$。
- `intersection(other)`：两个张成空间的交 $V\cap W$；$O(B^2)$。
##### 区间、线段树与回滚
- `PrefixLinearBasis(a)` / `push(x)`：建全部前缀 $O(nB)$ / 追加一个 $O(B)$。
- `range_rank(l,r)`、`range_contains(l,r,x)`、`range_max(l,r,x=0)`、`range_min(l,r,x)`：$O(B)$，全部是半开区间。
- `BasisInfo<Basis>()` / `BasisInfo<Basis>(x)`：空单位元 / 单元素；`operator+` 用 `merge()` 合并，$O(B^2)$。
- `RollbackLinearBasis::insert(x)`：每次都记录历史（包括无效插入）；$O(B)$。
- `snapshot()` / `rollback(snap)`：快照 $O(1)$，恢复 $O(\text{撤销的插入次数})$。
- `WitnessLinearBasis::insert(x)`：独立则 `nullopt`，依赖时返回一个**非空零异或子集掩码**；`represent(x)` 返回一组解的下标掩码，无法表示则 `nullopt`；均为 $O(B)$。
#### 注意事项
- `B` 表示参与运算的位数，输入必须 **`x < 2^B`**（`B=64` 时自然覆盖 `u64`）；代码不会自动检测高位截断。达到满秩时普通 `insert()` 可跳过后续插入，但**前缀版 `push()` 不可跳过**，因为主元时间戳还会变化。
- 秩为 $r$ 的异或空间有 $2^r$ 个不同值。原始有 $n$ 个带下标元素时，每个可达值恰好对应 $2^{n-r}$ 个下标子集（包括空集，包含重复元素也成立）。
- `kth(0)=0`，秩为 64 时 `u64` 的全部取值都可作为合法 `k`；若问**非空子集**第 `k` 小，需考虑存在非空 XOR 为 0 的子集时 `0` 的情况，不能直接套不同值排序。
- `min_xor(0)=0` 恒成立；求最小正可表示值调用 `min_positive()`。若原始元素非空，则最小**非空子集**异或值在 `n>rank()` 时为 `0`，否则是 `min_positive()`。
- `PrefixLinearBasis` 空区间只能表示 `0`，`range_max(l,l,x)=range_min(l,l,x)=x`，`range_rank(l,l)=0`。
- `merge` 与 `intersection` 只处理**张成空间**，不会记录原始元素个数或重复次数；合并后的第 `k` 小无法直接还原按子集计数的排列。
- `RollbackLinearBasis` 不能在回滚历史之间手动规范化主元或直接修改 `p`，否则仅记录一个主元位置不足以复原。
- `WitnessLinearBasis` 的掩码第 `i` 位对应第 `i` 次 `insert()`，只支持至多 64 个输入元素；`represent(0)` 合法返回 `0`（空集），要获得**非空**零异或集合应查看 `insert()` 的依赖返回。
- 如果需要解有 RHS 的 GF(2) 方程组，使用 11.1 的 `gauss_xor()`；如果只关心 XOR 张成空间，则本节更短。
#### 常见用法
##### 最大子集异或与第 k 小
```cpp
LinearBasis basis;
for (u64 x : a) basis.insert(x);
u64 mx = basis.max_xor();
auto kth = basis.kth(k); // 0base，包含 0
```
##### 区间查询和在线单点修改
```cpp
PrefixLinearBasis pre(a);
u64 ans = pre.range_max(l, r); // [l,r)
// 有在线修改时可用 BasisInfo<> 作为 SegTree 的 Info 类型。
```
##### 撤销 / 线段树分治
```cpp
RollbackLinearBasis lb;
int snap = lb.snapshot();
lb.insert(x);
lb.insert(y);
auto ans = lb.max_xor();
lb.rollback(snap);
```
##### 恢复表达方案和异或依赖
```cpp
WitnessLinearBasis wb;
auto dep0 = wb.insert(5); // nullopt
wb.insert(3);
auto dep = wb.insert(6);  // 0b111：5 ^ 3 ^ 6 = 0
auto mask = wb.represent(6); // 例如 0b011
```
##### 图上的最大路径异或（无向图）
从根 DFS 得到树路径异或 `dis[v]`，每条**非树边** `(u,v,w)` 插入 `dis[u] ^ dis[v] ^ w`，则从 `s` 到 `t` 的最大路径异或（允许重复经过边/点的**游走**）为：
```cpp
u64 ans = basis.max_xor(dis[s] ^ dis[t]);
```
若题目限定**简单路径**，不能直接套这条结论。

| 结构 / 操作 | 时间 | 空间 |
| --- | --- | --- |
| 普通 / 可撤销插入和查询 | $O(B)$ | $O(B)$；可撤销另有 $O(q)$ 历史 |
| 规范化 / 第 k 小 / 合并 / 求交 | $O(B^2)$ | $O(B)$ |
| 前缀线性基建表 | $O(nB)$ | $O(nB)$ |
| 前缀区间查询 | $O(B)$ | $O(1)$ |
