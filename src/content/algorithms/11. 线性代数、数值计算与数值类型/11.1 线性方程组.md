### 线性方程组
---
求解 $Ax=b$
#### 实数 Gauss–Jordan（特解、秩、零空间）
---
```cpp
struct RealLinearSolution {
    bool consistent;
    int rank;
    vector<ld> particular;
    vector<vector<ld>> nullspace;
};
RealLinearSolution gauss_real(vector<vector<ld>> a, int m, ld eps = 1E-12L) {
    int n = a.size();
    for (const auto &v : a) assert(int(v.size()) == m + 1);
    vector<int> where(m, -1);
    int r = 0;
    for (int col = 0; col < m && r < n; ++ col) {
        int p = r;
        for (int i = r + 1; i < n; ++ i) {
            if (abs(a[i][col]) > abs(a[p][col])) p = i;
        }
        if (abs(a[p][col]) <= eps) continue;
        swap(a[p], a[r]);
        where[col] = r;
        ld inv = 1 / a[r][col];
        for (int j = col; j <= m; ++ j) a[r][j] *= inv;
        for (int i = 0; i < n; ++ i) if (i != r) {
            ld t = a[i][col];
            if (abs(t) <= eps) continue;
            for (int j = col; j <= m; ++ j) a[i][j] -= t * a[r][j];
        }
        ++ r;
    }
    for (int i = 0; i < n; ++ i) {
        bool zero = true;
        for (int j = 0; j < m; ++ j) zero &= abs(a[i][j]) <= eps;
        if (zero && abs(a[i][m]) > eps) return {false, r, {}, {}};
    }
    vector<ld> x(m);
    for (int j = 0; j < m; ++ j) {
        if (where[j] != -1) x[j] = a[where[j]][m];
    }
    vector<vector<ld>> basis;
    for (int f = 0; f < m; ++ f) if (where[f] == -1) {
        vector<ld> v(m);
        v[f] = 1;
        for (int j = 0; j < m; ++ j) {
            if (where[j] != -1) v[j] = -a[where[j]][f];
        }
        basis.push_back(move(v));
    }
    return {true, r, move(x), move(basis)};
}
// 非空矩阵的便利接口；0 个方程时请显式传 m
RealLinearSolution gauss_real(vector<vector<ld>> a, ld eps = 1E-12L) {
    assert(!a.empty() && !a[0].empty());
    int m = int(a[0].size()) - 1;
    return gauss_real(move(a), m, eps);
}
```
### 素数模 Gauss–Jordan
---
```cpp
i64 linear_pow_mod(i64 a, i64 b, int mod) {
    i64 res = 1;
    for (; b; b >>= 1, a = a * a % mod) {
        if (b & 1) res = res * a % mod;
    }
    return res;
}
struct ModLinearSolution {
    bool consistent;
    int rank;
    vector<int> particular;
    vector<vector<int>> nullspace;
};
ModLinearSolution gauss_mod_prime(vector<vector<int>> a, int m, int mod) {
    assert(mod >= 2); // 必须为质数
    int n = a.size();
    for (auto &v : a) {
        assert(int(v.size()) == m + 1);
        for (int &x : v) x = (x % mod + (i64)mod) % mod;
    }
    vector<int> where(m, -1);
    int r = 0;
    for (int col = 0; col < m && r < n; ++ col) {
        int p = r;
        while (p < n && a[p][col] == 0) ++ p;
        if (p == n) continue;
        swap(a[p], a[r]);
        where[col] = r;
        i64 inv = linear_pow_mod(a[r][col], mod - 2, mod);
        for (int j = col; j <= m; ++ j) a[r][j] = a[r][j] * inv % mod;
        for (int i = 0; i < n; ++ i) if (i != r && a[i][col]) {
            i64 t = a[i][col];
            for (int j = col; j <= m; ++ j) {
                a[i][j] = (a[i][j] - t * a[r][j]) % mod;
                if (a[i][j] < 0) a[i][j] += mod;
            }
        }
        ++ r;
    }
    for (int i = r; i < n; ++ i) {
        if (a[i][m]) return {false, r, {}, {}};
    }
    vector<int> x(m);
    for (int j = 0; j < m; ++ j) {
        if (where[j] != -1) x[j] = a[where[j]][m];
    }
    vector<vector<int>> basis;
    for (int f = 0; f < m; ++ f) if (where[f] == -1) {
        vector<int> v(m);
        v[f] = 1;
        for (int j = 0; j < m; ++ j) if (where[j] != -1) {
            int t = a[where[j]][f];
            v[j] = t ? mod - t : 0;
        }
        basis.push_back(move(v));
    }
    return {true, r, move(x), move(basis)};
}
// 与原接口兼容；非空矩阵时从增广矩阵推断变量数。
ModLinearSolution gauss_mod_prime(vector<vector<int>> a, int mod) {
    assert(!a.empty() && !a[0].empty());
    int m = int(a[0].size()) - 1;
    return gauss_mod_prime(move(a), m, mod);
}
```
#### GF(2) / 异或方程组（`bitset`）
---
每行 `bit [0,m)` 为系数，`bit m` 为常数项；额外位必须为零。
```cpp
template<size_t M>
struct XorLinearSolution {
    bool consistent;
    int rank;
    bitset<M> particular;
    vector<bitset<M>> nullspace;
};
template<size_t M>
XorLinearSolution<M> gauss_xor(vector<bitset<M>> a, int m) {
    assert(0 <= m && m < int(M));
    int n = a.size(), r = 0;
    vector<int> where(m, -1);
    for (int col = 0; col < m && r < n; ++ col) {
        int p = r;
        while (p < n && !a[p][col]) ++ p;
        if (p == n) continue;
        swap(a[p], a[r]);
        where[col] = r;
        for (int i = 0; i < n; ++ i) {
            if (i != r && a[i][col]) a[i] ^= a[r];
        }
        ++ r;
    }
    for (int i = r; i < n; ++ i) {
        if (a[i][m]) return {false, r, {}, {}};
    }
    bitset<M> x;
    for (int j = 0; j < m; ++ j) {
        if (where[j] != -1) x[j] = a[where[j]][m];
    }
    vector<bitset<M>> basis;
    for (int f = 0; f < m; ++ f) if (where[f] == -1) {
        bitset<M> v;
        v[f] = 1;
        for (int j = 0; j < m; ++ j) {
            if (where[j] != -1) v[j] = a[where[j]][f];
        }
        basis.push_back(v);
    }
    return {true, r, x, move(basis)};
}
```
### 三对角线性方程组（Thomas / 追赶法）
---
```cpp
optional<vector<ld>> tridiagonal(vector<ld> lower,
        vector<ld> diagonal, vector<ld> upper, vector<ld> rhs,
        ld eps = 1E-18L) {
    int n = diagonal.size();
    assert(int(rhs.size()) == n);
    assert(int(lower.size()) == max(0, n - 1));
    assert(int(upper.size()) == max(0, n - 1));
    if (!n) return vector<ld>{};
    for (int i = 1; i < n; ++ i) {
        if (abs(diagonal[i - 1]) <= eps) return nullopt;
        ld t = lower[i - 1] / diagonal[i - 1];
        diagonal[i] -= t * upper[i - 1];
        rhs[i] -= t * rhs[i - 1];
    }
    if (abs(diagonal.back()) <= eps) return nullopt;
    vector<ld> x(n);
    x.back() = rhs.back() / diagonal.back();
    for (int i = n - 2; i >= 0; -- i) {
        x[i] = (rhs[i] - upper[i] * x[i + 1]) / diagonal[i];
    }
    return x;
}
```
#### 接口说明
四种算法均为 `0base`

| 接口 | 输入 | 返回 |
| --- | --- | --- |
| `gauss_real(a,m,eps)` | `n × (m+1)` 实数增广矩阵 | `consistent, rank, particular, nullspace` |
| `gauss_mod_prime(a,m,mod)` | `n × (m+1)` 模质数矩阵 | 同上，数值在 `[0,mod)` |
| `gauss_xor<M>(a,m)` | `bitset<M>` 行，`M>=m+1` | 同上，特解和零空间基为 `bitset<M>` |
| `tridiagonal(lower,diagonal,upper,rhs,eps)` | 下/主/上对角与右端 | `optional<vector<ld>>` |
- 若方程组有解，`particular` 是**自由变量设为零**时的一组特解，`nullspace` 有 `m-rank` 个向量，全部解为 $x=x_0+\sum_i c_i v_i$。
- 无解时 `consistent=false`，只保证 `rank` 有效；其他结果为空。
- 素数模域 $\mathbf F_p$ 中，若有解则解数为 $p^{m-r}$；GF(2) 中为 $2^{m-r}$。
- Thomas 数组：`lower[i]=A[i+1][i]`，`diagonal[i]=A[i][i]`，`upper[i]=A[i][i+1]`，`rhs[i]=b[i]`。
- `gauss_real(a,eps)` / `gauss_mod_prime(a,mod)` 是方便旧代码接入的非空矩阵重载；零方程、`m>0` 时一定使用带 `m` 的主接口。
#### 注意事项
- **浮点**：实数版采用列部分选主元，但使用绝对 `eps` 判断；系数数量级差异很大时需重新调整阈值，最好用原始 $A,b$ 检查残差。近奇异的秩与零空间基不保证高精度。
- **素数模**：必须验证题目的 `mod` 是质数；合数模的非零元素不保证有逆元。整数模的中间乘法在 `mod <= INT_MAX` 下由 `i64` 承担。
- **GF(2)**：`M` 是编译期容量（包含常数项），`m` 是实际变量数；`m` 及以上的额外未使用 bit 不能含有数据。
- **Thomas**：没有行交换；`nullopt` 只表示本追赶法遇到过小主元，**不等价于原系统无解**。若矩阵非严格对角占优，可能需要带选主元的带状消元。循环三对角不能直接使用。
- 只要 `rank(A)=rank([A|b])` 就有解；若又有 `rank(A)=m`，解唯一。零空间维数为 `m-r`。
- 若仅需秩或存在性，可删除三个 Gauss–Jordan 算法末尾构造 `particular/nullspace` 的代码，减少返回结果空间。
- 最小二乘和最小范数不由本节的“自由变量设为零”保证，使用 11.5 的 QR。
#### 常见用法
##### 实数解空间和唯一性
```cpp
vector<vector<ld>> a{{1, 1, 1}}; // x+y=1
auto sol = gauss_real(a, 2);
if (!sol.consistent) cout << "No solution\n";
else if (sol.rank == 2) cout << "Unique\n";
else cout << "Multiple\n";
// particular={1,0}, nullspace[0]={-1,1}
```
##### 素数模方程组
```cpp
vector<vector<int>> a{{1, 1, 3}, {1, 2, 5}};
auto sol = gauss_mod_prime(a, 998244353);
// consistent=true，particular={1,2}
```
##### GF(2) 开关类问题
```cpp
vector<bitset<101>> a(1);
a[0][0] = a[0][2] = a[0][4] = 1;
a[0][5] = 1; // x0 ^ x2 ^ x4 = 1，m=5
auto sol = gauss_xor(a, 5);
```
##### O(n) 解三对角系统
```cpp
auto x = tridiagonal({-1, -1}, {2, 2, 2},
                     {-1, -1}, {1, 0, 1});
// 有解时 x={1,1,1}
```
#### 复杂度

| 算法 | 时间 | 额外空间（含副本与输出） |
| --- | --- | --- |
| 实数 / 素数模消元 | $O(nm\min(n,m)+m(m-r))$，模数版另有 $O(r\log p)$ 逆元 | $O(nm+m(m-r))$ |
| GF(2) 消元 | 约 $O(nm\lceil M/w\rceil +m(m-r)\lceil M/w\rceil)$ 字操作 | $O((n+m-r)\lceil M/w\rceil)$ 字 |
| Thomas | $O(n)$ | $O(n)$ |
