### 矩阵求逆、行列式、最小二乘与最小范数
---
#### 实数矩阵求逆与行列式
---
```cpp
optional<vector<vector<ld>>> inverse_real(vector<vector<ld>> a, ld eps = 1E-12L) {
    int n = a.size();
    for (const auto &row : a) assert(int(row.size()) == n);
    vector aug(n, vector<ld>(2 * n));
    for (int i = 0; i < n; ++ i) {
        for (int j = 0; j < n; ++ j) aug[i][j] = a[i][j];
        aug[i][n + i] = 1;
    }
    for (int k = 0; k < n; ++ k) {
        int p = k;
        for (int i = k + 1; i < n; ++ i) {
            if (abs(aug[i][k]) > abs(aug[p][k])) p = i;
        }
        if (abs(aug[p][k]) <= eps) return nullopt;
        swap(aug[p], aug[k]);
        ld inv = 1 / aug[k][k];
        for (int j = 0; j < 2 * n; ++ j) aug[k][j] *= inv;
        for (int i = 0; i < n; ++ i) if (i != k) {
            ld t = aug[i][k];
            if (t == 0) continue;
            for (int j = 0; j < 2 * n; ++ j) aug[i][j] -= t * aug[k][j];
        }
    }
    vector res(n, vector<ld>(n));
    for (int i = 0; i < n; ++ i) {
        for (int j = 0; j < n; ++ j) res[i][j] = aug[i][n + j];
    }
    return res;
}
ld determinant_real(vector<vector<ld>> a, ld eps = 1E-12L) {
    int n = a.size();
    for (const auto &row : a) assert(int(row.size()) == n);
    ld det = 1;
    for (int k = 0; k < n; ++ k) {
        int p = k;
        for (int i = k + 1; i < n; ++ i) {
            if (abs(a[i][k]) > abs(a[p][k])) p = i;
        }
        if (abs(a[p][k]) <= eps) return 0;
        if (p != k) swap(a[p], a[k]), det = -det;
        ld pivot = a[k][k];
        det *= pivot;
        for (int i = k + 1; i < n; ++ i) {
            ld t = a[i][k] / pivot;
            for (int j = k + 1; j < n; ++ j) a[i][j] -= t * a[k][j];
            a[i][k] = 0;
        }
    }
    return det;
}
```
#### 素数模矩阵求逆与行列式
---
```cpp
i64 matrix_pow_mod(i64 a, i64 b, int mod) {
    i64 res = 1;
    for (; b; b >>= 1, a = a * a % mod) {
        if (b & 1) res = res * a % mod;
    }
    return res;
}
optional<vector<vector<int>>> inverse_mod_prime(vector<vector<int>> a, int mod) {
    assert(mod >= 2); // 必须是质数
    int n = a.size();
    for (const auto &row : a) assert(int(row.size()) == n);
    vector aug(n, vector<int>(2 * n));
    for (int i = 0; i < n; ++ i) {
        for (int j = 0; j < n; ++ j) {
            aug[i][j] = (a[i][j] % mod + (i64)mod) % mod;
        }
        aug[i][n + i] = 1;
    }
    for (int k = 0; k < n; ++ k) {
        int p = k;
        while (p < n && !aug[p][k]) ++ p;
        if (p == n) return nullopt;
        swap(aug[p], aug[k]);
        i64 inv = matrix_pow_mod(aug[k][k], mod - 2, mod);
        for (int j = 0; j < 2 * n; ++ j) aug[k][j] = aug[k][j] * inv % mod;
        for (int i = 0; i < n; ++ i) if (i != k && aug[i][k]) {
            i64 t = aug[i][k];
            for (int j = 0; j < 2 * n; ++ j) {
                aug[i][j] = (aug[i][j] - t * aug[k][j]) % mod;
                if (aug[i][j] < 0) aug[i][j] += mod;
            }
        }
    }
    vector res(n, vector<int>(n));
    for (int i = 0; i < n; ++ i) {
        for (int j = 0; j < n; ++ j) res[i][j] = aug[i][n + j];
    }
    return res;
}
int determinant_mod_prime(vector<vector<int>> a, int mod) {
    assert(mod >= 2); // 必须是质数
    int n = a.size();
    for (auto &row : a) {
        assert(int(row.size()) == n);
        for (int &x : row) x = (x % mod + (i64)mod) % mod;
    }
    i64 det = 1;
    for (int k = 0; k < n; ++ k) {
        int p = k;
        while (p < n && !a[p][k]) ++ p;
        if (p == n) return 0;
        if (p != k) swap(a[p], a[k]), det = (mod - det) % mod;
        i64 pivot = a[k][k];
        det = det * pivot % mod;
        i64 inv = matrix_pow_mod(pivot, mod - 2, mod);
        for (int i = k + 1; i < n; ++ i) if (a[i][k]) {
            i64 t = a[i][k] * inv % mod;
            for (int j = k + 1; j < n; ++ j) {
                a[i][j] = (a[i][j] - t * a[k][j]) % mod;
                if (a[i][j] < 0) a[i][j] += mod;
            }
            a[i][k] = 0;
        }
    }
    return det;
}
```
#### 任意模数行列式（无除逆元）
---
当模数可能为合数时，使用整数欧几里得式消元，不直接求非零主元的逆元。
```cpp
int determinant_mod_any(vector<vector<int>> a, int mod) {
    assert(mod >= 2);
    int n = a.size();
    for (auto &row : a) {
        assert(int(row.size()) == n);
        for (int &x : row) x = (x % mod + (i64)mod) % mod;
    }
    i64 det = 1;
    for (int k = 0; k < n; ++ k) {
        for (int i = k + 1; i < n; ++ i) {
            while (a[i][k]) {
                i64 t = a[k][k] / a[i][k];
                for (int j = k; j < n; ++ j) {
                    a[k][j] = (a[k][j] - t * a[i][j]) % mod;
                    if (a[k][j] < 0) a[k][j] += mod;
                }
                swap(a[k], a[i]);
                det = (mod - det) % mod;
            }
        }
        det = det * a[k][k] % mod;
    }
    return det;
}
```
#### Householder QR（满列秩，高矩阵）
---
`HouseholderQR` 和 `householder_qr()` 是下面两个求解器的**公共依赖**。如果只需要求逆或行列式，整段 QR 都可删除。
```cpp
struct HouseholderQR {
    int n, m;
    vector<vector<ld>> r, v;
    // Q^T x：反射从前往后应用
    vector<ld> apply_qt(vector<ld> x) const {
        assert(int(x.size()) == n);
        for (int k = 0; k < m; ++ k) {
            ld dot = 0;
            for (int i = k; i < n; ++ i) dot += v[k][i - k] * x[i];
            for (int i = k; i < n; ++ i) x[i] -= 2 * dot * v[k][i - k];
        }
        return x;
    }
    // Q x：反射从后往前应用
    vector<ld> apply_q(vector<ld> x) const {
        assert(int(x.size()) == n);
        for (int k = m - 1; k >= 0; -- k) {
            ld dot = 0;
            for (int i = k; i < n; ++ i) dot += v[k][i - k] * x[i];
            for (int i = k; i < n; ++ i) x[i] -= 2 * dot * v[k][i - k];
        }
        return x;
    }
};
optional<HouseholderQR> householder_qr(vector<vector<ld>> a,
        ld eps = 1E-12L) {
    int n = a.size(), m = n ? int(a[0].size()) : 0;
    assert(n >= m);
    for (const auto &row : a) assert(int(row.size()) == m);
    ld scale = 0;
    for (const auto &row : a) {
        for (ld x : row) scale = max(scale, abs(x));
    }
    HouseholderQR qr{n, m, {}, vector<vector<ld>>(m)};
    for (int k = 0; k < m; ++ k) {
        ld norm = 0;
        for (int i = k; i < n; ++ i) norm = hypot(norm, a[i][k]);
        if (norm <= eps * scale) return nullopt;
        ld alpha = a[k][k] >= 0 ? -norm : norm;
        vector<ld> v(n - k);
        for (int i = k; i < n; ++ i) v[i - k] = a[i][k];
        v[0] -= alpha;
        ld vn = 0;
        for (ld x : v) vn = hypot(vn, x);
        for (ld &x : v) x /= vn;
        for (int j = k; j < m; ++ j) {
            ld dot = 0;
            for (int i = k; i < n; ++ i) dot += v[i - k] * a[i][j];
            for (int i = k; i < n; ++ i) a[i][j] -= 2 * dot * v[i - k];
        }
        a[k][k] = alpha;
        for (int i = k + 1; i < n; ++ i) a[i][k] = 0;
        qr.v[k] = move(v);
    }
    qr.r = move(a);
    return qr;
}
struct LeastSquaresSolution {
    bool full_column_rank;
    vector<ld> x;
    ld residual_norm;
};
LeastSquaresSolution least_squares_qr(const vector<vector<ld>> &a,
        const vector<ld> &b, ld eps = 1E-12L) {
    int n = a.size(), m = n ? int(a[0].size()) : 0;
    assert(int(b.size()) == n && n >= m);
    auto qr = householder_qr(a, eps);
    if (!qr) return {false, {}, 0};
    vector<ld> y = qr->apply_qt(b), x(m);
    for (int i = m - 1; i >= 0; -- i) {
        ld s = y[i];
        for (int j = i + 1; j < m; ++ j) s -= qr->r[i][j] * x[j];
        x[i] = s / qr->r[i][i];
    }
    ld residual = 0;
    for (int i = m; i < n; ++ i) residual = hypot(residual, y[i]);
    return {true, move(x), residual};
}
struct MinimumNormSolution {
    bool full_row_rank;
    vector<ld> x;
};
MinimumNormSolution minimum_norm_qr(const vector<vector<ld>> &a,
        const vector<ld> &b, ld eps = 1E-12L) {
    int n = a.size(), m = n ? int(a[0].size()) : 0;
    assert(int(b.size()) == n && n <= m);
    vector at(m, vector<ld>(n));
    for (int i = 0; i < n; ++ i) {
        for (int j = 0; j < m; ++ j) at[j][i] = a[i][j];
    }
    auto qr = householder_qr(move(at), eps);
    if (!qr) return {false, {}};
    vector<ld> z(m);
    // R^T y = b，R^T 是下三角
    for (int i = 0; i < n; ++ i) {
        ld s = b[i];
        for (int j = 0; j < i; ++ j) s -= qr->r[j][i] * z[j];
        z[i] = s / qr->r[i][i];
    }
    return {true, qr->apply_q(move(z))};
}
```
#### 接口说明

| 接口 | 功能 | 前提 / 失败含义 |
| --- | --- | --- |
| `inverse_real(a,eps)` | 求实数逆矩阵 | 方阵；`nullopt` = 当前阈值下无法判定可逆 |
| `determinant_real(a,eps)` | 实数行列式 | 方阵，若主元过小返回 `0` |
| `matrix_pow_mod(a,b,p)` | 模快速幂 | `0<=a<mod`，`b>=0`，`mod>=2` |
| `inverse_mod_prime(a,p)` | 素数模逆矩阵 | 奇异时 `nullopt` |
| `determinant_mod_prime(a,p)` | 素数模行列式 | `p` 必须是质数 |
| `determinant_mod_any(a,m)` | 任意模数行列式 | `m>=2`，不依赖逆元 |
| `householder_qr(a,eps)` | 高矩阵 Householder QR | `n>=m` 且数值上满列秩；否则 `nullopt` |
| `least_squares_qr(a,b,eps)` | 最小化 $\|Ax-b\|_2$ | `n>=m`，必须满列秩 |
| `minimum_norm_qr(a,b,eps)` | 在 $Ax=b$ 中最小化 $\|x\|_2$ | `n<=m`，必须满行秩 |
`LeastSquaresSolution::residual_norm` 是残差的二范数；求解失败时 `x` 为空，残差无效。所有输入矩阵按值或 `const&` 传递，不修改调用者原数据。
#### 注意事项
- **求解 `Ax=b` 不应先求逆矩阵。** 若只是解方程，直接用 11.1 的高斯消元。
- 浮点逆矩阵、行列式的 `eps` 是绝对判零阈值；QR 的 `eps` 按输入最大元素的尺度进行近似判秩。所有浮点结果都应结合输入尺度、残差检验；病态矩阵仍可能误差较大。
- 素数模实现要求 `2 <= mod <= INT_MAX` 且模数为**质数**。模合数时求行列式可使用 `determinant_mod_any()`；模合数的逆矩阵不能直接按素数模算法求。
- 空方阵的行列式定义为 `1`，逆矩阵返回空矩阵；`determinant_mod_any()` 对模 `m` 返回 `1 % m`。
- 不要把最小二乘、最小范数混淆：高斯消元中“自由变量置零”未必是最小范数解。例如 $x+y=1$ 的最小范数解是 $(1/2,1/2)$。
- QR 本版只处理数值上**满秩**的对应矩阵：一般秩亏最小二乘、伪逆及最小范数最小二乘解需要带列主元 QR 或 SVD。
- `n=0` 时仅接受零列数的空矩阵；无法从空 `vector` 推断正的变量数。QR 满列秩还需确保输入中没有 `NaN/Inf`。
- **仅看行列式是否靠近零**不适合作为一般数值判秩方法；数量级过大或过小时，行列式本身可能上下溢。
- 若对整数矩阵要求**精确整数值**的行列式，不能由浮点结果四舍五入保证正确；应考虑 Bareiss（中间结果需防溢）或多模数 CRT。
##### 可直接复用的恒等式
- 满列秩最小二乘：$A^TAx=A^Tb$；满行秩最小范数：$x=A^T(AA^T)^{-1}b$。这两式仅用于推导，数值计算建议采用 QR，因为 $\kappa(A^TA)=\kappa(A)^2$（二范数条件数）。
- 矩阵行列式引理（域上 $A$ 可逆）：$\det(A+uv^T)=\det(A)(1+v^TA^{-1}u)$。
- Sherman–Morrison：若 $1+v^TA^{-1}u\ne0$，则 $(A+uv^T)^{-1}=A^{-1}-\dfrac{A^{-1}uv^TA^{-1}}{1+v^TA^{-1}u}$。更新逆矩阵时应先保存原矩阵对应的两侧向量。
#### 常见用法
##### 模行列式、矩阵树依赖
```cpp
int d = determinant_mod_prime({{1, 2}, {3, 4}}, 998244353); // p-2
int d2 = determinant_mod_any({{1, 2}, {3, 4}}, 1000);      // 998
```
##### 实数最小二乘：拟合直线 `y = ax+b`
```cpp
vector<vector<ld>> a(n, vector<ld>(2));
vector<ld> b(n);
for (int i = 0; i < n; ++ i) {
    a[i] = {xs[i], 1};
    b[i] = ys[i];
}
auto sol = least_squares_qr(a, b);
if (sol.full_column_rank) {
    ld slope = sol.x[0], intercept = sol.x[1];
    ld error = sol.residual_norm;
}
```
##### 最小范数解
```cpp
vector<vector<ld>> a{{1, 1}};
vector<ld> b{1};
auto sol = minimum_norm_qr(a, b);
// sol.x 近似为 {0.5, 0.5}
```
#### 复杂度
设 $A$ 为 $n\times m$，其中行列式与求逆时 $n=m$

| 接口 | 时间 | 额外空间 |
| --- | --- | --- |
| 实数 / 素数模求逆 | $O(n^3+n\log p)$（实数无幂项） | $O(n^2)$ |
| 实数 / 素数模行列式 | $O(n^3+n\log p)$ | $O(n^2)$ |
| 任意模数行列式 | $O(n^3\log mod)$（上界） | $O(n^2)$ |
| Householder QR、最小二乘（$n\ge m$） | $O(nm^2)$ | $O(nm)$ |
| 最小范数（$n\le m$，对 $A^T$ 分解） | $O(mn^2)$ | $O(mn)$ |
