### 数值积分、方程与单纯形
---
#### 方程求根（实根、二分、Newton）
---
```cpp
constexpr ld EPS = 1E-12L;
int sgn(ld x, ld eps = EPS) { return (x > eps) - (x < -eps); }
// 返回所有不同的实根（重根只返回一次）；恒等式 0=0 的情形无穷多根，返回空数组。
vector<ld> quadratic(ld a, ld b, ld c) {
    if (sgn(a) == 0) {
        if (sgn(b) == 0) return {};
        return {-c / b};
    }
    ld d = b * b - 4 * a * c;
    if (sgn(d) < 0) return {};
    if (sgn(d) == 0) return {-b / (2 * a)};
    ld root = sqrt(max((ld)0, d));
    ld q = -0.5L * (b + copysign(root, b));
    ld x = q / a, y = c / q;
    if (x > y) swap(x, y);
    return {x, y};
}
// 要求 [l,r] 两端异号或至少一端接近 0；只能保证找到一个根。
template<class F>
optional<ld> bisect_root(F f, ld l, ld r, ld eps = EPS,
        int iters = 120) {
    assert(l <= r);
    ld fl = f(l), fr = f(r);
    if (abs(fl) <= eps) return l;
    if (abs(fr) <= eps) return r;
    if ((fl < 0) == (fr < 0)) return nullopt;
    for (int it = 0; it < iters; ++ it) {
        ld m = midpoint(l, r), fm = f(m);
        if (abs(fm) <= eps || m == l || m == r) return m;
        if ((fl < 0) == (fm < 0)) l = m, fl = fm;
        else r = m;
    }
    return midpoint(l, r);
}
// 系数按最高次到常数项；给出 [l,r] 内全部数值可分辨的不同实根。
// 仅适用于低次数、系数尺度适中且不会在区间上发生严重数值消去的多项式。
vector<ld> polynomial_real_roots(vector<ld> a, ld l, ld r,
        ld eps = EPS) {
    assert(l <= r);
    while (a.size() > 1 && abs(a.front()) <= eps) a.erase(a.begin());
    int d = int(a.size()) - 1;
    if (d <= 0) return {};
    auto f = [&](ld x) {
        ld ans = 0;
        for (ld c : a) ans = ans * x + c;
        return ans;
    };
    if (d == 1) {
        ld x = -a[1] / a[0];
        return l - eps <= x && x <= r + eps ? vector<ld>{clamp(x, l, r)} : vector<ld>{};
    }
    vector<ld> der(d);
    for (int i = 0; i < d; ++ i) der[i] = a[i] * (d - i);
    vector<ld> critical = polynomial_real_roots(der, l, r, eps);
    vector<ld> points{l};
    for (ld x : critical) {
        if (x > l && x < r) points.push_back(x);
    }
    points.push_back(r);
    vector<ld> roots;
    auto add = [&](ld x) {
        if (roots.empty() || abs(roots.back() - x) > 16 * eps) {
            roots.push_back(clamp(x, l, r));
        }
    };
    for (int i = 0; i < int(points.size()); ++ i) {
        ld x = points[i], fx = f(x);
        if (abs(fx) <= eps) add(x); // 同时处理偶重根（导数根）
        if (i + 1 == int(points.size())) break;
        ld y = points[i + 1], fy = f(y);
        if ((fx < -eps && fy > eps) || (fx > eps && fy < -eps)) {
            auto root = bisect_root(f, x, y, eps);
            if (root) add(*root);
        }
    }
    return roots;
}
template<class F, class DF>
optional<ld> newton(F f, DF df, ld x, int iters = 100,
        ld eps = EPS) {
    for (int it = 0; it < iters; ++ it) {
        ld y = f(x), d = df(x);
        if (!isfinite(y) || !isfinite(d)) return nullopt;
        if (abs(y) <= eps) return x;
        if (abs(d) <= eps) return nullopt;
        ld nx = x - y / d;
        if (!isfinite(nx)) return nullopt;
        if (abs(nx - x) <= eps * max((ld)1, abs(nx))) {
            if (abs(f(nx)) <= eps) return nx;
        }
        x = nx;
    }
    return abs(f(x)) <= eps ? optional<ld>(x) : nullopt;
}
```
#### 自适应 Simpson 积分（缓存端点 / 中点值）
---
```cpp
template<class F>
ld adaptive_simpson_impl(F &f, ld l, ld r, ld fl, ld fm, ld fr, ld whole, ld eps, int depth) {
    ld m = midpoint(l, r);
    ld lm = midpoint(l, m), rm = midpoint(m, r);
    ld flm = f(lm), frm = f(rm);
    ld left = (fl + 4 * flm + fm) * (m - l) / 6;
    ld right = (fm + 4 * frm + fr) * (r - m) / 6;
    ld delta = left + right - whole;
    if (depth == 0 || m == l || m == r || abs(delta) <= 15 * eps) {
        return left + right + delta / 15;
    }
    return adaptive_simpson_impl(f, l, m, fl, flm, fm, left, eps / 2, depth - 1)
         + adaptive_simpson_impl(f, m, r, fm, frm, fr, right, eps / 2, depth - 1);
}
template<class F>
ld adaptive_simpson(F f, ld l, ld r, ld eps = 1E-10L, int max_depth = 25) {
    if (l > r) return -adaptive_simpson(f, r, l, eps, max_depth);
    if (l == r) return 0;
    ld fl = f(l), fr = f(r), fm = f(midpoint(l, r));
    ld whole = (fl + 4 * fm + fr) * (r - l) / 6;
    return adaptive_simpson_impl(f, l, r, fl, fm, fr, whole, eps, max_depth);
}
```
#### Romberg 积分（适合足够光滑的函数）
---
```cpp
template<class F>
ld romberg(F f, ld l, ld r, ld eps = 1E-10L, int max_level = 20) {
    assert(0 <= max_level && max_level <= 25);
    if (l > r) return -romberg(f, r, l, eps, max_level);
    if (l == r) return 0;
    vector<ld> prev{(f(l) + f(r)) * (r - l) / 2};
    ld trapezoid = prev[0];
    for (int level = 1; level <= max_level; ++ level) {
        int cnt = 1 << (level - 1);
        ld h = (r - l) / (1 << level), sum = 0;
        for (int i = 0; i < cnt; ++ i) sum += f(l + (2 * i + 1) * h);
        trapezoid = trapezoid / 2 + h * sum;
        vector<ld> cur(level + 1);
        cur[0] = trapezoid;
        ld p4 = 1;
        for (int j = 1; j <= level; ++ j) {
            p4 *= 4;
            cur[j] = cur[j - 1] + (cur[j - 1] - prev[j - 1]) / (p4 - 1);
        }
        if (abs(cur.back() - prev.back()) <= eps) return cur.back();
        prev = move(cur);
    }
    return prev.back();
}
```
#### 两阶段单纯形（最大化标准型）
---
求 $\max c^Tx$，满足 $Ax\le b,\ x\ge0$。返回 `-INF` = 不可行，`INF` = 无界，有限数 = 最优值
```cpp
struct LinearProgram {
    static constexpr ld EPS = 1E-10L;
    static constexpr ld INF = numeric_limits<ld>::infinity();
    int n, m;
    vector<int> basic, nonbasic;
    vector<vector<ld>> t;

    LinearProgram(const vector<vector<ld>> &a, const vector<ld> &b,
            const vector<ld> &c)
        : n(b.size()), m(c.size()), basic(n), nonbasic(m + 1),
          t(n + 2, vector<ld>(m + 2)) {
        for (const auto &row : a) assert(int(row.size()) == m);
        assert(int(a.size()) == n);
        for (int i = 0; i < n; ++ i) {
            basic[i] = m + i;
            t[i][m] = -1;
            t[i][m + 1] = b[i];
            for (int j = 0; j < m; ++ j) t[i][j] = a[i][j];
        }
        for (int j = 0; j < m; ++ j) {
            nonbasic[j] = j;
            t[n][j] = -c[j];
        }
        nonbasic[m] = -1;
        t[n + 1][m] = 1;
    }
    // ===== 内部转轴与两阶段迭代 =====
    void pivot(int r, int s) {
        ld inv = 1 / t[r][s];
        for (int i = 0; i < n + 2; ++ i) if (i != r) {
            for (int j = 0; j < m + 2; ++ j) if (j != s) {
                t[i][j] -= t[r][j] * t[i][s] * inv;
            }
        }
        for (int j = 0; j < m + 2; ++ j) if (j != s) t[r][j] *= inv;
        for (int i = 0; i < n + 2; ++ i) if (i != r) t[i][s] *= -inv;
        t[r][s] = inv;
        swap(basic[r], nonbasic[s]);
    }
    bool simplex(int phase) {
        int obj = phase == 1 ? n + 1 : n;
        while (true) {
            int s = -1;
            for (int j = 0; j <= m; ++ j) {
                if (phase == 2 && nonbasic[j] == -1) continue;
                if (s == -1 || t[obj][j] < t[obj][s] - EPS
                    || (abs(t[obj][j] - t[obj][s]) <= EPS
                        && nonbasic[j] < nonbasic[s])) s = j;
            }
            if (s == -1 || t[obj][s] >= -EPS) return true;
            int r = -1;
            for (int i = 0; i < n; ++ i) {
                if (t[i][s] <= EPS) continue;
                if (r == -1) r = i;
                else {
                    ld left = t[i][m + 1] / t[i][s];
                    ld right = t[r][m + 1] / t[r][s];
                    if (left < right - EPS || (abs(left - right) <= EPS
                            && basic[i] < basic[r])) r = i;
                }
            }
            if (r == -1) return false;
            pivot(r, s);
        }
    }
    // 只调用一次 solve；需要新题 / 新约束应重新构造对象
    ld solve(vector<ld> &x) {
        x.clear();
        if (m == 0) {
            for (int i = 0; i < n; ++ i) if (t[i][1] < -EPS) return -INF;
            return 0;
        }
        int r = 0;
        for (int i = 1; i < n; ++ i) {
            if (t[i][m + 1] < t[r][m + 1]) r = i;
        }
        if (n && t[r][m + 1] < -EPS) {
            pivot(r, m);
            if (!simplex(1) || t[n + 1][m + 1] < -EPS || abs(t[n + 1][m + 1]) > EPS) return -INF;
            auto it = find(basic.begin(), basic.end(), -1);
            if (it != basic.end()) {
                r = it - basic.begin();
                int s = 0;
                for (int j = 1; j <= m; ++ j) {
                    if (abs(t[r][j]) > abs(t[r][s])) s = j;
                }
                if (abs(t[r][s]) > EPS) pivot(r, s);
            }
        }
        if (!simplex(2)) return INF;
        x.assign(m, 0);
        for (int i = 0; i < n; ++ i) {
            if (basic[i] < m && basic[i] >= 0) x[basic[i]] = t[i][m + 1];
        }
        return t[n][m + 1];
    }
};
```
#### Kahan 补偿求和
---
```cpp
template<class Range>
ld kahan_sum(const Range &values) {
    ld sum = 0, c = 0;
    for (auto raw : values) {
        ld y = (ld)raw - c;
        ld t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}
```
#### 接口说明

| 接口 | 返回 / 约定 | 复杂度 |
| --- | --- | --- |
| `quadratic(a,b,c)` | 二次 / 退化一次方程的不同实根（升序） | $O(1)$ |
| `bisect_root(f,l,r,eps,iters)` | 存在异号端点时返回一个根的近似值；否则 `nullopt` | $O(I)$ 次函数求值 |
| `polynomial_real_roots(a,l,r,eps)` | 系数**降幂**排列，返回区间内不同实根 | 最坏约 $O(d^3 I)$ |
| `newton(f,df,x0,iters,eps)` | 单点 Newton 迭代；失败 `nullopt` | $O(I)$ |
| `adaptive_simpson(f,l,r,eps,max_depth)` | 定积分近似 | 自适应，最多 $O(2^D)$ 求值 |
| `romberg(f,l,r,eps,max_level)` | Romberg 积分近似 | $O(2^K+K^2)$ |
| `LinearProgram(a,b,c).solve(x)` | 标准型 LP 的最优值及一组 `x`；±∞ 两类失败 | 每转轴 $O(nm)$ |
| `kahan_sum(range)` | 按容器迭代顺序的补偿求和 | $O(N)$ |
`EPS` 是**绝对**误差阈值。多项式实根算法默认有限区间，且目标函数低次数、尺度适中；`l==r` 时只检查该端点
#### 注意事项
- `quadratic()` 采用防消去误差公式；判别式 `b²-4ac` 仍可能发生舍入或溢出。对于 `a=b=0` 返回空数组，无法用该返回区分“无根”和“恒等式”。
- 多项式实根通过**导数根分段**寻找单调区间，并检查驻点来覆盖偶重根；接近的根可能因 `eps` 合并，接近零的函数值可能产生伪根，无法保证高次数和病态系数下全部正确。
- `bisect_root()` 不要求函数单调，但要求区间连续并有变号信息；切线重根通常没有变号。`newton()` 只保证局部迭代，没有全局收敛保证。
- 自适应 Simpson 的 `eps` 是整体绝对误差预算，已知的断点、奇点应先分段；**达到深度上限不代表误差已满足要求**。对窄峰与非光滑函数，采样可能漏掉贡献。
- Romberg 对足够光滑函数有效；`max_level=25` 已可能需要约 $2^{25}$ 次函数求值，不要无约束调高。
- 单纯形只接受 $Ax\le b, x\ge0$ 的最大化形式。自由变量拆为 $x^+-x^-$；`=` 拆成两条 `<=`；`>=` 两侧同乘 `-1`；最小化用负目标函数转换。
- 单纯形采用浮点比较与防退化的编号平局规则，不提供精确整数最优值的保证；最坏转轴次数为指数级。有限返回值才可读取 `x`，调用对象应每组数据重建。
- `m=0` 的 LP 特判为空变量问题；`n=0` 且有变量时，目标存在正系数则无界，否则最优值为 `0`。
- Kahan 不能修复输入转换已丢失的精度，也不能消除 `NaN/Inf`。
- 单峰函数极值搜索复用 [[2.11 二分、三分与黄金分割]] 的 `golden_search`；0/1 分数规划可二分 $\lambda$ 并判断 $\max\sum_i(a_i-\lambda b_i)\ge0$（必须按原题选取约束下的方案，且分母为正）。
#### 常见用法
##### 求积分与全部低次数实根
```cpp
auto integral = adaptive_simpson([](ld x) { return exp(-x * x); }, 0, 1);
auto roots = polynomial_real_roots({1, 0, -1, 0}, -2, 2);
// x^3-x，在 [-2,2] 上的实根为 -1、0、1
```
##### 线性规划
```cpp
vector<vector<ld>> a{{1, 1}, {1, 0}, {0, 1}};
vector<ld> b{4, 2, 3}, c{3, 2}, x;
LinearProgram lp(a, b, c);
ld optimum = lp.solve(x);
if (optimum == -LinearProgram::INF) cout << "Infeasible\n";
else if (optimum == LinearProgram::INF) cout << "Unbounded\n";
else cout << optimum << '\n'; // 最优值 10，x={2,2}
```
#### 复杂度

| 算法 | 时间 / 求值次数 | 空间 |
| --- | --- | --- |
| 求根、Newton | 见接口表，递归多项式求根约 $O(d^3 I)$ | $O(d^2)$ |
| 自适应 Simpson | 取决于采样与误差，最多指数于 `max_depth` | $O(D)$ 调用栈 |
| Romberg | $O(2^K+K^2)$ | $O(K)$ |
| 单纯形 | 每转轴 $O(nm)$，转轴次数最坏指数级 | $O(nm)$ |
| Kahan | $O(N)$ | $O(1)$ |
