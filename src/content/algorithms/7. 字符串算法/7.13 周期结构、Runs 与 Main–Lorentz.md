### 周期结构、Runs 与 Main–Lorentz
---
#### Runs：枚举所有极大周期段
```cpp
struct Run {
    int period, l, r;
    int length() const { return r - l; }
};
template<class Seg>
vector<Run> enumerate_runs(const Seg &src) {
    using T = decay_t<decltype(*begin(src))>;
    vector<T> s(begin(src), end(src));
    int n{ int(s.size()) };
    // by_period[p] 暂存周期 p 对应的极大候选区间
    vector<vector<pair<int, int>>> by_period(n + 1);
    // 处理跨过左右分界线的周期段
    auto crossing = [&](const vector<T> &left, const vector<T> &right) {
        vector<tuple<int, int, int>> res;
        int nl{ int(left.size()) }, nr{ int(right.size()) };
        vector<T> a{ left }, b{ right };
        ranges::reverse(a);
        // 不需要分隔符：后续查询长度会被 nl / nr 截断
        b.insert(b.end(), left.begin(), left.end());
        b.insert(b.end(), right.begin(), right.end());
        auto zl = z_function(a);
        auto zr = z_function(b);
        for (int p = 1; p <= nl; ++ p) {
            int x{ p == nl ? p : min(p + zl[p], nl) };
            int y{ min(zr[nl + nr - p], nr) };
            if (x + y >= 2 * p) res.push_back({ p, x, y });
        }
        return res;
    };
    // ===== Main-Lorentz 风格分治：左右递归 + 处理跨中点周期 =====
    auto dfs = [&](auto self, int l, int r) -> void {
        if (r - l <= 1) return;
        int m{ (l + r) / 2 };
        self(self, l, m);
        self(self, m, r);
        vector<T> left(s.begin() + l, s.begin() + m);
        vector<T> right(s.begin() + m, s.begin() + r);
        for (auto [p, x, y] : crossing(left, right)) {
            by_period[p].push_back({ m - x, m + y });
        }
        ranges::reverse(left);
        ranges::reverse(right);
        for (auto [p, x, y] : crossing(right, left)) {
            by_period[p].push_back({ m - y, m + x });
        }
    };
    dfs(dfs, 0, n);
    // ===== 去除同周期下被包含的区间，并保留每个 Run 的最小周期 =====
    vector<Run> res;
    set<pair<int, int>> used;
    for (int p = 1; p <= n; ++ p) {
        auto &v = by_period[p];
        sort(v.begin(), v.end(), [](const auto &a, const auto &b) {
            if (a.first != b.first) return a.first < b.first;
            return a.second > b.second;
        });
        int mx{ -1 };
        for (auto [l, r] : v) {
            if (r <= mx) continue;
            mx = r;
            // 周期从小到大扫描，因此同一极大区间第一次出现即为最小周期
            if (used.insert({ l, r }).second) res.push_back({ p, l, r });
        }
    }
    return res;
}
```
#### Main–Lorentz：压缩表示所有相邻重复 `XX`
```cpp
struct TandemGroup {
    int period, l, r;
    // 对任意 start ∈ [l, r)：
    // s[start..start + 2 * period) = X + X
    int count() const { return r - l; }
};
template<class Seg>
vector<TandemGroup> main_lorentz(const Seg &src) {
    using T = decay_t<decltype(*begin(src))>;
    vector<T> a(begin(src), end(src));
    vector<TandemGroup> res;
    auto get_z = [](const vector<int> &z, int i) {
        return 0 <= i && i < int(z.size()) ? z[i] : 0;
    };
    // 用 optional 的 nullopt 作为绝不会与普通字符相等的分隔符
    auto with_sep = [](const vector<T> &x, const vector<T> &y) {
        vector<optional<T>> res;
        res.reserve(x.size() + y.size() + 1);
        for (const auto &v : x) res.emplace_back(v);
        res.emplace_back(nullopt);
        for (const auto &v : y) res.emplace_back(v);
        return res;
    };
    // 对固定半长 period，把所有合法 l1 压成连续的起点区间
    auto emit = [&](int shift, bool left, int center, int period, int k1, int k2) {
        int lo{ max(1, period - k2) }, hi{ min(period, k1) };
        // 防止与递归子问题重复统计
        if (left && hi == period) -- hi;
        if (lo > hi) return;
        int l, r;
        if (left) {
            l = shift + center - hi;
            r = shift + center - lo + 1;
        } else {
            l = shift + center - period - hi + 1;
            r = shift + center - period - lo + 2;
        }
        res.push_back({ period, l, r });
    };
    // ===== Main-Lorentz 分治 =====
    auto dfs = [&](auto self, const vector<T> &s, int shift) -> void {
        int n = s.size();
        if (n <= 1) return;
        int nl{ n / 2 }, nr{ n - nl };
        vector<T> left(s.begin(), s.begin() + nl);
        vector<T> right(s.begin() + nl, s.end());
        vector<T> rleft(left.rbegin(), left.rend());
        vector<T> rright(right.rbegin(), right.rend());
        self(self, left, shift);
        self(self, right, shift + nl);
        auto z1 = z_function(rleft);
        auto z2 = z_function(with_sep(right, left));
        auto z3 = z_function(with_sep(rleft, rright));
        auto z4 = z_function(right);
        for (int center = 0; center < n; ++ center) {
            int period, k1, k2;
            if (center < nl) {
                // 中心落在左半部分
                period = nl - center;
                k1 = get_z(z1, nl - center);
                k2 = get_z(z2, nr + 1 + center);
            } else {
                // 中心落在右半部分
                period = center - nl + 1;
                k1 = get_z(z3, n - (center - nl));
                k2 = get_z(z4, center - nl + 1);
            }
            if (k1 + k2 >= period) {
                emit(shift, center < nl, center, period, k1, k2);
            }
        }
    };
    dfs(dfs, a, 0);
    return res;
}
```
#### 基本定义
##### 周期
对于字符串区间：
```text
s[l..r)
```
长度：
$$L=r-l$$
若对于所有：
$$l+p\le i<r$$
都有：
$$s[i]=s[i-p]$$
则称 `p` 是该字符串的一个**周期**
等价地：
$$\boxed{ p\text{ 是周期} \iff s[l..r-p)=s[l+p..r) }$$
注意：
> 周期不要求整除字符串长度

例如：
```text
abababa
```
长度为 `7`，但 `2` 仍然是它的周期
##### 最小周期
所有周期中最小的一个称为：
$$p_{\min}$$
字符串的指数定义为：
$$⁡\boxed{ \operatorname{exp}(s)=\frac{|s|}{p_{\min}} }$$
指数不要求为整数
例如：
```text
ababa
```
最小周期为 `2`，指数为：
$$\frac52$$
##### Tandem repetition / Square
若：
$$S=XX$$
即两个完全相同的字符串相邻出现，则称其为 tandem repetition / square
若 `|X| = p`，则区间长度为：
$$2p$$
例如：
```text
abcabc
```
是半长为 `3` 的 tandem repetition
##### Run
一个 Run 表示：
```text
(period, l, r)
```
满足：
1. `[l,r)` 的**最小周期**为 `period`
2. 至少重复两次：
$$r-l\ge2\cdot period$$
3. 不能继续向左保持该周期：  
$$l=0 \quad\text{或}\quad s[l-1]\ne s[l-1+period]$$
4. 不能继续向右保持该周期：
$$r=n \quad\text{或}\quad s[r]\ne s[r-period]$$
也是：
$$\boxed{\text{Run = 极大的指数至少为 2 的周期段}}$$
#### 接口说明
##### `Run`
```cpp
struct Run {
    int period, l, r;
};
```
表示：
```text
s[l..r)
```
是一个 Run，最小周期为 `period`
- `length()`：返回 `r - l`
指数为：
```cpp
double exponent = 1.0 * run.length() / run.period;
```
比较指数时通常不需要浮点数
##### `enumerate_runs(s)`
```cpp
auto runs = enumerate_runs(s);
```
返回字符串所有 Runs
每个结果：
```cpp
run.period
run.l
run.r
```
满足：
```text
s[run.l..run.r)
```
是一个极大周期段
结果按照 `period` 非降序产生
##### `TandemGroup`
```cpp
struct TandemGroup {
    int period, l, r;
};
```
表示对于：
```cpp
for (int start = l; start < r; ++ start)
```
都有：
```text
s[start .. start + period) == s[start + period .. start + 2 * period)
```
也就是说，一个 `TandemGroup` 压缩了一批：
```text
XX
```
重复
##### `main_lorentz(s)`
返回所有 tandem repetition 的压缩表示：
```cpp
auto groups = main_lorentz(s);
```
不会直接展开所有答案
这是必要的，因为所有 tandem repetition 的数量本身可能达到：
$$\Theta(n^2)$$
#### 注意事项
- 所有区间均为 `0base` 的 `[l,r)`
- `Run::period` 是**最小周期**
- `TandemGroup::period` 是 `XX` 中单个 `X` 的长度，不保证它是 `XX` 的最小周期
- `enumerate_runs()` 只需要元素支持相等比较
- `main_lorentz()` 使用 `optional<T>` 构造安全分隔符，因此同样不要求字符集中预留特殊字符
- 两份算法都是静态算法；字符串修改后需要重新计算
- Run 的数量是线性的，但所有普通周期子串、所有 tandem repetition 的数量都可能是平方级
- 因此不要看到 Runs 数量少，就错误地认为“展开所有周期子串”也是 $O(n)$
- 如果题目只判断一个指定 `p` 是否为某区间周期，直接用哈希 / LCP 等工具通常更简单
- 如果需要枚举整个字符串的周期结构、最大重复段、最大指数等，Runs 才真正有优势
#### 常见用法
##### 枚举所有 Runs
```cpp
auto runs = enumerate_runs(s);
for (auto [p, l, r] : runs) {
    // s[l..r) 的最小周期为 p
    // 且 [l,r) 已无法继续向左右扩展
}
```
##### 求最大指数
Run 的指数为：
$$\frac{r-l}{p}$$
如果只比较大小，不需要浮点数：
```cpp
auto runs = enumerate_runs(s);
Run best{ 1, 0, 0 };
for (auto run : runs) {
    if (1LL * run.length() * best.period > 1LL * best.length() * run.period) {
        best = run;
    }
}
```
若 `runs` 可能为空，应单独处理
##### 判断是否存在指数至少为 `k` 的重复段
```cpp
bool ok{ false };
for (auto [p, l, r] : enumerate_runs(s)) {
    if (r - l >= k * p) {
        ok = true;
        break;
    }
}
```
因为任何指数至少为 `2` 的周期子串，都可以沿其周期继续向两边扩展，最终落入某个极大周期结构中
##### 最长 Run
```cpp
auto runs = enumerate_runs(s);
int ans{ 0 };
for (auto [p, l, r] : runs) {
    ans = max(ans, r - l);
}
```
得到最长的极大重复周期段长度
##### 按周期分类
```cpp
vector<vector<pair<int, int>>> by_period(s.size() + 1);
for (auto [p, l, r] : enumerate_runs(s)) {
    by_period[p].push_back({ l, r });
}
```
之后可以快速处理：
- 某周期 `p` 出现在哪些极大周期段
- 对周期长度加权
- 对不同周期做离线统计
##### Main–Lorentz：统计所有 `XX` 数量
不需要真正展开答案：
```cpp
auto groups = main_lorentz(s);
i64 ans{ 0 };
for (auto g : groups) {
    ans += g.count();
}
```
这样即使真实 `XX` 数量达到平方级，也仍然可以直接求总数，而不用保存每个区间
##### Main–Lorentz：求最长 `XX`
对于一个 `TandemGroup`，其中所有重复长度都是：
$$2\cdot period$$
因此：
```cpp
int ans{ 0 };
for (auto g : main_lorentz(s)) {
    ans = max(ans, 2 * g.period);
}
```
##### Main–Lorentz：找到任意一个 `XX`
```cpp
auto groups = main_lorentz(s);
if (!groups.empty()) {
    auto [p, l, r] = groups[0];
    int start{ l };
    // s[start..start + p) == s[start + p..start + 2p)
}
```
##### 展开所有 tandem repetition
只有题目真的要求输出所有答案时才这样做：
```cpp
for (auto [p, l, r] : main_lorentz(s)) {
    for (int start = l; start < r; ++ start) {
        int L{ start }, R{ start + 2 * p };
        // s[L..R) = XX
    }
}
```
复杂度必须计入输出量：
$$O(n\log n+K)$$
其中 `K` 是实际 tandem repetition 数量，最坏可为：
$$\Theta(n^2)$$
#### 二级结论
##### 周期与 Border
长度为 `n` 的字符串具有周期 `p`，当且仅当它存在长度：
$$n-p$$
的 Border
即：
$$\boxed{ p\text{ 是周期} \iff s[0..n-p)=s[p..n) }$$
因此：
```text
period p
<->
border n - p
```
是同一个结构的两种描述
##### 前缀函数与最小周期
设：
```cpp
pi[n - 1]
```
为整个字符串最长真 Border 长度
则：
$$\boxed{ p=n-\pi[n-1] }$$
是字符串的最小周期
但若要求字符串恰好由某个块完整重复：
```text
X X X ... X
```
还需要：
$$n\bmod p=0$$
所以要区分：
```text
最小周期
```
与：
```text
最短循环节长度
```
后者通常还要求整除
##### Fine–Wilf 定理
若长度为 `L` 的字符串同时具有周期 `p` 和 `q`，且：
$$L\ge p+q-\gcd(p,q)$$
则：
$$\boxed{ \gcd(p,q) }$$
也是它的周期
这是周期题中非常重要的结构定理
它说明：
> 一个足够长的字符串如果同时表现出两种周期性，那么两种周期会被强制合并成更小的公共周期
##### 为什么 Runs 能压缩周期结构
普通周期子串可能极多
例如：
```text
aaaaaaaa...
```
存在平方级数量的周期子串和 tandem repetition
但是极大周期段 Run 的数量满足：
$$\boxed{\#Runs=O(n)}$$
事实上 Runs Theorem 给出了更强的线性上界
因此很多题可以把：
```text
枚举所有周期子串
```
改写成：
```text
枚举所有 Runs
+
在每个 Run 内做数学统计
```
这也是 Runs 最核心的价值
##### Tandem repetition 与 Run
一个：
$$XX$$
必然具有周期：
$$|X|$$
可以沿这个周期向左右继续扩展，直到无法继续，于是得到一个极大周期区域
因此：
> tandem repetition 是局部重复； 
> Run 是把这种重复尽可能向两边扩展后的极大结构

区别在于：
- 一个 Run 可以压缩大量 tandem repetitions
- Run 只保留最小周期
- Main–Lorentz 可以直接描述所有具体 `XX`
所以通常：

| 需求        | 优先算法             |
| --------- | ---------------- |
| 极大周期段     | Runs             |
| 最大指数      | Runs             |
| 存在长重复周期结构 | Runs             |
| 枚举所有 `XX` | Main–Lorentz     |
| 统计所有 `XX` | Main–Lorentz 压缩组 |
| 单次判断某区间周期 | Hash / LCP       |
##### 一个 Run 内固定最小周期的基本重复
设 Run：
```text
(p,l,r)
```
长度：
$$L=r−l$$
那么最基本的长度 `2p` tandem repetition 的起点可以位于：
$$l,l+1,\ldots,r-2p$$
共有：
$$\boxed{ L-2p+1 }$$
个位置
但如果要统计**所有可能半长**的 tandem repetitions，不能简单只用这个公式；不同半长和局部更小周期会产生更复杂的情况，直接使用 `main_lorentz()` 更稳妥
#### Main–Lorentz 思想快速回忆
将当前串分成：
```text
u | v
```
递归解决：
```text
完全位于 u 中的重复
完全位于 v 中的重复
```
只剩下：
```text
跨越中点的重复
```
对于固定候选周期 `p`：
- 向左能匹配多少：LCS
- 向右能匹配多少：LCP
如果：
$$LCS+LCP\ge p$$
就存在长度至少 `2p` 的重复结构
LCP / LCS 可以通过若干 Z 函数一次性在线性时间求完，因此每层分治总复杂度为：
$$O(n)$$
递归深度：
$$O(\log n)$$
最终：
$$\boxed{O(n\log n)}$$

##### Main–Lorentz
构造压缩组：
```cpp
main_lorentz(s)
```
时间：
$$O(n\log n)$$
压缩组数量为：
$$O(n\log n)$$
若只做：
- 计数
- 求最长 repetition
- 判断是否存在
无需展开全部答案，仍保持 $O(n\log n)$
若展开所有 tandem repetitions：
$$O(n\log n+K)$$
其中：
$$K=\text{实际输出数量}$$
且最坏：
$$K=\Theta(n^2)$$
