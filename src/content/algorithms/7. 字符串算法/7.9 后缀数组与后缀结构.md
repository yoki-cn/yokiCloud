### SA（后缀数组）
---
```cpp
struct SuffixArray {
    struct MinRMQ {
        static constexpr int B{ 64 };
        using u64 = unsigned long long;
        int n{};
        vector<u64> stk;
        vector<vector<int>> st;
        void build(const vector<int> &a) {
            n = a.size();
            stk.assign(n, 0);
            st.clear();
            if (!n) return;
            int m{ (n + B - 1) / B };
            st.assign(__lg(m) + 1, vector<int>(m));
            for (int b = 0; b < m; ++ b) {
                int l{ b * B }, r{ min(n, l + B) };
                st[0][b] = *min_element(a.begin() + l, a.begin() + r);
                u64 s{};
                for (int i = l; i < r; ++ i) {
                    while (s && a[i] < a[l + __lg(s)]) s ^= 1ULL << __lg(s);
                    s |= 1ULL << (i - l);
                    stk[i] = s;
                }
            }
            for (int k = 1; k < st.size(); ++ k) {
                for (int i = 0; i + (1 << k) <= m; ++ i) {
                    st[k][i] = min(st[k - 1][i], st[k - 1][i + (1 << (k - 1))]);
                }
            }
        }
        int small(const vector<int> &a, int l, int r) const {
            return a[l + __builtin_ctzll(stk[r - 1] >> (l - l / B * B))];
        }
        int query(const vector<int> &a, int l, int r) const {
            if (l / B == (r - 1) / B) return small(a, l, r);
            int lb{ l / B + 1 }, rb{ (r - 1) / B };
            int ans{ min(small(a, l, lb * B), small(a, rb * B, r)) };
            if (lb < rb) {
                int k{ __lg(rb - lb) };
                ans = min({ ans, st[k][lb], st[k][rb - (1 << k)] });
            }
            return ans;
        }
    };
    int n{};
    vector<int> sa, rk, lc;
    MinRMQ rmq;
    // ===== 基础构造：构造 sa / rk / lc，核心代码不建议删除 =====
    template<class Seq>
    explicit SuffixArray(const Seq &a) {
        n = a.size();
        sa.resize(n);
        rk.resize(n);
        lc.assign(n ? n - 1 : 0, 0);
        if (!n) return;
        ranges::iota(sa, 0);
        ranges::sort(sa, [&](int i, int j) { return a[i] < a[j]; });
        rk[sa[0]] = 0;
        for (int i = 1; i < n; ++ i) rk[sa[i]] = rk[sa[i - 1]] + (a[sa[i - 1]] != a[sa[i]]);
        vector<int> tmp, cnt(n);
        tmp.reserve(n);
        for (int k = 1; rk[sa.back()] != n - 1; k <<= 1) {
            tmp.clear();
            for (int i = n - k; i < n; ++ i) tmp.push_back(i);
            for (int i : sa) if (i >= k) tmp.push_back(i - k);
            ranges::fill(cnt, 0);
            for (int x : rk) ++ cnt[x];
            for (int i = 1; i < n; ++ i) cnt[i] += cnt[i - 1];
            for (int x : views::reverse(tmp)) sa[-- cnt[rk[x]]] = x;
            swap(rk, tmp);
            rk[sa[0]] = 0;
            for (int i = 1; i < n; ++ i) {
                int x{ sa[i - 1] }, y{ sa[i] };
                int x2{ x + k < n ? tmp[x + k] : -1 };
                int y2{ y + k < n ? tmp[y + k] : -1 };
                rk[y] = rk[x] + (tmp[x] != tmp[y] || x2 != y2);
            }
        }
        // Kasai：由相邻后缀在线性时间内构造 LCP 数组
        for (int i = 0, k = 0; i < n; ++ i) {
            if (!rk[i]) {
                k = 0;
                continue;
            }
            if (k) -- k;
            int j{ sa[rk[i] - 1] };
            while (i + k < n && j + k < n && a[i + k] == a[j + k]) ++ k;
            lc[rk[i] - 1] = k;
        }
        // 若删除 lcp() / cmp()，同步删除本行、rmq 和 MinRMQ
        rmq.build(lc);
    }
    // ===== LCP / 子串比较：不需要时可整段删除 =====
    int lcp(int i, int j) const {
        if (i == j) return n - i;
        if (i == n || j == n) return 0;
        int x{ rk[i] }, y{ rk[j] };
        if (x > y) swap(x, y);
        return rmq.query(lc, x, y);
    }
    // 比较原序列的 a[l1,r1) 与 a[l2,r2)，返回 -1 / 0 / 1
    int cmp(int l1, int r1, int l2, int r2) const {
        int n1{ r1 - l1 }, n2{ r2 - l2 };
        int k{ min({ lcp(l1, l2), n1, n2 }) };
        if (k == min(n1, n2)) {
            if (n1 == n2) return 0;
            return n1 < n2 ? -1 : 1;
        }
        return rk[l1 + k] < rk[l2 + k] ? -1 : 1;
    }
    // ===== 模式匹配：不依赖 RMQ；不需要时可整段删除 =====
    // a 必须与构造 SuffixArray 时使用的原序列相同
    template<class Seq, class Pat>
    pair<int, int> range(const Seq &a, const Pat &t) const {
        auto compare = [&](int p) {
            int m = t.size();
            for (int i = 0; i < m; ++ i) {
                if (p + i == n) return -1;
                if (a[p + i] != t[i]) return a[p + i] < t[i] ? -1 : 1;
            }
            return 0;
        };
        int l{ 0 }, r{ n };
        while (l < r) {
            int m{ l + r >> 1 };
            if (compare(sa[m]) < 0) l = m + 1;
            else r = m;
        }
        int L{ l };
        r = n;
        while (l < r) {
            int m{ l + r >> 1 };
            if (compare(sa[m]) <= 0) l = m + 1;
            else r = m;
        }
        return { L, l };
    }
};
```
#### 成员变量
- `n`：原序列长度
- `sa[i]`：字典序第 `i` 小的后缀起点，即后缀 `a[sa[i], n)`
- `rk[i]`：后缀 `a[i, n)` 在 `sa` 中的排名，满足 `sa[rk[i]] = i`
- `lc[i]`：后缀 `a[sa[i], n)` 与 `a[sa[i + 1], n)` 的最长公共前缀长度，长度为 `max(n - 1, 0)`
- `rmq`：`lc` 上的静态最小值结构，仅 `lcp()` / `cmp()` 使用
#### 接口
- `SuffixArray(a)`：构造 `sa / rk / lc` 并预处理 LCP 查询，$O(n\log n)$
    - `a` 需要支持 `size()`、`operator[]`
    - 元素需要支持 `<`、`==`、`!=`
    - 可直接用于 `string`、`vector<int>`、`vector<i64>` 等可比较序列
- `lcp(i, j)`：返回后缀 `a[i,n)` 与 `a[j,n)` 的最长公共前缀长度，$O(1)$
    - 要求 `0 <= i,j <= n`
    - `i = n` 或 `j = n` 表示空后缀
- `cmp(l1, r1, l2, r2)`：按字典序比较 `a[l1,r1)` 与 `a[l2,r2)`，返回 `-1 / 0 / 1`，$O(1)$
- `range(a, t)`：返回所有以模式序列 `t` 为前缀的后缀在 `sa` 中构成的区间 `[L,R)`，$O(|t|\log n)$
    - 出现次数为 `R - L`
    - `sa[L..R)` 即各次出现的起点
#### 注意事项
- 所有下标均为 `0-based`，区间均为左闭右开 `[l,r)`
- `sa` **不包含空后缀**，因此长度恒为 `n`
- 本板子的 `lc[i]` 表示 `LCP(sa[i], sa[i + 1])`；有些资料的 `height[i]` 定义为 `LCP(sa[i - 1], sa[i])`，下标相差一位
- `lcp(n, j)` 中的 `n` 只表示查询时的空后缀，不能访问 `rk[n]`
- `range(a, t)` 中传入的 `a` 必须与构造 `SuffixArray` 时的原序列一致；结构体故意不保存原序列副本，以减少空间和复制开销
- 空模式 `t` 的 `range()` 返回 `[0,n)`，即所有非空后缀；若题目把空串在原串中的出现次数定义为 `n + 1`，需要单独处理
- `range()` 得到的出现位置按**后缀字典序**排列；若题目要求位置从小到大，需要额外排序
- 序列元素的比较关系决定字典序；处理自定义类型时应保证比较关系一致
- 不要在构造后修改 `sa / rk / lc / rmq`
- 若只需要 `sa / rk / lc`：
    - 删除 `MinRMQ
    - 删除成员 `rmq`
    - 删除构造函数末尾的 `rmq.build(lc)`
    - 删除 `lcp()` / `cmp()`
    - `range()` 可以独立保留
#### 常见用法
##### 基础构造
```cpp
string s;
cin >> s;
SuffixArray sa(s);
for (int p : sa.sa) {
    cout << p << ' ';
}
cout << '\n';
```
##### 本质不同子串个数
长度为 `n` 的字符串共有 $\frac{n(n+1)}2$ 个子串计重；按后缀顺序加入时，`lc` 恰好统计与前一个后缀重复的前缀：
```cpp
i64 ans = 1LL * sa.n * (sa.n + 1) / 2;
for (int x : sa.lc) ans -= x;
cout << ans << '\n';
```
##### 最长重复子串
最长重复子串长度等于 `lc` 的最大值：
```cpp
int ans = 0;
for (int x : sa.lc) ans = max(ans, x);
cout << ans << '\n';
```
若还需要一个出现位置：
```cpp
int len = 0, pos = -1;
for (int i = 0; i < sa.lc.size(); ++ i) {
    if (sa.lc[i] > len) {
        len = sa.lc[i];
        pos = sa.sa[i];
    }
}
```
##### 查询模式串出现次数
```cpp
string t;
cin >> t;
auto [L, R] = sa.range(s, t);
cout << R - L << '\n';
```
判断是否出现：
```cpp
auto [L, R] = sa.range(s, t);
bool exists = L < R;
```
##### 枚举模式串出现位置
```cpp
auto [L, R] = sa.range(s, t);
vector<int> pos(sa.sa.begin() + L, sa.sa.begin() + R);
ranges::sort(pos);
```
##### 整数序列模式匹配
不需要手动转成字符串，也不要求值域连续：
```cpp
vector<int> a(n), t(m);
for (int &x : a) cin >> x;
for (int &x : t) cin >> x;
SuffixArray sa(a);
auto [L, R] = sa.range(a, t);
cout << R - L << '\n';
```
#### 复杂度
设原序列长度为 `n`，模式长度为 `m`

|操作|时间复杂度|
|---|--:|
|构造 `sa / rk`|$O(n\log n)$|
|构造 `lc`|$O(n)$|
|构造 LCP RMQ|$O\left(n+\frac n{64}\log\frac n{64}\right)$|
|`lcp(i,j)`|$O(1)$|
|`cmp(...)`|$O(1)$|
|`range(a,t)`|$O(m\log n)$|
|枚举 `range()` 的全部出现位置|$O(R-L)$|
核心 `sa / rk / lc` 空间复杂度为 $O(n)$
带 LCP RMQ 后总空间复杂度为
$$O\left(n+\frac n{64}\log\frac n{64}\right)$$
