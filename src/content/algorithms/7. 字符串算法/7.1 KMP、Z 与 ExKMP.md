### KMP
#### 函数版
```cpp
template<class Seg>
vector<int> kmp(const Seg &s) {
    int n = s.size();;
    vector<int> f(n + 1);
    for (int i = 1, j = 0; i < n; ++ i) {
        while (j && s[i] != s[j]) j = f[j];
        if (s[i] == s[j]) ++ j;
        f[i + 1] = j;
    }
    return f;
}
template<class Seg>
vector<int> match(const Seg &p, const Seg &s) {
    int n = p.size(), m = s.size();
    vector<int> f = kmp(p), pos;
    if (n == 0) return pos;
    for (int i = 0, j = 0; i < m; ++ i) {
        while (j && s[i] != p[j]) j = f[j];
        if (s[i] == p[j]) ++ j;
        if (j == n) {
            pos.push_back(i - n + 1);
            j = f[j];
        }
    }
    return pos;
}
template<class Seg>
int period(const Seg &s) {
    if (s.empty()) return 0;
    int n = s.size();
    auto f = kmp(s);
    return n - f[n];
}
template<class Seg>
int cycle(const Seg &s) {
    int n = s.size(), p{ period(s) };
    return p && n % p == 0 ? p : n;
}
template<class Seg>
vector<int> borders(const Seg &s) {
    auto f = kmp(s);
    vector<int> res;
    for (int x = f.back(); x; x = f[x]) res.push_back(x);
    ranges::reverse(res);
    return res;
}
```
- `kmp(s)`：求前缀函数，返回 `f`，`f[i]` 表示 `s[0..i)` 的最长相等真前后缀长度，$O(n)$
- `match(p, s)`：求模式串 `p` 在文本串 `s` 中出现的所有位置，返回 `0base` 下标，$O(n+m)$
- `period(s)`：返回字符串最小正周期，等于 `n - f[n]`，$O(n)$
- `cycle(s)`：返回字符串最小循环节长度；若不能由短串重复构成，则返回 `n`，$O(n)$
- `borders(s)`：返回 `s` 所有 border 的长度，升序排列，$O(n)$
#### 固定模式串版
```cpp
template<class Seg>
struct KMP {
    int n{ 0 };
    Seg p;
    vector<int> f;
    KMP() = default;
    KMP(const Seg &p_) : p(p_) { build(); }
    void build() {
        n = p.size();
        f.assign(n + 1, 0);
        for (int i = 1, j = 0; i < n; ++ i) {
            while (j && p[i] != p[j]) j = f[j];
            if (p[i] == p[j]) ++ j;
            f[i + 1] = j;
        }
    }
    vector<int> match(const Seg &s) const {
        int m = s.size();
        vector<int> pos;
        if (n == 0) return pos;
        for (int i = 0, j = 0; i < m; ++ i) {
            while (j && s[i] != p[j]) j = f[j];
            if (s[i] == p[j]) ++ j;
            if (j == n) {
                pos.push_back(i - n + 1);
                j = f[j];
            }
        }
        return pos;
    }
    int period() const { return n ? n - f.back() : 0; }
    int cycle() const {
        int q{ period() };
        return q && n % q == 0 ? q : n;
    }
    vector<int> borders() const {
        vector<int> res;
        for (int x = f.back(); x; x = f[x]) res.push_back(x);
        ranges::reverse(res);
        return res;
    }
};
```
- `KMP<T> kmp(p)`：以模式串 `p` 构造 KMP 匹配器，$O(n)$
- `kmp.build()`：重新构造当前 `p` 的前缀函数，$O(n)$
- `kmp.match(s)`：求当前模式串 `p` 在文本串 `s` 中出现的所有位置，$O(m)$
- `kmp.period()`：返回模式串 `p` 的最小正周期，$O(1)$
- `kmp.cycle()`：返回模式串 `p` 的最小循环节长度，$O(1)$
- `kmp.borders()`：返回模式串 `p` 的所有 border 长度，$O(n)$
#### 注意事项
- `f` 的长度是 `n + 1`，不是 `n`；字符串 `s` 的最小正周期为 `n - f[n]`
- `period(s)` 不要求整除 `n`；`cycle(s)` 才要求循环节长度整除 `n`
- 拼接法 `p + '#' + s` 也能查找子串，但分隔符必须保证不在字符集中；本板子用直接扫描，避免该问题
- `T` 只需要支持 `size()` 和 `operator[]`，可用于 `string`、`vector<int>` 等序列
- 若固定一个模式串匹配多个文本串，优先用结构体版；只临时求一次前缀函数或周期，优先用函数版
#### 常见用法
##### 统计本质不同子串数量
```cpp
string s;
cin >> s;
i64 ans{ 0 };
string cur;
for (char c : s) {
    string t = (cur += c);
    ranges::reverse(t);
    auto f = kmp(t);
    int mx{ *ranges::max_element(f) };
    ans += (int)cur.size() - mx;
}
cout << ans << '\n';
```
### Z 函数
```cpp
template<class Seg>
vector<int> z_function(const Seg &s) {
    int n = s.size();
    vector<int> z(n);
    for (int i = 1, l = 0, r = 0; i < n; ++ i) {
        if (i <= r) z[i] = min(r - i + 1, z[i - l]);
        while (i + z[i] < n && s[z[i]] == s[i + z[i]]) ++ z[i];
        if (i + z[i] - 1 > r) l = i, r = i + z[i] - 1;
    }
    if (n) z[0] = n;
    return z;
}
```
- `z_function(s)`：返回 `z` 数组，`z[i]` 表示 `s` 与 `s[i..]` 的最长公共前缀长度，$O(n)$
#### 注意事项
- 规定 `z[0] = s.size()`
- `Seq` 需要支持 `size()` 和 `operator[]`，如 `string`、`vector<int>`
- 若使用 `p + sep + s` 查找模式串，`sep` 必须不在原字符集中出现
- 若不想处理分隔符，优先使用 `ExKMP`
#### 常见用法
```cpp
auto z = z_function(s);
{
    int n = s.size();
    if (z[i] >= len) {} // 前缀在某位置出现, 即 s[0..len) == s[i..i+len)
    if (z[n - len] == len) {} // 判断前缀是否为后缀, 即 s[0..len) == s[n-len..n)
}
{// 求最小正周期
    int p{ n };
    for (int d = 1; d < n; ++ d) if (n % d == 0 && z[d] == n - d) {
        p = d;
        break;
    }
}
{ // 查找模式串 p 出现位置
    string t = p + '#' + s;
    auto z = z_function(t);
    vector<int> pos;
    int n = p.size();
    for (int i = n + 1; i < t.size(); ++ i) if (z[i] >= n) pos.push_back(i - n - 1);
}
```
### ExKMP
```cpp
template<class Seg>
vector<int> exkmp(const Seg &p, const Seg &s) {
    int n = p.size(), m = s.size();
    vector<int> z = z_function(p), ext(m);
    if (n == 0) return ext;
    for (int i = 0, l = 0, r = -1; i < m; ++ i) {
        if (i <= r) ext[i] = min(r - i + 1, z[i - l]);
        while (ext[i] < n && i + ext[i] < m && p[ext[i]] == s[i + ext[i]]) ++ ext[i];
        if (i + ext[i] - 1 > r) l = i, r = i + ext[i] - 1;
    }
    return ext;
}
```
- `exkmp(p, s)`：返回 `ext` 数组，`ext[i]` 表示 `p` 与 `s[i..]` 的最长公共前缀长度，$O(|p| + |s|)$
#### 注意事项
- 依赖 `z_function()`
- `ext.size() == s.size()`
- `Seq` 需要支持 `size()` 和 `operator[]`，如 `string`、`vector<int>`
- 若 `p` 为空串，当前实现返回全 `0`；算法竞赛中通常不讨论空模式串
- 只需要单串前缀与后缀 LCP 时，使用 `Z 函数` 即可
#### 常见用法
```cpp
auto ext = exkmp(p, s);
// 查找模式串出现位置
vector<int> pos;
for (int i = 0; i < s.size(); ++ i) if (ext[i] == p.size()) pos.push_back(i);
// 求模式串与文本每个后缀的 LCP
int lcp{ ext[i] }; // p 与 s[i..] 的 LCP
// 比较模式串和文本后缀
int l{ ext[i] };
if (l == p.size()) { // p 是 s[i..] 的前缀
} else if (i + l == s.size()) { // s[i..] 是 p 的前缀
} else if (p[l] < s[i + l]) { // p < s[i..]
} else { // p > s[i..]
}
```
