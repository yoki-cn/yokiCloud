### 圆关系与交点
**解决问题：** 判断点、直线、线段、两个圆之间的位置关系，并求有限个交点
```cpp
struct Circle {
    Point ctr;
    ld rad{ 0 };
};
enum class CirRel {
    SEPARATE,         // 外离
    EXTERNAL_TANGENT, // 外切
    INTERSECT,        // 相交于两点
    INTERNAL_TANGENT, // 内切
    CONTAIN,          // 内含但不内切
    COINCIDE          // 重合
};
// 返回点相对闭圆盘的位置：圆内 / 圆上 / 圆外为 -1 / 0 / 1
int pt_cir_rel(Point p, Circle cir) { return sgn(distance(p, cir.ctr) - cir.rad); }
CirRel cir_rel(Circle a, Circle b) {
    ld dis{ distance(a.ctr, b.ctr) };
    if (sgn(dis) == 0 && sgn(a.rad - b.rad) == 0) return CirRel::COINCIDE;
    int outer{ sgn(dis - a.rad - b.rad) }, inner{ sgn(dis - fabsl(a.rad - b.rad)) };
    if (outer > 0) return CirRel::SEPARATE;
    if (outer == 0) return CirRel::EXTERNAL_TANGENT;
    if (inner > 0) return CirRel::INTERSECT;
    if (inner == 0) return CirRel::INTERNAL_TANGENT;
    return CirRel::CONTAIN;
}
// 返回 0 / 1 / 2 个交点；顺序沿 ln.dir 方向递增
vector<Point> line_cir_int(Line ln, Circle cir) {
    Point foot{ proj(cir.ctr, ln) };
    ld rem{ cir.rad * cir.rad - norm2(foot - cir.ctr) };
    if (sgn(rem) < 0) return {};
    if (sgn(rem) == 0) return {foot};
    Point unit{ ln.dir / length(ln.dir) }, off{ unit * sqrtl(max(0.L, rem)) };
    return {foot - off, foot + off};
}
// 返回 0 / 1 / 2 个交点；重合圆因交点无穷多而返回空数组
vector<Point> cir_int(Circle a, Circle b) {
    Point dir{ b.ctr - a.ctr };
    ld dis{ length(dir) };
    if (sgn(dis) == 0) return {};
    ld x{ (a.rad * a.rad - b.rad * b.rad + dis * dis) / (2 * dis) };
    ld h2{ a.rad * a.rad - x * x };
    if (sgn(h2) < 0) return {};
    Point mid{ a.ctr + dir * (x / dis) };
    if (sgn(h2) == 0) return { mid };
    Point off{ rotate90(dir) * (sqrtl(max(0.L, h2)) / dis) };
    return { mid - off, mid + off };
}
// 判断闭线段是否与闭圆盘有公共点
bool seg_disk(Segment seg, Circle cir) { return sgn(dis_seg(cir.ctr, seg) - cir.rad) <= 0; }
// 判断闭线段是否与圆周有公共点；整段严格位于圆内时返回 false
bool seg_cir(Segment seg, Circle cir) {
    ld near{ dis_seg(cir.ctr, seg) };
    ld far{ max(distance(seg.a, cir.ctr), distance(seg.b, cir.ctr)) };
    return sgn(near - cir.rad) <= 0 && sgn(far - cir.rad) >= 0;
}
// 返回圆上离 p 最近、最远的点；p 等于圆心时两者均不唯一
optional<pair<Point, Point>> near_far(Point p, Circle cir) {
    Point dir{ p - cir.ctr };
    if (sgn(norm2(dir)) == 0) return nullopt;
    dir = dir / length(dir);
    return pair(cir.ctr + dir * cir.rad, cir.ctr - dir * cir.rad);
}
```
#### 接口说明
- `pt_cir_rel(p,cir)`：返回 `-1 / 0 / 1`，分别表示点在圆内、圆上、圆外，$O(1)$
- `cir_rel(a,b)`：返回两圆六种关系之一，$O(1)$
- `line_cir_int(ln,cir)`：返回直线与圆周的有限交点；要求 `ln.dir` 非零，$O(1)$
- `cir_int(a,b)`：返回两圆周的有限交点；重合圆返回空数组，调用前可先用 `cir_rel()` 区分，$O(1)$
- `seg_disk()`：判断线段是否接触闭圆盘，$O(1)$
- `seg_cir()`：判断线段是否接触圆周，$O(1)$
- `near_far()`：返回最近点和最远点；查询点为圆心时返回 `nullopt`，$O(1)$
#### 注意事项
- `Circle::rad` 必须非负
- 直线与圆交点依赖投影，退化直线没有“沿方向的两个交点”语义，调用前保证 `ln.dir` 非零
- `cir_int()` 的空数组可能表示无交点，也可能表示重合圆；需要区分时先调用 `cir_rel()`
- `seg_disk()` 与 `seg_cir()` 语义不同：整条线段位于圆内时，前者为 `true`，后者为 `false`
- 查询点等于圆心时，圆上最近点和最远点都不唯一，可按题意任取 `ctr + Point(rad,0)`
#### 常见用法
```cpp
CirRel rel = cir_rel(a, b);
vector<Point> isecs = cir_int(a, b);
if (seg_cir(seg, cir)) {} // 线段碰到圆周
```
#### 复杂度
所有接口时间复杂度与额外空间均为 $O(1)$，不计返回数组
### 圆的切线与公切线
**解决问题：** 求圆外一点向圆引出的切线，以及两个圆的全部公切线
```cpp
// 返回 p 到 cir 的切点；圆内无切点，圆上一个，圆外两个
vector<Point> tan_pts(Point p, Circle cir) {
    Point dir{ p - cir.ctr };
    ld dis2{ norm2(dir) }, r2{ cir.rad * cir.rad };
    int rel{ sgn(dis2 - r2) };
    if (rel < 0) return {};
    if (rel == 0) return {p};
    Point base{ cir.ctr + dir * (r2 / dis2) };
    Point off{ rotate90(dir) * (cir.rad * sqrtl(max(0.L, dis2 - r2)) / dis2) };
    return { base + off, base - off };
}
// 返回 p 到 cir 的切线；圆内无切线，圆上一条，圆外两条
vector<Line> tan_line(Point p, Circle cir) {
    vector<Point> tans{ tan_pts(p, cir) };
    vector<Line> res;
    for (Point tan : tans) {
        if (same_pt(p, tan)) res.emplace_back(p, rotate90(p - cir.ctr));
        else res.push_back(Line::through(p, tan));
    }
    return res;
}
// 返回每条公切线在 a、b 上的切点；切点对可能重合
vector<pair<Point, Point>> com_tan_pts(Circle a, Circle b) {
    vector<pair<Point, Point>> res;
    Point dir{ b.ctr - a.ctr };
    ld dis2{ norm2(dir) };
    if (sgn(dis2) == 0) return res;
    // sign = 1：外公切线；sign = -1：内公切线
    for (int sign : {-1, 1}) {
        ld dr{ a.rad - sign * b.rad };
        ld h2{ dis2 - dr * dr };
        if (sgn(h2) < 0) continue;
        for (int side : {-1, 1}) {
            if (sgn(h2) == 0 && side == 1) continue;
            Point nor{ (dir * dr + rotate90(dir) * (sqrtl(max(0.L, h2)) * side)) / dis2 };
            res.push_back({ a.ctr + nor * a.rad, b.ctr + nor * (sign * b.rad) });
        }
    }
    return res;
}
```
#### 接口说明
- `tan_pts(p,cir)`：返回切点，数量为 $0/1/2$，$O(1)$
- `tan_line(p,cir)`：返回对应切线，数量为 $0/1/2$，$O(1)$
- `com_tan_pts(a,b)`：返回全部公切线在两圆上的切点对，数量为 $0$ 到 $4$，$O(1)$
- `com_tan_pts()` 中外公切线和内公切线均可能出现；相切时两种方向中某一组会退化为一条
#### 注意事项
- 两圆同心时没有唯一可枚举的公切线，本接口返回空数组；等圆重合时实际有无穷多条公切线
- 两圆相切时，某条公切线的两个切点可能重合。此时不能用 `Line::through(p1,p2)` 构造直线，应使用半径的垂线方向：`rotate90(p1-a.ctr)`
- 半径为 $0$ 的“圆”会退化为点，切线语义通常需要按题意单独处理
#### 常见用法
```cpp
for (Point tan : tan_pts(q, cir)) {
    Line ln = Line::through(q, tan);
}
for (auto [pa, pb] : com_tan_pts(a, b)) {
    Line tan = same_pt(pa, pb) ? Line(pa, rotate90(pa - a.ctr)) : Line::through(pa, pb);
}
```
#### 复杂度
所有接口时间复杂度与额外空间均为 $O(1)$，不计返回数组
### 三角形相关圆与最小覆盖圆
**解决问题：** 求三角形外接圆、内切圆，以及覆盖点集的最小圆
```cpp
optional<Circle> out_cir(Point a, Point b, Point c) {
    ld den{ 2 * cross(b - a, c - a) };
    if (sgn(den) == 0) return nullopt;
    ld b2{ norm2(b - a) }, c2{ norm2(c - a) };
    Point ctr{ a + Point((b2 * (c.y - a.y) - c2 * (b.y - a.y)) / den, (c2 * (b.x - a.x) - b2 * (c.x - a.x)) / den) };
    return Circle{ctr, distance(ctr, a)};
}
optional<Circle> in_cir(Point a, Point b, Point c) {
    ld sd_a{ distance(b, c) }, sd_b{ distance(c, a) };
    ld side_c{ distance(a, b) }, peri{ sd_a + sd_b + side_c };
    if (sgn(peri) == 0 || sgn(cross(a, b, c)) == 0) return nullopt;
    Point ctr{ (a * sd_a + b * sd_b + c * side_c) / peri };
    return Circle{ctr, dis_line(ctr, Line::through(a, b))};
}
Circle dia_cir(Point a, Point b) {
    return Circle{ (a + b) / 2, distance(a, b) / 2};
}
bool cir_has(Circle cir, Point p) { return sgn(distance(cir.ctr, p) - cir.rad) <= 0; }
// 随机增量最小覆盖圆；空点集返回圆心 (0,0)、半径 0
Circle min_cir(vector<Point> pts) {
    static mt19937_64 gen(chrono::steady_clock::now().time_since_epoch().count());
    ranges::shuffle(pts, gen);
    Circle res{{0, 0}, -1};
    auto out = [&](Point p) { return res.rad < 0 || !cir_has(res, p); };
    for (int i = 0; i < pts.size(); ++ i) if (out(pts[i])) {
        res = Circle{pts[i], 0};
        for (int j = 0; j < i; ++ j) if (out(pts[j])) {
            res = dia_cir(pts[i], pts[j]);
            for (int k = 0; k < j; ++ k) if (out(pts[k])) {
                auto cand = out_cir(pts[i], pts[j], pts[k]);
                if (cand) {
                    res = *cand;
                } else {
                    // 三点共线时取最远点对的直径圆
                    array<pair<Point, Point>, 3> pairs{{
                        {pts[i], pts[j]},
                        {pts[i], pts[k]},
                        {pts[j], pts[k]}
                    }};
                    res = dia_cir(pairs[0].first, pairs[0].second);
                    for (auto [a, b] : pairs) {
                        Circle cir{ dia_cir(a, b) };
                        if (cir.rad > res.rad) res = cir;
                    }
                }
            }
        }
    }
    if (pts.empty()) return Circle{{0, 0}, 0};
    return res;
}
```
#### 接口说明
- `out_cir(a,b,c)`：求非共线三点的外接圆；共线时返回 `nullopt`，$O(1)$
- `in_cir(a,b,c)`：求非退化三角形的内切圆；退化时返回 `nullopt`，$O(1)$
- `dia_cir(a,b)`：返回以 $ab$ 为直径的圆，$O(1)$
- `cir_has(cir,p)`：判断点是否位于闭圆盘内，$O(1)$
- `min_cir(pts)`：返回覆盖全部点的最小圆；期望 $O(n)$
#### 注意事项
- 外接圆和内切圆均要求三点不共线
- 最小覆盖圆使用随机增量；期望复杂度为 $O(n)$，最坏复杂度可达 $O(n^3)$
- 最小覆盖圆的边界最多由三个输入点确定：一个点、两个直径端点，或三个外接圆点
- 输入中允许重复点；若浮点近重点会影响稳定性，可在调用前按统一 `EPS` 清理
- 三点共线时，覆盖它们的最小圆由最远点对作为直径
#### 常见用法
```cpp
Circle cover = min_cir(pts);
auto outer = out_cir(a, b, c);
auto inner = in_cir(a, b, c);
```
#### 复杂度

| 操作 | 时间复杂度 | 额外空间 |
|---|---:|---:|
| 外接圆 / 内切圆 | $O(1)$ | $O(1)$ |
| 最小覆盖圆 | 期望 $O(n)$，最坏 $O(n^3)$ | $O(n)$ |
### 固定半径圆覆盖最多点
**解决问题：** 给定平面点集与半径 $r$，平移一个半径固定的圆，使闭圆盘覆盖的输入点数量最大
固定一个输入点 $p_i$ 位于圆周上。对另一个点 $p_j$，圆心绕 $p_i$ 的极角必须落在一个闭区间中；扫描这些角区间的最大重叠数即可
```cpp
int max_pts_cir(const vector<Point> &pts, ld rad) {
    assert(sgn(rad) >= 0);
    if (pts.empty()) return 0;
    if (sgn(rad) == 0) {
        int ans{ 1 };
        for (Point anc : pts) {
            int count{ 0 };
            for (Point p : pts) count += same_pt(anc, p);
            ans = max(ans, count);
        }
        return ans;
    }
    int ans{ 1 };
    for (int i = 0; i < pts.size(); ++ i) {
        vector<pair<ld, int>> evs;
        int act0{ 1 }; // 锚点 pts[i]
        for (int j = 0; j < pts.size(); ++ j) if (i != j) {
            ld dis{ distance(pts[i], pts[j]) };
            if (sgn(dis - 2 * rad) > 0) continue;
            ld mid{ atan2l(pts[j].y - pts[i].y, pts[j].x - pts[i].x) };
            if (mid < 0) mid += 2 * PI;
            ld delta{ acosl(clamp(dis / (2 * rad), -1.L, 1.L)) }, l{ mid - delta }, r{ mid + delta };
            while (l < 0) l += 2 * PI;
            while (r >= 2 * PI) r -= 2 * PI;
            if (l <= r) {
                evs.push_back({l, 1});
                evs.push_back({r, -1});
            } else {
                ++ act0;
                evs.push_back({r, -1});
                evs.push_back({l, 1});
            }
        }
        // 闭区间：同角度先加入，再删除
        ranges::sort(evs, [](auto a, auto b) {
            return a.first != b.first ? a.first < b.first : a.second > b.second;
        });
        int act{ act0 };
        ans = max(ans, act);
        for (auto [ang, delta] : evs) {
            (void)ang;
            act += delta;
            ans = max(ans, act);
        }
    }
    return ans;
}
```
#### 接口说明
- `max_pts_cir(pts,rad)`：返回一个半径为 `rad` 的闭圆盘最多能覆盖多少个输入点，$O(n^2\log n)$
- 圆心位置不固定，接口只返回最大数量，不恢复圆心
- `rad=0` 时返回某个位置上近重点的最大数量
#### 注意事项
- 最优圆可以平移到至少有一个输入点位于边界，因此枚举边界锚点不会漏解
- 角区间是闭区间；事件同角时必须先处理 `+1` 再处理 `-1`，否则会漏掉相切情形
- 距离大于 $2r$ 的点不可能与锚点同时被覆盖
- 若需要恢复圆心，在更新答案时保存锚点与当前事件角，圆心为 `anc + Point(cos(ang),sin(ang))*rad`
#### 常见用法
```cpp
int ans = max_pts_cir(pts, rad);
```
#### 复杂度
时间复杂度：$O(n^2\log n)$
额外空间：$O(n)$
### 圆与多边形面积交
**解决问题：** 求两个圆的公共面积，以及一个圆与任意简单多边形的公共面积
对多边形每条有向边 $(a,b)$，把三角形 $Oab$ 按与圆的交点分段；圆内部分使用叉积面积，圆外部分使用扇形面积
```cpp
ld sec_area(Point a, Point b, ld rad) { return rad * rad * atan2l(cross(a, b), dot(a, b)) / 2; }
// 圆心固定为原点，返回有向三角形 Oab 与圆盘的有向面积交
ld cir_tri_int(Point a, Point b, ld rad) {
    Point dir{ b - a };
    vector<ld> pars{0, 1};
    ld qa{ norm2(dir) }, qb{ 2 * dot(a, dir) };
    ld qc{ norm2(a) - rad * rad }, disc{ qb * qb - 4 * qa * qc };
    if (sgn(qa) != 0 && sgn(disc) >= 0) {
        ld root{ sqrtl(max(0.L, disc)) };
        for (ld t : {(-qb - root) / (2 * qa), (-qb + root) / (2 * qa)}) {
            if (t > EPS && t < 1 - EPS) pars.push_back(t);
        }
    }
    ranges::sort(pars);
    ld area{ 0 };
    for (int i = 0; i + 1 < pars.size(); ++ i) {
        Point l{ a + dir * pars[i] }, r{ a + dir * pars[i + 1] };
        Point mid{ (l + r) / 2 };
        if (sgn(length(mid) - rad) <= 0) area += cross(l, r) / 2;
        else area += sec_area(l, r, rad);
    }
    return area;
}
ld cir_poly_area(Circle cir, const vector<Point> &poly) {
    ld area{ 0 };
    for (int i = 0, n = poly.size(); i < n; ++ i) {
        area += cir_tri_int(poly[i] - cir.ctr, poly[(i + 1) % n] - cir.ctr, cir.rad);
    }
    return fabsl(area);
}
ld cir_int_area(Circle a, Circle b) {
    ld dis{ distance(a.ctr, b.ctr) };
    if (sgn(dis - a.rad - b.rad) >= 0) return 0;
    if (sgn(dis - fabsl(a.rad - b.rad)) <= 0) {
        ld rad{ min(a.rad, b.rad) };
        return PI * rad * rad;
    }
    ld ang_a{ 2 * acosl(clamp((a.rad * a.rad + dis * dis - b.rad * b.rad) / (2 * a.rad * dis), -1.L, 1.L)) };
    ld ang_b{ 2 * acosl(clamp((b.rad * b.rad + dis * dis - a.rad * a.rad) / (2 * b.rad * dis), -1.L, 1.L)) };
    return (a.rad * a.rad * (ang_a - sinl(ang_a)) + b.rad * b.rad * (ang_b - sinl(ang_b))) / 2;
}
```
#### 接口说明
- `sec_area(a,b,rad)`：返回从向量 `a` 逆时针/顺时针转到 `b` 的有向扇形面积，角度范围由 `atan2` 决定，$O(1)$
- `cir_tri_int(a,b,rad)`：圆心为原点时，返回有向三角形 $Oab$ 与圆盘的有向面积交，$O(1)$
- `cir_poly_area(cir,poly)`：返回圆盘与简单多边形的公共面积，$O(n)$
- `cir_int_area(a,b)`：返回两个闭圆盘的公共面积，$O(1)$
#### 注意事项
- 多边形顶点可以顺时针或逆时针排列，最终接口会取绝对值；底层三角贡献必须保留方向
- `cir_poly_area()` 适用于简单多边形，不要求凸
- 半径必须非负；两圆面积交公式中的反三角函数参数必须 `clamp` 到 $[-1,1]$
- 相离和相切时公共面积为 $0$；包含时公共面积为较小圆面积
#### 常见用法
```cpp
ld com_cir = cir_int_area(a, b);
ld com_poly = cir_poly_area(cir, poly);
```
#### 复杂度

| 操作 | 时间复杂度 | 额外空间 |
|---|---:|---:|
| 两圆面积交 | $O(1)$ | $O(1)$ |
| 圆与 $n$ 边形面积交 | $O(n)$ | $O(1)$ |
### 圆面积并与可见圆弧
**解决问题：** 求多个圆盘的面积并、并集边界长度，以及每个圆未被其他圆覆盖的可见圆弧
每个圆只对其未被其他圆覆盖的弧做 Green 公式积分。重复圆通过下标打破平局，只保留一个圆贡献边界
```cpp
// 返回第 idx 个圆未被其他圆覆盖的角区间，区间位于 [0,2PI]
vector<pair<ld, ld>> vis_arcs(const vector<Circle> &cirs, int idx) {
    Circle cir{ cirs[idx] };
    if (sgn(cir.rad) == 0) return {};
    vector<pair<ld, ld>> cov;
    for (int j = 0; j < cirs.size(); ++ j) if (j != idx) {
        Circle oth{ cirs[j] };
        ld dis{ distance(cir.ctr, oth.ctr) };
        // cir 被严格更大的圆或更早出现的重复圆完全覆盖
        if (sgn(dis + cir.rad - oth.rad) <= 0 && (sgn(cir.rad - oth.rad) < 0 || idx > j)) return {};
        // 外离/外切，或 oth 完全位于 cir 内部，不覆盖 cir 的边界弧
        if (sgn(dis - cir.rad - oth.rad) >= 0 || sgn(dis + oth.rad - cir.rad) <= 0) continue;
        ld mid{ atan2l(oth.ctr.y - cir.ctr.y, oth.ctr.x - cir.ctr.x) };
        if (mid < 0) mid += 2 * PI;
        ld delta{ acosl(clamp((cir.rad * cir.rad + dis * dis
            - oth.rad * oth.rad) / (2 * cir.rad * dis), -1.L, 1.L)) };
        ld l{ mid - delta }, r{ mid + delta };
        while (l < 0) l += 2 * PI, r += 2 * PI;
        while (l >= 2 * PI) l -= 2 * PI, r -= 2 * PI;
        if (r <= 2 * PI) {
            cov.push_back({l, r});
        } else {
            cov.push_back({l, 2 * PI});
            cov.push_back({0, r - 2 * PI});
        }
    }
    ranges::sort(cov);
    vector<pair<ld, ld>> mgd;
    for (auto itv : cov) {
        if (mgd.empty() || itv.first > mgd.back().second + EPS) mgd.push_back(itv);
        else mgd.back().second = max(mgd.back().second, itv.second);
    }
    vector<pair<ld, ld>> vis;
    ld st{ 0 };
    for (auto [l, r] : mgd) {
        if (l > st + EPS) vis.push_back({st, l});
        st = max(st, r);
    }
    if (st < 2 * PI - EPS) vis.push_back({st, 2 * PI});
    return vis;
}
ld cir_uni_area(const vector<Circle> &cirs) {
    ld area{ 0 };
    for (int i = 0; i < cirs.size(); ++ i) {
        Circle cir{ cirs[i] };
        for (auto [l, r] : vis_arcs(cirs, i)) {
            Point p1{ cir.ctr + Point(cosl(l), sinl(l)) * cir.rad };
            Point p2{ cir.ctr + Point(cosl(r), sinl(r)) * cir.rad };
            area += (cross(cir.ctr, p2 - p1) + cir.rad * cir.rad * (r - l)) / 2;
        }
    }
    return max(0.L, area);
}
ld cir_uni_peri(const vector<Circle> &cirs) {
    ld peri{ 0 };
    for (int i = 0; i < cirs.size(); ++ i) {
        for (auto [l, r] : vis_arcs(cirs, i)) {
            peri += cirs[i].rad * (r - l);
        }
    }
    return peri;
}
```
#### 接口说明
- `vis_arcs(cirs,idx)`：返回第 `idx` 个圆未被其他圆盘覆盖的角区间，$O(n\log n)$
- `cir_uni_area(cirs)`：返回所有闭圆盘的面积并，$O(n^2\log n)$
- `cir_uni_peri(cirs)`：返回圆盘并集边界总长度，$O(n^2\log n)$
- 若只需要某个圆的可见弧长度，累加 `rad*(r-l)` 即可
#### 注意事项
- 半径必须非负；半径为 $0$ 的圆不贡献面积和边界长度
- 重复圆通过输入下标去重：只保留下标较小者的完整边界
- 相切只产生单点，不改变面积并与边界长度，因此按“不覆盖弧”处理
- `vis_arcs()` 每次都会重新扫描全部圆；若同时求面积和周长，建议先缓存每个圆的可见区间，避免重复计算
- Green 公式使用逆时针圆弧；角区间统一存为 `l < r` 且位于 $[0,2\pi]$
#### 常见用法
```cpp
ld area = cir_uni_area(cirs);
ld peri = cir_uni_peri(cirs);
```
#### 复杂度
时间复杂度：$O(n^2\log n)$
除返回区间外，单次扫描额外空间为 $O(n)$
### 圆反演与常见结论
**解决问题：** 快速调用圆反演、Apollonius 轨迹、最近圆对与圆弧/扇形重心公式
#### 圆反演
以 $O$ 为中心、幂为 $R^2$ 的反演满足
$$
OP\cdot OP'=R^2,
\qquad
P'=O+(P-O)\frac{R^2}{|P-O|^2}
$$
```cpp
// power = R^2；反演中心本身没有有限像
optional<Point> inv_pt(Point p, Point o, ld power) {
    Point dir{ p - o };
    ld dis2{ norm2(dir) };
    if (sgn(dis2) == 0) return nullopt;
    return o + dir * (power / dis2);
}
```
不过反演中心的圆仍变为不过反演中心的圆；过反演中心的圆变为不过反演中心的直线，反向关系也成立。切点不为反演中心时，相切关系保持

#### Apollonius 圆
满足
$$
\frac{|PA|}{|PB|}=k
$$
的点集在 $k=1$ 时为 $AB$ 的垂直平分线，在 $k\ne1$ 时为圆
```cpp
// A、B 必须不同，k 必须非负；返回垂直平分线或 Apollonius 圆
optional<variant<Line, Circle>> apol_loc(Point a, Point b, ld k) {
    if (same_pt(a, b) || sgn(k) < 0) return nullopt;
    if (sgn(k - 1) == 0) {
        Point mid{ (a + b) / 2 };
        return Line(mid, rotate90(b - a));
    }
    ld k2{ k * k }, den{ 1 - k2 }, rad{ k * distance(a, b) / fabsl(den) };
    Point ctr{ (a - b * k2) / den };
    return Circle{ ctr, rad };
}
```
#### 最近圆对
两个闭圆盘之间的距离为
$$
\max(0,|c_i-c_j|-r_i-r_j)
$$
```cpp
// 返回 {最短距离,i,j}；圆少于两个时返回 {INF,-1,-1}
tuple<ld, int, int> close_cir(const vector<Circle> &cirs) {
    ld ans{ numeric_limits<ld>::infinity() };
    int a{ -1 }, b{ -1 };
    for (int i = 0; i < cirs.size(); ++ i) {
        for (int j = i + 1; j < cirs.size(); ++ j) {
            ld cur{ max(0.L, distance(cirs[i].ctr, cirs[j].ctr) - cirs[i].rad - cirs[j].rad) };
            if (cur < ans) {
                ans = cur;
                a = i;
                b = j;
            }
        }
    }
    return {ans, a, b};
}
```
#### 扇形与圆弧重心
圆心角为 $\theta$（弧度，$0<\theta\le2\pi$）时：
- 均匀扇形薄片的重心到圆心距离：
  $$\frac{4r\sin(\theta/2)}{3\theta}$$
- 均匀圆弧线的重心到圆心距离：
  $$\frac{2r\sin(\theta/2)}{\theta}$$
```cpp
ld sec_ctr_dis(ld rad, ld ang) {
    if (sgn(ang) == 0) return 2 * rad / 3;
    return 4 * rad * sinl(ang / 2) / (3 * ang);
}
ld arc_ctr_dis(ld rad, ld ang) {
    if (sgn(ang) == 0) return rad;
    return 2 * rad * sinl(ang / 2) / ang;
}
```
#### 接口说明
- `inv_pt(p,o,power)`：反演点；反演中心返回 `nullopt`，$O(1)$
- `apol_loc(a,b,k)`：返回 `Line` 或 `Circle`；非法输入返回 `nullopt`，$O(1)$
- `close_cir(cirs)`：暴力求最近圆对，$O(n^2)$
- `sec_ctr_dis()`、`arc_ctr_dis()`：返回重心沿角平分线到圆心的距离，$O(1)$
#### 注意事项
- 反演的 `power` 通常取正数 $R^2$；反演中心没有有限像
- `apol_loc()` 返回 `variant<Line,Circle>`，使用 `holds_alternative<Line>()` 区分
- `A=B` 时距离比轨迹退化，接口返回 `nullopt`
- 扇形薄片与圆弧线的重心公式不同，不要混用
- 圆上整点属于数论问题。只有化为整数圆心下的 $x^2+y^2=r^2$ 后，才能使用平方和表示数等结论
- 大量圆的最近圆对需要空间分治、网格或包围盒结构；本页只提供可靠的 $O(n^2)$ 兜底
#### 常见用法
```cpp
if (auto image = inv_pt(p, o, rad * rad)) {
    Point inv = *image;
}
auto locus = apol_loc(a, b, ratio);
```
#### 复杂度
除最近圆对为 $O(n^2)$ 外，其余接口均为 $O(1)$
