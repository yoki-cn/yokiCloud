### Andrew 凸包
**解决问题：** 求包含给定点集的最小凸多边形，作为凸包含、直径、旋转卡壳、Minkowski 和等算法的预处理
按 $(x,y)$ 字典序扫描点集。若当前点使凸链发生顺时针转向，就不断删除链尾
$$
(p_{k-1}-p_{k-2})\times(p-p_{k-1})<0
$$
`keep_col=false` 时还会删除边上的中间点，得到严格凸包；`true` 时保留所有边界点
```cpp
vector<Point> conv_hull(vector<Point> pts, bool keep_col = false) {
    ranges::sort(pts);
    pts.erase(ranges::unique(pts).begin(), pts.end());
    if (pts.size() <= 1) return pts;
    bool all_col{ true };
    for (int i = 2; i < pts.size(); ++ i) {
        if (sgn(cross(pts[0], pts[1], pts[i])) != 0) {
            all_col = false;
            break;
        }
    }
    if (all_col) return keep_col ? pts : vector<Point>{ pts.front(), pts.back() };
    vector<Point> lo, up;
    auto push = [&](vector<Point> &ch, Point p) {
        while (ch.size() >= 2) {
            int turn{ sgn(cross(ch[ch.size() - 2], ch.back(), p)) };
            if (turn > 0 || (keep_col && turn == 0)) break;
            ch.pop_back();
        }
        ch.push_back(p);
    };
    for (Point p : pts) push(lo, p);
    for (auto it = pts.rbegin(); it != pts.rend(); ++ it) push(up, *it);
    lo.pop_back();
    up.pop_back();
    lo.insert(lo.end(), up.begin(), up.end());
    return lo;
}
```
输出按逆时针排列，不重复存储首点。除全部共线且 `keep_col=true` 外，起点是 $(x,y)$ 字典序最小点
### Graham 扫描（可选）
**解决问题：** 当题目本身还需要围绕一个基点进行极角排序时，可复用排序结果构造凸包。普通求凸包优先使用更短、更稳定的 Andrew
```cpp
// ===== Graham 扫描：若只使用 Andrew，以下代码可直接删除 =====
vector<Point> gra_hull(vector<Point> pts, bool keep_col = false) {
    ranges::sort(pts, less{});
    pts.erase(ranges::unique(pts).begin(), pts.end());
    if (pts.size() <= 1) return pts;
    bool all_col{ true };
    for (int i = 2; i < pts.size(); ++ i) {
        if (sgn(cross(pts[0], pts[1], pts[i])) != 0) {
            all_col = false;
            break;
        }
    }
    if (all_col) return keep_col ? pts : vector<Point>{ pts.front(), pts.back() };
    int piv{ ranges::min_element(pts, [](Point a, Point b) {
            return a.y != b.y ? a.y < b.y : a.x < b.x;
        }) - pts.begin() };
    swap(pts[0], pts[piv]);
    Point o{ pts[0] };
    sort(pts.begin() + 1, pts.end(), [&](Point a, Point b) {
        return arg_less(a - o, b - o);
    });
    // 保留共线点时，最后一条同角射线必须反转，才能沿凸包边界依次经过这些点。
    if (keep_col) {
        int first{ int(pts.size()) - 1 };
        while (first > 1 && sgn(cross(pts[first - 1] - o, pts.back() - o)) == 0) -- first;
        ranges::reverse(pts.begin() + first, pts.end());
    }
    vector<Point> hull;
    for (Point p : pts) {
        while (hull.size() >= 2) {
            int turn{ sgn(cross(hull[hull.size() - 2], hull.back(), p)) };
            if (turn > 0 || (keep_col && turn == 0)) break;
            hull.pop_back();
        }
        hull.push_back(p);
    }
    return hull;
}
```
### 直线与凸包的交点（线性兜底）
**解决问题：** 求一条直线与凸多边形边界的全部交点。一般位置下返回 $0$、$1$ 或 $2$ 个点；直线与某条边重合时返回该重合边的端点
```cpp
/* ===== 浮点版专属：若不需要直线与凸包求交，以下代码可直接删除 ===== */
vector<Point> line_cv_int(const vector<Point> &hull, Line ln) {
    vector<Point> res;
    auto add = [&](Point p) {
        for (Point old : res) if (same_pt(old, p)) return;
        res.push_back(p);
    };
    for (int i = 0, n = hull.size(); i < n; ++ i) {
        Point a{ hull[i] }, b{ hull[(i + 1) % n] };
        int sd_a{ pt_side(a, ln) }, sd_b{ pt_side(b, ln) };
        if (sd_a == 0) add(a);
        if (sd_a * sd_b < 0) {
            auto p = line_int(ln, Line::through(a, b));
            if (p) add(*p);
        }
    }
    return res;
}
```
点是否位于凸包内直接使用上一篇的 `in_conv()`；两个凸多边形求交直接使用 `conv_int()`
### 凸包切点（可靠线性版）
**解决问题：** 给定凸多边形外一点 `q`，求经过该点的两条支撑线与凸包接触的顶点
对严格凸多边形的顶点 $p_i$，若相邻点 $p_{i-1},p_{i+1}$ 位于有向直线 `q -> p[i]` 的同侧，则 $p_i$ 是切点
```cpp
// hull 应为无共线冗余点的逆时针凸包；q 应严格位于凸包外
vector<int> tan_ver(const vector<Point> &hull, Point q) {
    int n = hull.size();
    if (n <= 2) {
        vector<int> res(n);
        ranges::iota(res, 0);
        return res;
    }
    vector<int> res;
    for (int i = 0; i < n; ++ i) {
        Point dir{ hull[i] - q };
        int pre{ sgn(cross(dir, hull[(i + n - 1) % n] - q)) };
        int nxt{ sgn(cross(dir, hull[(i + 1) % n] - q)) };
        if (pre == 0 || nxt == 0 || pre == nxt) res.push_back(i);
    }
    return res;
}
```
严格凸包且查询点严格在外部时，返回数组恰有两个下标。若保留了共线边界点，接触位置可能是一整段边，应该先调用 `norm_conv()`
### 凸包快速查询
**解决问题：** 对严格凸多边形进行方向极值点、直线求交边和凸包外点切点的 $O(\log n)$ 查询
以下接口要求 `poly` 是无共线冗余点的逆时针严格凸多边形。若只进行少量查询，前面的线性版本更短且更容易处理退化情况
```cpp
// 返回使 dot(poly[i],dir) 最大的顶点下标；dir 必须非零
int ext_ver(const vector<Point> &poly, Point dir) {
    int n = poly.size();
    if (n <= 1) return n - 1;
    if (n == 2) return sgn(dot(poly[1] - poly[0], dir)) > 0 ? 1 : 0;
    auto cmp = [&](int i, int j) { return sgn(cross(rotate90(dir), poly[i % n] - poly[j % n])); };
    auto ext = [&](int i) { return cmp(i + 1, i) >= 0 && cmp(i, i - 1 + n) < 0; };
    if (ext(0)) return 0;
    int lo{ 0 }, hi{ n };
    while (lo + 1 < hi) {
        int mid{ (lo + hi) >> 1 };
        if (ext(mid)) return mid;
        int ls{ cmp(lo + 1, lo) }, ms{ cmp(mid + 1, mid) };
        if (ls < ms || (ls == ms && ls == cmp(lo, mid))) hi = mid;
        else lo = mid;
    }
    return lo;
}
// 返回直线与凸包相交位置对应的边下标；ln.dir 必须非零
array<int, 2> line_hull(const vector<Point> &poly, Line ln) {
    int n = poly.size();
    Point a{ ln.p }, b{ ln.p + ln.dir };
    auto cmp = [&](int i) { return sgn(cross(a, poly[i], b)); };
    int ea{ ext_ver(poly, rotate90(a - b)) }, eb{ ext_ver(poly, rotate90(b - a)) };
    if (cmp(ea) < 0 || cmp(eb) > 0) return {-1, -1};
    array<int, 2> res;
    for (int z = 0; z < 2; ++ z) {
        int lo{ eb }, hi{ ea };
        while ((lo + 1) % n != hi) {
            int mid{ ((lo + hi + (lo < hi ? 0 : n)) >> 1) % n };
            if (cmp(mid) == cmp(eb)) lo = mid;
            else hi = mid;
        }
        res[z] = (lo + (cmp(hi) == 0)) % n;
        swap(ea, eb);
    }
    if (res[0] == res[1]) return {res[0], -1};
    if (cmp(res[0]) == 0 && cmp(res[1]) == 0) {
        int d{ (res[0] - res[1] + n + 1) % n };
        if (d == 0) return {res[0], res[0]};
        if (d == 2) return {res[1], res[1]};
    }
    return res;
}
// q 必须严格位于凸包外；返回两条切线的代表切点下标
array<int, 2> tan_fast(const vector<Point> &poly, Point q) {
    int n{ int(poly.size()) };
    auto vis_e = [&](int i) {
        return sgn(cross(poly[i] - q, poly[(i + 1) % n] - poly[i])) < 0;
    };
    auto vis = [&](int i) { return vis_e(i) || vis_e((i - 1 + n) % n); };
    auto rgt = [&](int i) { return sgn(cross(poly[0] - q, poly[i] - q)) < 0; };
    int lo, hi, mid, ta{ -1 }, tb{ -1 };
    if (vis(0)) {
        lo = 0, hi = n;
        while (hi - lo > 1) {
            mid = (lo + hi) >> 1;
            if (vis(mid) && rgt(mid)) lo = mid;
            else hi = mid;
        }
        tb = lo, lo = 0, hi = n;
        while (hi - lo > 1) {
            mid = (lo + hi) >> 1;
            if (vis((tb + n - mid) % n)) lo = mid;
            else hi = mid;
        }
        ta = (tb + n - lo) % n;
    } else {
        lo = 0, hi = n;
        while (hi - lo > 1) {
            mid = (lo + hi) >> 1;
            if (!vis(mid) && !rgt(mid)) lo = mid;
            else hi = mid;
        }
        ta = (lo + 1) % n;
        lo = 0, hi = n;
        while (hi - lo > 1) {
            mid = (lo + hi) >> 1;
            if (vis((ta + mid) % n)) lo = mid;
            else hi = mid;
        }
        tb = (ta + lo) % n;
    }
    return {ta, tb};
}
```
### 对踵点与凸包直径
**解决问题：** 求凸多边形的最远点对，以及枚举互相存在平行支撑线的一组对踵顶点
固定边 $p_i p_{i+1}$，与其距离最远的顶点使
$$
\left|(p_{i+1}-p_i)\times(p_j-p_i)\right|
$$
最大。边按逆时针旋转时，最远顶点也只会沿凸包单调前进，因此总扫描次数为线性级别
```cpp
using HullArea2 = decltype(cross(Point{}, Point{}));
using HullDist2 = decltype(norm2(Point{}));
// 返回去重后的对踵顶点对；排序去重使本接口总复杂度为 O(n log n)
vector<pair<int, int>> anti_pair(const vector<Point> &hull) {
    int n = hull.size();
    if (n <= 1) return {};
    if (n == 2) return {{0, 1}};
    vector<pair<int, int>> res;
    auto area = [&](int i, int j, int k) -> HullArea2 {
        HullArea2 value{ cross(hull[j] - hull[i], hull[k] - hull[i]) };
        return value < 0 ? -value : value;
    };
    auto add = [&](int a, int b) {
        if (a == b) return;
        if (a > b) swap(a, b);
        res.push_back({ a, b });
    }
    int j{ 1 };
    for (int i = 0; i < n; ++ i) {
        int nxt_i{ (i + 1) % n };
        while (sgn(area(i, nxt_i, (j + 1) % n) - area(i, nxt_i, j)) > 0) j = (j + 1) % n;
        add(i, j);
        add(nxt_i, j);
        int nxt_j{ (j + 1) % n };
        if (sgn(area(i, nxt_i, nxt_j) - area(i, nxt_i, j)) == 0) {
            add(i, nxt_j);
            add(nxt_i, nxt_j);
        }
    }
    ranges::sort(res);
    res.erase(ranges::unique(res).begin(), res.end());
    return res;
}
// 返回 {最大距离平方, 端点下标 u, 端点下标 v}
tuple<HullDist2, int, int> conv_diam(const vector<Point> &hull) {
    int n = hull.size();
    if (n == 0) return { HullDist2{}, -1, -1 };
    if (n == 1) return { HullDist2{}, 0, 0 };
    if (n == 2) return { dist2(hull[0], hull[1]), 0, 1 };
    HullDist2 best{};
    int ans_u{ 0 }, ans_v{ 0 };
    auto area = [&](int i, int j, int k) -> HullArea2 {
        HullArea2 value{ cross(hull[j] - hull[i], hull[k] - hull[i]) };
        return value < 0 ? -value : value;
    };
    auto upd = [&](int u, int v) {
        HullDist2 cur{ dist2(hull[u], hull[v]) };
        if (cur > best) {
            best = cur;
            ans_u = u;
            ans_v = v;
        }
    };
    int j{ 1 };
    for (int i = 0; i < n; ++ i) {
        int nxt_i{ (i + 1) % n };
        while (sgn(area(i, nxt_i, (j + 1) % n) - area(i, nxt_i, j)) > 0) j = (j + 1) % n;
        upd(i, j);
        upd(nxt_i, j);
        int nxt_j{ (j + 1) % n };
        if (sgn(area(i, nxt_i, nxt_j) - area(i, nxt_i, j)) == 0) {
            upd(i, nxt_j);
            upd(nxt_i, nxt_j);
        }
    }
    return { best, ans_u, ans_v };
}
```
`conv_diam()` 返回距离平方，可直接用于整数版；只有最终输出欧氏距离时才开方
### 凸包最大面积三角形
**解决问题：** 从凸多边形顶点中选择三个点，使三角形面积最大
固定 $i,j$ 后，$|\triangle p_i p_j p_k|$ 随 $k$ 先增后减；对固定 $i$ 依次移动 $j$ 时，最优 $k$ 单调前进
```cpp
// 返回 {最大面积的两倍, 顶点下标 i, j, k}
tuple<HullArea2, int, int, int> max_tri_a2(const vector<Point> &hull) {
    int n = hull.size();
    if (n < 3) return { HullArea2{}, -1, -1, -1 };
    auto area = [&](int i, int j, int k) -> HullArea2 {
        HullArea2 value{ cross(hull[i], hull[j], hull[k]) };
        return value < 0 ? -value : value;
    };
    HullArea2 best{};
    int ans_i{ -1 }, ans_j{ -1 }, ans_k{ -1 };
    for (int i = 0; i + 2 < n; ++ i) {
        int k{ i + 2 };
        for (int j = i + 1; j + 1 < n; ++ j) {
            if (k <= j) k = j + 1;
            while (k + 1 < n && sgn(area(i, j, k + 1) - area(i, j, k)) >= 0) ++ k;
            HullArea2 cur{ area(i, j, k) };
            if (cur > best) {
                best = cur;
                ans_i = i;
                ans_j = j;
                ans_k = k;
            }
        }
    }
    return { best, ans_i, ans_j, ans_k };
}
```
整数版保留“两倍面积”可以避免除法；实际面积为 `area2 / 2`
### 最小宽度与最小覆盖矩形
**解决问题：** 求凸多边形的最小宽度，以及边允许任意旋转时的最小面积、最小周长覆盖矩形

![旋转卡壳求覆盖矩形时的四条支撑线|387](/algorithm-assets/rotating-calipers-four-supports.png)

凸多边形的最小宽度一定对应某条凸包边的方向。对边向量 $e_i=p_{i+1}-p_i$，该方向对应的宽度为
$$
w_i=\max_j\frac{e_i\times(p_j-p_i)}{|e_i|}
$$
最优覆盖矩形至少有一条边与凸包边共线。固定方向单位向量 $u$，令 $v$ 为其逆时针法向量，只需维护点在 $u,v$ 上的最大、最小投影
```cpp
/* ===== 浮点版专属：若不需要宽度与覆盖矩形，以下代码可直接删除 ===== */
ld conv_width(const vector<Point> &hull) {
    int n = hull.size();
    if (n < 3) return 0.L;
    ld ans{ numeric_limits<ld>::infinity() };
    int j{ 1 };
    for (int i = 0; i < n; ++ i) {
        int nxt_i{ (i + 1) % n };
        Point edge{ hull[nxt_i] - hull[i] };
        auto h2 = [&](int k) { return cross(edge, hull[k] - hull[i]); };
        while (sgn(h2((j + 1) % n) - h2(j)) > 0) j = (j + 1) % n;
        ans = min(ans, max(0.L, h2(j)) / length(edge));
    }
    return ans;
}
struct CoverRect {
    ld area{ 0 }, peri{ 0 };
    array<Point, 4> cor{}; // 逆时针四个顶点
};
// 返回 {最小面积覆盖矩形, 最小周长覆盖矩形}
pair<CoverRect, CoverRect> min_cov_rec(const vector<Point> &hull) {
    int n = hull.size();
    CoverRect min_a, min_p;
    if (n == 0) return { min_a, min_p };
    if (n == 1) {
        min_a.cor.fill(hull[0]);
        min_p = min_a;
        return { min_a, min_p };
    }
    if (n == 2) {
        ld len{ distance(hull[0], hull[1]) };
        min_a.peri = 2 * len;
        min_a.cor = { hull[0], hull[1], hull[1], hull[0] };
        min_p = min_a;
        return {min_a, min_p};
    }
    min_a.area = numeric_limits<ld>::infinity();
    min_p.peri = numeric_limits<ld>::infinity();
    Point fst_e{ hull[1] - hull[0] };
    int top{ 0 }, r{ 0 }, l{ 0 };
    for (int i = 1; i < n; ++ i) {
        if (cross(fst_e, hull[i] - hull[0]) > cross(fst_e, hull[top] - hull[0])) top = i;
        if (dot(fst_e, hull[i] - hull[0]) > dot(fst_e, hull[r] - hull[0])) r = i;
        if (dot(fst_e, hull[i] - hull[0]) < dot(fst_e, hull[l] - hull[0])) l = i;
    }
    for (int i = 0; i < n; ++ i) {
        int nxt_i{ (i + 1) % n };
        Point edge{ hull[nxt_i] - hull[i] };
        auto norm_pr = [&](int k) { return cross(edge, hull[k] - hull[i]); };
        auto edge_pr = [&](int k) { return dot(edge, hull[k] - hull[i]); };
        while (sgn(norm_pr((top + 1) % n) - norm_pr(top)) > 0) top = (top + 1) % n;
        while (sgn(edge_pr((r + 1) % n) - edge_pr(r)) > 0) r = (r + 1) % n;
        while (sgn(edge_pr(l) - edge_pr((l + 1) % n)) > 0) l = (l + 1) % n;
        Point ax{ edge / length(edge) }, ay{ rotate90(ax) };
        ld mn_x{ dot(hull[l], ax) }, mx_x{ dot(hull[r], ax) };
        ld mn_y{ dot(hull[i], ay) }, mx_y{ dot(hull[top], ay) };
        if (mn_x > mx_x) swap(mn_x, mx_x);
        if (mn_y > mx_y) swap(mn_y, mx_y);
        ld w{ mx_x - mn_x }, h{ mx_y - mn_y };
        CoverRect cur;
        cur.area = w * h;
        cur.peri = 2 * (w + h);
        cur.cor = {
            ax * mn_x + ay * mn_y, ax * mx_x + ay * mn_y,
            ax * mx_x + ay * mx_y, ax * mn_x + ay * mx_y
        };
        if (sgn(cur.area - min_a.area) < 0) min_a = cur;
        if (sgn(cur.peri - min_p.peri) < 0) min_p = cur;
    }
    return {min_a, min_p};
}
```
### 最远 Manhattan 距离
**解决问题：** 求点集中两点的最大 Manhattan 距离。该问题不需要先求凸包
二维情况下
$$
|x_1-x_2|+|y_1-y_2|
=\max\left\{
|(x_1+y_1)-(x_2+y_2)|,
|(x_1-y_1)-(x_2-y_2)|
\right\}
$$
```cpp
// ===== 整数版专属 =====
i128 far_manh(const vector<Point> &pts) {
    if (pts.empty()) return 0;
    i128 min_s{ i128(pts[0].x) + pts[0].y }, max_s{ min_s };
    i128 min_dif{ i128(pts[0].x) - pts[0].y }, max_dif{ min_dif };
    for (Point p : pts) {
        i128 sum{ i128(p.x) + p.y }, dif{ i128(p.x) - p.y };
        min_s = min(min_s, sum);
        max_s = max(max_s, sum);
        min_dif = min(min_dif, dif);
        max_dif = max(max_dif, dif);
    }
    return max(max_s - min_s, max_dif - min_dif);
}
```
### 接口说明
#### 凸包构造
- `conv_hull(pts, keep_col)`：Andrew 凸包。默认删除边上中间点，返回逆时针凸包且不重复首点，$O(n\log n)$
- `gra_hull(pts, keep_col)`：Graham 扫描。仅在还需要极角序时建议保留，$O(n\log n)$
- 两个接口都只删除坐标完全相同的点。浮点输入存在近重复点时，应在调用前按统一 `EPS` 清理
#### 查询与切点
- `line_cv_int(hull, ln)`：求直线与凸包边界交点；仅浮点版，$O(n)$
- `tan_ver(hull, q)`：求凸包外一点的切点下标；严格凸包返回两个切点，$O(n)$
- `ext_ver(poly,dir)`：返回指定方向的支撑顶点，$O(\log n)$
- `line_hull(poly,ln)`：返回直线经过的凸包边下标，$O(\log n)$
- `tan_fast(poly,q)`：返回凸包外点的两个代表切点，$O(\log n)$
- 点包含直接调用 `in_conv()`，单次 $O(\log n)$
- 点到凸包的最近距离不需要新板子：遍历每条边并调用 `dis_seg()`，总复杂度 $O(n)$
#### 旋转卡壳
- `anti_pair(hull)`：返回去重后的对踵顶点对，$O(n\log n)$；卡壳扫描本身为 $O(n)$，排序去重为 $O(n\log n)$
- `conv_diam(hull)`：返回 `{距离平方,u,v}`，$O(n)$
- `max_tri_a2(hull)`：返回 `{最大面积的两倍,i,j,k}`，$O(n^2)$
- `conv_width(hull)`：返回最小宽度；仅浮点版，$O(n)$
- `min_cov_rec(hull)`：同时返回最小面积和最小周长覆盖矩形；仅浮点版，$O(n)$
- `far_manh(pts)`：返回最大 Manhattan 距离；仅整数版，$O(n)$，不依赖凸包
### 注意事项
- 卡壳、切点和凸包含均应使用逆时针凸包。除特别需要边界点外，构造时使用默认 `keep_col=false`；已有凸多边形可先调用 `norm_conv()`
- `conv_hull(..., true)` 在全部共线时按字典序返回所有点，此时结果不是一个环；旋转卡壳接口不能直接使用该退化结果
- 浮点排序不能把 `EPS` 写入 `operator<` 或比较器。近重复点应先量化、聚类或按题意单独清理
- `line_cv_int()` 要求 `ln.dir` 非零。直线与凸包边重合时返回该边端点，不表示无穷多个交点
- `tan_ver()` 和 `tan_fast()` 的查询点应严格位于凸包外。点位于内部或边界时，“两条外切线”不满足接口语义
- 三个快速接口要求至少三个顶点的严格凸多边形；含共线冗余点时应先调用 `norm_conv()`
- `conv_diam()` 和 `max_tri_a2()` 返回平方距离或两倍面积，整数版不要提前转成浮点数
- 最小宽度与覆盖矩形需要构造单位向量和非整数顶点，因此只放在浮点版
- 动态插入、删除凸包属于独立数据结构问题。本页只处理静态点集；不要把静态 Andrew 反复重建误认为动态凸包
### 常见用法
#### 求点集直径
```cpp
vector<Point> hull = conv_hull(pts);
auto [distance2, u, v] = conv_diam(hull);
```
#### 多次查询点是否位于凸包
直接使用上一页的凸包含接口：
```cpp
vector<Point> hull = norm_conv(conv_hull(pts));
for (Point q : qs) Contain where = in_conv(hull, q);
```
#### 求点到凸包的最近距离
无需新增高度相似的函数，直接扫描基础板子的点到线段距离：
```cpp
// [浮点版专属]
ld ans = numeric_limits<ld>::infinity();
for (int i = 0, n = hull.size(); i < n; ++ i) {
    ans = min(ans, dis_seg(q, Segment(hull[i], hull[(i + 1) % n])));
}
```
若 `q` 位于凸包内部或边界，区域距离应为 $0$，可先调用 `in_conv()` 判断
#### 求凸包外一点的两条切线
```cpp
vector<Point> hull = norm_conv(conv_hull(pts));
vector<int> tangent = tan_ver(hull, q);
if (tangent.size() == 2) {
    Line first = Line::through(q, hull[tangent[0]]);
    Line second = Line::through(q, hull[tangent[1]]);
}
```
#### 求最小宽度与覆盖矩形
```cpp
// [浮点版专属]
vector<Point> hull = norm_conv(conv_hull(pts));
ld w = conv_width(hull);
auto [min_a, min_p] = min_cov_rec(hull);
```
