### 有向面积
**解决问题：** 判断多边形顶点方向，计算多边形面积，并作为重心、凸多边形规范化、Pick 定理等算法的基础
设多边形顶点依次为 $p_0,p_1,\dots,p_{n-1}$，并令 $p_n=p_0$，则有向面积的两倍为
$$
S_2=\sum_{i=0}^{n-1}p_i\times p_{i+1}
$$
- $S_2>0$：顶点按逆时针排列
- $S_2<0$：顶点按顺时针排列
- 实际面积为 $\dfrac{|S_2|}{2}$
```cpp
using Area2 = decltype(cross(Point{}, Point{}));
// 返回多边形有向面积的两倍；整数版为 i128，浮点版为 ld
Area2 poly_area2(const vector<Point> &poly) {
    Area2 area2{};
    for (int i = 0, n = poly.size(); i < n; ++ i) {
        area2 += cross(poly[i], poly[(i + 1) % n]);
    }
    return area2;
}
```
### 凸多边形规范化与凸性判断
**解决问题：** 统一凸多边形方向和起点，删除重复点与共线冗余点，使凸包含、Minkowski 和等接口满足前提；也可判断一个按边界排列的多边形是否为凸多边形
`norm_conv()` 进行以下处理：
1. 删除重复末点和连续重复点
2. 非退化多边形统一为逆时针
3. 删除位于相邻两点之间的共线冗余点
4. 将起点旋转为 $(y,x)$ 字典序最小的点
5. 全部共线时只保留线段的两个端点
```cpp
// 规范化按边界顺序给出的凸多边形；输入不是凸集边界时结果没有意义
vector<Point> norm_conv(vector<Point> poly) {
    if (poly.size() > 1 && poly.front() == poly.back()) poly.pop_back();
    vector<Point> pts;
    for (Point p : poly) {
        if (pts.empty() || p != pts.back()) pts.push_back(p);
    }
    if (pts.size() > 1 && pts.front() == pts.back()) pts.pop_back();
    if (pts.size() <= 1) return pts;
    int dir{ sgn(poly_area2(pts)) };
    if (dir == 0) {
        ranges::sort(pts, less{});
        pts.erase(ranges::unique(pts).begin(), pts.end());
        if (pts.size() <= 1) return pts;
        return { pts.front(), pts.back() };
    }
    if (dir < 0) ranges::reverse(pts);
    vector<Point> res;
    int n = pts.size();
    for (int i = 0; i < n; ++ i) {
        Point pre{ pts[(i + n - 1) % n] }, cur{ pts[i] }, nxt{ pts[(i + 1) % n] };
        // cur 位于线段 [pre,nxt] 上时，它是凸多边形边界上的冗余点
        if (sgn(cross(pre, cur, nxt)) == 0 && sgn(dot(cur, pre, nxt)) <= 0) continue;
        res.push_back(cur);
    }
    int st{ ranges::min_element(res, [](Point a, Point b) {
        return a.y != b.y ? a.y < b.y : a.x < b.x;
    }) - res.begin() };
    ranges::rotate(res, res.begin() + st);
    return res;
}
// 判断按边界顺序给出的多边形是否为凸多边形；全部共线不视为凸多边形
bool is_conv(const vector<Point> &poly, bool allo_col = true) {
    int n = poly.size();
    if (n < 3) return false;
    int dir{ 0 };
    for (int i = 0; i < n; ++ i) {
        int turn{ sgn(cross(poly[i], poly[(i + 1) % n], poly[(i + 2) % n])) };
        if (turn == 0) {
            if (!allo_col) return false;
        } else if (dir == 0) dir = turn;
        else if (dir != turn) return false;
    }
    return dir != 0;
}
```
### 点与多边形的包含关系
**解决问题：** 判断查询点位于简单多边形外部、边界还是内部；凸多边形经一次规范化后可将单次查询优化到 $O(\log n)$
普通版本使用射线法：从查询点向右作射线，若与边界的有效穿越次数为奇数，则点在内部。代码使用半开纵坐标条件处理顶点，避免同一个顶点被相邻两条边重复计算
凸多边形版本把多边形划分为以 `poly[0]` 为公共顶点的三角形扇，先判断查询点是否位于首尾两条射线之间，再二分所在扇区
```cpp
enum class Contain {
    OUTSIDE,  // 多边形外部
    BOUNDARY, // 多边形边界
    INSIDE    // 多边形内部
};
// 整数版精确比较；浮点版按 EPS 判断两个点是否重合
bool same_geo(Point a, Point b) {
    if constexpr (is_integral_v<decltype(Point{}.x)>) return a == b;
    else return sgn(a.x - b.x) == 0 && sgn(a.y - b.y) == 0;
}
// 判断点与简单多边形的包含关系；多边形方向可以为顺时针或逆时针
Contain in_poly(const vector<Point> &poly, Point p) {
    int wind{ 0 };
    for (int i = 0, n = poly.size(); i < n; ++ i) {
        Point a{ poly[i] }, b{ poly[(i + 1) % n] };
        if (on_seg(p, Segment(a, b))) return Contain::BOUNDARY;
        int turn{ sgn(cross(a, b, p)) };
        if (a.y <= p.y) {
            if (b.y > p.y && turn > 0) ++ wind;
        } else if (b.y <= p.y && turn < 0) -- wind;
    }
    return wind == 0 ? Contain::OUTSIDE : Contain::INSIDE;
}
// poly 必须是 norm_conv() 的结果；支持空集、点、线段和严格凸多边形
Contain in_conv(const vector<Point> &poly, Point p) {
    int n = poly.size();
    if (n == 0) return Contain::OUTSIDE;
    if (n == 1) return same_geo(p, poly[0]) ? Contain::BOUNDARY : Contain::OUTSIDE;
    if (n == 2) return on_seg(p, Segment(poly[0], poly[1])) ? Contain::BOUNDARY : Contain::OUTSIDE;
    int l{ sgn(cross(poly[0], poly[1], p)) }, r{ sgn(cross(poly[0], poly.back(), p)) };
    if (l < 0 || r > 0) return Contain::OUTSIDE;
    if (l == 0) return on_seg(p, Segment(poly[0], poly[1])) ? Contain::BOUNDARY : Contain::OUTSIDE;
    if (r == 0) return on_seg(p, Segment(poly[0], poly.back())) ? Contain::BOUNDARY : Contain::OUTSIDE;
    int low{ 1 }, high{ n - 1 };
    while (high - low > 1) {
        int mid{ (low + high) >> 1 };
        if (sgn(cross(poly[0], poly[mid], p)) >= 0) low = mid;
        else high = mid;
    }
    int side{ sgn(cross(poly[low], poly[high], p)) };
    if (side < 0) return Contain::OUTSIDE;
    if (side == 0) return Contain::BOUNDARY;
    return Contain::INSIDE;
}
```
> [!warning] 浮点版中，`in_poly()` 的边界判定使用 `EPS`，射线的半开纵坐标比较仍使用原始 `y` 值。不要把半开比较随意改成多个互不一致的 `EPS` 判断，否则顶点附近可能重复计数
### 周长与重心
**解决问题：** 计算多边形边界总长度，以及均匀薄片多边形的几何重心
周长为
$$
L=\sum_{i=0}^{n-1}|p_{i+1}-p_i|
$$
非零面积均匀多边形的重心为
$$
C=\frac{\sum_{i=0}^{n-1}(p_i+p_{i+1})(p_i\times p_{i+1})}{3S_2}
$$
```cpp
// 返回闭合多边形的周长；空集、单点的周长为 0
ld poly_peri(const vector<Point> &poly) {
    ld peri{ 0 };
    for (int i = 0, n = poly.size(); i < n; ++ i) {
        peri += length(poly[(i + 1) % n] - poly[i]);
    }
    return peri;
}
// 返回均匀薄片多边形的重心；有向面积接近 0 时返回 nullopt
optional<Point> poly_ctr(const vector<Point> &poly) {
    ld area2{ poly_area2(poly) };
    if (sgn(area2) == 0) return nullopt;
    Point num;
    for (int i = 0, n = poly.size(); i < n; ++ i) {
        Point a{ poly[i] }, b{ poly[(i + 1) % n] };
        num += (a + b) * cross(a, b);
    }
    return num / (3 * area2);
}
```
### 线段是否完全位于简单多边形内
**解决问题：** 判断闭线段的每一个点是否都位于简单多边形内部或边界，适用于“可见线段是否越出区域”“运动路径是否合法”等问题。
将线段参数化为
$$
q(t)=a+t(b-a),\qquad 0\le t\le1
$$
收集线段与多边形边界相交处的全部参数 $t$，排序后，任意相邻参数之间的开区间不会再次穿越边界，因此只需检查每段中点
```cpp
// ===== 浮点版专属：依赖 line_int()，整数版若需要应改用有理数 =====
// 边界视为多边形的一部分；poly 必须是无重复末点的简单多边形
bool seg_in(const vector<Point> &poly, Segment seg) {
    if (in_poly(poly, seg.a) == Contain::OUTSIDE || in_poly(poly, seg.b) == Contain::OUTSIDE) {
        return false;
    }
    Point dir{ seg.b - seg.a };
    ld len2{ norm2(dir) };
    if (sgn(len2) == 0) return true;
    vector<ld> pars{0.L, 1.L};
    auto add_par = [&](Point p) {
        ld t{ dot(p - seg.a, dir) / len2 };
        if (t >= -EPS && t <= 1 + EPS) pars.push_back(clamp(t, 0.L, 1.L));
    };
    for (int i = 0, n = int(poly.size()); i < n; ++ i) {
        Segment edge(poly[i], poly[(i + 1) % n]);
        if (on_seg(edge.a, seg)) add_par(edge.a);
        if (on_seg(edge.b, seg)) add_par(edge.b);
        if (seg_int(seg, edge, true)) {
            auto isec = line_int(Line::through(seg.a, seg.b), Line::through(edge.a, edge.b));
            if (isec) add_par(*isec);
        }
    }
    ranges::sort(pars);
    pars.erase(unique(pars.begin(), pars.end(), [](ld a, ld b) {
        return sgn(a - b) == 0;
    }), pars.end());
    for (int i = 0; i + 1 < pars.size(); ++ i) {
        ld mid{ (pars[i] + pars[i + 1]) / 2 };
        Point p{ seg.a + dir * mid };
        if (in_poly(poly, p) == Contain::OUTSIDE) return false;
    }
    return true;
}
```
### 有向直线切割多边形
**解决问题：** 用一个半平面裁剪多边形；连续对凸多边形的每条边裁剪，可求两个凸多边形的交。
对于有向直线 `ln.p + t * ln.dir`，保留其左侧闭半平面：
$$
H=\{p\mid ln.dir\times(p-ln.p)\ge0\}.
$$
```cpp
// ===== 浮点版专属：交点一般不是整数点，依赖 line_int() =====
保留有向直线左侧和直线上的部分；输入顶点按边界顺序排列
vector<Point> cut_poly(const vector<Point> &poly, Line ln) {
    vector<Point> res;
    for (int i = 0, n = poly.size(); i < n; ++ i) {
        Point a{ poly[i] }, b{ poly[(i + 1) % n] };
        int sd_a{ pt_side(a, ln) }, sd_b{ pt_side(b, ln) };
        if (sd_a >= 0) res.push_back(a);
        if (sd_a * sd_b < 0) {
            auto isec = line_int(Line::through(a, b), ln);
            if (isec) res.push_back(*isec);
        }
    }
    return res;
}
求两个非退化凸多边形的交；返回结果可能为空、点、线段或凸多边形
vector<Point> conv_int(vector<Point> poly, vector<Point> clip) {
    poly = norm_conv(move(poly));
    clip = norm_conv(move(clip));
    if (poly.size() < 3 || clip.size() < 3) return {};
    for (int i = 0, n = clip.size(); i < n && !poly.empty(); ++ i) {
        poly = cut_poly(poly, Line::through(clip[i], clip[(i + 1) % n]));
    }
    return norm_conv(move(poly));
}
```
### Ear Clipping 三角剖分
**解决问题：** 将无孔简单多边形划分为 $n-2$ 个内部互不相交的三角形，便于面积积分、渲染或后续动态规划。
一个凸顶点 $b$ 与相邻顶点 $a,c$ 构成“耳”，当且仅当三角形 $\triangle abc$ 内部和边界上不含当前多边形的其他顶点。剪去一只耳后继续处理剩余多边形。
```cpp
// 返回三角形的原始顶点下标；失败或输入退化时返回空数组
vector<array<int, 3>> ear_clip(vector<Point> poly) {
    int n = poly.size();
    vector<array<int, 3>> tris;
    if (n < 3 || sgn(poly_area2(poly)) == 0) return tris;
    vector<int> org(n);
    ranges::iota(org);
    if (sgn(poly_area2(poly)) < 0) {
        ranges::reverse(poly);
        ranges::reverse(org);
    }
    vector<int> idx(n);
    ranges::iota(idx);
    auto in_tri = [&](Point p, Point a, Point b, Point c) {
        return sgn(cross(a, b, p)) >= 0 && sgn(cross(b, c, p)) >= 0 && sgn(cross(c, a, p)) >= 0;
    };
    while (idx.size() > 3) {
        bool found{ false };
        for (int at = 0; at < idx.size(); ++ at) {
            int pre{ idx[(at + idx.size() - 1) % idx.size()] };
            int cur{ idx[at] }, nxt{ idx[(at + 1) % idx.size()] };
            if (sgn(cross(poly[pre], poly[cur], poly[nxt])) <= 0) continue;
            bool has{ false };
            for (int v : idx) {
                if (v == pre || v == cur || v == nxt) continue;
                if (in_tri(poly[v], poly[pre], poly[cur], poly[nxt])) {
                    has = true;
                    break;
                }
            }
            if (has) continue;
            tris.push_back({org[pre], org[cur], org[nxt]});
            idx.erase(idx.begin() + at);
            found = true;
            break;
        }
        if (!found) return {};
    }
    tris.push_back({org[idx[0]], org[idx[1]], org[idx[2]]});
    return tris;
}
```
### Pick 定理与格点计数
**解决问题：** 对顶点均为整点的简单多边形，计算边界格点数和内部格点数
对边 $a\to b$，令 $\Delta x=b_x-a_x$、$\Delta y=b_y-a_y$，该边对边界格点总数的贡献为
$$
\gcd(|\Delta x|,|\Delta y|)
$$
设内部格点数为 $I$、边界格点数为 $B$、实际面积为 $S$，Pick 定理为
$$
S=I+\frac{B}{2}-1,
\qquad
I=\frac{|S_2|-B+2}{2}
$$
```cpp
// ===== 整数版专属：浮点坐标不适用 Pick 定理 =====
i128 gcd128(i128 a, i128 b) {
    if (a < 0) a = -a;
    if (b < 0) b = -b;
    while (b) {
        a %= b;
        swap(a, b);
    }
    return a;
}
// 返回 { 边界格点数 B, 内部格点数 I }
pair<i128, i128> lat_cnt(const vector<Point> &poly) {
    i128 bnd{ 0 }, area2{ 0 };
    for (int i = 0, n = poly.size(); i < n; ++ i) {
        Point a{ poly[i] }, b{ poly[(i + 1) % n] };
        bnd += gcd128(i128(b.x) - a.x, i128(b.y) - a.y);
        area2 += cross(a, b);
    }
    if (area2 < 0) area2 = -area2;
    return { bnd, (area2 - bnd + 2) / 2 };
}
```
### Minkowski 和
**解决问题：** 合成两个凸集合的所有向量和，常用于凸多边形平移、碰撞判定、距离问题以及把“两点分别取自两个集合”的问题转化为点包含
两个集合的 Minkowski 和定义为
$$
A\oplus B=\{a+b\mid a\in A,\ b\in B\}
$$
凸多边形的边向量按极角有序，$A\oplus B$ 的边界向量正是两组边向量的归并；同方向边应合并为向量和，反方向边不能合并
```cpp
// 求两个凸多边形、线段或点集凸边界的 Minkowski 和
vector<Point> mink_sum(vector<Point> a, vector<Point> b) {
    a = norm_conv(move(a));
    b = norm_conv(move(b));
    if (a.empty() || b.empty()) return {};
    if (a.size() == 1) {
        for (Point &p : b) p += a[0];
        return b;
    }
    if (b.size() == 1) {
        for (Point &p : a) p += b[0];
        return a;
    }
    vector<Point> edge_a(a.size()), edge_b(b.size());
    for (int i = 0; i < a.size(); ++ i) edge_a[i] = a[(i + 1) % a.size()] - a[i];
    for (int i = 0; i < b.size(); ++ i) edge_b[i] = b[(i + 1) % b.size()] - b[i];
    vector<Point> res{ a[0] + b[0] };
    int i{ 0 }, j{ 0 };
    while (i < edge_a.size() || j < edge_b.size()) {
        Point edge;
        if (i == edge_a.size()) edge = edge_b[j ++];
        else if (j == edge_b.size()) edge = edge_a[i ++];
        else {
            bool same_dir{ sgn(cross(edge_a[i], edge_b[j])) == 0 && sgn(dot(edge_a[i], edge_b[j])) > 0 };
            if (same_dir) edge = edge_a[i ++] + edge_b[j ++];
            else if (arg_less(edge_a[i], edge_b[j])) edge = edge_a[i ++];
            else edge = edge_b[j ++];
        }
        res.push_back(res.back() + edge);
    }
    res.pop_back(); // 最后一个点与起点重合
    return norm_conv(move(res));
}
```
### 多边形核
**解决问题：** 判断简单多边形是否存在一个点，使该点能够看见多边形内的所有点，并求出所有这类点组成的区域。
若简单多边形按逆时针排列，则其核为每条有向边左侧闭半平面的交：
$$
\operatorname{Kernel}(P)=\bigcap_{i=0}^{n-1}
\{x\mid (p_{i+1}-p_i)\times(x-p_i)\ge0\}
$$
将每条边构造成 `Line::through(poly[i], poly[(i + 1) % n])`，调用半平面交板子即可；交集为空表示多边形没有核。
### 接口说明
#### 公共约定
- 多边形顶点按边界顺序给出，不重复存储末尾的首点；空集、点和线段仅在接口明确说明时允许
- `poly_area2()`、`in_poly()`、`is_conv()`、`ear_clip()`、`norm_conv()` 和 `mink_sum()` 可用于整数版与浮点版
- 浮点版的几何判定统一通过基础板子的 `sgn()` 使用 `EPS`；排序仍使用原始坐标比较
- `norm_conv()` 只负责规范化已经按凸集边界排列的输入，不会把任意点集求成凸包
#### 面积、方向与规范化
- `poly_area2(poly)`：返回有向面积的两倍；正数表示逆时针，负数表示顺时针，$O(n)$
- `norm_conv(poly)`：规范化凸多边形、线段或点；删除共线冗余点并统一方向、起点，$O(n\log n)$，其中只有退化为共线点集时需要排序，非退化输入为 $O(n)$
- `is_conv(poly, allo_col)`：判断按边界排列的多边形是否凸；`allo_col=false` 时任意连续三点共线都会返回 `false`，$O(n)$
#### 包含关系
- `Contain::OUTSIDE / BOUNDARY / INSIDE`：分别表示外部、边界和内部
- `same_geo(a,b)`：整数版精确判等，浮点版按 `EPS` 判断点重合，$O(1)$
- `in_poly(poly, p)`：判断点与简单多边形的关系；支持顺、逆时针，$O(n)$
- `in_conv(poly, p)`：判断点与规范化凸集的关系；调用前应执行一次 `poly = norm_conv(poly)`，单次查询 $O(\log n)$
- `seg_in(poly, seg)`：判断闭线段是否完全位于简单多边形的内部或边界；仅浮点版，$O(n\log n)$
#### 构造与分解
- `poly_peri(poly)`：返回周长；仅浮点版，$O(n)$
- `poly_ctr(poly)`：返回均匀薄片多边形的重心；面积接近零时返回 `nullopt`，仅浮点版，$O(n)$
- `cut_poly(poly, ln)`：保留有向直线左侧闭半平面内的部分；仅浮点版，$O(n)$
- `conv_int(a, b)`：求两个非退化凸多边形的交；返回数组可能表示空集、点、线段或凸多边形，仅浮点版，逐边裁剪最坏 $O(nm)$
- `ear_clip(poly)`：返回 `n-2` 个三角形的原始顶点下标；输入不合法或无法继续剪耳时返回空数组，$O(n^3)$
#### 格点与 Minkowski 和
- `lat_cnt(poly)`：返回 `{B,I}`，即边界与内部格点数；仅整数版，要求整点简单多边形，$O(n)$
- `mink_sum(a,b)`：返回两个凸边界的 Minkowski 和并自动规范化；允许输入为空集、点、线段或非退化凸多边形，$O(n+m)$，退化输入规范化排序除外
### 注意事项
- `in_poly()` 只保证简单多边形语义；自交多边形会按非零环绕数规则分类，不应把它当作普通内部判定
- `in_conv()` 的 $O(\log n)$ 前提是输入已由 `norm_conv()` 处理；直接传入带首尾共线冗余点的普通凸多边形可能破坏扇形二分
- `poly_ctr()` 计算的是均匀面积薄片的重心，不是顶点坐标平均值；面积为零时没有该意义下的重心
- `seg_in()` 把边界视为合法区域；只要求严格位于内部时，还需要排除端点或中点落在 `BOUNDARY` 的情况。
- `ear_clip()` 要求无孔简单多边形。连续重复点、共线冗余点会阻止剪耳，应在保留原始下标映射的前提下预处理；带孔或约束三角剖分应使用 CDT，而不是本函数
- `cut_poly()` 与 `conv_int()` 会构造交点，因此默认只放在浮点版；整数输入的交点也可能是分数
- Pick 定理仅适用于整点简单多边形；`|S_2|-B+2` 理论上必为非负偶数，若不是，通常说明输入或面积、边界计数发生错误
- Minkowski 和要求输入按凸边界顺序给出。非凸多边形必须先做凸包或凸分解，不能直接归并原边向量
- `norm_conv()` 使用 `Point::operator==` 删除完全重复点；浮点输入若存在数值误差造成的近重复点，应在调用前按统一 `EPS` 清理
- 多边形核只对按逆时针排列的边使用左半平面；顺时针输入应先反转
### 常见用法
#### 计算面积并统一为逆时针
```cpp
Area2 area2 = poly_area2(poly);
if (sgn(area2) < 0) ranges::reverse(poly);
// 实际面积为 abs(area2) / 2；整数版可能出现半整数面积
```
#### 判断点是否位于简单多边形内或边界
```cpp
Contain where = in_poly(poly, q);
if (where != Contain::OUTSIDE) {} // 内部或边界
```
#### 多次查询点是否位于凸多边形
```cpp
poly = norm_conv(move(poly)); // 只预处理一次
for (Point q : qs) {
    Contain where = in_conv(poly, q);
}
```
#### 判断线段是否完全留在区域中

```cpp
// [浮点版专属]
if (seg_in(poly, Segment(a, b))) {} // 允许贴着边界移动
```
#### 用直线裁掉凸多边形的一部分
```cpp
// [浮点版专属]
Line ln = Line::through(a, b);
poly = cut_poly(poly, ln); // 只保留有向直线 a -> b 左侧
```
#### 求两个凸多边形交集的面积
```cpp
// [浮点版专属]
vector<Point> isec = conv_int(a, b);
ld area = fabsl(poly_area2(isec)) / 2;
```
#### 对简单多边形三角剖分
```cpp
auto tris = ear_clip(poly);
if (poly.size() >= 3 && tris.empty()) {} // 输入退化、非简单或存在未清理问题
for (auto [a, b, c] : tris) {
    // 三角形顶点为 poly[a]、poly[b]、poly[c]
}
```
#### 使用 Pick 定理
```cpp
// [整数版专属]
auto [bnd, is_in] = lat_cnt(poly);
```
#### 计算 Minkowski 和
```cpp
vector<Point> sum = mink_sum(a, b);
```
#### 用 Minkowski 和判断两个凸多边形是否相交
$A$ 与 $B$ 相交，当且仅当原点属于 $A\oplus(-B)$：
```cpp
vector<Point> neg_b = b;
for (Point &p : neg_b) p = -p;
vector<Point> dif = mink_sum(a, move(neg_b));
bool hit = in_conv(dif, Point{}) != Contain::OUTSIDE;
```
#### 求多边形核
```cpp
if (sgn(poly_area2(poly)) < 0) ranges::reverse(poly);
vector<Line> hps;
for (int i = 0, n = poly.size(); i < n; ++ i) {
    hps.push_back(Line::through(poly[i], poly[(i + 1) % n]));
}
// 将 hps 交给“半平面交”板子；结果为空表示无核
```
### 复杂度

| 操作 | 时间复杂度 |
|---|---:|
| `poly_area2()` / `poly_peri()` / `poly_ctr()` | $O(n)$ |
| `norm_conv()` | 非退化 $O(n)$；全部共线 $O(n\log n)$ |
| `is_conv()` / `in_poly()` | $O(n)$ |
| `in_conv()` | $O(\log n)$ |
| `seg_in()` | $O(n\log n)$ |
| `cut_poly()` | $O(n)$ |
| `conv_int()` | 最坏 $O(nm)$ |
| `ear_clip()` | $O(n^3)$ |
| `lat_cnt()` | $O(n)$ |
| `mink_sum()` | $O(n+m)$，不计退化输入排序 |
除返回结果外，各判定函数额外空间为 $O(1)$；需要构造新多边形的接口空间复杂度与输出点数同阶
