### 点、向量、直线与线段
---
#### 数值与版本约定
整数版的 `Point` 加减、数乘结果必须可由 `i64` 表示，叉积、距离平方及其累加结果必须可由有符号 `i128` 表示，转换为 `i128` 后计算不等于支持任意满范围 `i64` 输入的所有几何运算
浮点版使用 `EPS = 1E-12L` 进行几何分支判定，`EPS` 不是题目输出误差标准，最终答案通常按
$$
\frac{|a-b|}{\max(1,|b|)}\le 10^{-6}
$$
判断，其中 $a$ 为输出值，$b$ 为标准值。本仓库验证使用更严格的 `1e-9` 阈值；建议至少输出 `12` 位小数，输出位数本身不能保证数值算法精度
叉积、长度、长度平方具有不同量纲，不要通过 `sgn(norm2(v)) == 0` 判断方向向量是否退化
#### 整数结构体
```cpp
int sgn(i128 x) { return (x > 0) - (x < 0); }

struct Point {
    i64 x{ 0 }, y{ 0 };
    Point() = default;
    Point(i64 x_, i64 y_) : x(x_), y(y_) {}
    Point operator+() const { return *this; }
    Point operator-() const { return Point(-x, -y); }
    Point &operator+=(Point p) { x += p.x; y += p.y; return *this; }
    Point &operator-=(Point p) { x -= p.x; y -= p.y; return *this; }
    friend Point operator+(Point a, Point b) { return a += b; }
    friend Point operator-(Point a, Point b) { return a -= b; }
    Point &operator*=(i128 k) { x *= k; y *= k; return *this; }
    friend Point operator*(Point p, i128 k) { return p *= k; }
    friend Point operator*(i128 k, Point p) { return p *= k; }
    friend bool operator==(Point a, Point b) { return a.x == b.x && a.y == b.y; }
    friend bool operator!=(Point a, Point b) { return !(a == b); }
    // 原始坐标字典序
    friend auto operator<=>(Point a, Point b) = default;
};
bool same_pt(Point a, Point b) { return a == b; }
struct Line {
    Point p, dir;
    Line() = default;
    Line(Point p_, Point dir_) : p(p_), dir(dir_) {}
    // 构造经过 a、b 的直线，a == b 时退化为单点 a
    static Line through(Point a, Point b) { return Line(a, b - a); }
};
struct Segment {
    Point a, b;
    Segment() = default;
    Segment(Point a_, Point b_) : a(a_), b(b_) {}
};
```
#### 通用函数
##### 点与向量
```cpp
i128 dot(Point a, Point b) { return i128(a.x) * b.x + i128(a.y) * b.y; }
i128 cross(Point a, Point b) { return i128(a.x) * b.y - i128(a.y) * b.x; }
i128 dot(Point o, Point a, Point b) {
    return (i128(a.x) - o.x) * (i128(b.x) - o.x) + (i128(a.y) - o.y) * (i128(b.y) - o.y);
}
i128 cross(Point o, Point a, Point b) {
    return (i128(a.x) - o.x) * (i128(b.y) - o.y) - (i128(a.y) - o.y) * (i128(b.x) - o.x);
}
i128 norm2(Point p) { return dot(p, p); }
i128 dist2(Point a, Point b) {
    i128 dx{ i128(a.x) - b.x }, dy{ i128(a.y) - b.y };
    return dx * dx + dy * dy;
}
Point rotate90(Point p) { return Point(-p.y, p.x); }
// ===== 极角、方向分类 =====
// 返回极角所在半平面，上半平面为 0，下半平面为 1
// 浮点版同样使用原始比较，不能使用 sgn()，否则可能破坏严格弱序
int arg_half(Point p) { return p.y < 0 || (p.y == 0 && p.x < 0); }
// 从正 x 轴开始逆时针比较极角，同方向按长度升序
bool arg_less(Point a, Point b) {
    int ha{ arg_half(a) }, hb{ arg_half(b) };
    if (ha != hb) return ha < hb;
    auto turn{ cross(a, b) };
    if (turn != 0) return turn > 0;
    return norm2(a) < norm2(b);
}
enum class CCW {
    CLOCKWISE = -1,        // p 在有向直线 a -> b 右侧
    ON_SEGMENT = 0,        // p 在线段 [a,b] 上
    COUNTER_CLOCKWISE = 1, // p 在有向直线 a -> b 左侧
    ONLINE_BACK = 2,       // p 与 a、b 共线，但在 a 后方
    ONLINE_FRONT = -2      // p 与 a、b 共线，但越过 b
};
CCW ccw(Point a, Point b, Point p) {
    int turn{ sgn(cross(a, b, p)) };
    if (turn > 0) return CCW::COUNTER_CLOCKWISE;
    if (turn < 0) return CCW::CLOCKWISE;
    if (sgn(dot(a, b, p)) < 0) return CCW::ONLINE_BACK;
    if (sgn(dist2(a, p) - dist2(a, b)) > 0) return CCW::ONLINE_FRONT;
    return CCW::ON_SEGMENT;
}
```
##### 直线
```cpp
auto dir_cross(Line a, Line b) { return cross(a.dir, b.dir); }
auto dir_dot(Line a, Line b) { return dot(a.dir, b.dir); }
bool parallel(Line a, Line b) { return sgn(dir_cross(a, b)) == 0; }
bool perp(Line a, Line b) { return sgn(dir_dot(a, b)) == 0; }
int pt_side(Point p, Line ln) { return sgn(cross(ln.dir, p - ln.p)); }
bool on_line(Point p, Line ln) {
    if (ln.dir == Point{}) return same_pt(p, ln.p);
    return pt_side(p, ln) == 0;
}
enum class LineRel {
    INTERSECT, // 有唯一交点
    PARALLEL,  // 平行但不重合
    COINCIDENT // 重合
};
// 要求两条直线均非退化
LineRel line_rel(Line a, Line b) {
    if (!parallel(a, b)) return LineRel::INTERSECT;
    return on_line(b.p, a) ? LineRel::COINCIDENT : LineRel::PARALLEL;
}
```
##### 线段
```cpp
// 判断点是否位于闭线段 [a,b] 上，允许线段退化为单点
bool on_seg(Point p, Segment s) {
    return sgn(cross(s.a, s.b, p)) == 0 && sgn(dot(p, s.a, s.b)) <= 0;
}
// 判断两点是否位于直线同侧，stct = true 时不允许点在线上
bool same_sd(Point a, Point b, Line ln, bool stct = true) {
    int sa{ pt_side(a, ln) }, sb{ pt_side(b, ln) };
    if (stct) return sa != 0 && sa == sb;
    return sa == 0 || sb == 0 || sa == sb;
}
// 判断两条线段是否相交，stct = true 时只判断内部严格交叉
bool seg_int(Segment a, Segment b, bool stct = false) {
    int c1{ sgn(cross(a.a, a.b, b.a)) }, c2{ sgn(cross(a.a, a.b, b.b)) };
    int c3{ sgn(cross(b.a, b.b, a.a)) }, c4{ sgn(cross(b.a, b.b, a.b)) };
    bool prop{ c1 != 0 && c2 != 0 && c1 != c2 && c3 != 0 && c4 != 0 && c3 != c4 };
    if (stct) return prop;
    if (prop) return true;
    return (c1 == 0 && on_seg(b.a, a)) || (c2 == 0 && on_seg(b.b, a))
        || (c3 == 0 && on_seg(a.a, b)) || (c4 == 0 && on_seg(a.b, b));
}
```

#### 浮点 Point 与新增函数
浮点版使用本节结构体整体替换整数结构体，然后复制上面的通用函数，其中 `dot()`、`cross()`、`norm2()`、`dist2()` 的返回类型及函数体内的显式转换均由 `i128` 改为 `ld`
##### 浮点结构体
```cpp
using ld = long double;
constexpr ld EPS{ 1E-12L };
const ld PI = acosl(-1.L);
int sgn(ld x, ld eps = EPS) { return (x > eps) - (x < -eps); }
struct Point {
    ld x{ 0 }, y{ 0 };
    Point() = default;
    Point(ld x_, ld y_) : x(x_), y(y_) {}
    Point operator+() const { return *this; }
    Point operator-() const { return Point(-x, -y); }
    Point &operator+=(Point p) { x += p.x; y += p.y; return *this; }
    Point &operator-=(Point p) { x -= p.x; y -= p.y; return *this; }
    friend Point operator+(Point a, Point b) { return a += b; }
    friend Point operator-(Point a, Point b) { return a -= b; }
    Point &operator*=(ld k) { x *= k; y *= k; return *this; }
    Point &operator/=(ld k) { x /= k; y /= k; return *this; }
    friend Point operator*(Point p, ld k) { return p *= k; }
    friend Point operator*(ld k, Point p) { return p *= k; }
    friend Point operator/(Point p, ld k) { return p /= k; }
    // 原始值精确判等，几何判等使用 same_pt()
    friend bool operator==(Point a, Point b) { return a.x == b.x && a.y == b.y; }
    friend bool operator!=(Point a, Point b) { return !(a == b); }
    // 原始坐标字典序，不能引入 EPS
    friend auto operator<=>(Point a, Point b) = default;
};
bool same_pt(Point a, Point b) { return sgn(a.x - b.x) == 0 && sgn(a.y - b.y) == 0; }
struct Line {
    Point p, dir;
    Line() = default;
    Line(Point p_, Point dir_) : p(p_), dir(dir_) {}
    static Line through(Point a, Point b) { return Line(a, b - a); }
    optional<ld> y_at(ld x) const {
        if (sgn(dir.x) == 0) return nullopt;
        return p.y + dir.y * (x - p.x) / dir.x;
    }
    optional<ld> x_at(ld y) const {
        if (sgn(dir.y) == 0) return nullopt;
        return p.x + dir.x * (y - p.y) / dir.y;
    }
    // 构造 ax + by = c，a、b 均接近 0 时返回 nullopt
    static optional<Line> equation(ld a, ld b, ld c) {
        if (sgn(a) == 0 && sgn(b) == 0) return nullopt;
        Point q = fabsl(a) > fabsl(b) ? Point(c / a, 0) : Point(0, c / b);
        return Line(q, Point(-b, a));
    }
};
struct Segment {
    Point a, b;
    Segment() = default;
    Segment(Point a_, Point b_) : a(a_), b(b_) {}
};
```
##### 浮点新增函数
```cpp
// ===== 长度、角度与旋转 =====
ld length(Point p) { return sqrtl(norm2(p)); }
ld distance(Point a, Point b) { return sqrtl(dist2(a, b)); }
Point rotate(Point p, ld ang) {
    ld c{ cosl(ang) }, s{ sinl(ang) };
    return Point(p.x * c - p.y * s, p.x * s + p.y * c);
}
// 两个非零向量的无向夹角，范围为 [0, PI]
ld vec_ang(Point a, Point b) { return atan2l(fabsl(cross(a, b)), dot(a, b)); }
// ===== 直线交点、投影与距离 =====
optional<Point> line_int(Line a, Line b) {
    ld den{ cross(a.dir, b.dir) };
    if (sgn(den) == 0) return nullopt;
    ld t{ cross(b.p - a.p, b.dir) / den };
    return a.p + a.dir * t;
}
Point proj(Point p, Line ln) {
    if (ln.dir == Point{}) return ln.p;
    ld t{ dot(p - ln.p, ln.dir) / norm2(ln.dir) };
    return ln.p + ln.dir * t;
}
Point refl(Point p, Line ln) { return 2.L * proj(p, ln) - p; }
ld dis_line(Point p, Line ln) {
    if (ln.dir == Point{}) return distance(p, ln.p);
    return fabsl(cross(ln.dir, p - ln.p)) / length(ln.dir);
}
// 两条非退化直线的无向夹角，范围为 [0, PI]
ld line_ang(Line a, Line b) { return vec_ang(a.dir, b.dir); }
// ===== 线段距离 =====
ld dis_seg(Point p, Segment s) {
    Point dir{ s.b - s.a };
    if (dir == Point{}) return distance(p, s.a);
    ld t{ clamp(dot(p - s.a, dir) / norm2(dir), 0.L, 1.L) };
    return distance(p, s.a + dir * t);
}
ld seg_dis(Segment a, Segment b) {
    if (seg_int(a, b)) return 0.L;
    return min({ dis_seg(a.a, b), dis_seg(a.b, b), dis_seg(b.a, a), dis_seg(b.b, a) });
}
```
#### 接口说明
##### 基础结构
- `Point(x,y)`：构造点或向量
- `Line(p,dir)`：由一点和方向向量构造直线
- `Line::through(a,b)`：构造经过 `a,b` 的直线，$O(1)$
- `Segment(a,b)`：构造闭线段 $[a,b]$
##### 通用接口
- `dot(a,b)`：向量点积，$O(1)$
- `cross(a,b)`：向量叉积，$O(1)$
- `dot(o,a,b)`：$\overrightarrow{oa}\cdot\overrightarrow{ob}$，$O(1)$
- `cross(o,a,b)`：$\overrightarrow{oa}\times\overrightarrow{ob}$，$O(1)$
- `norm2(p)`：向量长度平方，$O(1)$
- `dist2(a,b)`：两点距离平方，$O(1)$
- `rotate90(p)`：向量逆时针旋转 $90^\circ$，$O(1)$
- `arg_less(a,b)`：从正 $x$ 轴开始逆时针极角比较，同向按长度升序
- `ccw(a,b,p)`：判断 `p` 相对有向线段 `a -> b` 的位置，$O(1)$
- `parallel(a,b)`：判断两直线方向是否平行，$O(1)$
- `perp(a,b)`：判断两直线方向是否垂直，$O(1)$
- `pt_side(p,ln)`：返回点在线的左侧 / 线上 / 右侧对应的 `1 / 0 / -1`，$O(1)$
- `on_line(p,ln)`：判断点是否在线上，支持退化直线，$O(1)$
- `line_rel(a,b)`：判断非退化直线唯一相交 / 平行 / 重合，$O(1)$
- `on_seg(p,s)`：判断点是否在闭线段上，$O(1)$
- `same_sd(a,b,ln,stct)`：判断两点是否位于直线同侧，$O(1)$
- `seg_int(a,b,stct)`：判断两线段是否相交，`stct = true` 时只接受内部严格交叉，$O(1)$
##### 浮点新增接口
- `same_pt(a,b)`：按 `EPS` 判断两点重合，$O(1)$
- `length(p)`：向量长度，$O(1)$
- `distance(a,b)`：两点欧氏距离，$O(1)$
- `rotate(p,ang)`：逆时针旋转 `ang` 弧度，$O(1)$
- `vec_ang(a,b)`：两个非零向量的无向夹角，$O(1)$
- `Line::equation(a,b,c)`：构造 $ax+by=c$，非法时返回 `nullopt`，$O(1)$
- `x_at(y)` / `y_at(x)`：根据一个坐标求直线上另一坐标，不存在唯一值时返回 `nullopt`，$O(1)$
- `line_int(a,b)`：两条非平行直线交点，平行或重合返回 `nullopt`，$O(1)$
- `proj(p,ln)`：点在线上的正交投影，$O(1)$
- `refl(p,ln)`：点关于直线的对称点，$O(1)$
- `dis_line(p,ln)`：点到直线距离，$O(1)$
- `line_ang(a,b)`：两条非退化直线的无向夹角，$O(1)$
- `dis_seg(p,s)`：点到线段距离，$O(1)$
- `seg_dis(a,b)`：两线段最短距离，$O(1)$
#### 注意事项
- 整数版的加减和数乘结果必须能由 `i64` 表示，叉积、点积、距离平方及其相关中间结果必须能由 `i128` 表示
- 浮点版的 `operator==` 和 `<=>` 比较原始坐标，几何意义下的点重合使用 `same_pt()`
- 浮点排序不能在比较器中使用 `EPS`，`arg_half()`、`arg_less()` 和字典序比较均保持原始值比较，否则可能破坏严格弱序
- `EPS` 只服务于几何分支，不代表最终答案允许误差
- 最终数值答案通常按 $\frac{|a-b|}{\max(1,|b|)}\le 10^{-6}$ 判断，建议输出至少 `12` 位小数
- `parallel()` 对零方向也会返回真，退化直线参与方向夹角、平行、垂直等问题时应根据题意单独处理
- `line_rel()` 要求两条直线均非退化
- `line_int()` 的 `nullopt` 同时表示平行与重合，需要区分时先调用 `line_rel()`
- `vec_ang()` 要求两个向量均非零，`line_ang()` 要求两条直线均非退化
- 判断浮点方向是否退化应使用 `v == Point{}`，只把精确零向量视为退化，不要使用 `sgn(norm2(v)) == 0`
#### 常见用法
##### 排序
```cpp
ranges::sort(p);                  // 原始坐标字典序
ranges::sort(v, arg_less);        // 从正 x 轴开始逆时针极角排序
```
##### 点积判断夹角
```cpp
int rel = sgn(dot(u, v));
if (rel > 0) {}       // 锐角
else if (rel < 0) {}  // 钝角
else {}               // 垂直
```
##### 叉积判断方向
```cpp
int turn = sgn(cross(a, b, p));
if (turn > 0) {}       // p 在 a -> b 左侧
else if (turn < 0) {}  // p 在 a -> b 右侧
else {}                // 共线
```
##### 构造直线并判断关系
```cpp
Line a = Line::through(p1, p2);
Line b = Line::through(q1, q2);
if (parallel(a, b)) {}
if (perp(a, b)) {}
LineRel rel = line_rel(a, b);
```
##### 判断点与线段
```cpp
if (on_line(p, ln)) {}
if (on_seg(p, seg)) {}
if (seg_int(a, b)) {}
if (seg_int(a, b, true)) {} // 仅内部严格交叉
```
##### 浮点旋转、交点与距离
```cpp
Point q = rotate(p, PI / 3);
ld ang = vec_ang(u, v);
auto isec = line_int(a, b);
if (isec) {}
Point foot = proj(p, ln);
ld d1 = dis_line(p, ln);
ld d2 = dis_seg(p, seg);
```
##### 浮点输出
```cpp
auto fix0 = [](ld x) { return sgn(x) == 0 ? 0.L : x; };
cout << fixed << setprecision(12) << fix0(p.x) << ' ' << fix0(p.y) << '\n';
```
