### 点、向量
```cpp
// [浮点版追加]
// using ld = long double;
// constexpr ld EPS{ 1E-12L };
// const ld PI = acosl(-1.L);

// 返回符号：负数 / 零 / 正数分别返回 -1 / 0 / 1
// [浮点版] 改为：
// int sgn(ld x, ld eps = EPS) { return (x > eps) - (x < -eps); }
int sgn(i128 x) { return (x > 0) - (x < 0); }
struct Point {
    i64 x{ 0 }, y{ 0 };
    Point() = default;
    Point(i64 x_, i64 y_) : x(x_), y(y_) {}
    Point operator+() const { return *this; }
    Point operator-() const { return Point(-x, -y); }
    Point &operator+=(Point p) { x += p.x; y += p.y; return *this; }
    Point &operator-=(Point p) { x -= p.x; y -= p.y; return *this; }
    friend Point operator+(Point a, Point b) { return a += b; }
    friend Point operator-(Point a, Point b) { return a -= b; }
    Point &operator*=(i128 k) { x *= k; y *= k; return *this; }
    friend Point operator*(Point p, i128 k) { return p *= k; }
    friend Point operator*(i128 k, Point p) { return p *= k; }
/*  [浮点版追加]
    标量除法
    Point &operator/=(ld k) { x /= k; y /= k; return *this; }
    friend Point operator/(Point p, ld k) { return p /= k; }
*/
    // [浮点版] 仍是原始值精确判等；几何判等应使用 same_pt()
    friend bool operator==(Point a, Point b) { return a.x == b.x && a.y == b.y; }
    friend bool operator!=(Point a, Point b) { return !(a == b); }
    // 原始坐标字典序，浮点版也不能加入 EPS
    friend bool operator<(Point a, Point b) { return a.x != b.x ? a.x < b.x : a.y < b.y; }
};
// 点积
i128 dot(Point a, Point b) { return i128(a.x) * b.x + i128(a.y) * b.y; }
// 叉积
i128 cross(Point a, Point b) { return i128(a.x) * b.y - i128(a.y) * b.x; }
// 以 o 为原点，计算向量 oa 与 ob 的点积
i128 dot(Point o, Point a, Point b) { return (i128(a.x) - o.x) * (i128(b.x) - o.x) + (i128(a.y) - o.y) * (i128(b.y) - o.y); }
// 以 o 为原点，计算向量 oa 与 ob 的叉积
i128 cross(Point o, Point a, Point b) { return (i128(a.x) - o.x) * (i128(b.y) - o.y) - (i128(a.y) - o.y) * (i128(b.x) - o.x); }
// 返回向量模长的平方
i128 norm2(Point p) { return dot(p, p); }
// 返回两点欧氏距离的平方
i128 dist2(Point a, Point b) {
    i128 dx{ i128(a.x) - b.x }, dy{ i128(a.y) - b.y };
    return dx * dx + dy * dy;
}
/* [浮点版追加]
向量模长
ld length(Point p) { return sqrtl(norm2(p)); }
[浮点版追加] 两点之间的欧氏距离
ld distance(Point a, Point b) { return length(a - b); }
[浮点版追加] 按 EPS 判断两个点是否重合
bool same_pt(Point a, Point b) { return sgn(a.x - b.x) == 0 && sgn(a.y - b.y) == 0; }
*/
// 将向量逆时针旋转 90°；整数版和浮点版均可直接使用
Point rotate90(Point p) { return Point(-p.y, p.x); }
/* [浮点版追加]
将向量逆时针旋转 ang 弧度
Point rotate(Point p, ld ang) {
    ld c{ cosl(ang) }, s{ sinl(ang) };
    return Point(p.x * c - p.y * s, p.x * s + p.y * c);
}
*/

// ===== 极角、夹角：若不需要方向分类，以下代码可直接删除 =====
// 返回极角所在半平面：上半平面为 0，下半平面为 1
// 浮点版保持原始比较，不能使用 sgn()，否则可能破坏严格弱序
int arg_half(Point p) { return p.y < 0 || (p.y == 0 && p.x < 0); }
// 从正 x 轴开始逆时针比较极角，同方向按长度升序；供 sort 使用
bool arg_less(Point a, Point b) {
    int ha{ arg_half(a) }, hb{ arg_half(b) };
    if (ha != hb) return ha < hb;
    // 浮点版仍使用 turn != 0，不能改为 sgn(turn) != 0
    i128 turn{ cross(a, b) };
    if (turn != 0) return turn > 0;
    return norm2(a) < norm2(b);
}
enum class CCW {
    CLOCKWISE = -1,        // p 在有向直线 a -> b 右侧
    ON_SEGMENT = 0,        // p 在线段 [a,b] 上
    COUNTER_CLOCKWISE = 1, // p 在有向直线 a -> b 左侧
    ONLINE_BACK = 2,       // p 与 a,b 共线，但在 a 的后方
    ONLINE_FRONT = -2      // p 与 a,b 共线，但越过了 b
};
// 将 p 相对有向线段 a -> b 分为左侧、右侧及三种共线位置
CCW ccw(Point a, Point b, Point p) {
    // 整数版精确判断；浮点版会自动调用带 EPS 的 sgn(ld)
    int turn{ sgn(cross(a, b, p)) };
    if (turn > 0) return CCW::COUNTER_CLOCKWISE;
    if (turn < 0) return CCW::CLOCKWISE;
    // [浮点版] 改为：
    // if (sgn(dot(a, b, p)) < 0) return CCW::ONLINE_BACK;
    if (dot(a, b, p) < 0) return CCW::ONLINE_BACK;
    // [浮点版] 改为：
    // if (sgn(dist2(a, p) - dist2(a, b)) > 0) return CCW::ONLINE_FRONT;
    if (dist2(a, p) > dist2(a, b)) return CCW::ONLINE_FRONT;
    return CCW::ON_SEGMENT;
}
// 
/* [浮点版追加]
返回两个非零向量的无向夹角，范围为 [0,PI]
ld vec_ang(Point a, Point b) { return atan2l(fabsl(cross(a, b)), dot(a, b)); }
*/
```
### 直线
```cpp
// ===== 直线：若不需要直线判定与浮点构造，以下代码可直接删除 =====
struct Line {  
    Point p, dir;  
    Line() = default;  
    Line(Point p_, Point d_) : p(p_), dir(d_) {}  
    // 构造经过 a、b 的直线；a == b 时退化为单点 a  
    static Line through(Point a, Point b) { return Line(a, b - a); }  
/* [浮点版追加]  
    // 给定 x，返回直线上对应的 y；竖直线无法唯一确定时返回 nullopt
    optional<ld> y_at(ld x) const {
        if (sgn(dir.x) == 0) return nullopt;
        return p.y + dir.y * (x - p.x) / dir.x;
    }
    // 给定 y，返回直线上对应的 x；水平线无法唯一确定时返回 nullopt
    optional<ld> x_at(ld y) const {
        if (sgn(dir.y) == 0) return nullopt;
        return p.x + dir.x * (y - p.y) / dir.y;
    }
    // 构造 ax + by = c；a、b 均为 0 时返回 nullopt
    static optional<Line> equation(ld a, ld b, ld c) {
        if (sgn(a) == 0 && sgn(b) == 0) return nullopt;
        Point p(fabsl(a) > fabsl(b) ? Point(c / a, 0) : Point(0, c / b));
        return Line(p, Point(-b, a));
    }
*/
};
// 返回两条直线方向向量的叉积
i128 dir_cross(Line a, Line b) { return cross(a.dir, b.dir); }
// 返回两条直线方向向量的点积
i128 dir_dot(Line a, Line b) { return dot(a.dir, b.dir); }
// 判断两条直线是否平行, 方向向量为零时，也会与任意方向判为平行
bool parallel(Line a, Line b) { return sgn(dir_cross(a, b)) == 0; }
// 判断两条直线的方向向量是否垂直
// 退化直线的方向为零，也会与任意方向点积为零，但没有真实垂直含义
bool perp(Line a, Line b) { return sgn(dir_dot(a, b)) == 0; }
// 返回点相对有向直线的位置：左侧 / 线上 / 右侧为 1 / 0 / -1
int pt_side(Point p, Line ln) {
    i128 dx{ i128(p.x) - ln.p.x }, dy{ i128(p.y) - ln.p.y };
    return sgn(i128(ln.dir.x) * dy - i128(ln.dir.y) * dx);
}
// 判断点是否位于直线上；直线退化为点时仅该点位于其上
bool on_line(Point p, Line ln) {
    if (sgn(norm2(ln.dir)) == 0) return p == ln.p;
    // [浮点版] 改为：return same_pt(p, ln.p);
    return pt_side(p, ln) == 0;
}
enum class LineRel {
    INTERSECT, // 相交，有唯一交点
    PARALLEL,  // 平行但不重合，没有交点
    COINCIDENT // 重合，有无穷多个交点
};
// 判断两条非退化直线相交、平行或重合
LineRel line_rel(Line a, Line b) {
    if (!parallel(a, b)) return LineRel::INTERSECT;
    return on_line(b.p, a) ? LineRel::COINCIDENT : LineRel::PARALLEL;
}
/* ===== 浮点版专属：若只需要整数几何判定，以下代码不用添加 =====
求两条非平行直线的交点；平行或重合均返回 nullopt
optional<Point> line_int(Line a, Line b) {
    ld den{ cross(a.dir, b.dir) };
    if (sgn(den) == 0) return nullopt;
    ld t{ cross(b.p - a.p, b.dir) / den };
    return a.p + a.dir * t;
}
求点在线上的正交投影；直线退化为点时返回 ln.p
Point proj(Point p, Line ln) {
    ld len2{ norm2(ln.dir) };
    if (sgn(len2) == 0) return ln.p;
    ld t{ dot(p - ln.p, ln.dir) / len2 };
    return ln.p + ln.dir * t;
}
求点关于直线的对称点；直线退化为点时按 ln.p 中心对称
Point refl(Point p, Line ln) { return 2 * proj(p, ln) - p; }
求点到直线的距离；直线退化为点时返回两点距离
ld dis_line(Point p, Line ln) {
    ld len2{ norm2(ln.dir) };
    if (sgn(len2) == 0) return distance(p, ln.p);
    return fabsl(cross(ln.dir, p - ln.p)) / sqrtl(len2);
}
求两条非退化直线方向向量的无向夹角，范围为 [0, PI]
ld line_ang(Line a, Line b) { return vec_ang(a.dir, b.dir); }
*/
```
### 线段
```cpp
// ===== 线段：若不需要线段判定与距离，以下代码可直接删除 =====
struct Segment {
    Point a, b;
    Segment() = default;
    Segment(Point a_, Point b_) : a(a_), b(b_) {}
};
// 判断点是否位于闭线段 [a,b] 上，包含端点并允许线段退化为单点
bool on_seg(Point p, Segment s) {
    return sgn(cross(s.a, s.b, p)) == 0 && sgn(dot(p, s.a, s.b)) <= 0;
}
// 判断两点是否位于直线同侧；stct = true 时不允许点在线上
bool same_sd(Point a, Point b, Line ln, bool stct = true) {
    int sa{ pt_side(a, ln) }, sb{ pt_side(b, ln) };
    if (stct) return sa != 0 && sa == sb;
    return sa == 0 || sb == 0 || sa == sb;
}
// 判断两条线段是否相交；stct = true 时不含端点接触和共线重叠
bool seg_int(Segment a, Segment b, bool stct = false) {
    int c1{ sgn(cross(a.a, a.b, b.a)) }, c2{ sgn(cross(a.a, a.b, b.b)) };
    int c3{ sgn(cross(b.a, b.b, a.a)) }, c4{ sgn(cross(b.a, b.b, a.b)) };
    bool prop{ c1 != 0 && c2 != 0 && c1 != c2 && c3 != 0 && c4 != 0 && c3 != c4 };
    if (stct) return prop;
    if (prop) return true;
    return (c1 == 0 && on_seg(b.a, a)) || (c2 == 0 && on_seg(b.b, a))
        || (c3 == 0 && on_seg(a.a, b)) || (c4 == 0 && on_seg(a.b, b));
}
/* ===== 浮点版专属：若只需要整数几何判定，以下代码不要添加 =====
求点到线段的最短距离，允许线段退化为一个点
ld dis_seg(Point p, Segment s) {
    Point dir{ s.b - s.a };
    ld len2{ norm2(dir) };
    if (sgn(len2) == 0) return distance(p, s.a);
    return distance(p, s.a + dir * clamp(dot(p - s.a, dir) / len2, 0.L, 1.L));
}
求两条线段的最短距离；相交时返回 0
ld seg_dis(Segment a, Segment b) {
    if (seg_int(a, b)) return 0.L;
    return min({ dis_seg(a.a, b), dis_seg(a.b, b), dis_seg(b.a, a), dis_seg(b.b, a) });
}
*/
```
### 常见用法
#### 排序
`Point` 已重载 `<`，两个版本都可以直接排序：
```cpp
ranges::sort(p);           // 按 { x, y } 字典序排序
ranges::sort(v, arg_less); // 极角排序
// 从正 x 轴开始逆时针排序，同方向的向量按长度从小到大排列
```
> [!warning]  浮点版的 `<`、`arg_half()` 和 `arg_less()` 仍使用原始值比较，不能改为 `sgn()`，否则可能破坏 `sort` 要求的严格弱序
#### 点积判断夹角
```cpp
int rel = sgn(dot(u, v));
if (rel > 0) {}      // 夹角为锐角
else if (rel < 0) {} // 夹角为钝角
else {}              // 两个非零向量垂直
```
两个版本写法相同：
- 整数版的 `sgn(i128)` 精确判断
- 浮点版的 `sgn(ld)` 按 `EPS` 判断
#### 叉积判断方向
```cpp
int turn = sgn(cross(a, b, p));
if (turn > 0) {}      // p 在有向直线 a -> b 左侧
else if (turn < 0) {} // p 在有向直线 a -> b 右侧
else {}               // a、b、p 共线
```
#### 判断向量平行或垂直
```cpp
Point u = b - a, v = d - c;
bool is_par = sgn(cross(u, v)) == 0; // 平行
bool is_perp = sgn(dot(u, v)) == 0;  // 垂直
```
这里应保证需要判断垂直的两个向量均不是零向量
#### 向量旋转
逆时针旋转 $90^\circ$ 两个版本均可使用：
```cpp
Point q = rotate90(p);
```
任意角旋转只适用于浮点版：
```cpp
// [浮点版专属]
Point q1 = rotate(p, PI / 3);          // 逆时针旋转 60°
Point q2 = o + rotate(p - o, ang);  // 绕点 o 逆时针旋转 ang
Point q3 = rotate(p, -PI / 2);        // 顺时针旋转 90°
```
#### 计算两个向量的夹角
```cpp
// [浮点版专属]
Point u = b - a, v = d - c;
ld ang = vec_ang(u, v); // 弧度
ld deg = ang * 180 / PI;  // 角度制
```
需要保证 `u`、`v` 均为非零向量
#### 判断点相对有向线段的位置
```cpp
CCW pos = ccw(a, b, p);
switch (pos) {
case CCW::COUNTER_CLOCKWISE: // p 在有向直线 a -> b 左侧
    break;
case CCW::CLOCKWISE: // p 在有向直线 a -> b 右侧
    break;
case CCW::ON_SEGMENT: // p 在线段 [a,b] 上，包括端点
    break;
case CCW::ONLINE_BACK: // p、a、b 共线，且 p 位于 a 的后方
    break;
case CCW::ONLINE_FRONT: // p、a、b 共线，且 p 越过了 b
    break;
}
```
只判断点是否在线段上时，也可以直接使用：
```cpp
if (ccw(a, b, p) == CCW::ON_SEGMENT) {} // p 在线段 [a,b] 上
```
不过已有 `on_seg()` 时，直接使用线段接口更简洁
#### 将点绕指定中心按极角排序
将点集按照相对于点 `o` 的极角排序：
```cpp
ranges::sort(p, [&](Point a, Point b) {
    return arg_less(a - o, b - o);
});
```
同方向时，当前 `arg_less()` 会按照到 `o` 的距离从小到大排序
#### 直接构造直线
由直线上一点和方向向量构造：
```cpp
Line ln(Point(1, 2), Point(3, 4));
```
它表示：$(1,2)+t(3,4)$
经过两点构造：
```cpp
Line ln = Line::through(a, b);
```
> [!warning]  当 $a=b$ 时，直线退化为点 $a$。部分函数有退化处理，但平行、垂直和普通直线关系不再具有完整的几何意义，应尽量避免。
#### 由一般式构造直线
浮点版可以直接由 $ax+by=c$ 构造直线
例如构造：$2x-3y=6$
```cpp
// [浮点版专属]
auto ln = Line::equation(2, -3, 6);
if (ln) {} // 使用 *ln、ln->p 或 ln->dir
```
完整写法：
```cpp
// [浮点版专属]
auto res = Line::equation(a, b, c);
if (!res) {} // a、b 均接近 0，无法表示一条直线
else {
    Line ln = *res;
}
```
整数版的 `Line { p, dir }` 要求 `p` 为整数点，而一般的整数系数直线不一定经过整数点，因此默认不提供该构造函数。
#### 判断直线平行、垂直与点在线上
```cpp
Line l1 = Line::through(a, b);
Line l2 = Line::through(c, d);
if (parallel(l1, l2)) {}      // 两条非退化直线平行或重合
if (perp(l1, l2)) {} // 两条非退化直线垂直
if (on_line(p, l1)) {}        // p 位于 l1 上
```
两个版本写法相同：
- 整数版精确判断
- 浮点版通过 `sgn(ld)` 按 `EPS` 判断
#### 判断两条直线的位置关系
```cpp
LineRel rel = line_rel(l1, l2);
switch (rel) {
case LineRel::INTERSECT: // 两条直线相交，有唯一交点
    break;
case LineRel::PARALLEL: // 两条直线平行但不重合
    break;
case LineRel::COINCIDENT: // 两条直线重合
    break;
}
```
也可以直接判断：
```cpp
if (line_rel(l1, l2) == LineRel::COINCIDENT) {} // 两条直线重合
```
> [!warning]  `line_rel()` 要求 `l1`、`l2` 均为非退化直线
#### 求两条直线的交点
```cpp
// [浮点版专属]
Line l1 = Line::through(a, b), l2 = Line::through(c, d);
auto p = line_int(l1, l2);
if (p) cout << fixed << setprecision(12) << p->x << ' ' << p->y << '\n';
else {} // 两条直线平行或重合
```
需要区分平行与重合时：
```cpp
// [浮点版专属]
auto p = line_int(l1, l2);
if (p) {} // 唯一交点为 *p
else if (line_rel(l1, l2) == LineRel::COINCIDENT) {} // 两条直线重合
else {} // 两条直线平行但不重合
```
整数坐标直线的交点可能是分数，因此默认整数版不提供返回 `Point<i64>` 的交点函数
#### 求点到直线的投影
已知直线上的一点 $A$ 和直线的**方向向量** $\vec{d}$，以及直线外的一点 $P$：
$$ P' = A + \frac{\vec{AP}\cdot \vec{d}}{\vec{d} \cdot \vec{d}} d $$
仅适用于二维平面，已知二维平面上的直线方程为一般式 $Ax + By + C = 0$，且直线外有一点 $P(x_0, y_0)$：
$$(x', y') = (x_0 - A \frac{Ax_0 + By_0 + C}{A^2 + B^2},\quad y_0 - B \frac{Ax_0 + By_0 + C}{A^2 + B^2})$$

```cpp
// [浮点版专属]
Line ln = Line::through(a, b);
Point foot = proj(p, ln);
```
`foot` 是从 `p` 向直线作垂线得到的垂足
直线退化为点时，投影结果为 `ln.p`：
```cpp
if (same_pt(foot, ln.p)) {} // 两点在 EPS 意义下重合
```
#### 求点关于直线的对称点
```cpp
// [浮点版专属]
Line ln = Line::through(a, b);
Point q = refl(p, ln);
```
此时 `proj(p,ln)` 是线段 `pq` 的中点
直线退化为点时，结果为 `p` 关于 `ln.p` 的中心对称点
#### 求点到直线的距离
```cpp
// [浮点版专属]
Line ln = Line::through(a, b);
ld dis = dis_line(p, ln);
```
直线退化为点时，返回：
```cpp
distance(p, ln.p)
```
#### 求两条直线的夹角
```cpp
// [浮点版专属]
Line l1 = Line::through(a, b), l2 = Line::through(c, d);
ld ang = line_ang(l1, l2); // 弧度
ld deg = ang * 180 / PI;  // 角度制
```
返回范围为 $[0,\pi]$
若只需要两条无向直线间的较小夹角：
```cpp
// [浮点版专属]
ld ang = line_ang(l1, l2);
ld sm_ang = min(ang, PI - ang);
```
此时 `sm_ang` 的范围为：$\left[0,\frac{\pi}{2}\right]$
#### 创建线段
```cpp
Segment s(a, b);
```
线段允许退化为单点：
```cpp
Segment pt_seg(a, a);
```
#### 判断点是否在线段上
```cpp
Segment s(a, b);
if (on_seg(p, s)) {} // p 在线段 [a,b] 上，包括两个端点
```
整数版精确判断，浮点版按 `EPS` 判断
#### 判断两点是否位于直线同侧
```cpp
if (same_sd(a, b, ln)) {} // a、b 严格位于 ln 同一侧
if (same_sd(a, b, ln, false)) {} // a、b 位于同侧，或至少一个点在线上
```
判断两个点是否严格位于直线异侧：
```cpp
int sa = pt_side(a, ln), sb = pt_side(b, ln);
if (sa * sb < 0) {} // a、b 严格位于直线两侧
```
> [!warning]  直线退化为点时不存在通常意义上的“同侧”和“异侧”
#### 判断两条线段是否相交
包含端点接触和共线重叠：
```cpp
Segment s1(a, b), s2(c, d);
if (seg_int(s1, s2)) {} // 两条线段至少有一个公共点
```
只判断严格相交：
```cpp
if (seg_int(s1, s2, true)) {} // 两条线段在各自内部交叉
```
以下情况在非严格模式中算相交，在严格模式中不算：
```text
端点接触
共线且部分重叠
完全重合
退化线段落在另一条线段上
```
#### 求点到线段的距离
```cpp
// [浮点版专属]
Segment s(a, b);
ld dis = dis_seg(p, s);
```
线段退化为点时，会自动返回：
```cpp
distance(p, a)
```
也可以取得线段上距离 `p` 最近的点：
```cpp
// [浮点版专属]
Point dir = s.b - s.a, near;
ld len2 = norm2(dir);
if (sgn(len2) == 0) near = s.a;
else near = s.a + dir * clamp(dot(p - s.a, dir) / len2, 0.L, 1.L);
```
#### 求两条线段的最短距离
```cpp
// [浮点版专属]
Segment s1(a, b), s2(c, d);
ld dis = seg_dis(s1, s2);
```
两条线段相交、端点接触或共线重叠时，结果均为 `0`
#### 输出浮点结果并消除负零
```cpp
// [浮点版]
auto fix0 = [](ld x) { return sgn(x) == 0 ? 0.L : x; };
cout << fixed << setprecision(12) << fix0(p.x) << ' ' << fix0(p.y) << '\n';
```
