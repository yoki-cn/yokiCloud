### Delaunay 三角剖分
**解决问题：** 将平面点集连接成满足空圆性质的平面三角网，并导出原输入下标表示的边与三角形
Delaunay 三角形满足：其外接圆内部不含其他输入点。四点共圆时剖分不唯一

![Delaunay 三角剖分与空外接圆性质|353](/algorithm-assets/delaunay-empty-circumcircles.png)

```cpp
struct Delaunay {
    struct Vertex {
        Point p;
        int id{ -1 };
    };
private:
    struct Edge {
        int to{ 0 };
        list<Edge>::iterator rev;
        explicit Edge(int to_ = 0) : to(to_) {}
    };
    vector<list<Edge>> wk;
    static Point3 lift(Point p) { return Point3(p.x, p.y, norm2(p)); }
    // 返回 -1 / 0 / 1：p 在 abc 外接圆内 / 圆上 / 圆外
    static int in_cir(Point a, Point b, Point c, Point p) {
        if (sgn(cross(a, b, c)) < 0) swap(b, c);
        Point3 a3{ lift(a) }, b3{ lift(b) }, c3{ lift(c) }, p3{ lift(p) };
        b3 -= a3;
        c3 -= a3;
        p3 -= a3;
        return sgn(dot(p3, cross(b3, c3)));
    }
    void add_edge(int u, int v) {
        wk[u].push_front(Edge(v));
        wk[v].push_front(Edge(u));
        wk[u].begin()->rev = wk[v].begin();
        wk[v].begin()->rev = wk[u].begin();
    }
    bool prop_int(int a, int b, int c, int d) const {
        return seg_int(Segment(vs[a].p, vs[b].p), Segment(vs[c].p, vs[d].p), true);
    }
    void divide(int l, int r) {
        if (r - l <= 2) {
            for (int i = l; i <= r; ++ i) {
                for (int j = i + 1; j <= r; ++ j) add_edge(i, j);
            }
            return;
        }
        int mid{ (l + r) >> 1 };
        divide(l, mid);
        divide(mid + 1, r);
        int lo_l{ l }, lo_r{ r };
        while (true) {
            bool mov{ false };
            Point p_l{ vs[lo_l].p };
            Point p_r{ vs[lo_r].p };
            for (auto it = wk[lo_l].begin(); it != wk[lo_l].end(); ++ it) {
                Point cand{ vs[it->to].p };
                int turn{ sgn(cross(p_r, p_l, cand)) };
                if (turn > 0 || (turn == 0 && norm2(cand - p_r) < norm2(p_l - p_r))) {
                    lo_l = it->to;
                    mov = true;
                    break;
                }
            }
            if (mov) continue;
            for (auto it = wk[lo_r].begin(); it != wk[lo_r].end(); ++ it) {
                Point cand{ vs[it->to].p };
                int turn{ sgn(cross(p_l, p_r, cand)) };
                if (turn < 0 || (turn == 0 && norm2(cand - p_l) < norm2(p_r - p_l))) {
                    lo_r = it->to;
                    mov = true;
                    break;
                }
            }
            if (!mov) break;
        }
        add_edge(lo_l, lo_r);
        while (true) {
            int cho{ -1 }, side{ 0 };
            for (auto it = wk[lo_l].begin(); it != wk[lo_l].end(); ++ it) {
                int v{ it->to };
                if (sgn(cross(vs[lo_l].p, vs[lo_r].p, vs[v].p)) > 0
                        && (cho == -1 || in_cir(vs[lo_l].p, vs[lo_r].p, vs[cho].p, vs[v].p) < 0)) {
                    cho = v;
                    side = -1;
                }
            }
            for (auto it = wk[lo_r].begin(); it != wk[lo_r].end(); ++ it) {
                int v{ it->to };
                if (sgn(cross(vs[lo_r].p, vs[v].p, vs[lo_l].p)) > 0
                        && (cho == -1 || in_cir(vs[lo_l].p, vs[lo_r].p, vs[cho].p, vs[v].p) < 0)) {
                    cho = v;
                    side = 1;
                }
            }
            if (cho == -1) break;
            if (side == -1) {
                for (auto it = wk[lo_l].begin(); it != wk[lo_l].end();) {
                    int v{ it->to };
                    if (prop_int(lo_l, v, lo_r, cho)) {
                        wk[v].erase(it->rev);
                        wk[lo_l].erase(it ++);
                    } else ++ it;
                }
                lo_l = cho;
                add_edge(lo_l, lo_r);
            } else {
                for (auto it = wk[lo_r].begin(); it != wk[lo_r].end();) {
                    int v{ it->to };
                    if (prop_int(lo_r, v, lo_l, cho)) {
                        wk[v].erase(it->rev);
                        wk[lo_r].erase(it++);
                    } else ++ it;
                }
                lo_r = cho;
                add_edge(lo_l, lo_r);
            }
        }
    }
    void fin_g() {
        g.assign(vs.size(), {});
        for (int u = 0; u < vs.size(); ++ u) {
            for (const Edge &edge : wk[u]) g[u].push_back(edge.to);
            ranges::sort(g[u], [&](int a, int b) {
                return arg_less(vs[a].p - vs[u].p, vs[b].p - vs[u].p);
            });
        }
        wk.clear();
    }
public:
    vector<Vertex> vs;      // 去重、排序后的内部点
    vector<vector<int>> g;    // 内部下标邻接表，邻居按逆时针极角排列
    void build(vector<Vertex> pts) {
        ranges::sort(pts, [](const Vertex &a, const Vertex &b) { return a.p < b.p; });
        vs.clear();
        for (Vertex v : pts) {
            if (vs.empty() || !same_pt(vs.back().p, v.p)) vs.push_back(v);
            else vs.back().id = min(vs.back().id, v.id);
        }
        wk.assign(vs.size(), {});
        if (!vs.empty()) divide(0, int(vs.size()) - 1);
        fin_g();
    }
    void build(const vector<Point> &pts) {
        vector<Vertex> idxs(pts.size());
        for (int i = 0; i < pts.size(); ++ i) idxs[i] = Vertex{pts[i], i};
        build(move(idxs));
    }
    vector<pair<int, int>> edges() const {
        vector<pair<int, int>> res;
        for (int u = 0; u < vs.size(); ++ u) {
            for (int v : g[u]) if (u < v) {
                res.push_back({ vs[u].id, vs[v].id });
            }
        }
        return res;
    }
    // 利用平面嵌入遍历所有逆时针有界面；Delaunay 有界面应为三角形
    vector<array<int, 3>> tris() const {
        int n = vs.size();
        vector<unordered_map<int, int>> pos(n);
        vector<vector<char>> used(n);
        for (int u = 0; u < n; ++ u) {
            used[u].assign(g[u].size(), false);
            pos[u].reserve(g[u].size() * 2 + 1);
            for (int i = 0; i < g[u].size(); ++ i) {
                pos[u][g[u][i]] = i;
            }
        }
        vector<array<int, 3>> res;
        for (int st_u = 0; st_u < n; ++ st_u) {
            for (int st_idx = 0; st_idx < g[st_u].size(); ++ st_idx) {
                if (used[st_u][st_idx]) continue;
                vector<int> face;
                int u{ st_u }, idx{ st_idx };
                while (!used[u][idx]) {
                    used[u][idx] = true;
                    face.push_back(u);
                    int v{ g[u][idx] }, rev_idx{ pos[v].at(u) };
                    u = v;
                    idx = (rev_idx - 1 + g[v].size()) % int(g[v].size());
                }
                if (face.size() != 3) continue;
                vector<Point> poly;
                for (int v : face) poly.push_back(vs[v].p);
                if (sgn(poly_area2(poly)) <= 0) continue; // 排除外部面与退化面
                array<int, 3> tri{ vs[face[0]].id, vs[face[1]].id, vs[face[2]].id };
                ranges::sort(tri);
                res.push_back(tri);
            }
        }
        ranges::sort(res);
        res.erase(ranges::unique(res).begin(), res.end());
        return res;
    }
};
```
#### 接口说明
- `Delaunay::Vertex{p,id}`：显式保存原编号；`id` 应唯一且非负
- `build(vector<Point>)`：原输入下标自动作为 `id`
- `build(vector<Vertex>)`：保留调用者给出的 `id`；按坐标排序并用 `same_pt()` 去重
- `vs`：内部点，已排序去重；不要把内部下标当作原输入编号
- `g[u]`：内部下标邻接表，邻居按绕 `vs[u]` 的逆时针极角排列
- `edges()`：返回原 `id` 表示的无向 Delaunay 边，每条边一次
- `tris()`：返回原 `id` 表示的有界三角面，三元组内部排序
#### 注意事项
- 近共线、近共圆或恶意构造数据需要精确/自适应 `orient2d`、`in_cir` 谓词
- 共线或四点共圆时 Delaunay 剖分不唯一，不要依赖某一条对角线必然出现
- 全部共线时没有二维三角形，但仍可能返回连接相邻点的图边
- 近重点会被合并，只保留该组最小 `id`；必须保留重复点时先自行分组并保存映射
- `tris()` 依赖 15.2 的 `poly_area2()` 判断面方向；若只需要边，可删除整个 `tris()`
- Voronoi 是 Delaunay 的对偶：每个三角形外心是 Voronoi 顶点，凸包边对应无界射线；裁剪到边界框时可调用 15.2 的 `cut_poly()` 或 15.4 的半平面交
#### 常见用法
```cpp
Delaunay del;
del.build(pts);
vector<pair<int, int>> edges = del.edges();
vector<array<int, 3>> tris = del.tris();
```
#### 复杂度
- 构造：常规/非退化数据下 $O(n\log n)$
- 平面 Delaunay 图边数：$O(n)$
- `edges()`：$O(n)$
- `tris()`：$O(n)$ 期望/平面图规模，哈希表额外空间 $O(n)$
### 矩形面积并与周长并
**解决问题：** 求大量轴对齐矩形的面积并与并集周长
矩形统一视为半开区域 $[x_1,x_2)\times[y_1,y_2)$；退化矩形忽略。扫描线沿 $x$ 轴移动，线段树维护当前纵截面覆盖长度、连续覆盖段数和左右端覆盖状态
```cpp
struct Rectangle {
    i64 x1, y1, x2, y2;
};
struct RectUnion {
    i128 area{ 0 }, peri{ 0 };
};
RectUnion rec_union(const vector<Rectangle> &recs) {
    struct Event {
        i64 x, y1, y2;
        int delta;
    };
    vector<Event> evs;
    vector<i64> ys;
    for (Rectangle rec : recs) {
        if (rec.x1 >= rec.x2 || rec.y1 >= rec.y2) continue;
        evs.push_back({ rec.x1, rec.y1, rec.y2, 1 });
        evs.push_back({ rec.x2, rec.y1, rec.y2, -1 });
        ys.push_back(rec.y1);
        ys.push_back(rec.y2);
    }
    if (evs.empty()) return {};
    ranges::sort(evs, [](const Event &a, const Event &b) {
        if (a.x != b.x) return a.x < b.x;
        return a.delta > b.delta; // 同 x 先加入，再删除
    });
    ranges::sort(ys);
    ys.erase(ranges::unique(ys).begin(), ys.end());
    int m{ int(ys.size()) - 1 };
    vector<int> cover(4 * max(1, m)), segs(4 * max(1, m));
    vector<i128> len(4 * max(1, m));
    vector<char> l_cov(4 * max(1, m)), r_cov(4 * max(1, m));
    auto pull = [&](auto self, int p, int l, int r) -> void {
        if (cover[p] > 0) {
            len[p] = i128(ys[r]) - ys[l];
            segs[p] = 1;
            l_cov[p] = r_cov[p] = true;
            return;
        }
        if (r - l == 1) {
            len[p] = segs[p] = 0;
            l_cov[p] = r_cov[p] = false;
            return;
        }
        int lc{ p << 1 }, rc{ p << 1 | 1 };
        len[p] = len[lc] + len[rc];
        segs[p] = segs[lc] + segs[rc] - (r_cov[lc] && l_cov[rc]);
        l_cov[p] = l_cov[lc];
        r_cov[p] = r_cov[rc];
    };
    auto upd = [&](auto self, int p, int l, int r, int ql, int qr, int delta) -> void {
        if (ql <= l && r <= qr) {
            cover[p] += delta;
            pull(pull, p, l, r);
            return;
        }
        int mid{ (l + r) >> 1 };
        if (ql < mid) self(self, p << 1, l, mid, ql, qr, delta);
        if (mid < qr) self(self, p << 1 | 1, mid, r, ql, qr, delta);
        pull(pull, p, l, r);
    };
    RectUnion res;
    i64 pre_x{ evs[0].x };
    for (int i = 0; i < evs.size(); ) {
        i64 x{ evs[i].x };
        i128 dx{ i128(x) - pre_x };
        res.area += len[1] * dx;
        res.peri += i128(2) * segs[1] * dx;
        int j{ i };
        while (j < evs.size() && evs[j].x == x) {
            i128 bef{ len[1] };
            int l{ ranges::lower_bound(ys, evs[j].y1) - ys.begin() };
            int r{ ranges::lower_bound(ys, evs[j].y2) - ys.begin() };
            upd(upd, 1, 0, m, l, r, evs[j].delta);
            res.peri += len[1] > bef ? len[1] - bef : bef - len[1];
            ++ j;
        }
        pre_x = x;
        i = j;
    }
    return res;
}
```
#### 接口说明
- `Rectangle{x1,y1,x2,y2}`：表示半开矩形 $[x_1,x_2)\times[y_1,y_2)$
- `rec_union(recs)`：返回 `{area,peri}`，两者均为 `i128`，$O(n\log n)$
- 面积与周长按几何长度计算；重叠边、内部边均不会重复计入
#### 注意事项
- 同一横坐标的事件必须先处理进入事件，再处理离开事件；否则首尾相接的矩形会产生虚假的公共竖边
- 水平边贡献为 `2 * 当前连续覆盖段数 * dx`
- 每次区间修改前后的覆盖长度差，恰好是该事件产生的竖边贡献；同坐标先加后删可正确抵消公共边
- 若只求面积，可删除 `segs / l_cov / r_cov` 与所有周长累加
#### 常见用法
```cpp
RectUnion res = rec_union(recs);
cout << res.area << ' ' << res.peri << '\n';
```
#### 复杂度
时间复杂度：$O(n\log n)$
空间复杂度：$O(n)$
### 矩形覆盖层数
**解决问题：** 求被恰好 $k$ 个矩形覆盖的面积，或被至少 $k$ 个矩形覆盖的面积
```cpp
struct RectCover {
    vector<i128> exact;    // exact[k]：恰好覆盖 k 次的面积
    vector<i128> ge; // ge[k]：至少覆盖 k 次的面积
};
RectCover rec_cov(const vector<Rectangle> &recs) {
    struct Event {
        i64 x, y1, y2;
        int delta;
    };
    vector<Event> evs;
    vector<i64> ys;
    for (Rectangle rec : recs) {
        if (rec.x1 >= rec.x2 || rec.y1 >= rec.y2) continue;
        evs.push_back({rec.x1, rec.y1, rec.y2, 1});
        evs.push_back({rec.x2, rec.y1, rec.y2, -1});
        ys.push_back(rec.y1);
        ys.push_back(rec.y2);
    }
    int n = recs.size();
    RectCover res;
    res.exact.assign(n + 1, 0);
    res.ge.assign(n + 1, 0);
    if (evs.empty()) return res;
    ranges::sort(evs, [](const Event &a, const Event &b) { return a.x < b.x; });
    ranges::sort(ys);
    ys.erase(ranges::unique(ys).begin(), ys.end());
    vector<int> cover(ys.size() - 1);
    i64 pre_x{ evs[0].x };
    for (int i = 0; i < evs.size();) {
        i64 x{ evs[i].x };
        i128 dx{ i128(x) - pre_x };
        for (int j = 0; j + 1 < ys.size(); ++ j) {
            if (cover[j] == 0) continue;
            i128 h{ i128(ys[j + 1]) - ys[j] };
            res.exact[cover[j]] += h * dx;
        }
        int nxt{ i };
        while (nxt < evs.size() && evs[nxt].x == x) {
            int l{ ranges::lower_bound(ys, evs[nxt].y1) - ys.begin() };
            int r{ ranges::lower_bound(ys, evs[nxt].y2) - ys.begin() };
            for (int j = l; j < r; ++ j) cover[j] += evs[nxt].delta;
            ++ nxt;
        }
        pre_x = x;
        i = nxt;
    }
    for (int k = n; k >= 1; -- k) {
        res.ge[k] = res.exact[k] + (k < n ? res.ge[k + 1] : 0);
    }
    return res;
}
```
#### 接口说明
- `exact[k]`：被恰好 $k$ 个有效矩形覆盖的面积
- `ge[k]`：被至少 $k$ 个有效矩形覆盖的面积
- `rec_cov()`：返回全部层数，时间复杂度 $O(n^2)$
- 面积并等于 `ge[1]`；恰好覆盖两次等于 `exact[2]`
#### 注意事项
- 本版逐个维护离散纵段，适合中等 $n$；只求面积并/周长并时优先使用 15.6.2 的 $O(n\log n)$ 线段树版
- `exact` 和 `ge` 的长度均为 `recs.size()+1`；退化矩形不会贡献面积
- 先统计当前横条面积，再处理该横坐标的全部事件
- 原始“大于等于层数”逐层累加会退化到 $O(n^3)$；本版先统计 `exact[k]`，最后做一次后缀和，因此为 $O(n^2)$
#### 常见用法
```cpp
RectCover res = rec_cov(recs);
i128 uni_a = res.ge[1];
i128 ex2 = res.exact[2];
```
#### 复杂度
时间复杂度：$O(n^2)$
空间复杂度：$O(n)$
### 三角形面积并
**解决问题：** 求任意若干三角形的面积并；三角形之间可以相交、包含或退化
把所有顶点横坐标与不同三角形边的**真交点**横坐标作为分段点。相邻分段内部边界次序不变，竖截面区间并长度是一次函数，可用梯形公式精确积分
```cpp
optional<ld> tri_edge_x(Point a, Point b, Point c, Point d) {
    Segment s1(a, b), s2(c, d);
    if (!seg_int(s1, s2, true)) return nullopt;
    auto p = line_int(Line::through(a, b), Line::through(c, d));
    if (!p) return nullopt;
    return p->x;
}
ld tri_slice(const vector<array<Point, 3>> &tris, ld x) {
    vector<pair<ld, ld>> itvs;
    for (const auto &tri : tris) {
        vector<ld> ys;
        for (int edge = 0; edge < 3; ++ edge) {
            Point a{ tri[edge] }, b{ tri[(edge + 1) % 3] };
            if (sgn(a.x - b.x) == 0) continue;
            if (!(min(a.x, b.x) < x && x < max(a.x, b.x))) continue;
            ld ratio{ (x - a.x) / (b.x - a.x) };
            ys.push_back(a.y + (b.y - a.y) * ratio);
        }
        if (ys.size() >= 2) {
            auto [lo, hi] = ranges::minmax_element(ys);
            itvs.push_back({*lo, *hi});
        }
    }
    if (itvs.empty()) return 0;
    ranges::sort(itvs);
    ld ans{ 0 }, l{ itvs[0].first }, r{ itvs[0].second };
    for (int i = 1; i < itvs.size(); ++ i) {
        if (itvs[i].first <= r + EPS) r = max(r, itvs[i].second);
        else {
            ans += r - l;
            l = itvs[i].first;
            r = itvs[i].second;
        }
    }
    return ans + r - l;
}
ld tri_area(vector<array<Point, 3>> tris) {
    tris.erase(remove_if(tris.begin(), tris.end(), [](const auto &tri) {
        return sgn(cross(tri[0], tri[1], tri[2])) == 0;
    }), tris.end());
    vector<ld> xs;
    for (const auto &tri : tris) {
        for (Point p : tri) xs.push_back(p.x);
    }
    for (int i = 0; i < tris.size(); ++ i) {
        for (int j = i + 1; j < tris.size(); ++ j) {
            for (int a = 0; a < 3; ++ a) {
                for (int b = 0; b < 3; ++ b) {
                    auto x = tri_edge_x(
                        tris[i][a], tris[i][(a + 1) % 3],
                        tris[j][b], tris[j][(b + 1) % 3]);
                    if (x) xs.push_back(*x);
                }
            }
        }
    }
    ranges::sort(xs);
    xs.erase(ranges::unique(xs).begin(), [](ld a, ld b) {
        return sgn(a - b) == 0;
    }), xs.end());
    ld area{ 0 };
    for (int i = 0; i + 1 < xs.size(); ++ i) {
        ld w{ xs[i + 1] - xs[i] };
        if (sgn(w) == 0) continue;
        ld x1{ xs[i] + w / 3 }, x2{ xs[i] + 2 * w / 3 };
        area += w * (tri_slice(tris, x1) + tri_slice(tris, x2)) / 2;
    }
    return area;
}
```
#### 接口说明
- `tri_area(tris)`：返回任意三角形集合的面积并；自动忽略退化三角形
- `tri_slice()`：返回竖线 `x=常数` 与所有三角形相交区间的并长度，主要供主算法调用
- 边交点直接复用基础板子的严格线段相交与直线交点，不重复维护另一套几何谓词
#### 注意事项
- 必须加入不同三角形边的交点横坐标；只按顶点横坐标分段会漏掉截面端点次序变化
- 端点相交的横坐标已经作为某个三角形顶点加入，无需额外作为“真交点”处理
- 共线重叠边不产生唯一交点，其端点横坐标已经包含在分段集合中
- 本实现依赖统一 `EPS` 合并近重合横坐标，适合普通竞赛浮点数据，不适合极端病态尺度
- 复杂度较高，只适合三角形数量较小的题目
#### 常见用法
```cpp
ld area = tri_area(tris);
```
#### 复杂度
最多产生 $O(n^2)$ 个分段横坐标，每段合并 $O(n)$ 个区间：
- 时间复杂度：$O(n^3\log n)$
- 空间复杂度：$O(n^2)$
### 平面最近点对
**解决问题：** 在平面点集中找出欧氏距离最近的一对点，并返回距离平方与两个端点
```cpp
using CloseDis2 = decltype(norm2(Point{}));
struct ClosePair {
    CloseDis2 dis2;
    Point a, b;
};
void upd(optional<ClosePair> &ans, Point a, Point b) {
    CloseDis2 cur{ norm2(a - b) };
    if (!ans || cur < ans->dis2) ans = ClosePair{cur, a, b};
}
optional<ClosePair> close_impl(vector<Point> &p, vector<Point> &b, int l, int r) {
    if (r - l <= 3) {
        optional<ClosePair> ans;
        for (int i = l; i < r; ++ i) {
            for (int j = i + 1; j < r; ++ j) {
                upd(ans, p[i], p[j]);
            }
        }
        sort(p.begin() + l, p.begin() + r, [](Point a, Point b) {
            return a.y != b.y ? a.y < b.y : a.x < b.x;
        });
        return ans;
    }
    int mid{ (l + r) >> 1 };
    auto mid_x{ p[mid].x };
    auto ans_l{ close_impl(p, b, l, mid) };
    auto ans_r{ close_impl(p, b, mid, r) };
    optional<ClosePair> ans{ ans_l };
    if (ans_r && (!ans || ans_r->dis2 < ans->dis2)) ans = ans_r;
    merge(p.begin() + l, p.begin() + mid, p.begin() + mid, p.begin() + r, b.begin(), [](Point a, Point b) {
        return a.y != b.y ? a.y < b.y : a.x < b.x;
    });
    copy(b.begin(), b.begin() + r - l, p.begin() + l);
    vector<Point> strip;
    for (int i = l; i < r; ++ i) {
        auto dx{ p[i].x - mid_x };
        CloseDis2 dx2{ CloseDis2(dx) * dx };
        if (ans && dx2 >= ans->dis2) continue;
        for (int j = int(strip.size()) - 1; j >= 0; -- j) {
            auto dy{ p[i].y - strip[j].y };
            CloseDis2 dy2{ CloseDis2(dy) * dy };
            if (ans && dy2 >= ans->dis2) break;
            upd(ans, p[i], strip[j]);
        }
        strip.push_back(p[i]);
    }
    return ans;
}
optional<ClosePair> close_pair(vector<Point> p) {
    if (p.size() < 2) return nullopt;
    ranges::sort(p);
    vector<Point> b(p.size());
    return close_impl(p, b, 0, p.size());
}
```
#### 接口说明
- `close_pair(pts)`：返回最近点对；点数少于 $2$ 时返回 `nullopt`
- `dis2` 是距离平方
- `a`、`b` 是最近点本身；若需要原下标，可把输入改为 `{Point,id}` 并同步比较坐标
#### 注意事项
- 顶层先按 $(x,y)$ 排序；递归过程中各区间会改为按 $y$ 排序，这是算法的一部分
- 输入存在重复点时会正确返回距离平方 $0$
- 整数版保留平方距离即可避免开方误差；输出欧氏距离时最后再转换为 `ld` 并 `sqrtl()`
- 不要在按 $y$ 归并后再次使用当前位置推断原始按 $x$ 下标
#### 常见用法
```cpp
auto res = close_pair(pts);
if (res) {
    auto dis2 = res->dis2;
    Point a = res->a;
    Point b = res->b;
}
```
#### 复杂度
时间复杂度：$O(n\log n)$
额外空间：$O(n)$
### 欧氏最小生成树
**解决问题：** 以平面点为完全图顶点、欧氏距离为边权，求最小生成树总长度与所选边
```cpp
struct EuclidMST {  
    ld w{ 0 };  
    vector<pair<int, int>> edges;  
};  
EuclidMST euc_mst(const vector<Point> &p) {  
    int n = p.size();
    EuclidMST res;  
    if (n == 0) return res;  
    vector<ld> best(n, numeric_limits<ld>::infinity());  
    vector<int> pa(n, -1);
    vector<char> used(n);  
    best[0] = 0;
    for (int it = 0; it < n; ++ it) {
        int u{ -1 };
        for (int i = 0; i < n; ++ i) if (!used[i] && (u == -1 || best[i] < best[u])) {  
            u = i;
        }  
        used[u] = true;  
        res.w += sqrtl(best[u]);  
        if (pa[u] != -1) res.edges.push_back({pa[u], u});  
        for (int v = 0; v < n; ++ v) if (!used[v]) {  
            ld dis2{ ld(norm2(p[u] - p[v])) };  
            if (dis2 < best[v]) {  
                best[v] = dis2;  
                pa[v] = u;  
            }  
        }  
    }  
    return res;  
}
```
#### 接口说明
- `euc_mst(pts)`：返回最小生成树总长度与 $n-1$ 条原输入下标边
- 空点集返回权值 $0$、空边集；单点同样返回空边集
- 内部比较使用距离平方，只在边被加入答案时开方
#### 大规模做法
欧氏 MST 一定是 Delaunay 图的子图。点数较大时：
1. 调用 15.6.1 的 `Delaunay::edges()` 得到 $O(n)$ 条候选边
2. 用已有 `DSU` 和 Kruskal 按欧氏距离排序
#### 注意事项
- 本版是完全图 Prim，适合 $n$ 为几千量级的正确兜底
- 若坐标为整数，`norm2()` 可先精确计算，再转换为 `ld`；坐标过大时确认转换精度是否满足题意
- Delaunay 大规模方案要处理重复点映射与浮点退化
#### 常见用法
```cpp
EuclidMST res = euc_mst(pts);
cout << fixed << setprecision(10) << res.w << '\n';
```
#### 复杂度
时间复杂度：$O(n^2)$
空间复杂度：$O(n)$，不计返回边
### 六边形轴坐标
**解决问题：** 在规则六边形网格上表示格子，计算距离并枚举六个相邻格
使用 axial 坐标 $(q,r)$，隐含第三个 cube 坐
$$
s=-q-r
$$
两格距离为
$$
\frac{|\Delta q|+|\Delta r|+|\Delta s|}{2}
$$
```cpp
struct Hex {  
    i64 q{ 0 }, r{ 0 };  
    friend bool operator==(Hex a, Hex b) { return a.q == b.q && a.r == b.r; }  
    friend bool operator<(Hex a, Hex b) { return tie(a.q, a.r) < tie(b.q, b.r); }  
    Hex &operator+=(Hex oth) { q += oth.q; r += oth.r; return *this; }  
    friend Hex operator+(Hex a, Hex b) { return a += b; }  
    friend Hex operator-(Hex a, Hex b) { return Hex{a.q - b.q, a.r - b.r}; }  
};  
i64 hex_dis(Hex a, Hex b) {  
    i64 dq{ a.q - b.q }, dr{ a.r - b.r }, ds{ -dq - dr };  
    return (llabs(dq) + llabs(dr) + llabs(ds)) / 2;  
}  
array<Hex, 6> hex_nb(Hex cell) {  
    constexpr array<Hex, 6> dir {{  
        {1, 0}, {1, -1}, {0, -1},  
        {-1, 0}, {-1, 1}, {0, 1}  
    }};  
    array<Hex, 6> res;
    for (int i = 0; i < 6; ++ i) res[i] = cell + dir[i];  
    return res;  
}
```
#### 接口说明
- `Hex{q,r}`：一个 axial 坐标格子
- `hex_dis(a,b)`：返回最少相邻移动次数，$O(1)$
- `hex_nb(cell)`：按固定环绕顺序返回六个相邻格，$O(1)$
#### 注意事项
- axial 坐标只有两个显式分量，但距离计算必须补出 $s=-q-r$
- 不同题目的屏幕坐标、行列坐标与 axial 坐标方向可能不同；先确定输入到 $(q,r)$ 的映射
- 若需要哈希，可对 `(q,r)` 使用现有 pair 哈希，不必另写几何哈希
#### 常见用法
```cpp
i64 dis = hex_dis(st, tar);
for (Hex v : hex_nb(cur)) {
    // BFS / DFS
}
```
#### 复杂度
所有接口时间复杂度与额外空间均为 $O(1)$
