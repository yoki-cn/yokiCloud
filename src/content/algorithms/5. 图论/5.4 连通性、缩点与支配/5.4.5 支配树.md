### 支配树
---
使用 Lengauer--Tarjan 算法，只处理从 `root` 可达的部分
```cpp
struct DominatorTree {
    int n, root, timer{};
    vector<vector<int>> g, tree;
    vector<int> dfn, rev, parent, semi, idom_idx, uf, label;
    vector<int> idom, tin, tout;
    bool built{};
    DominatorTree(int n_, int root_) : n(n_), root(root_), g(n), tree(n),
        dfn(n, -1), rev(n), parent(n, -1), idom(n, -1) {}
    void add_edge(int u, int v) {
        assert(!built);
        g[u].push_back(v);
    }
private:
    // ===== DFS 编号 =====
    void dfs(int u) {
        dfn[u] = timer;
        rev[timer ++] = u;
        for (int v : g[u]) if (dfn[v] == -1) {
            dfs(v);
            parent[dfn[v]] = dfn[u];
        }
    }
    // ===== Lengauer--Tarjan 并查集 eval/link =====
    void compress(int v) {
        if (uf[uf[v]] == uf[v]) return;
        compress(uf[v]);
        if (semi[label[uf[v]]] < semi[label[v]]) label[v] = label[uf[v]];
        uf[v] = uf[uf[v]];
    }
    int eval(int v) {
        if (uf[v] == v) return label[v];
        compress(v);
        return label[v];
    }
    void dfs_tree(int u, int &clk) {
        tin[u] = clk ++;
        for (int v : tree[u]) dfs_tree(v, clk);
        tout[u] = clk;
    }
public:
    const vector<int> &build() {
        if (built) return idom;
        assert(0 <= root && root < n);
        dfs(root);
        int r{ timer };
        vector<vector<int>> pred(r), bucket(r);
        for (int u : views::iota(0, n)) if (dfn[u] != -1) {
            for (int v : g[u]) if (dfn[v] != -1) pred[dfn[v]].push_back(dfn[u]);
        }
        semi.resize(r);
        idom_idx.assign(r, -1);
        uf.resize(r);
        label.resize(r);
        ranges::iota(semi, 0);
        ranges::iota(uf, 0);
        ranges::iota(label, 0);
        // ===== 半支配点 =====
        for (int w = r - 1; w >= 1; -- w) {
            for (int v : pred[w]) semi[w] = min(semi[w], semi[eval(v)]);
            bucket[semi[w]].push_back(w);
            uf[w] = parent[w];
            int p{ parent[w] };
            for (int v : bucket[p]) {
                int u{ eval(v) };
                idom_idx[v] = semi[u] < semi[v] ? u : p;
            }
            bucket[p].clear();
        }
        // ===== 立即支配点 =====
        for (int w = 1; w < r; ++ w) {
            if (idom_idx[w] != semi[w]) idom_idx[w] = idom_idx[idom_idx[w]];
        }
        idom[root] = root;
        for (int w = 1; w < r; ++ w) idom[rev[w]] = rev[idom_idx[w]];
        // ===== 支配树与 O(1) 支配查询 =====
        for (int v : views::iota(0, n)) {
            if (v != root && idom[v] != -1) tree[idom[v]].push_back(v);
        }
        tin.assign(n, -1);
        tout.assign(n, -1);
        int clk = 0;
        dfs_tree(root, clk);
        built = true;
        return idom;
    }
    bool reachable(int u) const {
        assert(built);
        return idom[u] != -1;
    }
    bool dominates(int u, int v) const {
        assert(built);
        if (tin[u] == -1 || tin[v] == -1) return false;
        return tin[u] <= tin[v] && tout[v] <= tout[u];
    }
};
```
#### 定义
在以 `root` 为起点的有向图中，若从 `root` 到 `v` 的每一条路径都经过 `u`，则称 `u` 支配 `v`
记为
$$u\operatorname{dom}v$$
除 `v` 本身外最靠近 `v` 的支配点称为立即支配点
$$
idom[v]
$$
全部立即支配关系构成支配树
#### 接口
`build()` 返回 `idom`
```cpp
auto &idom = dom.build();
```
约定
```cpp
idom[root] = root;
idom[v] = -1; // root 无法到达 v
```
`tree[u]` 为支配树中 `u` 的儿子
`reachable(u)` 判断 `root` 是否可以到达 `u`
`dominates(u,v)` 判断 `u` 是否支配 `v`
利用支配树性质
$$u\operatorname{dom}v$$
当且仅当 `u` 是支配树中 `v` 的祖先，因此 Euler 序预处理后查询为 $O(1)$
#### 常见用法
```cpp
DominatorTree dom(n, root);
for (auto [u, v] : edges) dom.add_edge(u, v);
dom.build();
if (dom.dominates(u, v)) {
    // 所有 root -> v 路径都必须经过 u
}
for (int v : views::iota(0, n)) {
    if (dom.idom[v] == -1) continue;
    // v 从 root 可达
}
```
常见应用
* 控制流图
* 判断某点是否是所有路径必经点
* 删除某点后哪些节点必然与根断开
* CFG 优化
* 有向图必经路径问题
#### 注意事项
* 支配关系相对于固定根 `root`
* 不可达节点不属于支配树，`idom[v] == -1`
* `root` 支配所有可达节点，并规定 `idom[root] = root`
* 支持重边和自环
* 构造完成后不能继续加边
* 当前 DFS 与支配树 DFS 都使用递归，极长链可能需要扩大栈或改显式栈
#### 复杂度
Lengauer--Tarjan 配合并查集路径压缩近似线性
$$
O((n+m)\alpha(n))
$$
空间
$$
O(n+m)
$$
`dominates()`
$$
O(1)
$$
