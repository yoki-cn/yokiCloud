### SCC 与缩点 DAG
---
#### Tarjan SCC
```cpp
struct SCC {
    int n, cur{}, cnt{};
    vector<vector<int>> g, comp;
    vector<int> stk, dfn, low, bel, siz;
    bool built{};
    explicit SCC(int n_) : n(n_), g(n), dfn(n, -1), low(n), bel(n, -1) {
        stk.reserve(n);
        comp.reserve(n);
        siz.reserve(n);
    }
    void add_edge(int u, int v) {
        assert(!built);
        g[u].push_back(v);
    }
private:
    // ===== Tarjan 核心 =====
    void dfs(int u) {
        dfn[u] = low[u] = cur ++;
        stk.push_back(u);
        for (int v : g[u]) {
            if (dfn[v] == -1) {
                dfs(v);
                low[u] = min(low[u], low[v]);
            } else if (bel[v] == -1) {
                // 只有仍在 Tarjan 栈中的点才能更新 low。
                low[u] = min(low[u], dfn[v]);
            }
        }
        if (dfn[u] != low[u]) return;
        comp.push_back({});
        while (true) {
            int v{ stk.back() };
            stk.pop_back();
            bel[v] = cnt;
            comp.back().push_back(v);
            if (v == u) break;
        }
        siz.push_back(comp.back().size());
        ++ cnt;
    }
public:
    int build() {
        if (built) return cnt;
        for (int u : views::iota(0, n)) {
            if (dfn[u] == -1) dfs(u);
        }
        built = true;
        return cnt;
    }
    bool same(int u, int v) const {
        assert(built);
        return bel[u] == bel[v];
    }
    // ===== 缩点 DAG：若不需要可以删除 =====
    vector<vector<int>> dag() const {
        assert(built);
        vector<vector<int>> res(cnt);
        vector<int> seen(cnt, -1);
        for (int x : views::iota(0, cnt)) {
            for (int u : comp[x]) {
                for (int v : g[u]) {
                    int y{ bel[v] };
                    if (x == y || seen[y] == x) continue;
                    seen[y] = x;
                    res[x].push_back(y);
                }
            }
        }
        return res;
    }
};
```
#### 非递归 Kosaraju
若 `n` 很大、链状图可能导致递归栈溢出，可以替换为这一版
接口和编号方向与上面的 Tarjan 保持一致
```cpp
struct KosarajuSCC {
    int n, cnt{};
    vector<vector<int>> g, rg, comp;
    vector<int> bel, siz;
    bool built{};
    explicit KosarajuSCC(int n_) : n(n_), g(n), rg(n), bel(n, -1) {
        comp.reserve(n);
        siz.reserve(n);
    }
    void add_edge(int u, int v) {
        assert(!built);
        g[u].push_back(v);
        rg[v].push_back(u);
    }
    int build() {
        if (built) return cnt;
        vector<char> vis(n);
        vector<int> it(n), order, stk;
        order.reserve(n);
        stk.reserve(n);
        // ===== 第一遍：反图退出序 =====
        for (int s : views::iota(0, n)) {
            if (vis[s]) continue;
            vis[s] = true;
            stk.push_back(s);
            while (!stk.empty()) {
                int u{ stk.back() };
                if (it[u] == int(rg[u].size())) {
                    order.push_back(u);
                    stk.pop_back();
                    continue;
                }
                int v{ rg[u][it[u] ++] };
                if (!vis[v]) {
                    vis[v] = true;
                    stk.push_back(v);
                }
            }
        }
        // ===== 第二遍：原图划分 SCC =====
        for (int s : order | views::reverse) {
            if (bel[s] != -1) continue;
            comp.push_back({});
            bel[s] = cnt;
            stk.push_back(s);
            while (!stk.empty()) {
                int u{ stk.back() };
                stk.pop_back();
                comp.back().push_back(u);
                for (int v : g[u]) if (bel[v] == -1) {
                    bel[v] = cnt;
                    stk.push_back(v);
                }
            }
            siz.push_back(comp.back().size());
            ++ cnt;
        }
        built = true;
        return cnt;
    }
    bool same(int u, int v) const {
        assert(built);
        return bel[u] == bel[v];
    }
    vector<vector<int>> dag() const {
        assert(built);
        vector<vector<int>> res(cnt);
        vector<int> seen(cnt, -1);
        for (int x : views::iota(0, cnt)) {
            for (int u : comp[x]) {
                for (int v : g[u]) {
                    int y{ bel[v] };
                    if (x == y || seen[y] == x) continue;
                    seen[y] = x;
                    res[x].push_back(y);
                }
            }
        }
        return res;
    }
};
```
#### 接口说明
两种实现统一提供：
* `add_edge(u,v)`：添加有向边 $u\to v$
* `build()`：构造 SCC，返回强连通分量数量
* `bel[u]`：原点 `u` 所属 SCC
* `comp[id]`：SCC `id` 包含的原节点
* `siz[id]`：SCC 大小
* `same(u,v)`：判断两点是否强连通
* `dag()`：返回去重后的缩点 DAG
必须先：
```cpp
scc.build();
```
才能访问 `bel / comp / siz` 或调用 `same()/dag()`
构造完成后不再允许 `add_edge()`
#### SCC 编号性质
这两个版本特意保持相同编号方向
对于缩点 DAG 中任意一条边：
$$
x\rightarrow y
$$
都有：
$$
x>y
$$
因此：
$$
0,1,\ldots,cnt-1
$$
是**反拓扑序**，从汇 SCC 向源 SCC
而：
$$
cnt-1,cnt-2,\ldots,0
$$
是拓扑序，从源 SCC 向汇 SCC
所以若：
$$
dp[x]
$$
依赖所有后继 `y`：
```cpp
for (int x = 0; x < scc.cnt; ++ x) {
    for (int y : dag[x]) {
        // y < x，dp[y] 已经完成
        dp[x] = max(dp[x], dp[y]);
    }
}
```
若从源 SCC 向后传播：
```cpp
for (int x = scc.cnt - 1; x >= 0; -- x) {
    for (int y : dag[x]) {
        // x -> y
    }
}
```
#### 常见用法
##### 强连通判断
```cpp
SCC scc(n);
for (auto [u, v] : edges) {
    scc.add_edge(u, v);
}
scc.build();
if (scc.same(u, v)) {
    // u、v 强连通
}
```
##### 缩点 DAG
```cpp
auto dag = scc.dag();
for (int x : views::iota(0, scc.cnt)) {
    for (int y : dag[x]) {
        // x -> y，并且 x > y
    }
}
```
##### 源 SCC 与汇 SCC
```cpp
auto dag = scc.dag();
vector<int> indeg(scc.cnt);
int sources = 0, sinks = 0;
for (int x : views::iota(0, scc.cnt)) {
    if (dag[x].empty()) ++ sinks;
    for (int y : dag[x]) ++ indeg[y];
}
for (int x : views::iota(0, scc.cnt)) {
    sources += indeg[x] == 0;
}
```
若 `scc.cnt > 1`，添加最少边使整张图强连通：
$$
\max(sources,sinks)
$$
若只有一个 SCC，则答案为 `0`
##### 所有点都能到达的节点
缩点 DAG 恰好存在一个汇 SCC 时，其中所有原节点都可以被原图任意节点到达
```cpp
auto dag = scc.dag();
int sink{ -1 };
for (int x : views::iota(0, scc.cnt)) {
    if (!dag[x].empty()) continue;
    if (sink != -1) {
        sink = -2;
        break;
    }
    sink = x;
}
int ans = sink >= 0 ? scc.siz[sink] : 0;
```
#### 注意事项
* Tarjan 中访问到已经离开栈的节点时，**不能**再更新 `low`；本实现通过
```cpp
bel[v] == -1
```
判断节点是否仍在栈中
* 自环和重边不会影响 SCC 划分
* `comp[id]` 内部节点顺序只与搜索过程有关
* `dag()` 自动删除 SCC 之间的重边，但不排序
* Tarjan 只存一份原图，常数和内存较小，通常优先使用
* 非递归 Kosaraju 同时保存原图和反图，边存储量约为两倍，但不会因深 DFS 链爆递归栈
* Kosaraju 第一遍必须记录**退出序**，所以非递归实现需要 `it[u]` 保存当前邻接表扫描位置；不能在第一次访问节点时直接加入 `order`
* 不要根据“Tarjan / Kosaraju”名字猜测 SCC 编号的拓扑方向，应以具体实现为准
#### 复杂度
设点数为 $n$，边数为 $m$
Tarjan：
$$
O(n+m)
$$
空间：
$$
O(n+m)
$$
Kosaraju：
$$
O(n+m)
$$
空间：
$$
O(n+m)
$$
但需要同时保存原图和反图
