### 仙人掌图
---
**解决问题：** 判定并分解仙人掌图，提取所有环，构造圆方树，并处理环上距离、两点最短路、路径经过环数等问题
#### 定义
无向图是仙人掌图，当且仅当每条边至多属于一个简单环
等价地，每个点双连通分量只能是
* 一条桥
* 一个简单环
不同简单环可以共享一个割点，但不能共享边，也不能共享两个及以上不同顶点
支持**不连通图**，此时每个连通分量都必须是仙人掌，也可以称为仙人掌森林

默认边权为 `i64`，不传权值时为 `1`
```cpp
template<class W = i64>
struct Cactus {
    struct Edge {
        int u, v;
        W w;
    };
    struct Cycle {
        // edges[i] 连接 vertices[i] -> vertices[(i + 1) % size()]
        vector<int> vertices, edges;
        vector<W> pref;
        W total{};
        int size() const { return vertices.size(); }
        // 从位置 i 沿保存方向走到 j，以及走反方向的长度
        pair<W, W> arc_lengths(int i, int j) const {
            W forward = i <= j ? pref[j] - pref[i] : total - (pref[i] - pref[j]);
            return {forward, total - forward};
        }
        W shortest_arc(int i, int j) const {
            auto [a, b] = arc_lengths(i, j);
            return min(a, b);
        }
    };
    // tree 中：
    // edge != -1：原图桥，对应原边编号
    // pos  != -1：原点与环方点的关联，表示原点在该环中的位置
    struct Link {
        int to, edge{-1}, pos{-1};
    };
    int n, timer{}, lg{}, components{};
    vector<Edge> edges;
    vector<vector<int>> adj;
    // ===== 仙人掌分解 =====
    vector<int> dfn, low, estk, cycle_of_edge, component_id;
    vector<char> bridge;
    vector<Cycle> cycles;
    // belong[u] = {环编号, u 在该环中的位置}
    // 一个割点可以属于多个环
    vector<vector<pair<int, int>>> belong;
    // ===== 仙人掌圆方森林 =====
    // [0, n)             原图节点
    // [n, n + cycles)    环方点
    vector<vector<Link>> tree;
    // ===== 可裁剪：路径查询 =====
    vector<vector<int>> up;
    vector<int> dep, tree_comp, parent_pos, cycle_pref, anchor_pos;
    vector<W> dist, parent_weight;
    bool built{}, valid{}, query_ready{};
    explicit Cactus(int n_) : n(n_), adj(n) {}
    int add_edge(int u, int v, W w = W{1}) {
        assert(!built && u != v);
        int id = edges.size();
        edges.push_back({u, v, w});
        adj[u].push_back(id);
        adj[v].push_back(id);
        return id;
    }
private:
    int other(int id, int u) const { return edges[id].u ^ edges[id].v ^ u; }
    vector<int> tmp_deg;
    vector<vector<int>> tmp_adj;
    // ===== 将一个点双判定为桥或简单环 =====
    bool make_block(vector<int> ids) {
        if (ids.size() == 1) {
            bridge[ids[0]] = true;
            return true;
        }
        vector<int> vertices;
        for (int id : ids) {
            auto [u, v, w] = edges[id];
            if (tmp_deg[u] ++ == 0) vertices.push_back(u);
            if (tmp_deg[v] ++ == 0) vertices.push_back(v);
            tmp_adj[u].push_back(id);
            tmp_adj[v].push_back(id);
        }
        // 简单环满足 |E| = |V|，且块内每个点度数为 2
        bool ok{ ids.size() == vertices.size() };
        for (int u : vertices) ok &= tmp_deg[u] == 2;
        if (!ok) {
            for (int u : vertices) {
                tmp_deg[u] = 0;
                tmp_adj[u].clear();
            }
            return false;
        }
        // ===== 按环上的实际顺序恢复点和边 =====
        Cycle cyc;
        int s{ ranges::min(vertices) }, u{ s }, prev{ -1 };
        for (int step = 0; step < ids.size(); ++ step) {
            cyc.vertices.push_back(u);
            auto &inc = tmp_adj[u];
            int id{ inc[0] != prev ? inc[0] : inc[1] };
            cyc.edges.push_back(id);
            int v{ other(id, u) };
            prev = id;
            u = v;
        }
        ok &= u == s;
        if (!ok) {
            for (int v : vertices) {
                tmp_deg[v] = 0;
                tmp_adj[v].clear();
            }
            return false;
        }
        int cid = cycles.size();
        cyc.pref.assign(cyc.size() + 1, W{});
        for (int i : views::iota(0, cyc.size())) {
            cyc.pref[i + 1] = cyc.pref[i] + edges[cyc.edges[i]].w;
            cycle_of_edge[cyc.edges[i]] = cid;
            belong[cyc.vertices[i]].push_back({cid, i});
        }
        cyc.total = cyc.pref.back();
        cycles.push_back(move(cyc));
        for (int v : vertices) {
            tmp_deg[v] = 0;
            tmp_adj[v].clear();
        }
        return true;
    }
    // ===== 点双 Lowlink，边栈保存当前块 =====
    void dfs(int u, int parent_edge, int cid) {
        component_id[u] = cid;
        dfn[u] = low[u] = ++ timer;
        for (int id : adj[u]) {
            if (id == parent_edge) continue;
            int v{ other(id, u) };
            if (!dfn[v]) {
                estk.push_back(id);
                dfs(v, id, cid);
                low[u] = min(low[u], low[v]);
                if (low[v] >= dfn[u]) {
                    vector<int> ids;
                    while (true) {
                        int e{ estk.back() };
                        estk.pop_back();
                        ids.push_back(e);
                        if (e == id) break;
                    }
                    valid &= make_block(move(ids));
                }
            } else if (dfn[v] < dfn[u]) {
                // 无向边只把返祖边压栈一次
                estk.push_back(id);
                low[u] = min(low[u], dfn[v]);
            }
        }
    }
    // ===== 环缩成方点，桥直接保留 =====
    void build_tree() {
        tree.assign(n + cycles.size(), {});
        for (int id : views::iota(0, int(edges.size()))) {
            if (!bridge[id]) continue;
            auto [u, v, w] = edges[id];
            tree[u].push_back({v, id, -1});
            tree[v].push_back({u, id, -1});
        }
        for (int cid : views::iota(0, int(cycles.size()))) {
            int c{ n + cid };
            for (int i : views::iota(0, cycles[cid].size())) {
                int u{ cycles[cid].vertices[i] };
                tree[c].push_back({u, -1, i});
                tree[u].push_back({c, -1, i});
            }
        }
    }
    // ===== 以下全部是可裁剪的路径查询部分 =====
    void build_lca() {
        int N = tree.size();
        lg = bit_width((unsigned)max(1, N));
        up.assign(lg, vector<int>(N));
        dep.assign(N, -1);
        tree_comp.assign(N, -1);
        parent_pos.assign(N, -1);
        cycle_pref.assign(N, 0);
        anchor_pos.assign(cycles.size(), -1);
        dist.assign(N, W{});
        parent_weight.assign(N, W{});
        for (int s : views::iota(0, n)) {
            if (dep[s] != -1) continue;
            dep[s] = 0;
            tree_comp[s] = component_id[s];
            up[0][s] = s;
            vector<int> stk{s};
            while (!stk.empty()) {
                int u{ stk.back() };
                stk.pop_back();
                for (auto [v, edge, pos] : tree[u]) {
                    if (dep[v] != -1) continue;
                    dep[v] = dep[u] + 1;
                    tree_comp[v] = tree_comp[u];
                    up[0][v] = u;
                    cycle_pref[v] = cycle_pref[u] + (v >= n);
                    W w{};
                    if (u < n && v < n) {
                        // 桥
                        w = edges[edge].w;
                    } else if (u < n && v >= n) {
                        // 从根方向第一次进入这个环，u 是该环锚点
                        anchor_pos[v - n] = pos;
                    } else if (u >= n && v < n) {
                        // 环方点 -> 环上其他点
                        int cid{ u - n };
                        parent_pos[v] = pos;
                        w = cycles[cid].shortest_arc(anchor_pos[cid], pos);
                    }
                    parent_weight[v] = w;
                    dist[v] = dist[u] + w;
                    stk.push_back(v);
                }
            }
        }
        for (int k = 1; k < lg; ++ k) {
            for (int u : views::iota(0, N)) {
                up[k][u] = up[k - 1][up[k - 1][u]];
            }
        }
    }
    int climb(int u, int k) const {
        for (int b = 0; k; ++ b, k >>= 1) {
            if (k & 1) u = up[b][u];
        }
        return u;
    }
public:
    // ===== 主构造：只分解仙人掌并建立圆方森林，线性复杂度 =====
    bool build() {
        if (built) return valid;
        built = true;
        valid = true;
        int m = edges.size();
        dfn.assign(n, 0);
        low.assign(n, 0);
        bridge.assign(m, false);
        cycle_of_edge.assign(m, -1);
        component_id.assign(n, -1);
        belong.assign(n, {});
        tmp_deg.assign(n, 0);
        tmp_adj.assign(n, {});
        for (int u : views::iota(0, n)) {
            if (dfn[u]) continue;
            int cid{ components ++ };
            dfs(u, -1, cid);
            estk.clear();
        }
        if (!valid) return false;
        build_tree();
        return true;
    }
    // ===== 基础查询 =====
    bool is_bridge(int id) const {
        assert(built && valid);
        return bridge[id];
    }
    bool edge_on_cycle(int id) const {
        assert(built && valid);
        return cycle_of_edge[id] != -1;
    }
    bool connected(int u, int v) const {
        assert(built && valid);
        return component_id[u] == component_id[v];
    }
    int component_count() const {
        assert(built && valid);
        return components;
    }
    int cycle_count() const {
        assert(built && valid);
        return cycles.size();
    }
    // ===== 可裁剪：LCA / 最短路 / 路径经过环数 =====
    void prepare_queries() {
        assert(built && valid);
        if (query_ready) return;
        build_lca();
        query_ready = true;
    }
    int lca(int u, int v) const {
        assert(query_ready && tree_comp[u] == tree_comp[v]);
        if (dep[u] < dep[v]) swap(u, v);
        u = climb(u, dep[u] - dep[v]);
        if (u == v) return u;
        for (int k = lg - 1; k >= 0; -- k) {
            if (up[k][u] == up[k][v]) continue;
            u = up[k][u];
            v = up[k][v];
        }
        return up[0][u];
    }
    // 非负边权下的两点最短距离
    optional<W> distance(int u, int v) const {
        assert(query_ready && 0 <= u && u < n && 0 <= v && v < n);
        if (!connected(u, v)) return nullopt;
        int p{ lca(u, v) };
        W ans{ dist[u] + dist[v] - W{2} * dist[p] };
        // LCA 是原图节点时，普通树上距离公式已经正确
        if (p < n) return ans;
        // LCA 是环方点时，需要把“分别走向环锚点”
        // 替换成两个实际入环点之间的最短环弧
        int a{ climb(u, dep[u] - dep[p] - 1) };
        int b{ climb(v, dep[v] - dep[p] - 1) };
        int cid{ p - n };
        ans -= parent_weight[a] + parent_weight[b];
        ans += cycles[cid].shortest_arc(parent_pos[a], parent_pos[b]);
        return ans;
    }
    // u -> v 的结构路径经过多少个环
    int cycles_on_path(int u, int v) const {
        assert(query_ready && connected(u, v));
        int p{ lca(u, v) };
        return cycle_pref[u] + cycle_pref[v] - 2 * cycle_pref[p] + (p >= n);
    }
};
```
#### 判定原理
Tarjan 点双分解时使用边栈
当 DFS 树边 `u -> v` 满足
$$
low[v]\ge dfn[u]
$$
时，从边栈中弹出直到当前树边，即得到一个完整点双连通分量
仙人掌的关键性质是每个点双只能有两种形式
##### 桥
只有一条边
```text
u ----- v
```
##### 简单环
若一个点双包含多条边，则必须满足
$$
|E|=|V|
$$
并且块内每个点度数都等于 $2$
因此代码直接检查
```cpp
ids.size() == vertices.size()
```
以及
```cpp
tmp_deg[u] == 2
```
只要某个点双不是桥也不是环，整张图就不是仙人掌
例如 theta 图
```text
      a
     / \
s --     -- t
     \ /
      b
```
存在两点之间三条内部不交路径，其中某些边会属于多个简单环，因此会被拒绝

#### 环信息
所有环保存在
```cpp
cactus.cycles
```
##### 遍历全部环
```cpp
for (int cid : views::iota(0, cactus.cycle_count())) {
    const auto &cyc = cactus.cycles[cid];
    for (int u : cyc.vertices) {
        // 环上的点
    }
    for (int id : cyc.edges) {
        // 环上的原边编号
    }
}
```
`vertices` 与 `edges` 一一对应
```text
vertices[0] --edges[0]--> vertices[1]
vertices[1] --edges[1]--> vertices[2]
...
vertices[k-1] --edges[k-1]--> vertices[0]
```
这里的方向只是为了保存顺序
原图仍然是无向图
#### 一个点属于哪些环
```cpp
for (auto [cid, pos] : cactus.belong[u]) {
    // u 位于第 cid 个环
    // 并且 u == cycles[cid].vertices[pos]
}
```
仙人掌中一个割点可以属于多个环，因此不能只保存一个
```cpp
cycle_id[u]
```
这也是为什么我们使用
```cpp
belong[u]
```
保存多个环归属
#### 判断边类型
##### 是否为桥
```cpp
if (cactus.is_bridge(id)) {}
```
##### 是否属于某个环
```cpp
if (cactus.edge_on_cycle(id)) {
    int cid = cactus.cycle_of_edge[id];
}
```
在合法仙人掌中，每条边恰好属于以下一种类型
```text
桥
或
某一个简单环
```
#### 环上距离
对于
```cpp
const auto &cyc = cactus.cycles[cid];
```
若已知两个点在环中的位置 `i,j`
```cpp
auto [clockwise, counterclockwise] = cyc.arc_lengths(i, j);
```
得到沿两个方向的弧长
两条弧满足
$$
d_1+d_2=cycle\_length
$$
其中
```cpp
cyc.total
```
就是环总长度
非负边权下的最短环上距离
```cpp
W d = cyc.shortest_arc(i, j);
```
即
$$
\min(d_1,d_2)
$$
#### 仙人掌圆方树
本板将每个环缩成一个新的方点
节点编号为
```text
[0, n)                    原图节点
[n, n + cycle_count())    环方点
```
桥直接保留
例如
```text
    1
   / \
  0   2 ----- 3
   \ /
```
其中 `0-1-2-0` 为环
会转换成
```text
      1
      |
0 --- C --- 2 --- 3
```
`C` 是这个环对应的方点
##### 为什么桥不再建立方点
标准 block-cut tree 可以给每个点双都建立方点
但桥点双只有两个端点
```text
u - square - v
```
完全可以直接压缩为
```text
u - v
```
因此这里仅为环建立方点，结构更小
这仍然是一棵森林
对于连通仙人掌则是一棵树
#### 遍历圆方树

```cpp
for (int u : views::iota(0, int(cactus.tree.size()))) {
    for (auto e : cactus.tree[u]) {
        int v = e.to;
    }
}
```
判断节点类型
```cpp
if (u < n) {
    // 原图节点
} else {
    int cid = u - n;
    // 第 cid 个环的方点
}
```
很多仙人掌 DP 到这里就已经变成普通树 DP
#### 连通分量
```cpp
int cc = cactus.component_count();
```
判断两个原图节点是否连通
```cpp
if (cactus.connected(u, v)) {}
```
若题目保证普通意义上的仙人掌图必须连通，只需要检查
```cpp
cactus.build() && cactus.component_count() == 1
```
#### 路径查询预处理
若只需要
* 判定仙人掌
* 找环
* 建圆方树
* 做自己的树 DP
到这里就可以停止
只有需要下面的 LCA、最短路、路径环数时才调用
```cpp
cactus.prepare_queries();
```
这部分可以整体裁掉
#### 为什么仙人掌最短路不能直接给圆方树边赋固定权值
对于一个环
```text
      a
    /   \
   x     y
    \   /
      b
```
如果简单地将环缩成方点 `C`
```text
a - C - b
```
那么 `a -> C -> b` 的固定两条边无法同时表达环上任意两点之间
$$
\min(\text{顺时针弧长},\text{逆时针弧长})
$$
因此普通加权树距离公式并不能直接使用
我们采用一个根方向上的**环锚点**解决这个问题
#### 根方向锚点
把圆方树任意定根
每个环方点都有唯一一个朝向根的原图节点
记为该环的 `anchor`
对于环中的其他节点 `v`
将方点到 `v` 的临时树权定义为
$$
dist_{cycle}(anchor,v)
$$
这样从根到任意节点的最短距离都是正确的
只有一种情况需要特殊修正
> 两点的 LCA 本身是环方点

此时两条路径从环中的不同位置进入，不能都绕回 `anchor`
#### 两点最短路
要求边权非负
预处理
```cpp
cactus.prepare_queries();
```
查询
```cpp
auto d = cactus.distance(u, v);
if (!d) {
    // 不连通
} else {
    // *d 为最短距离
}
```
查询复杂度
$$
O(\log n)
$$
##### LCA 是原图节点
此时普通树距离公式成立
$$
dist(u,v)
=
dist[u]+dist[v]-2dist[lca]
$$
##### LCA 是环方点
设两条路径从该环上的 `a,b` 两点离开
普通树公式实际上计算了
$$
dist(anchor,a)+dist(anchor,b)
$$
但真正需要的是
$$
dist_{cycle}(a,b)
$$
因此修正为
$$
\begin{aligned}
ans
={}&dist[u]+dist[v]-2dist[lca]\\
&-dist(anchor,a)-dist(anchor,b)\\
&+dist_{cycle}(a,b)
\end{aligned}
$$
对应代码
```cpp
ans -= parent_weight[a] + parent_weight[b];
ans += cycles[cid].shortest_arc(parent_pos[a], parent_pos[b]);
```
#### 路径经过多少个环
```cpp
int k = cactus.cycles_on_path(u, v);
```
表示从 `u` 到 `v` 的圆方树路径上有多少个环方点
查询复杂度
$$
O(\log n)
$$
这个信息非常实用
#### 两点间简单路径数量
在仙人掌中，每经过一个环，都有两种方式从环的一侧走到另一侧
因此若圆方树路径经过 $k$ 个环，则两点间不同简单路径数量为
$$
2^k
$$
例如模 `mod` 统计
```cpp
int k = cactus.cycles_on_path(u, v);
i64 ans = qpow(2, k, mod);
```
前提是 `u,v` 连通
#### 环贡献 DP
圆方树上做 DP 时
##### 原图节点
通常像普通树节点一样合并儿子
##### 环方点
不能把所有邻点独立处理
因为它们在原图中实际位于同一个环上
此时通常需要
* 按 `cycles[cid].vertices` 的环顺序 DP
* 做前缀 / 后缀
* 环形 DP
* 顺时针与逆时针分别转移
因此看到
```cpp
u >= n
```
时应想到
```cpp
int cid = u - n;
const auto &cyc = cactus.cycles[cid];
```
而不是把它当普通树节点无脑合并
#### 与 VBCC 的关系
仙人掌判定实际上可以概括成一句话
> 对所有 VBCC，检查它是单边还是简单环

但已有的通用 VBCC 板子主要保存的是**点集合**
仙人掌常常还需要
* 块内边编号
* 环的实际顺序
* 环上前缀距离
* 原边属于哪个环
因此我们内部重新做了一遍边栈 Lowlink
并不是因为算法不同，而是为了保存更适合仙人掌题目的信息
#### 重边
本板使用**父边编号**而不是父节点，因此可以正确处理平行边
两条平行边
```text
u === v
```
会被视为长度为 `2` 的环
三条及以上平行边则不再是仙人掌，因为每条边会参与多个简单环
如果题目明确保证简单图，则无需关心这一情况
#### 自环
当前 `add_edge()` 直接要求
```cpp
u != v
```
也就是不支持自环
虽然某些定义会把自环视为长度为 `1` 的环，但竞赛中仙人掌通常讨论无自环图
若题目真的允许自环，应按题目定义单独处理
#### 常见用法
##### 判定并枚举环
```cpp
Cactus<> cactus(n);
for (auto [u, v] : edges) cactus.add_edge(u, v);
if (!cactus.build()) {
    cout << "Not a cactus\n";
} else {
    for (int cid : views::iota(0, cactus.cycle_count())) {
        for (int u : cactus.cycles[cid].vertices) {
            // 第 cid 个环上的节点
        }
    }
}
```
##### 判断边是桥还是环边
```cpp
for (int id : views::iota(0, m)) {
    if (cactus.is_bridge(id)) {
        // 桥
    } else {
        int cid = cactus.cycle_of_edge[id];
        // 第 cid 个环
    }
}
```
##### 最短路
```cpp
if (!cactus.build()) return;
cactus.prepare_queries();
auto ans = cactus.distance(u, v);
if (!ans) {
    // 不连通
} else {
    cout << *ans << '\n';
}
```
##### 路径上的环数
```cpp
cactus.prepare_queries();
int k = cactus.cycles_on_path(u, v);
```
若需要简单路径条数
```cpp
i64 ways = qpow(2, k, mod);
```
#### 快速回忆
```text
仙人掌
    每条边至多属于一个简单环

Tarjan 边栈
    每个点双：
        1 条边 -> 桥
        多条边 ->
            |E| = |V|
            且每点块内度 = 2
            -> 简单环

环
    vertices[i]
    edges[i]
    pref[i]
    total

圆方树
    原点 [0,n)
    环方点 [n,n+C)
    桥直接保留
    环替换成方点

只分解
    build()
    O(n+m)

需要路径查询
    prepare_queries()

两点最短路
    LCA 是原点 -> 普通树距离
    LCA 是环方点 -> 替换一次环内距离

路径经过 k 个环
    不同简单路径数 = 2^k
```
#### 注意事项
* 本板处理无向图
* 支持不连通仙人掌森林
* 支持重边
* 不支持自环
* 一个割点可以同时属于多个环
* `cycle_of_edge[id] == -1` 表示该边不是环边
* 合法仙人掌中的非环边一定是桥
* 环保存方向是任意的，只用于固定顺序
* `Cycle::shortest_arc()` 和 `distance()` 的标准最短路语义要求边权非负
* 若只需要结构分解，不要调用 `prepare_queries()`
* `tree` 是仅给环建立方点的压缩圆方树，不是“每个 VBCC 都建立方点”的完整 block-cut tree
* 圆方树上的环方点做 DP 时通常需要结合原环顺序，不能总当成普通树节点
* Lowlink 使用递归 DFS，极长链需要注意系统栈空间
* 构造完成后不能继续 `add_edge()`
#### 复杂度
设原图有 $n$ 个点、$m$ 条边
##### 仙人掌判定与环分解
每条边只进入和离开边栈一次
$$
O(n+m)
$$
所有环的点和边总规模也是
$$
O(n+m)
$$
因此 `build()` 总时间复杂度
$$
O(n+m)
$$
空间复杂度
$$
O(n+m)
$$
##### 圆方树规模
若环数量为 $c$
圆方树节点数为
$$
N=n+c
$$
并且
$$
N=O(n+m)
$$
##### 路径查询预处理
倍增 LCA
$$
O(N\log N)
$$
时间和空间均为
$$
O(N\log N)
$$
##### 单次查询
`connected()`
$$
O(1)
$$
`is_bridge()`
$$
O(1)
$$
环上两个已知位置的距离
$$
O(1)
$$
`lca()`
$$
O(\log N)
$$
`distance()`
$$
O(\log N)
$$
`cycles_on_path()`
$$
O(\log N)
$$
