### 函数图
---
**解决问题：** 每个点恰有一条出边时，分解环与反向树，并处理入环点、到环距离、第 $k$ 次跳转、有向距离和两点同步首次相遇
```cpp
struct FunctionalGraph {
    int n, lg;
    vector<int> to;
    vector<vector<int>> rev, cycles, up;
    vector<char> in_cycle;
    vector<int> cycle_id, cycle_pos, entry, depth, comp_size;
    explicit FunctionalGraph(vector<int> to_) : n(to_.size()), to(move(to_)), rev(n),
        in_cycle(n), cycle_id(n, -1), cycle_pos(n, -1), entry(n, -1), depth(n, -1) {
        build();
    }
private:
    int lift(int u, int k) const {
        for (int b = 0; k; ++ b, k >>= 1) if (k & 1) u = up[b][u];
        return u;
    }
    // ===== 入度剥离 + 找环 + 反图 BFS + 倍增 =====
    void build() {
        vector<int> indeg(n);
        for (int u : views::iota(0, n)) {
            assert(0 <= to[u] && to[u] < n);
            rev[to[u]].push_back(u);
            ++ indeg[to[u]];
        }
        queue<int> q;
        for (int u : views::iota(0, n)) if (!indeg[u]) q.push(u);
        ranges::fill(in_cycle, true);
        while (!q.empty()) {
            int u{ q.front() };
            q.pop();
            in_cycle[u] = false;
            if (-- indeg[to[u]] == 0) q.push(to[u]);
        }
        // 剥离后剩余点全部位于环上
        for (int s : views::iota(0, n)) if (in_cycle[s] && cycle_id[s] == -1) {
            int cid{ cycles.size() }, u{ s };
            cycles.push_back({});
            do {
                cycle_id[u] = cid;
                cycle_pos[u] = cycles.back().size();
                entry[u] = u;
                depth[u] = 0;
                cycles.back().push_back(u);
                u = to[u];
            } while (u != s);
            comp_size.push_back(cycles.back().size());
        }
        // 从所有环点沿反图向外扩展
        for (int u : views::iota(0, n)) if (in_cycle[u]) q.push(u);
        while (!q.empty()) {
            int u{ q.front() };
            q.pop();
            for (int v : rev[u]) {
                if (depth[v] != -1) continue;
                cycle_id[v] = cycle_id[u];
                entry[v] = entry[u];
                depth[v] = depth[u] + 1;
                ++ comp_size[cycle_id[v]];
                q.push(v);
            }
        }
        lg = bit_width((unsigned)max(1, n));
        up.assign(lg, vector<int>(n));
        if (n) up[0] = to;
        for (int k = 1; k < lg; ++ k) {
            for (int u : views::iota(0, n)) up[k][u] = up[k - 1][up[k - 1][u]];
        }
    }
public:
    bool on_cycle(int u) const { return in_cycle[u]; }
    bool same_component(int u, int v) const { return cycle_id[u] == cycle_id[v]; }
    int component_count() const { return cycles.size(); }
    int cycle_size(int u) const { return cycles[cycle_id[u]].size(); }
    int component_size(int u) const { return comp_size[cycle_id[u]]; }
    int orbit_size(int u) const { return depth[u] + cycle_size(u); }
    // ===== 任意大的 k 次跳转 =====
    int jump(int u, u64 k) const {
        if (k <= u64(depth[u])) return lift(u, int(k));
        k -= depth[u];
        u = entry[u];
        const auto &cyc = cycles[cycle_id[u]];
        return cyc[(cycle_pos[u] + k % cyc.size()) % cyc.size()];
    }
    // ===== u 沿有向边到 v 的最少步数，不可达返回 -1 =====
    i64 distance(int u, int v) const {
        if (!same_component(u, v)) return -1;
        if (!on_cycle(v)) {
            if (depth[u] < depth[v]) return -1;
            int d{ depth[u] - depth[v] };
            return lift(u, d) == v ? d : -1;
        }
        int len{ cycle_size(u) };
        int around{ (cycle_pos[v] - cycle_pos[entry[u]] + len) % len };
        return i64(depth[u]) + around;
    }
    // ===== 两点每秒同时沿 to 移动，返回首次相遇的 {时间, 节点} =====
    optional<pair<i64, int>> first_meeting(int u, int v) const {
        if (!same_component(u, v)) return nullopt;
        if (u == v) return pair<i64, int>{0, u};
        // 深度相同时，只可能在同一棵反向树中先汇合
        if (depth[u] == depth[v]) {
            if (entry[u] != entry[v]) return nullopt;
            int a{ u }, b{ v }, walked{ 0 }, d{ depth[u] };
            for (int k = lg - 1; k >= 0; -- k) {
                int step = 1 << k;
                if (walked + step <= d && up[k][a] != up[k][b]) {
                    a = up[k][a];
                    b = up[k][b];
                    walked += step;
                }
            }
            return pair<i64, int>{walked + 1, up[0][a]};
        }
        // 深度不同不可能在树上同步相遇，只可能进入环后相遇
        int len{ cycle_size(u) };
        auto phase = [&](int x) {
            int p = (cycle_pos[entry[x]] - depth[x]) % len;
            return p < 0 ? p + len : p;
        };
        if (phase(u) != phase(v)) return nullopt;
        i64 t = max(depth[u], depth[v]);
        return pair<i64, int>{t, jump(u, t)};
    }
};
```
#### 结构
函数图满足每个节点恰有一条出边
```cpp
to[u]
```
表示
$$
u\to to[u]
$$
每个弱连通分量都恰好由
```text
一个有向环
+
若干棵指向环的树
```
组成
因此可以通过入度剥离一次性得到全部环
```text
反复删除入度为 0 的点
→ 最后没有被删除的点恰好是所有环点
```
之后从所有环点沿反图 BFS，即可确定每个树点所属的环和入环位置
#### 成员变量
##### `cycle_id[u]`
节点 `u` 所属弱连通分量的环编号
两个点位于同一弱连通分量当且仅当
```cpp
cycle_id[u] == cycle_id[v]
```
##### `cycles[id]`
第 `id` 个环按有向边顺序保存的全部节点
```cpp
for (int u : fg.cycles[id]) {}
```
若
```cpp
fg.cycles[id] = {a, b, c};
```
则有
```text
a -> b -> c -> a
```
##### `cycle_pos[u]`
仅对环点有意义
表示环点 `u` 在 `cycles[cycle_id[u]]` 中的位置
非环点保持 `-1`
##### `entry[u]`
从 `u` 不断沿 `to` 前进时首次到达的环点
若 `u` 本身位于环上
```cpp
entry[u] == u
```
##### `depth[u]`
从 `u` 到环的距离
$$
depth[u]=\min\{k\ge0:f^k(u)\text{ 位于环上}\}
$$
因此
```cpp
fg.on_cycle(u)
```
等价于
```cpp
fg.depth[u] == 0
```
##### `comp_size[id]`
环 `id` 所在整个弱连通分量的节点数
##### `rev[u]`
函数图的反图邻接表
```cpp
v ∈ rev[u]
```
表示
```text
to[v] = u
```
去掉环与环之间的边以后，`rev` 中每个环点都是一棵树的根
#### 常用接口
##### 判断是否在环上
```cpp
if (fg.on_cycle(u)) {}
```
##### 判断是否属于同一弱连通分量
```cpp
if (fg.same_component(u, v)) {}
```
##### 环长
```cpp
int len = fg.cycle_size(u);
```
##### 连通分量大小
```cpp
int siz = fg.component_size(u);
```
##### 从一个点出发最终访问多少个不同节点
```cpp
int cnt = fg.orbit_size(u);
```
因为先经过 `depth[u]` 个树点，再完整经过所属环
$$
orbit\_size(u)=depth[u]+cycle\_size(u)
$$
##### 第 $k$ 次跳转
```cpp
int v = fg.jump(u, k);
```
返回
$$
f^k(u)
$$
`k` 使用 `u64`，可以远大于 `n`
实现先跳到环
```text
u -> ... -> entry[u]
```
剩余步数再对环长取模，因此不需要建立 $64$ 层倍增
例如
```cpp
u64 k = 1000000000000000000ULL;
int v = fg.jump(u, k);
```
仍然可以直接处理
##### 为什么倍增只需要 $O(\log n)$ 层
倍增只负责从树点跳到环
最多需要
$$
depth[u]\le n
$$
一旦进入环，剩余步数直接对环长取模
因此只需要
$$
O(\log n)
$$
层
#### 有向距离
```cpp
i64 d = fg.distance(u, v);
```
返回从 `u` 不断沿 `to` 前进首次到达 `v` 所需的步数
不可达返回 `-1`
##### `v` 不在环上
此时 `v` 必须位于 `u` 到环的唯一路径上
首先必须满足
$$
depth[u]\ge depth[v]
$$
令
$$
d=depth[u]-depth[v]
$$
再检查
```cpp
fg.jump(u, d) == v
```
##### `v` 位于环上
先从 `u` 到达 `entry[u]`
花费
$$
depth[u]
$$
再沿环走到 `v`
因此总距离为
$$
depth[u]+\text{环上顺时针距离}
$$
#### 两点同步首次相遇
```cpp
auto meet = fg.first_meeting(u, v);
```
表示 `u` 与 `v` 每次同时执行
```cpp
u = to[u];
v = to[v];
```
返回首次相遇的
```cpp
{时间, 相遇节点}
```
永远不会相遇则返回 `nullopt`
##### 初始深度相同
若
$$
depth[u]=depth[v]
$$
只有 `entry[u] == entry[v]` 时才可能相遇
此时两条树链最终汇合，可以倍增寻找第一次汇合位置
##### 初始深度不同
若
$$
depth[u]\ne depth[v]
$$
两点不可能在树上同步相遇
因为每经过一步，两者剩余到环距离同时减一，深度差始终不变
只能等两点都进入环以后判断是否处于同一个同步相位
对环长 $L$ 定义
$$
phase(x)=cycle\_pos[entry[x]]-depth[x]\pmod L
$$
若
$$
phase(u)\ne phase(v)
$$
则永远不会相遇
否则首次相遇时刻为
$$
\max(depth[u],depth[v])
$$
#### 常见用法
##### 基础分解
```cpp
vector<int> to(n);
for (int &v : to) {
    cin >> v;
    -- v;
}
FunctionalGraph fg(to);
for (int u : views::iota(0, n)) {
    cout << "cycle = " << fg.cycle_id[u] << '\n';
    cout << "entry = " << fg.entry[u] << '\n';
    cout << "depth = " << fg.depth[u] << '\n';
}
```
##### 遍历所有环
```cpp
for (int id : views::iota(0, fg.component_count())) {
    for (int u : fg.cycles[id]) {
        // u 是该环上的节点
    }
}
```
##### 大步跳转
```cpp
int v = fg.jump(u, k);
```
##### 判断一个点能否沿有向边到达另一个点
```cpp
i64 d = fg.distance(u, v);
if (d == -1) {
    // 不可达
} else {
    // 恰好 d 步到达
}
```
##### 两个人同步移动首次相遇
```cpp
auto res = fg.first_meeting(u, v);
if (!res) {
    // 永不相遇
} else {
    auto [time, node] = *res;
}
```
#### 快速回忆
```text
每点出度 = 1

入度剥离
    未被删除的点 = 环点

环点反图 BFS
    cycle_id = 属于哪个环
    entry    = 从哪个环点入环
    depth    = 到环距离

jump(u,k)
    k <= depth[u]：倍增
    否则：到 entry 后对环长取模

distance(u,v)
    v 在树上：检查 v 是否是 u 的祖先
    v 在环上：depth[u] + 环上距离

first_meeting(u,v)
    depth 相同：同 entry 后找树上首次汇合
    depth 不同：比较环上同步 phase
```
#### 注意事项
* 函数图要求每个节点**恰好一条出边**
* 自环是长度为 $1$ 的合法环
* 不同弱连通分量之间永远不可达也永远不会相遇
* 每个弱连通分量恰好有一个有向环
* `cycle_pos[u]` 仅对环点有效，树点应使用 `cycle_pos[entry[u]]`
* `depth[u]` 是到环距离，不是一般意义上的树深度
* `rev` 包含环内部的反向边，若只遍历挂在环上的树，需要跳过已经位于环上的邻点
* 函数图与无向图中的基环树不同，不要混用板子
* 当前构造函数完成全部预处理，修改 `to` 后必须重新构造
* 若只需要找环和入环信息，可以删除 `up/lift/jump/distance/first_meeting`
* 若只需要第 $k$ 次跳转，可以保留入度剥离、环分解与 `jump()`，删除同步相遇相关代码
#### 复杂度
设节点数为 $n$
入度剥离与环分解
$$
O(n)
$$
反图 BFS
$$
O(n)
$$
倍增预处理
$$
O(n\log n)
$$
因此总预处理
$$
O(n\log n)
$$
空间复杂度
$$
O(n\log n)
$$
查询复杂度

| 接口                 |         复杂度 |
| ------------------ | ----------: |
| `on_cycle()`       |      $O(1)$ |
| `same_component()` |      $O(1)$ |
| `cycle_size()`     |      $O(1)$ |
| `component_size()` |      $O(1)$ |
| `orbit_size()`     |      $O(1)$ |
| `jump()`           | $O(\log n)$ |
| `distance()`       | $O(\log n)$ |
| `first_meeting()`  | $O(\log n)$
