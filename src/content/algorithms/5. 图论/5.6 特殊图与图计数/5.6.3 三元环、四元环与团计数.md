### 三元环、四元环与团计数
---
#### 三元环与四元环
二者共用一次按度数排序后的定向
```cpp
struct SmallCycleCounter {
    int n;
    vector<pair<int, int>> edges;
    vector<vector<int>> g, out;
    vector<int> deg, rk;
    bool built{};
    explicit SmallCycleCounter(int n_) : n(n_), g(n), out(n), deg(n), rk(n) {}
    void add_edge(int u, int v) {
        assert(!built && u != v);
        edges.push_back({u, v});
        g[u].push_back(v);
        g[v].push_back(u);
        ++ deg[u], ++ deg[v];
    }
    void build() {
        if (built) return;
        vector<int> order(n);
        ranges::iota(order, 0);
        ranges::sort(order, {}, [&](int u) { return pair{deg[u], u}; });
        for (int i : views::iota(0, n)) rk[order[i]] = i;
        for (auto [u, v] : edges) {
            if (rk[u] > rk[v]) swap(u, v);
            out[u].push_back(v);
        }
        built = true;
    }
    i64 count_triangles() {
        build();
        vector<int> mark(n, -1);
        i64 ans = 0;
        for (int u : views::iota(0, n)) {
            for (int v : out[u]) mark[v] = u;
            for (int v : out[u]) {
                for (int w : out[v]) ans += mark[w] == u;
            }
        }
        return ans;
    }
    i64 count_four_cycles() {
        build();
        vector<int> cnt(n);
        i64 ans = 0;
        for (int u : views::iota(0, n)) {
            for (int v : g[u]) {
                for (int w : out[v]) if (rk[w] > rk[u]) ans += cnt[w] ++;
            }
            for (int v : g[u]) {
                for (int w : out[v]) if (rk[w] > rk[u]) -- cnt[w];
            }
        }
        return ans;
    }
};
```
##### 定向规则
定义点序为
```cpp
pair{deg[u], u}
```
较小的点指向较大的点
```text
低度点 -> 高度点
```
度数相同则按编号定向
该定向满足每个点出度为 $O(\sqrt m)$
##### 三元环
```cpp
i64 ans = counter.count_triangles();
```
对每个 `u`
```text
标记 u 的全部出邻居
枚举 u -> v -> w
检查 w 是否被 u 标记
```
每个三元环恰好统计一次
###### 复杂度
$$O(m\sqrt m)$$
##### 四元环
```cpp
i64 ans = counter.count_four_cycles();
```
对于固定 `u` 枚举
```text
u - v -> w
```
并要求
```cpp
rk[w] > rk[u]
```
`cnt[w]` 表示此前已有多少条合法的 `u -> ... -> w` 长度为 $2$ 的路径
每加入一条新路径
```cpp
ans += cnt[w] ++;
```
就与此前所有路径分别形成一个四元环
###### 复杂度
$$O(m\sqrt m)$$
##### 注意事项
- 输入必须是简单无向图
- 不允许自环
- 不允许重复边
- 有对角线的四点子图可能包含多个不同四元环，本函数按环的边集合分别计数
- `build()` 后不能再加边
- 三元环与四元环计数可以分别或重复调用
#### `n <= 64` 最大团与极大团
Bron--Kerbosch + pivot
```cpp
struct Clique64 {
    struct Result {
        int maximum{};
        i64 maximum_count{}, maximal_count{};
        u64 maximum_mask{};
    };
    int n;
    vector<u64> adj;
    Result res;
    explicit Clique64(int n_) : n(n_), adj(n) {
        assert(0 <= n && n <= 64);
    }
    void add_edge(int u, int v) {
        assert(u != v);
        adj[u] |= 1ULL << v;
        adj[v] |= 1ULL << u;
    }
private:
    void dfs(u64 r, u64 p, u64 x) {
        if (!p && !x) {
            int siz{ popcount(r) };
            ++ res.maximal_count;
            if (siz > res.maximum) {
                res.maximum = siz;
                res.maximum_count = 1;
                res.maximum_mask = r;
            } else if (siz == res.maximum) ++ res.maximum_count;
            return;
        }
        int pivot{ -1 }, score{ -1 };
        for (u64 s = p | x; s; s &= s - 1) {
            int u{ countr_zero(s) };
            int cur{ popcount(p & adj[u]) };
            if (cur > score) score = cur, pivot = u;
        }
        u64 cand = pivot == -1 ? p : p & ~adj[pivot];
        while (cand) {
            int v{ countr_zero(cand) };
            u64 bit = 1ULL << v;
            cand ^= bit;
            dfs(r | bit, p & adj[v], x & adj[v]);
            p ^= bit;
            x |= bit;
        }
    }
public:
    Result solve() {
        res = {};
        u64 full = n == 64 ? ~0ULL : (1ULL << n) - 1;
        dfs(0, full, 0);
        return res;
    }
};
```
##### 返回值
```cpp
auto res = clique.solve();
```
`res.maximum`：最大团大小
`res.maximum_count`：最大团数量
`res.maximal_count`：全部极大团数量
`res.maximum_mask`：任意一个最大团
恢复节点
```cpp
for (int u : views::iota(0, clique.n)) {
    if (res.maximum_mask >> u & 1ULL) {
        // u 位于一个最大团
    }
}
```
##### 最大团与极大团
最大团要求点数最大
极大团只要求无法继续加入新节点
每个最大团都是极大团，但极大团不一定最大
原版本的 `mcnt` 实际统计的是极大团数量，本版将
- 最大团大小
- 最大团数量
- 极大团数量
分别返回
##### 为什么不能按 `best` 剪枝
如果只求最大团，可以使用
```text
当前团大小 + 候选点数量 <= best
```
等分支限界
但这里要求准确统计所有极大团
若按照最大团大小剪枝，会遗漏较小极大团，使 `maximal_count` 错误
因此原来的恒假条件
```cpp
if (popcount(r) + popcount(p) <= best && mcnt < 0) return;
```
应直接删除
若只求最大团，应另用专门的最大团分支限界板子
##### 注意事项
- 要求 `n <= 64`
- 输入按简单无向图处理
- 自环没有意义
- `solve()` 会枚举全部极大团，最坏复杂度必然指数级
- pivot 能大幅降低实际搜索量，但不改变指数级最坏界
- 最大团一定是极大团，因此可以在枚举极大团时顺便统计最大团
- `maximum_mask` 只保存任意一个最大团
##### 复杂度
极大团数量经典上界为
$$O(3^{n/3})$$
Bron--Kerbosch + pivot 的运行时间与输出的极大团数量同为指数级量级

邻接位集空间
$$O(n)$$
递归深度
$$O(n)$$
