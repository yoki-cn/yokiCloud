### 弦图与完美消除序列
---
MCS 使用桶维护，生成候选序与验证均为线性复杂度
```cpp
struct ChordalGraph {
    int n;
    vector<vector<int>> g;
    explicit ChordalGraph(int n_) : n(n_), g(n) {}
    void add_edge(int u, int v) {
        assert(u != v);
        g[u].push_back(v);
        g[v].push_back(u);
    }
    pair<bool, vector<int>> perfect_elimination_order() const {
        if (!n) return {true, {}};
        // ===== MCS =====
        vector<int> label(n), order(n), pos(n);
        vector<char> used(n);
        vector<vector<int>> bucket(n + 1);
        bucket[0].resize(n);
        ranges::iota(bucket[0], 0);
        int mx{ 0 };
        for (int i = n - 1; i >= 0; -- i) {
            while (true) {
                while (!bucket[mx].empty()) {
                    int u{ bucket[mx].back() };
                    if (!used[u] && label[u] == mx) break;
                    bucket[mx].pop_back();
                }
                if (!bucket[mx].empty()) break;
                -- mx;
            }
            int u{ bucket[mx].back() };
            bucket[mx].pop_back();
            used[u] = true;
            order[i] = u;
            for (int v : g[u]) if (!used[v]) {
                ++ label[v];
                bucket[label[v]].push_back(v);
                mx = max(mx, label[v]);
            }
        }
        for (int i : views::iota(0, n)) pos[order[i]] = i;
        // ===== 线性验证 PEO =====
        vector<int> parent(n, -1), mark(n, -1);
        vector<vector<int>> child(n);
        for (int u : views::iota(0, n)) {
            for (int v : g[u]) {
                if (pos[v] > pos[u] && (parent[u] == -1 || pos[v] < pos[parent[u]])) parent[u] = v;
            }
            if (parent[u] != -1) child[parent[u]].push_back(u);
        }
        for (int p : views::iota(0, n)) if (!child[p].empty()) {
            mark[p] = p;
            for (int v : g[p]) mark[v] = p;
            for (int u : child[p]) {
                for (int v : g[u]) {
                    if (pos[v] > pos[u] && mark[v] != p) return {false, move(order)};
                }
            }
        }
        return {true, move(order)};
    }
    int maximum_clique_size(const vector<int> &peo) const {
        vector<int> pos(n);
        for (int i : views::iota(0, n)) pos[peo[i]] = i;
        int ans{ 0 };
        for (int u : views::iota(0, n)) {
            int cur{ 1 };
            for (int v : g[u]) cur += pos[v] > pos[u];
            ans = max(ans, cur);
        }
        return ans;
    }
    pair<int, vector<int>> optimal_coloring(const vector<int> &peo) const {
        vector<int> color(n, -1), mark(n, -1);
        int colors{ 0 };
        for (int i = n - 1; i >= 0; -- i) {
            int u{ peo[i] };
            for (int v : g[u]) if (color[v] != -1) mark[color[v]] = u;
            int c{ 0 };
            while (c < colors && mark[c] == u) ++ c;
            if (c == colors) ++ colors;
            color[u] = c;
        }
        return {colors, move(color)};
    }
};
```
#### 定义与接口
完美消除序列 PEO 满足
> 对序列中的每个点 `u`，所有排在 `u` 后面的邻点构成团

无向简单图是弦图，当且仅当存在 PEO
```cpp
auto [ok, peo] = chordal.perfect_elimination_order();
```
`ok == true` 时 `peo` 是一组合法 PEO
最大团
```cpp
int omega = chordal.maximum_clique_size(peo);
```
最优染色
```cpp
auto [chromatic, color] = chordal.optimal_coloring(peo);
```
弦图是完美图，因此
$$
\chi(G)=\omega(G)
$$
#### 注意事项
* 输入必须是简单无向图
* 不允许自环
* 不应重复加入同一条边，否则 MCS 的 `label` 会被重复统计
* 输入可能有重边时应先去重
* MCS 得到的候选序只有验证成功后才能作为 PEO
* `maximum_clique_size()` 与 `optimal_coloring()` 要求传入合法 PEO
#### 复杂度
MCS
$$
O(n+m)
$$
PEO 验证
$$
O(n+m)
$$
最大团与染色
$$
O(n+m)
$$
空间
$$
O(n+m)
$$
### 稳定婚姻
---
```cpp
struct StableMarriageResult {
    vector<int> a_to_b, b_to_a;
};
StableMarriageResult stable_marriage(const vector<vector<int>> &pref_a, const vector<vector<int>> &pref_b) {
    int n = pref_a.size();
    assert(pref_b.size() == n);
    vector rank_b(n, vector<int>(n));
    for (int b : views::iota(0, n)) {
        for (int i : views::iota(0, n)) rank_b[b][pref_b[b][i]] = i;
    }
    vector<int> a_to_b(n, -1), b_to_a(n, -1), nxt(n), free(n);
    ranges::iota(free, 0);
    while (!free.empty()) {
        int a{ free.back() }, b{ pref_a[a][nxt[a] ++] };
        int old{ b_to_a[b] };
        if (old == -1) {
            a_to_b[a] = b;
            b_to_a[b] = a;
            free.pop_back();
        } else if (rank_b[b][a] < rank_b[b][old]) {
            a_to_b[old] = -1;
            a_to_b[a] = b;
            b_to_a[b] = a;
            free.back() = old;
        }
    }
    return {move(a_to_b), move(b_to_a)};
}
```
#### 接口与性质
```cpp
auto [a_to_b, b_to_a] = stable_marriage(pref_a, pref_b);
```
偏好表要求每行都是 `[0,n)` 的完整排列
A 方主动求婚，因此得到
* A 方最优稳定匹配
* B 方最差稳定匹配
这里的最优和最差均限定在全部稳定匹配中
若希望 B 方最优，交换双方后运行即可
#### 复杂度
每个 A 最多向每个 B 求婚一次
$$
O(n^2)
$$
额外空间
$$
O(n^2)
$$
主要由 `rank_b` 占用
### 图度序列判定
---
#### Erdős--Gallai
```cpp
bool graphical_degree_sequence(vector<int> d) {
    int n = d.size();
    if (ranges::any_of(d, [&](int x) { return x < 0 || x >= n; })) return false;
    ranges::sort(d, greater{});
    vector<i64> pre(n + 1);
    for (int i : views::iota(0, n)) pre[i + 1] = pre[i] + d[i];
    if (pre[n] & 1) return false;
    for (int k : views::iota(1, n + 1)) {
        int p = lower_bound(d.begin() + k, d.end(), k, greater<int>()) - d.begin();
        i64 rhs = 1LL * k * (k - 1) + 1LL * (p - k) * k + pre[n] - pre[p];
        if (pre[k] > rhs) return false;
    }
    return true;
}
```
判断非负整数序列能否作为某个简单无向图的度数序列
降序排列后，对所有 $1\le k\le n$ 检查
$$
\sum_{i=1}^{k}d_i\le k(k-1)+\sum_{i=k+1}^{n}\min(d_i,k)
$$
并要求
$$
\sum_i d_i\equiv0\pmod2
$$
复杂度
$$
O(n\log n)
$$
#### Landau 竞赛图得分序列
```cpp
bool tournament_score_sequence(vector<int> s) {
    int n = s.size();
    ranges::sort(s);
    i64 sum { 0 };
    for (int k : views::iota(1, n + 1)) {
        if (s[k - 1] < 0 || s[k - 1] >= n) return false;
        sum += s[k - 1];
        if (k < n && sum < 1LL * k * (k - 1) / 2) return false;
    }
    return sum == 1LL * n * (n - 1) / 2;
}
```
升序得分序列满足
$$
\sum_{i=1}^{k}s_i\ge\binom{k}{2}\qquad(1\le k<n)
$$
且
$$
\sum_{i=1}^{n}s_i=\binom{n}{2}
$$
#### 复杂度
$$
O(n\log n)
$$
