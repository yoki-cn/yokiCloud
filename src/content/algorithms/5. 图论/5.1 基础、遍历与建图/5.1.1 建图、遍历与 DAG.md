### 建图、遍历与 DAG
---
```cpp
template<class W = i64>
struct Graph {
    struct Edge {
        int to, id;
        W w;
    };
    int n, edge_cnt{ 0 };
    vector<vector<Edge>> g;
    explicit Graph(int n_) : n(n_), g(n) {}
    // ===== 基础建图：以下代码是图结构核心，不建议删除 =====
    int add_directed(int u, int v, W w = 1) {
        int id{ edge_cnt ++ };
        g[u].push_back({v, id, w});
        return id;
    }
    int add_undirected(int u, int v, W w = 1) {
        int id{ edge_cnt ++ };
        g[u].push_back({v, id, w});
        g[v].push_back({u, id, w});
        return id;
    }
    // ===== 基础遍历：若只把 Graph 当邻接表使用，可以整段删除 =====
    template<ranges::input_range R>
    vector<int> bfs(R &&source) const {
        vector<int> dis(n, -1), q;
        q.reserve(n);
        for (int s : source) if (dis[s] == -1) {
            dis[s] = 0;
            q.push_back(s);
        }
        for (int h = 0; h < q.size(); ++ h) {
            int u = q[h];
            for (const auto &e : g[u]) if (dis[e.to] == -1) {
                dis[e.to] = dis[u] + 1;
                q.push_back(e.to);
            }
        }
        return dis;
    }
    vector<int> bfs(int s) const { return bfs(array{s}); }
    vector<int> dfs_order(int s) const {
        vector<int> order, stk{s};
        vector<char> vis(n);
        vis[s] = true;
        while (!stk.empty()) {
            int u = stk.back();
            stk.pop_back();
            order.push_back(u);
            for (const auto &e : g[u] | views::reverse) if (!vis[e.to]) {
                vis[e.to] = true;
                stk.push_back(e.to);
            }
        }
        return order;
    }
    // ===== DAG：若不需要拓扑排序或 DAG DP，以下代码可整段删除 =====
    optional<vector<int>> topo_order() const {
        vector<int> deg(n), order;
        order.reserve(n);
        for (int u : views::iota(0, n)) {
            for (const auto &e : g[u]) ++ deg[e.to];
        }
        for (int u : views::iota(0, n)) if (deg[u] == 0) order.push_back(u);
        for (int h = 0; h < order.size(); ++ h) {
            int u{ order[h] };
            for (const auto &e : g[u]) {
                if (-- deg[e.to] == 0) order.push_back(e.to);
            }
        }
        if (order.size() != n) return nullopt;
        return order;
    }
    template<class T = i64, ranges::input_range R>
    optional<vector<T>> dag_path_count(R &&source) const {
        auto order = topo_order();
        if (!order) return nullopt;
        vector<T> dp(n);
        for (int s : source) dp[s] = T{ 1 };
        for (int u : *order) {
            for (const auto &e : g[u]) dp[e.to] += dp[u];
        }
        return dp;
    }
    template<class T = i64>
    optional<vector<T>> dag_path_count(int s) const {
        return dag_path_count<T>(array{s});
    }
    template<ranges::input_range R>
    optional<vector<W>> dag_longest_path(R &&source) const {
        auto order = topo_order();
        if (!order) return nullopt;
        const W neg = numeric_limits<W>::lowest() / 4;
        vector<W> dis(n, neg);
        for (int s : source) dis[s] = W{};
        for (int u : *order) if (dis[u] != neg) {
            for (const auto &e : g[u]) {
                dis[e.to] = max(dis[e.to], dis[u] + e.w);
            }
        }
        return dis;
    }
    optional<vector<W>> dag_longest_path(int s) const {
        return dag_longest_path(array{s});
    }
    template<ranges::input_range R>
    optional<vector<W>> dag_shortest_path(R &&source) const {
        auto order = topo_order();
        if (!order) return nullopt;
        const W inf = numeric_limits<W>::max() / 4;
        vector<W> dis(n, inf);
        for (int s : source) dis[s] = W{};
        for (int u : *order) if (dis[u] != inf) {
            for (const auto &e : g[u]) {
                dis[e.to] = min(dis[e.to], dis[u] + e.w);
            }
        }
        return dis;
    }
    optional<vector<W>> dag_shortest_path(int s) const {
        return dag_shortest_path(array{s});
    }
    // ===== 可达性：若不需要传递闭包，以下代码可整段删除 =====
    template<size_t N>
    vector<bitset<N>> transitive_closure() const {
        assert(n <= int(N));
        vector<bitset<N>> reach(n);
        for (int u : views::iota(0, n)) {
            reach[u].set(u);
            for (const auto &e : g[u]) reach[u].set(e.to);
        }
        for (int k : views::iota(0, n)) {
            for (int u : views::iota(0, n)) {
                if (reach[u][k]) reach[u] |= reach[k];
            }
        }
        return reach;
    }
    template<size_t N>
    optional<vector<bitset<N>>> dag_transitive_closure() const {
        assert(n <= int(N));
        auto order = topo_order();
        if (!order) return nullopt;
        vector<bitset<N>> reach(n);
        for (int u : views::iota(0, n)) reach[u].set(u);
        for (int u : *order | views::reverse) {
            for (const auto &e : g[u]) reach[u] |= reach[e.to];
        }
        return reach;
    }
};
```
#### 成员变量
- `n`：点数，点编号为 `0..n-1`
- `edge_cnt`：逻辑边数量
- `g[u]`：从 `u` 出发的邻接边
- `Edge::to`：终点
- `Edge::id`：逻辑边编号
- `Edge::w`：边权
- 无向边的两条邻接边共享同一个 `id`
#### 接口
- `Graph<W>(n)`：建立 `n` 个点的空图，$O(n)$
- `add_directed(u, v, w)`：加入有向边 `u -> v`，返回边号，均摊 $O(1)$
- `add_undirected(u, v, w)`：加入无向边，两方向共用边号，均摊 $O(1)$
- `bfs(s)`：单源 BFS，返回无权最短距离，不可达为 `-1`，$O(n+m)$
- `bfs(source)`：多源 BFS，`source` 可以是 `vector<int>`、`array` 等 range，$O(n+m)$
- `dfs_order(s)`：返回从 `s` 可达部分的非递归 DFS 访问序，$O(n+m)$
- `topological_order()`：返回拓扑序；有环返回 `nullopt`，$O(n+m)$
- `dag_path_count<T>(s)`：统计从 `s` 出发到各点的路径条数；有环返回 `nullopt`，$O(n+m)$
- `dag_path_count<T>(source)`：多源版本，每个源点初值为 `1`
- `dag_longest_path(s)`：DAG 单源最长路；有环返回 `nullopt`，$O(n+m)$
- `dag_longest_path(source)`：DAG 多源最长路
- `dag_shortest_path(s)`：DAG 单源最短路，允许负边；有环返回 `nullopt`，$O(n+m)$
- `dag_shortest_path(source)`：DAG 多源最短路
- `transitive_closure<N>()`：一般有向图 Bitset 传递闭包
- `dag_transitive_closure<N>()`：DAG 专用传递闭包；有环返回 `nullopt`
#### 注意事项
- `bfs()` 完全忽略边权，只适合所有边代价相同的无权最短路
- `topological_order()` 每次重新计算入度，因此可以安全重复调用，也不会修改图
- 拓扑排序和全部 `dag_*` 接口要求图由**有向边**组成；加入一条普通无向边等价于加入两个反向弧，通常会形成长度为 `2` 的有向环
- `dag_path_count()` 将平行边视为不同路径；每个源点本身存在一条长度为 `0` 的路径。若计数需要取模，可令 `T` 为取模类
- DAG 最短路不要求边权非负；最长路同理。不可达点分别保留 `numeric_limits<W>::max() / 4` 与 `numeric_limits<W>::lowest() / 4`
- 距离相加前仍需估算数值范围，不能因为设置了无穷元就忽略溢出
- `transitive_closure<N>()` 中 `N` 是编译期最大点数，要求 `n <= N`；实际只分配 `n` 行 Bitset
- 若图确定为 DAG，优先使用 `dag_transitive_closure()`，通常明显快于 Floyd 式传递闭包
- `dfs_order()` 只保证得到一个 DFS 风格访问序，不维护灰点/黑点、进入时间和退出时间，不能直接拿来做有向 DFS 边分类
#### 常见用法
##### 无向图与边号
```cpp
Graph g(n);
int id = g.add_undirected(u, v, w);
for (const auto &[v, id, w] : g.g[u]) {
    // ...
}
```
桥、欧拉路等需要区分“父边”而不是“父点”时，可以直接比较 `id`
##### 单源 / 多源 BFS
```cpp
auto dis1 = g.bfs(s);
vector<int> source = {s1, s2, s3};
auto dis2 = g.bfs(source);
```
##### 拓扑排序
```cpp
auto order = g.topological_order();
if (!order) {
    // 有环
} else {
    for (int u : *order) {
        // ...
    }
}
```
##### DAG 最长路
```cpp
auto dis = g.dag_longest_path(s);
if (!dis) {
    // 不是 DAG
} else {
    cout << (*dis)[t] << '\n';
}
```
##### DAG 路径计数
```cpp
auto cnt = g.dag_path_count<i64>(s);
cout << (*cnt)[t] << '\n';
```
使用取模类：
```cpp
auto cnt = g.dag_path_count<Z>(s);
```
##### DAG 可达性
```cpp
auto reach = g.dag_transitive_closure<2000>();
if (reach && (*reach)[u][v]) {
    // u 可以到达 v
}
```
##### 网格图通常不显式建边
```cpp
constexpr array<pair<int, int>, 4> dir{{
    {1, 0}, {-1, 0}, {0, 1}, {0, -1}
}};
for (auto [dx, dy] : dir) {
    int nx = x + dx, ny = y + dy;
    if (0 <= nx && nx < n && 0 <= ny && ny < m) {
        // 枚举相邻格
    }
}
```
邻接矩阵只在点数较小且图很稠密时考虑：
```cpp
vector matrix(n, vector<i64>(n, INF));
```
#### 复杂度
设 `m` 为邻接表中保存的有向弧数量；无向边会保存为两条弧

| 操作                                                                         |         时间复杂度 |
| -------------------------------------------------------------------------- | ------------: |
| `add_directed()` / `add_undirected()`                                      |     均摊 $O(1)$ |
| `bfs()` / `dfs_order()` / `topological_order()` / DAG 路径计数 / DAG 最短路 / 最长路 |      $O(n+m)$ |
| `transitive_closure<N>()`                                                  |   $O(n^2N/w)$ |
| `dag_transitive_closure<N>()`                                              | $O((n+m)N/w)$ |
当 $N=\Theta(n)$ 时，一般传递闭包可写作：
$$O\left(\frac{n^3}{w}\right)$$

DAG 传递闭包为：
$$O\left(\frac{mn}{w}\right)$$
其中 `w` 为机器字长
基础图空间复杂度：
$$O(n+m)$$
传递闭包额外空间复杂度：
$$O\left(\frac{nN}{w}\right)$$
