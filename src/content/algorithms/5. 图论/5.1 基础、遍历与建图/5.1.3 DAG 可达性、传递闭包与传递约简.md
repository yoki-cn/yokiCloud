### DAG 可达性、传递闭包与传递约简
---
**解决问题：** 在 DAG 上批量判断可达性、构造传递闭包，并删除所有可由其他路径替代的冗余边得到唯一的传递约简
```cpp
struct DAGReachability {
    struct Edge { int u, v; };
    int n, words{};
    vector<Edge> edges;
    vector<vector<int>> out, in;
    vector<int> topo, pos;
    vector<vector<u64>> reach;
    bool built{}, dag{}, closure_ready{};
    explicit DAGReachability(int n_) : n(n_), out(n), in(n) {}
    int add_edge(int u, int v) {
        assert(!built);
        int id = edges.size();
        edges.push_back({u, v});
        out[u].push_back(id);
        in[v].push_back(id);
        return id;
    }
    // ===== 拓扑序：所有后续功能依赖 =====
    bool build() {
        if (built) return dag;
        vector<int> indeg(n);
        for (auto [u, v] : edges) ++ indeg[v];
        queue<int> q;
        for (int u : views::iota(0, n)) if (!indeg[u]) q.push(u);
        topo.reserve(n);
        while (!q.empty()) {
            int u{ q.front() };
            q.pop();
            topo.push_back(u);
            for (int id : out[u]) {
                int v{ edges[id].v };
                if (-- indeg[v] == 0) q.push(v);
            }
        }
        dag = topo.size() == n;
        if (dag) {
            pos.resize(n);
            for (int i : views::iota(0, n)) pos[topo[i]] = i;
        }
        built = true;
        return dag;
    }
    // ===== 完整传递闭包：中等规模使用 =====
    void build_closure() {
        assert(build());
        if (closure_ready) return;
        words = (n + 63) >> 6;
        reach.assign(n, vector<u64>(words));
        for (int u : topo | views::reverse) {
            reach[u][u >> 6] |= 1ULL << (u & 63);
            for (int id : out[u]) {
                int v{ edges[id].v };
                for (int b : views::iota(0, words)) reach[u][b] |= reach[v][b];
            }
        }
        closure_ready = true;
    }
    bool reachable(int u, int v) const {
        assert(closure_ready);
        return reach[u][v >> 6] >> (v & 63) & 1ULL;
    }
    int reachable_count(int u) const {
        assert(closure_ready);
        int ans{ 0 };
        for (u64 x : reach[u]) ans += popcount(x);
        return ans;
    }
    // ===== 大图离线询问：每 64 个询问并行传播 =====
    vector<char> answer_queries(const vector<pair<int, int>> &queries) const {
        assert(built && dag);
        int q = queries.size();
        vector<char> ans(q);
        if (closure_ready) {
            for (int i : views::iota(0, q)) {
                auto [s, t] = queries[i];
                ans[i] = reachable(s, t);
            }
            return ans;
        }
        vector<u64> mask(n);
        for (int l = 0; l < q; l += 64) {
            ranges::fill(mask, 0);
            int r{ min(l + 64, q) };
            for (int i : views::iota(l, r)) {
                auto [s, t] = queries[i];
                mask[s] |= 1ULL << (i - l);
            }
            for (int u : topo) {
                for (int id : out[u]) mask[edges[id].v] |= mask[u];
            }
            for (int i : views::iota(l, r)) {
                auto [s, t] = queries[i];
                ans[i] = mask[t] >> (i - l) & 1ULL;
            }
        }
        return ans;
    }
    // ===== 传递约简：返回保留的原边编号 =====
    vector<int> transitive_reduction() {
        build_closure();
        vector<int> kept;
        kept.reserve(edges.size());
        vector<u64> covered(words);
        for (int u : views::iota(0, n)) {
            vector<int> ids = out[u];
            ranges::sort(ids, {}, [&](int id) { return pos[edges[id].v]; });
            ranges::fill(covered, 0);
            for (int id : ids) {
                int v{ edges[id].v };
                if (covered[v >> 6] >> (v & 63) & 1ULL) continue;
                kept.push_back(id);
                for (int b : views::iota(0, words)) covered[b] |= reach[v][b];
            }
        }
        ranges::sort(kept);
        return kept;
    }
};
```
#### 基本结构
DAG 中存在拓扑序
$$
u\to v\Longrightarrow pos[u]<pos[v]
$$
因此可达性 DP 可以按照**逆拓扑序**进行
定义 `reach[u]` 为从 `u` 可达的全部节点组成的 Bitset，并约定节点可以零步到达自身
$$
u\in reach[u]
$$
对于每条边 $u\to v$
$$
reach[u]\mathrel{|=}reach[v]
$$
因为逆拓扑处理时 `reach[v]` 已经完整求出
```cpp
for (int u : topo | views::reverse) {
    reach[u][u >> 6] |= 1ULL << (u & 63);
    for (int id : out[u]) {
        int v{ edges[id].v };
        for (int b : views::iota(0, words)) reach[u][b] |= reach[v][b];
    }
}
```
最终 `reach` 就是图的传递闭包
#### 接口说明
##### `add_edge(u,v)`
加入有向边
```cpp
int id = dag.add_edge(u, v);
```
返回稳定的原边编号
允许重边，但图最终必须是 DAG
自环一定会使 `build()` 判定失败
##### `build()`
计算拓扑序并判断是否为 DAG
```cpp
if (!dag.build()) {
    // 原图存在环
}
```
成功后
```cpp
dag.topo
```
是拓扑序
```cpp
dag.pos[u]
```
是节点 `u` 的拓扑位置
调用 `build()` 后不能再加边
#### 完整传递闭包
```cpp
dag.build_closure();
```
得到
```cpp
dag.reach[u]
```
表示所有从 `u` 可达的节点
约定可达性包含长度为 $0$ 的路径，因此
```cpp
dag.reachable(u, u) == true
```
若题目只认为至少经过一条边才算可达，在 DAG 中只需额外处理 `u == v`
##### 单次可达查询
```cpp
bool ok = dag.reachable(u, v);
```
复杂度
$$
O(1)
$$
##### 可达节点数量
```cpp
int cnt = dag.reachable_count(u);
```
包含 `u` 自身
若题目不统计自身
```cpp
int cnt = dag.reachable_count(u) - 1;
```
#### 为什么 Bitset 有效
普通集合合并若逐点处理，一条边可能需要 $O(n)$
把 $n$ 个可达状态压入 $\left\lceil\frac n{64}\right\rceil$ 个 `u64` 后，每次集合合并只需 $O\left(\frac n{64}\right)$ 次按位 OR
因此特别适合
* 中等规模 DAG
* 大量任意两点可达询问
* 可达点计数
* 传递闭包
* 偏序关系
* 传递约简
#### 大图离线可达性询问
完整闭包需要
$$
O\left(\frac{n^2}{64}\right)
$$
个 `u64`
当 $n$ 很大时可能无法接受
如果只需要回答 $q$ 个离线询问
```cpp
vector<pair<int, int>> queries;
auto ans = dag.answer_queries(queries);
```
则不需要建立完整闭包
##### 原理
每次同时处理最多 $64$ 个询问
第 `i` 个询问
$$
s_i\to t_i
$$
对应 `u64` 中的一位
初始化
```cpp
mask[s_i] |= 1ULL << i;
```
随后按拓扑序传播
```cpp
mask[v] |= mask[u];
```
若第 `i` 位最终出现在 `mask[t_i]` 中，就说明
$$
s_i\leadsto t_i
$$
成立
因此 $64$ 个 DFS 式可达性传播被压成了一次 DAG DP
##### 使用
```cpp
DAGReachability dag(n);
for (auto [u, v] : edges) dag.add_edge(u, v);
if (!dag.build()) {
    // 不是 DAG
}
vector<pair<int, int>> queries{
    {0, 5},
    {3, 8},
    {7, 2}
};
auto ans = dag.answer_queries(queries);
for (int i : views::iota(0, int(queries.size()))) {
    if (ans[i]) {
        // queries[i].first 可以到达 queries[i].second
    }
}
```
如果已经调用过 `build_closure()`，`answer_queries()` 会自动直接使用闭包，以 $O(q)$ 回答
#### 传递闭包
图 $G$ 的传递闭包 $G^+$ 保留所有可达关系
若
$$
u\leadsto v
$$
则闭包中存在边
$$
u\to v
$$
本板不显式建立这些 $O(n^2)$ 条边，而是用
```cpp
reach[u]
```
保存闭包关系
若确实需要枚举所有可达点
```cpp
dag.build_closure();
for (int u : views::iota(0, n)) {
    for (int b : views::iota(0, dag.words)) {
        for (u64 s = dag.reach[u][b]; s; s &= s - 1) {
            int v{ 64 * b + countr_zero(s) };
            if (v < n) {
                // u 可以到达 v
            }
        }
    }
}
```
#### 传递约简
传递约简要删除所有可由其他路径替代的边，同时保持可达关系完全不变
例如
```text
0 -> 1 -> 2
 \------> 2
```
边
```text
0 -> 2
```
可以通过
```text
0 -> 1 -> 2
```
替代，因此属于冗余边
约简后只保留
```text
0 -> 1 -> 2
```
##### DAG 上的性质
对于 DAG，传递约简是唯一的
若 DAG 表示一个偏序关系，其传递约简就是对应的 Hasse 图
#### `transitive_reduction()`
```cpp
auto kept = dag.transitive_reduction();
```
返回需要保留的**原边编号**
例如
```cpp
for (int id : kept) {
    auto [u, v] = dag.edges[id];
    // u -> v 属于传递约简
}
```
需要直接建立约简后的邻接表
```cpp
vector<vector<int>> reduced(n);
for (int id : dag.transitive_reduction()) {
    auto [u, v] = dag.edges[id];
    reduced[u].push_back(v);
}
```
#### 传递约简原理
固定节点 `u`
将所有出边
$$
u\to v
$$
按照 `v` 的拓扑位置从小到大处理
维护 `covered`
表示已经可以通过此前保留的直接后继到达的全部节点
若当前 `v` 已经满足
$$
v\in covered
$$
说明存在此前某个后继 `w`
$$
u\to w\leadsto v
$$
于是边
$$
u\to v
$$
可以删除
否则必须保留该边，并执行
$$
covered\mathrel{|=}reach[v]
$$
代码对应
```cpp
for (int id : ids) {
    int v = edges[id].v;
    if (covered[v >> 6] >> (v & 63) & 1ULL) continue;
    kept.push_back(id);
    for (int b : views::iota(0, words)) covered[b] |= reach[v][b];
}
```
##### 为什么必须按拓扑序处理后继
若存在
$$
u\to a,\qquad u\to b,\qquad a\leadsto b
$$
必然有
$$
pos[a]<pos[b]
$$
必须先处理 `a`
这样处理 `b` 时才能发现
$$
b\in reach[a]
$$
并删掉冗余边 `u -> b`
如果顺序反过来，会错误保留这条边
#### 重边
闭包和离线询问都天然允许重边
传递约简中若存在多条完全相同的边
```text
u -> v
u -> v
```
只会保留其中第一条
后续同端点边会发现 `v` 已经处于 `covered`
因此无需提前去重
若不同重边携带不同权值或编号语义，需要自行决定希望保留哪一条
例如希望优先保留最小边权，应在加入图前先处理重边，或修改同终点边的处理顺序
#### 一般有向图怎么办
本板只处理 DAG
如果一般有向图存在 SCC，可以先使用前面的 `SCC` 板子缩点
```text
原有向图
    ↓ SCC
缩点 DAG
    ↓ 本板
可达性 / 闭包 / 传递约简
```
对于原图节点 `u,v`
若
```cpp
scc.bel[u] == scc.bel[v]
```
则二者互相可达
否则只需判断缩点 DAG 中
```text
bel[u] -> bel[v]
```
是否可达
注意：**缩点 DAG 的传递约简只约简 SCC 之间的关系**
它不会给出原有环图内部的最小等价边集
#### 常见用法
##### 中等规模全可达查询
```cpp
DAGReachability dag(n);
for (auto [u, v] : edges) dag.add_edge(u, v);
if (!dag.build()) {
    // 有环
}
dag.build_closure();
if (dag.reachable(u, v)) {
    // u 可以到达 v
}
```
##### 每个点能到达多少点
```cpp
dag.build_closure();
for (int u : views::iota(0, n)) {
    int cnt = dag.reachable_count(u);
}
```
##### 大图批量询问
```cpp
DAGReachability dag(n);
for (auto [u, v] : edges) dag.add_edge(u, v);
dag.build();
vector<pair<int, int>> query(q);
for (auto &[u, v] : query) cin >> u >> v;
auto ans = dag.answer_queries(query);
```
此时不要调用 `build_closure()`，否则失去低内存优势
##### 求传递约简
```cpp
DAGReachability dag(n);
for (int i = 0; i < m; ++ i) {
    int u, v;
    cin >> u >> v;
    dag.add_edge(u, v);
}
if (!dag.build()) {
    // 不是 DAG
}
auto kept = dag.transitive_reduction();
for (int id : kept) {
    auto [u, v] = dag.edges[id];
    cout << u << ' ' << v << '\n';
}
```
#### 快速回忆
```text
DAG + 全可达关系
    逆拓扑
    reach[u] |= reach[v]

中等 n
    n 个 Bitset
    单次查询 O(1)

大 n + q 个离线询问
    每 64 个询问占一个 u64
    沿拓扑序传播 mask

传递约简
    先求闭包
    固定 u
    出边终点按 topo 从小到大
    v 已被 covered 覆盖 -> 删除 u -> v
    否则保留并 covered |= reach[v]

一般有向图
    先 SCC 缩点成 DAG
```
#### 注意事项
* `build()` 后不能继续加入边
* `build()` 返回 `false` 表示存在有向环，后续 DAG 接口不能使用
* `reachable(u,u)` 定义为 `true`
* 完整闭包适合中等规模，空间通常比时间更先成为瓶颈
* `answer_queries()` 在没有闭包时使用 $64$ 路并行传播，只需 $O(n)$ 额外空间
* 传递约简依赖完整闭包，因此仍需要 $O(n^2/64)$ Bitset 空间
* DAG 的传递约简唯一
* 重边不会影响可达性，约简只会保留其中一条
* 如果边携带权值、容量等额外语义，传递约简只保证**可达关系**不变，不保证这些额外量保持不变
* 一般有向图先 SCC 缩点后可以处理分量之间的可达关系，但不能把得到的结果理解为原图完整传递约简
* 若只需要普通拓扑 DP，不需要可达矩阵，不应使用完整闭包
#### 复杂度
记
$$
B=\left\lceil\frac n{64}\right\rceil
$$
拓扑排序
$$
O(n+m)
$$
完整传递闭包
$$
O(n+mB)
$$
通常写作
$$
O\left(\frac{nm}{64}+\frac{n^2}{64}\right)
$$
空间复杂度
$$
O(nB+m)=O\left(\frac{n^2}{64}+m\right)
$$
闭包建立后的单次可达查询
$$
O(1)
$$
`reachable_count()`
$$
O(B)
$$
对于 $q$ 个离线询问，不建立完整闭包时
$$
O\left(\left\lceil\frac q{64}\right\rceil(n+m)\right)
$$
额外空间
$$
O(n)
$$
传递约简在已经具有闭包后，还需要对每个节点的出边按拓扑位置排序，并进行 Bitset 合并
$$
O(m\log m+mB)
$$
因此包含闭包构造后的总复杂度为
$$
O(n+mB+m\log m)
$$
总空间仍为
$$
O(nB+m)
$$
