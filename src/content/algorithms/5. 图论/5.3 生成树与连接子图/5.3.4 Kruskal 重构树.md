### Kruskal 重构树
---
```cpp
struct KruskalTree {
    struct Edge {
        int u, v;
        i64 w;
    };
    int n, tot{}, lg{};
    vector<i64> value;
    vector<array<int, 2>> child;
    vector<int> parent, depth, leaf_count, root_of, roots;
    vector<vector<int>> up;
    explicit KruskalTree(int n_) : n(n_) {}
    // ===== 构造重构森林 =====
    void build(vector<Edge> edges) {
        ranges::sort(edges, {}, &Edge::w);
        int cap = max(0, 2 * n - 1);
        value.assign(cap, 0);
        child.assign(cap, {-1, -1});
        parent.assign(cap, -1);
        leaf_count.assign(cap, 0);
        for (int u : views::iota(0, n)) leaf_count[u] = 1;
        DSU dsu(n);
        vector<int> top(n);
        ranges::iota(top, 0);
        tot = n;
        for (auto [u, v, w] : edges) {
            int x = dsu.find(u), y = dsu.find(v);
            if (x == y) continue;
            int a{ top[x] }, b{ top[y] }, p{ tot ++ };
            value[p] = w;
            child[p] = {a, b};
            parent[a] = parent[b] = p;
            leaf_count[p] = leaf_count[a] + leaf_count[b];
            dsu.unite(x, y);
            top[dsu.find(x)] = p;
        }
        value.resize(tot);
        child.resize(tot);
        parent.resize(tot);
        leaf_count.resize(tot);
        roots.clear();
        for (int u : views::iota(0, n)) {
            if (dsu.is_root(u)) roots.push_back(top[u]);
        }
        // ===== 倍增：若只需要重构森林本身，可以删除 =====
        depth.assign(tot, 0);
        root_of.assign(tot, -1);
        vector<int> order;
        order.reserve(tot);
        for (int r : roots) {
            root_of[r] = r;
            order.push_back(r);
        }
        for (int i = 0; i < order.size(); ++ i) {
            int u = order[i];
            for (int v : child[u]) if (v != -1) {
                depth[v] = depth[u] + 1;
                root_of[v] = root_of[u];
                order.push_back(v);
            }
        }
        lg = bit_width((unsigned)max(1, tot));
        up.assign(lg, vector<int>(tot));
        for (int u : views::iota(0, tot)) {
            up[0][u] = parent[u] == -1 ? u : parent[u];
        }
        for (int k = 1; k < lg; ++ k) {
            for (int u : views::iota(0, tot)) {
                up[k][u] = up[k - 1][up[k - 1][u]];
            }
        }
    }
    // ===== 常用查询 =====
    int lca(int u, int v) const {
        if (root_of[u] != root_of[v]) return -1;
        if (depth[u] < depth[v]) swap(u, v);
        int d = depth[u] - depth[v];
        for (int k = 0; k < lg; ++ k) {
            if (d >> k & 1) u = up[k][u];
        }
        if (u == v) return u;
        for (int k = lg - 1; k >= 0; -- k) {
            if (up[k][u] == up[k][v]) continue;
            u = up[k][u];
            v = up[k][v];
        }
        return up[0][u];
    }
    optional<i64> connect_threshold(int u, int v) const {
        assert(u != v);
        int p = lca(u, v);
        if (p == -1) return nullopt;
        return value[p];
    }
    bool connected(int u, int v, i64 limit) const {
        if (u == v) return true;
        auto w = connect_threshold(u, v);
        return w && *w <= limit;
    }
    int component_node(int u, i64 limit) const {
        for (int k = lg - 1; k >= 0; -- k) {
            int p{ up[k][u] };
            if (p != u && p >= n && value[p] <= limit) u = p;
        }
        return u;
    }
    int component_size(int u, i64 limit) const {
        return leaf_count[component_node(u, limit)];
    }
};
```
#### 接口说明
构造时将边按权值从小到大加入
原图节点编号：
$$
[0,n)
$$
重构得到的新节点编号：
$$
[n,tot)
$$
每次两个连通块第一次合并时建立一个新节点 `p`：
```cpp
value[p] = 当前边权;
child[p] = {两个连通块对应的重构树根};
```
因此从叶子向根：
$$
value
$$
单调不降
##### `build(edges)`
建立 Kruskal 重构森林
若原图连通：
```cpp
roots.size() == 1
```
否则会得到多棵树
最终：
$$
tot\le 2n-1
$$
##### `lca(u,v)`
求重构森林 LCA
若 `u,v` 原本不连通，返回：
```cpp
-1
```
对于两个不同的原图节点：
```cpp
int p = tree.lca(u, v);
```
若 `p != -1`，则：
```cpp
tree.value[p]
```
就是两点第一次连通所需的最小边权阈值。
##### `connect_threshold(u,v)`
对于不同节点，返回：
$$
\min\{W:\text{只保留 }w\le W\text{ 的边后 }u,v\text{ 连通}\}
$$
原图最终仍不连通则返回 `nullopt`
等价于两点之间的最小瓶颈值：
$$
\min_P\max_{e\in P}w_e
$$
##### `connected(u,v,limit)`
判断只允许使用：
$$
w\le limit
$$
的边时，两点是否连通
##### `component_node(u,limit)`
返回只保留：
$$
w\le limit
$$
的边以后，原点 `u` 所在连通块对应的最高重构树节点
##### `component_size(u,limit)`
返回该连通块中原图节点数量
#### 常见用法
##### 两点最小瓶颈
```cpp
auto ans = tree.connect_threshold(u, v);

if (!ans) {
    // 原图中 u、v 不连通
} else {
    cout << *ans << '\n';
}
```
##### 阈值连通
```cpp
if (tree.connected(u, v, x)) {
    // 只使用权值 <= x 的边已经可以连通
}
```
##### 某点在阈值下所在连通块大小
```cpp
cout << tree.component_size(u, x) << '\n';
```
这类询问正是 Kruskal 重构树非常典型的用途：合并节点权值单调，因此可以在叶子到根的链上倍增寻找最后一个不超过阈值的祖先
#### 注意事项
* 当前版本按边权**从小到大**构造，对应条件：
$$
w\le x
$$
* 相同权值的边可能形成若干连续的同权内部节点，这是正常的，不影响 LCA 权值和阈值查询
* 原图允许负边、重边和自环；自环不会导致新的合并
* `value[u]` 对原图叶子 `u < n` 没有实际意义
* `connect_threshold(u,u)` 的“首次连通阈值”没有统一定义，因此接口要求 `u != v`；`connected(u,u,limit)` 始终为真
* 若只需要构造树而不需要在线查询，可以删除 `depth/root_of/up` 以及后续查询函数
* 若题目按“保留权值至少为 `x` 的边”询问，可以建立**最大 Kruskal 重构树**：边按权值从大到小处理，并相应反转阈值判断
#### 复杂度
设原图边数为 `m`
构造：
$$
O(m\log m)
$$
重构森林节点数：
$$
O(n)
$$
倍增预处理：
$$
O(n\log n)
$$
空间：
$$
O(n\log n)
$$
查询：

| 操作                    |       时间复杂度 |
| --------------------- | ----------: |
| `lca()`               | $O(\log n)$ |
| `connect_threshold()` | $O(\log n)$ |
| `connected()`         | $O(\log n)$ |
| `component_node()`    | $O(\log n)$ |
| `component_size()`    | $O(\log n)$ |
