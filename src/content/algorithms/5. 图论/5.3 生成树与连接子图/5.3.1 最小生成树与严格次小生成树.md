### 最小生成树与严格次小生成树
---
#### Kruskal、指定连通块数与严格次小生成树
```cpp
struct MST {
    struct Edge {
        int u, v, id;
        i64 w;
    };
    struct Result {
        i64 cost{};
        vector<int> used;
        int components{};
    };
    int n;
    vector<Edge> edges;
    explicit MST(int n_) : n(n_) {}
    int add_edge(int u, int v, i64 w) {
        int id = edges.size();
        edges.push_back({ u, v, id, w });
        return id;
    }
    // ===== Kruskal：基础 MST / 最小生成森林，以下代码是核心 =====
    Result kruskal(int components = 1) const {
        assert(1 <= components && components <= n);
        vector<int> order(edges.size());
        ranges::iota(order, 0);
        ranges::sort(order, {}, [&](int id) {
            return pair{ edges[id].w, id };
        });
        DSU dsu(n);
        Result res;
        res.used.reserve(n - components);
        for (int id : order) {
            if (dsu.count() == components) break;
            const auto &[u, v, _, w] = edges[id];
            if (!dsu.unite(u, v)) continue;
            res.cost += w;
            res.used.push_back(id);
        }
        res.components = dsu.count();
        return res;
    }
private:
    // ===== 严格次小 MST：若不需要，可以连同 strict_second() 删除 =====
    struct Top2 {
        i64 first{}, second{};
        bool has_first{}, has_second{};
        void add(i64 x) {
            if (!has_first) {
                first = x;
                has_first = true;
            } else if (x == first) {
                return;
            } else if (x > first) {
                second = first;
                has_second = true;
                first = x;
            } else if (!has_second || x > second) {
                second = x;
                has_second = true;
            }
        }
    };
    static Top2 merge_top2(Top2 a, const Top2 &b) {
        if (b.has_first) a.add(b.first);
        if (b.has_second) a.add(b.second);
        return a;
    }
public:
    optional<i64> strict_second() const {
        auto mst = kruskal();
        if (mst.components != 1 || n <= 1) return nullopt;
        vector<char> in_mst(edges.size());
        vector<vector<pair<int, i64>>> g(n);
        for (int id : mst.used) {
            in_mst[id] = true;
            const auto &[u, v, _, w] = edges[id];
            g[u].push_back({ v, w });
            g[v].push_back({ u, w });
        }
        // 倍增维护路径上的最大值与严格次大值。
        int lg = bit_width((unsigned)n);
        vector dep(n, -1);
        vector up(lg, vector<int>(n));
        vector mx(lg, vector<Top2>(n));
        vector<int> order{ 0 };
        dep[0] = 0;
        for (int i = 0; i < order.size(); ++ i) {
            int u = order[i];
            for (auto [v, w] : g[u]) if (dep[v] == -1) {
                dep[v] = dep[u] + 1;
                up[0][v] = u;
                mx[0][v].add(w);
                order.push_back(v);
            }
        }
        for (int k = 1; k < lg; ++ k) {
            for (int u : views::iota(0, n)) {
                int p = up[k - 1][u];
                up[k][u] = up[k - 1][p];
                mx[k][u] = merge_top2(mx[k - 1][u], mx[k - 1][p]);
            }
        }
        auto path_top2 = [&](int u, int v) {
            Top2 res;
            if (dep[u] < dep[v]) swap(u, v);
            int d = dep[u] - dep[v];
            for (int k = 0; k < lg; ++ k) if (d >> k & 1) {
                res = merge_top2(res, mx[k][u]);
                u = up[k][u];
            }
            if (u == v) return res;
            for (int k = lg - 1; k >= 0; -- k) {
                if (up[k][u] == up[k][v]) continue;
                res = merge_top2(res, mx[k][u]);
                res = merge_top2(res, mx[k][v]);
                u = up[k][u];
                v = up[k][v];
            }
            res = merge_top2(res, mx[0][u]);
            res = merge_top2(res, mx[0][v]);
            return res;
        };
        optional<i64> ans;
        for (const auto &e : edges) {
            if (in_mst[e.id] || e.u == e.v) continue;
            auto top = path_top2(e.u, e.v);
            if (!top.has_first) continue;
            if (e.w > top.first) {
                i64 cur = mst.cost + e.w - top.first;
                if (!ans || cur < *ans) ans = cur;
            } else if (e.w == top.first && top.has_second) {
                i64 cur = mst.cost + e.w - top.second;
                if (!ans || cur < *ans) ans = cur;
            }
        }
        return ans;
    }
};
```
#### 稠密 Prim
```cpp
optional<i64> prim_dense(const vector<vector<i64>> &a) {
    int n = a.size();
    if (!n) return i64{};
    constexpr i64 INF = numeric_limits<i64>::max() / 4;
    vector<i64> best(n, INF);
    vector<char> used(n);
    best[0] = 0;
    i64 ans = 0;
    for (int it = 0; it < n; ++ it) {
        int u{ -1 };
        for (int v : views::iota(0, n)) {
            if (!used[v] && (u == -1 || best[v] < best[u])) u = v;
        }
        if (u == -1 || best[u] == INF) return nullopt;
        used[u] = true;
        ans += best[u];
        for (int v : views::iota(0, n)) if (!used[v]) {
            best[v] = min(best[v], a[u][v]);
        }
    }
    return ans;
}
```
#### 接口说明
##### `add_edge(u, v, w)`
加入无向边，并返回稳定的逻辑边编号
```cpp
int id = mst.add_edge(u, v, w);
```
`strict_second()` 内部也依靠这一编号区分 MST 边与非树边
##### `kruskal(components = 1)`
返回：
```cpp
MST::Result { cost, used, components }
```
其中：
- `cost`：实际得到的最小生成森林总权值
- `used`：所选原边编号
- `components`：最终连通块数量
普通 MST：
```cpp
auto res = mst.kruskal();
if (res.components != 1) {
    // 原图不连通
}
```
要求恰好留下 `k` 个连通块：
```cpp
auto res = mst.kruskal(k);
if (res.components != k) {
    // 原图自身的连通性不足，无法合并到 k 块
}
```
当能够达到 `k` 个连通块时，所得森林在所有恰有 `k` 个连通块的生成森林中权值最小
##### `strict_second()`
返回严格满足
$$W'>W_{\mathrm{MST}}$$
的最小生成树权值
不存在时返回 `nullopt`
这里求的是**严格次小生成树**，因此如果存在多棵权值完全相同的 MST，它们仍然都属于“最小”，不会被视为严格次小
##### `prim_dense(a)`
邻接矩阵 Prim
- `a[u][v]`：边权
- 无边：`INF`
- 返回 MST 总权值
- 图不连通：`nullopt`
#### 严格次小生成树原理
先得到一棵 MST
对于每条非树边：
$$e=(u,v,w)$$
加入它后会形成唯一环
如果删除环中最大权边 $mx_1$，新树权值为：
$$W+w-mx_1$$
根据 MST 性质：
$$w\ge mx_1$$
若：
$$w>mx_1$$
直接得到严格更大的候选
但如果：
$$w=mx_1$$
删除最大边只会得到另一棵同权 MST
因此必须删除路径上**严格小于 $mx_1$ 的最大边**，也就是严格次大值 $mx_2$：
$$W+w-mx_2$$
这就是倍增中维护两个**不同权值**最大值的原因
#### 注意事项
- MST / Prim / Kruskal 都允许负边
- 自环会被 DSU 自动忽略
- 重边正常处理
- `strict_second()` 维护的是“最大和严格次大**权值**”，重复出现的同权最大边只算一个值
- 如果非树边与路径最大边等权，而路径不存在严格次大值，则这条边只能产生另一棵同权 MST，无法贡献严格次小答案
- MST 权值和以及替换后的答案必须能够放入 `i64`
- 稀疏图通常直接 Kruskal；邻接矩阵或接近完全图时使用 `prim_dense()`
- 堆优化 Prim 的复杂度为 $O(m\log n)$，但若已经维护边表，Kruskal 通常足够，不必再专门保存一份板子
#### 常见用法
##### 普通 MST
```cpp
MST mst(n);
for (int i = 0; i < m; ++ i) {
    int u, v;
    i64 w;
    cin >> u >> v >> w;
    mst.add_edge(u, v, w);
}
auto res = mst.kruskal();
if (res.components != 1) {
    cout << "Impossible\n";
} else {
    cout << res.cost << '\n';
}
```
##### 指定连通块数
```cpp
int k;
cin >> k;
auto res = mst.kruskal(k);
if (res.components == k) {
    cout << res.cost << '\n';
}
```
##### 恢复 MST 边
```cpp
auto res = mst.kruskal();
for (int id : res.used) {
    auto [u, v, _, w] = mst.edges[id];
}
```
##### 严格次小 MST
```cpp
auto ans = mst.strict_second();
if (!ans) {
    // 原图不连通，或不存在严格次小生成树
} else {
    cout << *ans << '\n';
}
```
#### 复杂度
Kruskal：
$$O(m\log m)$$
指定连通块数不会改变最坏复杂度
严格次小生成树：
- Kruskal：
$$O(m\log m)$$
- 倍增预处理：
$$O(n\log n)$$
- 枚举所有非树边：
$$O(m\log n)$$
总复杂度：
$$O(m\log m+(n+m)\log n)$$
空间复杂度：
$$O(m+n\log n)$$
稠密 Prim：
$$O(n^2)$$
空间由邻接矩阵主导：
$$O(n^2)$$
