### 球面与地理计算
---
**依赖：** 复用 **3.1 浮点版**的 `Point3`、`ld`、`PI`、`dot()`、`cross()`、`triple()`、`length()`、`unit()` 和向量四则运算；不依赖 3.2～3.5。`ld` 等类型别名由全局板子提供，**这里不再重复 `using`**。整数经纬度输入可隐式转为 `ld`，但本节的空间向量必须使用浮点版 `Point3`。
**统一约定：** `GeoPoint{lon, lat}` 按 **经度、纬度** 顺序，单位为**角度（°）**；经度向东为正，纬度向北为正，纬度范围 `[-90°,90°]`。球心在坐标原点，`z` 轴指向北极，`x` 轴指向 `(0°,0°)`，`y` 轴指向 `(90°E,0°)`。输入半径 `r` 应为非负，距离计算要求 `r > 0`；同一次计算必须统一长度单位。
#### 经纬度与三维坐标
---
```cpp
struct GeoPoint {
    ld lon{ 0 }, lat{ 0 }; // 经度、纬度，单位 °
};
ld geo_rad(ld deg) { return deg * PI / 180.L; }
ld geo_deg(ld rad) { return rad * 180.L / PI; }
ld geo_norm_lon(ld lon) { // [-180, 180)
    lon = remainderl(lon, 360.L);
    if (lon >= 180.L) lon -= 360.L;
    return lon;
}
// ===== 经纬度 -> 单位球面坐标：核心基础 =====
Point3 geo_to_unit(GeoPoint g) {
    if (g.lat == 90.L) return {0, 0, 1};
    if (g.lat == -90.L) return {0, 0, -1};
    ld lon = geo_rad(geo_norm_lon(g.lon)), lat = geo_rad(g.lat);
    ld c = cosl(lat);
    return {c * cosl(lon), c * sinl(lon), sinl(lat)};
}
Point3 geo_to_xyz(GeoPoint g, ld r) { return geo_to_unit(g) * r; }
// ===== 空间坐标 -> 经纬度 =====
// 任意非零向量可还原方向；零向量返回 nullopt
optional<GeoPoint> geo_from_xyz(Point3 p) {
    if (p == Point3{}) return nullopt;
    ld lon = geo_deg(atan2l(p.y, p.x));
    ld lat = geo_deg(atan2l(p.z, hypotl(p.x, p.y)));
    return GeoPoint{geo_norm_lon(lon), lat};
}
```
#### 圆心角、弦长与球面最短距离
---
```cpp
// ===== 球面距离：只依赖 3.1 与 geo_to_unit；可独立保留 =====
// 返回 [0, PI]；内部用 atan2，避免 acos 在近重合点处损失精度
ld geo_angle(GeoPoint a, GeoPoint b) {
    Point3 u = geo_to_unit(a), v = geo_to_unit(b);
    return atan2l(length(cross(u, v)), dot(u, v));
}
// 沿大圆较短弧的球面距离；对跖点有无数条等长最短路径
ld geo_arc_dist(GeoPoint a, GeoPoint b, ld r) {
    return r * geo_angle(a, b);
}
// 球内两点的空间直线距离，即弦长
ld geo_chord_dist(GeoPoint a, GeoPoint b, ld r) {
    return 2.L * r * sinl(geo_angle(a, b) / 2.L);
}
```
#### 航向角、沿大圆前进与路径插值
---
```cpp
// ===== 航向角与目的地：可整体删除 =====
// 初始航向角：从正北顺时针量起，[0, 360) 度
// 两点重合或对跖时无唯一航向，返回 nullopt
optional<ld> geo_bearing(GeoPoint a, GeoPoint b) {
    ld ang = geo_angle(a, b);
    if (ang <= 1E-14L || PI - ang <= 1E-14L) return nullopt;
    ld p = geo_rad(a.lat), q = geo_rad(b.lat);
    ld d = geo_rad(geo_norm_lon(b.lon - a.lon));
    ld x = sinl(d) * cosl(q);
    ld y = cosl(p) * sinl(q) - sinl(p) * cosl(q) * cosl(d);
    ld ans = geo_deg(atan2l(x, y));
    if (ans < 0) ans += 360.L;
    if (ans >= 360.L) ans -= 360.L;
    return ans;
}
// 从 a 沿初始航向 bearing（角度）行进 dist；r > 0
// 可令 dist > PI*r（对应继续沿原大圆行进）
GeoPoint geo_destination(GeoPoint a, ld bearing, ld dist, ld r) {
    ld lon = geo_rad(geo_norm_lon(a.lon));
    ld lat = geo_rad(a.lat), az = geo_rad(bearing);
    Point3 u = geo_to_unit(a);
    Point3 east{-sinl(lon), cosl(lon), 0};
    Point3 north{-sinl(lat)*cosl(lon), -sinl(lat)*sinl(lon), cosl(lat)};
    Point3 dir = north * cosl(az) + east * sinl(az);
    ld t = remainderl(dist / r, 2.L * PI);
    Point3 p = u * cosl(t) + dir * sinl(t);
    return *geo_from_xyz(p);
}
// ===== 大圆短弧插值：只依赖 3.1，可整体删除 =====
// 输入非零向量，0 <= t <= 1；输出单位球面向量
// 对跖点没有唯一短弧，返回 nullopt
optional<Point3> geo_slerp(Point3 a, Point3 b, ld t) {
    auto x = unit(a), y = unit(b);
    if (!x || !y) return nullopt;
    ld ang = atan2l(length(cross(*x, *y)), dot(*x, *y));
    if (PI - ang <= 1E-14L) return nullopt;
    if (ang <= 1E-8L) {
        Point3 p = *x * (1.L - t) + *y * t;
        return p / length(p);
    }
    ld s = sinl(ang);
    Point3 p = *x * (sinl((1.L - t) * ang) / s)
             + *y * (sinl(t * ang) / s);
    return p / length(p);
}
optional<GeoPoint> geo_interpolate(GeoPoint a, GeoPoint b, ld t) {
    auto p = geo_slerp(geo_to_unit(a), geo_to_unit(b), t);
    if (!p) return nullopt;
    return geo_from_xyz(*p);
}
```
#### 球面三角形与球面多边形面积
---
球面三角形的边均采用**较短大圆弧**。下面使用单位球向量计算有向球面角盈（立体角）：
$$
\Omega(a,b,c)=2\operatorname{atan2}\!\left(
\det(a,b,c),\,1+a\cdot b+b\cdot c+c\cdot a\right),
\quad S=R^2\Omega.
$$
`Ω` 的符号随顶点顺序反转。多边形采用以首点为顶点的有向球面三角形扇形求和，适用于**由短大圆弧连接、无自交且所有顶点位于同一个开半球内**的简单球面多边形（可以非凸）。多边形顶点按边界依次给出，不重复首点。
```cpp
// ===== 球面三角形：依赖 3.1 浮点 dot/cross/triple/length/unit =====
// 返回有向角盈（球面单位球上的有向面积），范围约为 (-2PI, 2PI)
// 零向量、对跖边或无法唯一确定的退化边界返回 nullopt
optional<ld> sphere_tri_excess(Point3 a, Point3 b, Point3 c) {
    auto pa = unit(a), pb = unit(b), pc = unit(c);
    if (!pa || !pb || !pc) return nullopt;
    a = *pa; b = *pb; c = *pc;
    auto bad = [&](Point3 u, Point3 v) {
        return dot(u, v) < 0 && length(cross(u, v)) <= 1E-14L;
    };
    if (bad(a, b) || bad(b, c) || bad(c, a)) return nullopt;
    ld y = triple(a, b, c);
    ld x = 1.L + dot(a, b) + dot(b, c) + dot(c, a);
    if (y == 0 && x <= 0) return nullopt;
    return 2.L * atan2l(y, x);
}
// 有向球面三角形面积，r > 0；通常取绝对值获得普通面积
optional<ld> sphere_tri_area(Point3 a, Point3 b, Point3 c, ld r) {
    auto e = sphere_tri_excess(a, b, c);
    if (!e) return nullopt;
    return *e * r * r;
}
// ===== 球面顶角：只依赖 3.1，可独立删除 =====
// 球面三角形 abc 在 b 点的内角（0 到 PI），单位为弧度
// 要求 ba、bc 均确定唯一的短大圆弧
optional<ld> sphere_vertex_angle(Point3 a, Point3 b, Point3 c) {
    auto pa = unit(a), pb = unit(b), pc = unit(c);
    if (!pa || !pb || !pc) return nullopt;
    a = *pa; b = *pb; c = *pc;
    Point3 u = cross(b, cross(a, b));
    Point3 v = cross(b, cross(c, b));
    ld lu = length(u), lv = length(v);
    if (lu <= 1E-14L || lv <= 1E-14L) return nullopt;
    return atan2l(length(cross(u, v)), dot(u, v));
}
// ===== 球面多边形面积：依赖 sphere_tri_excess，可独立删除 =====
// 简单、短大圆弧边界、位于某个开半球内；允许凹多边形
// 返回有向面积：反转顶点顺序会反号；无效输入返回 nullopt
optional<ld> sphere_poly_signed_area(const vector<Point3> &p, ld r) {
    int n = p.size();
    if (n < 3 || r <= 0) return nullopt;
    vector<Point3> q(n);
    for (int i = 0; i < n; ++ i) {
        auto t = unit(p[i]);
        if (!t) return nullopt;
        q[i] = *t;
    }
    for (int i = 0; i < n; ++ i) {
        Point3 a = q[i], b = q[(i + 1) % n];
        ld ang = atan2l(length(cross(a, b)), dot(a, b));
        if (ang <= 1E-14L || PI - ang <= 1E-14L) return nullopt;
    }
    ld sum = 0;
    for (int i = 1; i + 1 < n; ++ i) {
        auto t = sphere_tri_excess(q[0], q[i], q[i + 1]);
        if (!t) return nullopt;
        sum += *t;
    }
    return sum * r * r;
}
optional<ld> sphere_poly_area(const vector<Point3> &p, ld r) {
    auto s = sphere_poly_signed_area(p, r);
    if (!s) return nullopt;
    return fabsl(*s);
}
```
#### 两大圆的交点
---
```cpp
// ===== 两条大圆的交点：可整体删除 =====
// (a,b)、(c,d) 各确定一条大圆；非重合时返回一对对跖交点
// 输入为球心在原点的非零位置向量；不判断交点是否落在给定短弧内
optional<array<Point3, 2>> geo_great_circle_int(Point3 a, Point3 b,
                                                 Point3 c, Point3 d) {
    auto ua = unit(a), ub = unit(b), uc = unit(c), ud = unit(d);
    if (!ua || !ub || !uc || !ud) return nullopt;
    Point3 n = cross(*ua, *ub), m = cross(*uc, *ud);
    ld ln = length(n), lm = length(m);
    if (ln <= 1E-14L || lm <= 1E-14L) return nullopt;
    n /= ln; m /= lm;
    Point3 v = cross(n, m);
    ld len = length(v);
    if (len <= 1E-14L) return nullopt; // 相同的大圆，无唯一交点对
    v /= len;
    return array<Point3, 2>{v, -v};
}
```
#### 球冠、球带面积
---
```cpp
// ===== 球冠 / 球带的曲面面积：可整体删除 =====
// h 为沿球轴方向的高度，要求 r >= 0、0 <= h <= 2r
// 球冠面积不包含切割产生的平面底面
ld sphere_cap_area(ld r, ld h) { return 2.L * PI * r * h; }
// 球带：上下两个平行截面之间球面的面积，h 为两平面间的轴向高度
ld sphere_zone_area(ld r, ld h) { return 2.L * PI * r * h; }
```
#### 接口说明

| 接口                                               | 返回 / 几何意义                         | 复杂度    |
| ------------------------------------------------ | --------------------------------- | ------ |
| `GeoPoint{lon,lat}`                              | 经度、纬度（**角度**，经度在前）                | —      |
| `geo_rad(deg)` / `geo_deg(rad)`                  | 角度与弧度互换                           | $O(1)$ |
| `geo_norm_lon(lon)`                              | 经度归一化至 `[-180,180)`               | $O(1)$ |
| `geo_to_unit(g)`                                 | 经纬度转单位球上的 `Point3`                | $O(1)$ |
| `geo_to_xyz(g,r)`                                | 经纬度转半径 `r` 球面的空间坐标                | $O(1)$ |
| `geo_from_xyz(p)`                                | 非零空间向量转经纬度；零向量返回 `nullopt`        | $O(1)$ |
| `geo_angle(a,b)`                                 | 圆心角，弧度，范围 `[0,PI]`                | $O(1)$ |
| `geo_arc_dist(a,b,r)`                            | 球面最短距离 `r*圆心角`                    | $O(1)$ |
| `geo_chord_dist(a,b,r)`                          | 球内直线弦长                            | $O(1)$ |
| `geo_bearing(a,b)`                               | 从正北顺时针的初始航向角，单位 °；非唯一返回 `nullopt` | $O(1)$ |
| `geo_destination(a,bearing,dist,r)`              | 从位置出发沿大圆走 `dist` 后的经纬度            | $O(1)$ |
| `geo_slerp(a,b,t)` / `geo_interpolate(a,b,t)`    | 两球面点之间短大圆弧的球面线性插值                 | $O(1)$ |
| `sphere_tri_excess(a,b,c)`                       | 三点构成的有向球面角盈，单位 `sr`（数值等于单位球面积）    | $O(1)$ |
| `sphere_tri_area(a,b,c,r)`                       | 有向球面三角形面积                         | $O(1)$ |
| `sphere_vertex_angle(a,b,c)`                     | 球面三角形在 `b` 处的内角，单位弧度              | $O(1)$ |
| `sphere_poly_signed_area(p,r)`                   | 按顶点顺序计算有向球面多边形面积                  | $O(n)$ |
| `sphere_poly_area(p,r)`                          | 球面多边形面积非负值                        | $O(n)$ |
| `geo_great_circle_int(a,b,c,d)`                  | 两不同大圆的一对对跖交点；无唯一交点对返回 `nullopt`   | $O(1)$ |
| `sphere_cap_area(r,h)` / `sphere_zone_area(r,h)` | 球冠曲面面积 / 球带曲面面积                   | $O(1)$ |
#### 注意事项
- **球面模型**：全部公式基于完美球体，而不是 WGS84 等椭球地球模型。纬度按地心球面纬度理解，不能直接保证真实地理测量精度。实际地球半径须按题目给定（如公里与米不可混用）。
- **角度单位**：`GeoPoint`、`geo_bearing`、`geo_destination` 的输入/输出地理角采用 **度**；`geo_angle`、`sphere_vertex_angle`、`sphere_tri_excess` 采用 **弧度 / 球面角盈**，勿混用。
- **经度跨界**：`179°E` 与 `179°W` 实际仅隔经差 `2°`，不要直接使用经度数值差计算路程。经度 `180°` 的返回规范化为 `-180°`。
- **极点**：在北极、南极，经度不是唯一的。`geo_from_xyz` 在恰好 `x=y=0` 时约定经度为 `0°`；`geo_destination` 从极点出发时的航向依据输入经度确定的局部方位系。
- **重合与对跖**：大圆最短距离对所有两点都存在；但重合点没有唯一初始航向，对跖点有无穷条最短大圆弧，`geo_bearing()`、`geo_interpolate()` 遇到这些情形返回 `nullopt`。`geo_destination()` 不需要终点，因此可以任意指定初始航向。
- **三角形与多边形**：采用短大圆弧作为边；对跖边、零向量无效。球面多边形实现只保证**简单、顶点都落在某个开半球、各边为短大圆弧**的情况，允许凹多边形；不能直接推广到绕球超过半球、边界自交、有洞、跨越对跖点、含不唯一大圆边的多边形。代码不额外检查“简单且同半球”这一全局前置条件。
- **有向面积**：三角形和多边形反转顶点顺序应反号，`sphere_poly_area()` 取绝对值；不按“经度增大就一定为正”来判断朝向。面积单位是 `r` 的长度单位的平方。
- **球面角盈**：凸球面多边形的面积满足 `S=R²(Σ内角-(n-2)π)`；代码采用有向三角形 `atan2` 求和，避免显式相减造成小面积时的消去误差。不要用普通平面鞋带公式计算球面面积。
- **浮点尺度**：`atan2(cross,dot)` 通常比直接 `acos(dot)` 更稳定，但浮点精度并非无限；近重合、近对跖以及接近退化三角形时仍可能丢失相对精度。本节 `1E-14L` 用于**角度退化**，与 3.1 的坐标差 `EPS` 判等概念不同。
- **球冠高度**：面积公式要求 `0<=h<=2r`；其中 `h=2r` 表示整个球面。
#### 常见用法
##### 经纬度距离：北京到另一点（球体近似）
```cpp
GeoPoint a{116.4L, 39.9L};
GeoPoint b{121.5L, 31.2L};
ld r = 6371.L;                      // 单位：km，示例球半径
ld ang = geo_angle(a, b);          // 弧度
ld arc = geo_arc_dist(a, b, r);    // 大圆最短弧长，km
ld chord = geo_chord_dist(a, b, r);// 穿过球内部的直线距离，km
```
##### 坐标转换与航向
```cpp
Point3 p = geo_to_unit(GeoPoint{90, 0}); // 约等于 (0,1,0)
auto g = geo_from_xyz(p);               // (90°,0°)
auto bearing = geo_bearing({0,0}, {90,0}); // 约为 90°（正东）
GeoPoint q = geo_destination({0,0}, 90, 1000, 6371); // 从赤道出发向东 1000 km
```

```cpp
vector<GeoPoint> g{{0,45}, {90,45}, {180,45}, {270,45}};
vector<Point3> p;
for (auto x : g) p.push_back(geo_to_unit(x));
```

#### 复杂度
除 `sphere_poly_signed_area()` / `sphere_poly_area()` 需要遍历 $n$ 个顶点、使用 $O(n)$ 额外空间外，其余所有操作均为 $O(1)$ 时间、$O(1)$ 额外空间
