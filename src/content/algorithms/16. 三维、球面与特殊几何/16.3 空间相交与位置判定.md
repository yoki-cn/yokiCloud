### 空间相交与位置判定
---
**依赖：** 按需要选择 3.1、3.2 的**同一种数值版本**。整数版使用 `Point3`、`Line3`、`Segment3`、`Plane3`、`cross()`、`orient()`、`sgn()`、`on_line()`、`on_seg()` 等；浮点版额外使用 `length()`、`distance()`、`signed_dis_plane()`、`dis_line()`、`seg_dis()` 等。
本节区分几何对象的**位置关系**与**交点坐标**：整数版提供精确的相交/包含判定，不做可能产生有理数的坐标计算；浮点版提供直线与直线、直线与平面、两平面的唯一交点/交线。线段与三角形、两个三角形默认**包含边界接触和共面重叠**
#### 整数版：空间线、面的位置关系
---
```cpp
// ===== 两条空间直线的关系 =====
enum class LineLineRel {
    INVALID,     // 无效：至少一条直线退化
    INTERSECT,   // 相交：唯一交点
    PARALLEL,    // 平行：不重合，无交点
    COINCIDENT,  // 重合：无穷多个交点
    SKEW         // 异面：不平行、不相交
};
// ===== 空间直线与平面的关系 =====
enum class LinePlaneRel {
    INVALID,     // 无效：直线或平面退化
    INTERSECT,   // 相交：唯一交点
    PARALLEL,    // 平行：不包含，无交点
    CONTAINED    // 包含：整条直线位于平面内
};
// ===== 两个空间平面的关系 =====
enum class PlanePlaneRel {
    INTERSECT,   // 相交：交集是一条直线
    PARALLEL,    // 平行：不重合，无交线
    COINCIDENT   // 重合：两个平面完全相同
};
// ===== 直线、平面分类 =====
LineLineRel line_rel(Line3 a, Line3 b) {
    if (a.dir == Point3{} || b.dir == Point3{}) return LineLineRel::INVALID;
    if (parallel(a.dir, b.dir)) {
        return on_line(b.p, a) ? LineLineRel::COINCIDENT : LineLineRel::PARALLEL;
    }
    auto n = cross(a.dir, b.dir);
    i128 v = n[0] * (i128(b.p.x) - a.p.x)
           + n[1] * (i128(b.p.y) - a.p.y)
           + n[2] * (i128(b.p.z) - a.p.z);
    return v == 0 ? LineLineRel::INTERSECT : LineLineRel::SKEW;
}
LinePlaneRel line_plane_rel(Line3 l, Plane3 pl) {
    if (l.dir == Point3{}) return LinePlaneRel::INVALID;
    auto n = cross(pl.a, pl.b, pl.c);
    if (n[0] == 0 && n[1] == 0 && n[2] == 0) return LinePlaneRel::INVALID;
    i128 den = n[0] * l.dir.x + n[1] * l.dir.y + n[2] * l.dir.z;
    if (den != 0) return LinePlaneRel::INTERSECT;
    return on_plane(l.p, pl) ? LinePlaneRel::CONTAINED : LinePlaneRel::PARALLEL;
}
PlanePlaneRel plane_rel(Plane3 a, Plane3 b) {
    // 两个平面均要求三个定义点不共线
    i128 u = orient(a.a, a.b, a.c, b.a);
    i128 v = orient(a.a, a.b, a.c, b.b);
    i128 w = orient(a.a, a.b, a.c, b.c);
    if (u != v || u != w) return PlanePlaneRel::INTERSECT;
    return u == 0 ? PlanePlaneRel::COINCIDENT : PlanePlaneRel::PARALLEL;
}
```
#### 整数版：线段和三角形相交
---
```cpp
struct Triangle3 {
    Point3 a, b, c;
    Triangle3() = default;
    Triangle3(Point3 a_, Point3 b_, Point3 c_) : a(a_), b(b_), c(c_) {}
};
// ===== 二维投影辅助：供空间线段/三角形相交使用 =====
// k=0/1/2 分别丢弃 x/y/z；均保持循环坐标次序
using P2 = pair<i64, i64>;
P2 proj2(Point3 p, int k) {
    if (k == 0) return {p.y, p.z};
    if (k == 1) return {p.z, p.x};
    return {p.x, p.y};
}
i128 cross2(P2 a, P2 b, P2 c) {
    return (i128(b.first) - a.first) * (i128(c.second) - a.second)
         - (i128(b.second) - a.second) * (i128(c.first) - a.first);
}
bool on_seg2(P2 p, P2 a, P2 b) {
    return cross2(a, b, p) == 0
        && min(a.first, b.first) <= p.first && p.first <= max(a.first, b.first)
        && min(a.second, b.second) <= p.second && p.second <= max(a.second, b.second);
}
bool seg_int2(P2 a, P2 b, P2 c, P2 d) {
    int x = sgn(cross2(a, b, c)), y = sgn(cross2(a, b, d));
    int z = sgn(cross2(c, d, a)), w = sgn(cross2(c, d, b));
    if (x == 0 && on_seg2(c, a, b)) return true;
    if (y == 0 && on_seg2(d, a, b)) return true;
    if (z == 0 && on_seg2(a, c, d)) return true;
    if (w == 0 && on_seg2(b, c, d)) return true;
    return x * y < 0 && z * w < 0;
}
int drop_axis(array<i128, 3> n) {
    auto ab = [](i128 x) -> __uint128_t {
        return x < 0 ? __uint128_t(-(x + 1)) + 1 : __uint128_t(x);
    };
    int k = 0;
    for (int i = 1; i < 3; ++ i) if (ab(n[i]) > ab(n[k])) k = i;
    return k;
}
// ===== 两空间闭线段相交：允许退化为点 =====
// 不能仅做二维投影相交：必须先排除异面直线
bool seg_int3(Segment3 a, Segment3 b) {
    if (a.a == a.b) return on_seg(a.a, b);
    if (b.a == b.b) return on_seg(b.a, a);
    if (orient(a.a, a.b, b.a, b.b) != 0) return false;
    auto n = cross(a.b - a.a, b.b - b.a);
    if (n[0] == 0 && n[1] == 0 && n[2] == 0) {
        if (!collinear(a.a, a.b, b.a)) return false;
        return on_seg(a.a, b) || on_seg(a.b, b)
            || on_seg(b.a, a) || on_seg(b.b, a);
    }
    int k = drop_axis(n);
    return seg_int2(proj2(a.a, k), proj2(a.b, k), proj2(b.a, k), proj2(b.b, k));
}
// ===== 点与空间三角形：允许三角形退化为三条边的并 =====
bool on_tri(Point3 p, Triangle3 t, bool stct = false) {
    auto n = cross(t.a, t.b, t.c);
    if (n[0] == 0 && n[1] == 0 && n[2] == 0) {
        if (stct) return false;
        return on_seg(p, {t.a, t.b}) || on_seg(p, {t.b, t.c})
            || on_seg(p, {t.c, t.a});
    }
    if (orient(t.a, t.b, t.c, p) != 0) return false;
    int k = drop_axis(n);
    P2 a = proj2(t.a, k), b = proj2(t.b, k);
    P2 c = proj2(t.c, k), q = proj2(p, k);
    int s = sgn(cross2(a, b, c));
    int x = s * sgn(cross2(a, b, q));
    int y = s * sgn(cross2(b, c, q));
    int z = s * sgn(cross2(c, a, q));
    return stct ? (x > 0 && y > 0 && z > 0)
                : (x >= 0 && y >= 0 && z >= 0);
}
// ===== 线段与空间三角形：包含共面、边界与退化情形 =====
bool seg_tri_int(Segment3 s, Triangle3 t) {
    auto n = cross(t.a, t.b, t.c);
    if (n[0] == 0 && n[1] == 0 && n[2] == 0) {
        return seg_int3(s, {t.a, t.b}) || seg_int3(s, {t.b, t.c})
            || seg_int3(s, {t.c, t.a});
    }
    int x = sgn(orient(t.a, t.b, t.c, s.a));
    int y = sgn(orient(t.a, t.b, t.c, s.b));
    if (x == 0 && on_tri(s.a, t)) return true;
    if (y == 0 && on_tri(s.b, t)) return true;
    if (x == 0 && y == 0) {
        return seg_int3(s, {t.a, t.b}) || seg_int3(s, {t.b, t.c})
            || seg_int3(s, {t.c, t.a});
    }
    if (x == 0 || y == 0 || x == y) return false;
    int u = sgn(orient(s.a, s.b, t.a, t.b));
    int v = sgn(orient(s.a, s.b, t.b, t.c));
    int w = sgn(orient(s.a, s.b, t.c, t.a));
    return (u >= 0 && v >= 0 && w >= 0)
        || (u <= 0 && v <= 0 && w <= 0);
}
// ===== 两三角形相交 =====
bool tri_tri_int(Triangle3 a, Triangle3 b) {
    for (auto s : {Segment3{a.a, a.b}, {a.b, a.c}, {a.c, a.a}}) {
        if (seg_tri_int(s, b)) return true;
    }
    for (auto s : {Segment3{b.a, b.b}, {b.b, b.c}, {b.c, b.a}}) {
        if (seg_tri_int(s, a)) return true;
    }
    return false;
}
```
#### 浮点版：直线与平面交点、两平面交线
---
依赖 3.1、3.2 浮点版的 `Line3`、`Plane3`、`Point3` 与基础函数；无须复制整数版的二维投影辅助代码
```cpp
// ===== 空间位置关系 =====
enum class LineLineRel {
    INVALID,     // 无效：至少一条直线退化
    INTERSECT,   // 相交：唯一交点
    PARALLEL,    // 平行：不重合，无交点
    COINCIDENT,  // 重合：无穷多个公共点
    SKEW         // 异面：不平行、不相交
};
enum class LinePlaneRel {
    INVALID,     // 无效：直线或平面退化
    INTERSECT,   // 相交：唯一交点
    PARALLEL,    // 平行：不包含，无交点
    CONTAINED    // 包含：整条直线位于平面内
};
enum class PlanePlaneRel {
    INVALID,     // 无效：至少一个平面退化
    INTERSECT,   // 相交：交集是一条直线
    PARALLEL,    // 平行：不重合，无交线
    COINCIDENT   // 重合：两个平面完全相同
};
// ===== 无限直线、平面位置分类：可独立删除 =====
LineLineRel line_rel(Line3 a, Line3 b) {
    ld la = length(a.dir), lb = length(b.dir);
    if (la == 0 || lb == 0) return LineLineRel::INVALID;
    Point3 u = a.dir / la, v = b.dir / lb;
    Point3 n = cross(u, v);
    if (length(n) <= EPS) {
        return dis_line(b.p, a) <= EPS
            ? LineLineRel::COINCIDENT : LineLineRel::PARALLEL;
    }
    ld d = fabsl(dot(b.p - a.p, n)) / length(n);
    return d <= EPS ? LineLineRel::INTERSECT : LineLineRel::SKEW;
}
optional<Point3> line_int(Line3 a, Line3 b) {
    if (line_rel(a, b) != LineLineRel::INTERSECT) return nullopt;
    Point3 w = b.p - a.p, n = cross(a.dir, b.dir);
    ld t = dot(cross(w, b.dir), n) / norm2(n);
    return a.p + a.dir * t;
}

struct LinePlaneHit {
    LinePlaneRel rel;
    optional<Point3> point; // 仅 INTERSECT 时存在
};
LinePlaneHit line_plane_int(Line3 l, Plane3 pl) {
    ld dl = length(l.dir), dn = length(pl.n);
    if (dl == 0 || dn == 0) return {LinePlaneRel::INVALID, nullopt};
    ld c = dot(l.dir / dl, pl.n / dn);
    if (fabsl(c) <= EPS) {
        bool inside = fabsl(signed_dis_plane(l.p, pl)) <= EPS;
        return {inside ? LinePlaneRel::CONTAINED : LinePlaneRel::PARALLEL, nullopt};
    }
    ld t = dot(pl.p - l.p, pl.n) / dot(l.dir, pl.n);
    return {LinePlaneRel::INTERSECT, l.p + l.dir * t};
}
struct PlanePlaneHit {
    PlanePlaneRel rel;
    optional<Line3> line; // 仅 INTERSECT 时存在
};
PlanePlaneHit plane_plane_int(Plane3 a, Plane3 b) {
    ld la = length(a.n), lb = length(b.n);
    if (la == 0 || lb == 0) return {PlanePlaneRel::INVALID, nullopt};
    Point3 n1 = a.n / la, n2 = b.n / lb;
    Point3 dir = cross(n1, n2);
    ld den = norm2(dir);
    if (length(dir) <= EPS) {
        bool inside = fabsl(signed_dis_plane(b.p, a)) <= EPS;
        return {inside ? PlanePlaneRel::COINCIDENT : PlanePlaneRel::PARALLEL, nullopt};
    }
    // 以 a.p 为局部原点，避免在世界坐标下直接构造大常数
    ld d = dot(b.p - a.p, n2);
    Point3 p = a.p + cross(dir, n1) * (d / den);
    return {PlanePlaneRel::INTERSECT, Line3{p, dir}};
}
```
#### 浮点版：三角形和线段相交
---
```cpp
struct Triangle3 {
    Point3 a, b, c;
    Triangle3() = default;
    Triangle3(Point3 a_, Point3 b_, Point3 c_) : a(a_), b(b_), c(c_) {}
};
// ===== 浮点空间线段相交：依赖 3.2 的 seg_dis() =====
bool seg_int3(Segment3 a, Segment3 b) { return seg_dis(a, b) <= EPS; }
// ===== 点与空间三角形：允许退化成线段或点 =====
bool on_tri(Point3 p, Triangle3 t, bool stct = false) {
    Point3 n = cross(t.b - t.a, t.c - t.a);
    ld ln = length(n);
    ld scale = max({distance(t.a, t.b), distance(t.b, t.c), distance(t.c, t.a)});
    if (scale == 0 || ln <= EPS * scale) {
        if (stct) return false;
        return on_seg(p, {t.a, t.b}) || on_seg(p, {t.b, t.c})
            || on_seg(p, {t.c, t.a});
    }
    Point3 u = n / ln;
    if (fabsl(dot(p - t.a, u)) > EPS) return false;
    auto edge_side = [&](Point3 a, Point3 b) -> ld {
        return dot(u, cross(b - a, p - a)) / distance(a, b);
    };
    ld x = edge_side(t.a, t.b), y = edge_side(t.b, t.c), z = edge_side(t.c, t.a);
    return stct ? (x > EPS && y > EPS && z > EPS)
                : (x >= -EPS && y >= -EPS && z >= -EPS);
}
// ===== 线段与空间三角形：兼容共面、边界及退化 =====
bool seg_tri_int(Segment3 s, Triangle3 t) {
    Point3 n = cross(t.b - t.a, t.c - t.a);
    ld ln = length(n);
    ld scale = max({distance(t.a, t.b), distance(t.b, t.c), distance(t.c, t.a)});
    if (scale == 0 || ln <= EPS * scale) {
        return seg_int3(s, {t.a, t.b}) || seg_int3(s, {t.b, t.c})
            || seg_int3(s, {t.c, t.a});
    }
    Point3 u = n / ln;
    ld d0 = dot(s.a - t.a, u), d1 = dot(s.b - t.a, u);
    int x = sgn(d0), y = sgn(d1);
    if (x == 0 && on_tri(s.a, t)) return true;
    if (y == 0 && on_tri(s.b, t)) return true;
    if (x == 0 && y == 0) {
        return seg_int3(s, {t.a, t.b}) || seg_int3(s, {t.b, t.c})
            || seg_int3(s, {t.c, t.a});
    }
    if (x == 0 || y == 0 || x == y) return false;
    Point3 p = s.a + (s.b - s.a) * (d0 / (d0 - d1));
    return on_tri(p, t);
}
// ===== 两三角形：内部交叉、共面包含、边界相切均为 true =====
bool tri_tri_int(Triangle3 a, Triangle3 b) {
    for (auto s : {Segment3{a.a, a.b}, {a.b, a.c}, {a.c, a.a}}) {
        if (seg_tri_int(s, b)) return true;
    }
    for (auto s : {Segment3{b.a, b.b}, {b.b, b.c}, {b.c, b.a}}) {
        if (seg_tri_int(s, a)) return true;
    }
    return false;
}
```
#### 公共接口

| 接口 | 含义 | 时间复杂度 |
|---|---|---|
| `Triangle3(a,b,c)` | 三角形三个顶点，允许退化为线段或点 | $O(1)$ |
| `line_rel(a,b)` | 无限直线：唯一交点 / 平行 / 重合 / 异面 / 非法 | $O(1)$ |
| `seg_int3(a,b)` | 两空间闭线段是否相交（含边界和退化） | $O(1)$ |
| `on_tri(p,t,stct)` | 点是否在三角形上；`stct=true` 仅非退化三角形相对内部 | $O(1)$ |
| `seg_tri_int(s,t)` | 闭线段与三角形相交（共面、包含、相切均计入） | $O(1)$ |
| `tri_tri_int(a,b)` | 两空间三角形相交（允许退化） | $O(1)$ |
#### 整数版额外接口

| 接口                     | 含义                       | 时间复杂度  |
| ---------------------- | ------------------------ | ------ |
| `line_plane_rel(l,pl)` | 唯一交点 / 平行 / 直线包含于平面 / 非法 | $O(1)$ |
| `plane_rel(a,b)`       | 两非退化平面相交 / 平行 / 重合       | $O(1)$ |
#### 浮点版额外接口

| 接口                     | 含义                                             | 时间复杂度  |
| ---------------------- | ---------------------------------------------- | ------ |
| `line_int(a,b)`        | 两空间直线唯一交点，其他情况 `nullopt`                       | $O(1)$ |
| `line_plane_int(l,pl)` | 返回 `LinePlaneHit{rel,point}`；仅唯一交点时 `point` 有值 | $O(1)$ |
| `plane_plane_int(a,b)` | 返回 `PlanePlaneHit{rel,line}`；仅相交成直线时 `line` 有值 | $O(1)$ |
**结果枚举：** `LineLineRel::{INVALID,INTERSECT,PARALLEL,COINCIDENT,SKEW}`；`LinePlaneRel::{INVALID,INTERSECT,PARALLEL,CONTAINED}`；浮点 `PlanePlaneRel` 另含 `INVALID`。`INTERSECT` 对直线/平面表示唯一点、对两平面表示唯一交线；不要仅通过 `optional` 空值区分平行与重合。
#### 注意事项
- **输入约定：** `Line3(p,dir)` 中 `dir` 是方向，不是第二个端点；整数 `Plane3(a,b,c)` 使用三点，浮点 `Plane3(p,n)` 使用一点与法向量，接口不可混用。
- 整数版 `line_rel()`、`line_plane_rel()` 对零方向返回 `INVALID`；`plane_rel()` 及平面分类要求平面非退化。整数版不返回除法得到的有理数交点。
- **共面分支不可省略：** `seg_tri_int()` 若线段两端都在三角形平面上，需要检查三角形内端点与三条边相交；只判断两端异侧会漏掉共面重叠。
- **空间线段必须先验共面：** 二维投影交叉不保证三维线段相交，异面时投影可能重合。
- **包含边界：** `seg_int3`、`seg_tri_int`、`tri_tri_int` 默认允许共享顶点、共享边、切触及部分/完全重合；如题目要求真正穿过内部，应另外构造严格谓词，不能简单地把 `<=` 全部改成 `<`。
- **退化三角形：** 本板子按三条闭边的并处理，允许退化为线段或点；`on_tri(...,true)` 对退化三角形返回 `false`。这不代表退化对象具有二维内部。
- **整数溢出：** `orient`、空间叉积、`line_rel` 的扩展混合积等每一步和累加都必须落在 `i128` 内；`seg_int3` 计算两线段方向 `b-a` 时，差值还需落在 `i64` 中。若输入到达 `i64` 极限，不能只靠 `i128` 保证所有公式安全。
- 浮点版以 `EPS` 控制**距离误差**与**无量纲角度误差**，不等价于精确几何谓词。近共面、近相切、极大坐标或极细长三角形可能需要更高精度/自适应谓词；分类容差可能将接近相交的对象视为相交。
- 浮点 `line_int`、`line_plane_int`、`plane_plane_int` 对近乎平行的输入可能数值敏感。特别是两平面夹角很小，交线坐标绝对误差可能显著放大。
- 浮点 `seg_int3` 直接复用 3.2 的 `seg_dis()`；如只复制三角形部分，必须保留 3.2 相应的最近点及距离函数；它和后续相交谓词共享同一 `EPS`。
- `on_tri()` 浮点版使用投影到所在平面后的边侧距离做包容判定，`stct=true` 要求三个边侧距离都严格大于 `EPS`；点至三角形所在平面的距离仍允许 `EPS` 误差。
#### 常见用法
##### 判断空间直线是否异面
```cpp
Line3 a = Line3::through(p1, p2);
Line3 b = Line3::through(q1, q2);
if (line_rel(a, b) == LineLineRel::SKEW) {
    // 两条无限直线异面，没有交点
}
```
##### 线段与空间三角形
```cpp
Triangle3 tri{a, b, c};
Segment3 seg{p, q};
```
