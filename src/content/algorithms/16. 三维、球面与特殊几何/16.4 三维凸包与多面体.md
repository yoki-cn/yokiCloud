### 三维凸包与多面体
---
**依赖 3.1：** 选择同一种数值版本的 `Point3`、`cross()`、`orient()`、`collinear()` 与 `sgn()`；浮点版还需 `length()`、`EPS` 和 `ld`。
本板子是 **增量法三维凸包**：输出统一朝外的三角面，并提供三维体积、表面积、体积重心、点包含与重心到表面最短距离；全点共面的情形退化为**单侧平面凸多边形**，分别提供平面面积及面积重心。输入允许重复点、共线和共面，但浮点近退化输入受 `EPS` 与数值精度限制
```cpp
// ===== 三维凸包增量法：依赖 3.1 同一种数值版本的 Point3 / orient / cross / collinear =====
// 整数版还需 i128、sgn(i128)；浮点版还需 ld、EPS、sgn(ld)、length()。
// f 的三角面法向朝外；顶点下标对应 h.p（已去重、重排），不是原始输入下标。
template<class P = Point3>
struct Hull3 {
    using T = decltype(P{}.x);
    using Wide = conditional_t<is_integral_v<T>, i128, ld>;
    vector<P> p;
    vector<array<int, 3>> f;
    vector<int> poly; // dim == 1 时为线段端点，dim == 2 时为共面凸多边形环
    int dim{ -1 };    // -1 空集；0 点；1 线段；2 共面多边形；3 三维实体
    Hull3() = default;
    explicit Hull3(vector<P> ps) { build(move(ps)); }
    // ===== 几何谓词：整数精确；浮点比较点到平面的有符号距离 =====
    static int side(P a, P b, P c, P d) {
        if constexpr (is_integral_v<T>) {
            return sgn(orient(a, b, c, d));
        } else {
            P n = cross(a, b, c);
            ld s = length(n);
            return s == 0 ? 0 : sgn(orient(a, b, c, d) / s);
        }
    }
    static ld coord(P a, int k) {
        return k == 0 ? ld(a.x) : k == 1 ? ld(a.y) : ld(a.z);
    }
    static pair<T, T> project(P a, int k) {
        if (k == 0) return {a.y, a.z};
        if (k == 1) return {a.z, a.x};
        return {a.x, a.y};
    }
    static int turn2(P a, P b, P c, int k) {
        auto [ax, ay] = project(a, k);
        auto [bx, by] = project(b, k);
        auto [cx, cy] = project(c, k);
        if constexpr (is_integral_v<T>) {
            i128 x = i128(bx) - ax, y = i128(by) - ay;
            return sgn(x * (i128(cy) - ay) - y * (i128(cx) - ax));
        } else {
            ld x = bx - ax, y = by - ay;
            ld len = hypotl(x, y);
            return len == 0 ? 0 : sgn((x * (cy - ay) - y * (cx - ax)) / len);
        }
    }
    static int drop_axis(P a, P b, P c) {
        if constexpr (is_integral_v<T>) {
            auto n = cross(a, b, c);
            auto ab = [](i128 x) -> __uint128_t {
                return x < 0 ? __uint128_t(-(x + 1)) + 1 : __uint128_t(x);
            };
            int k = 0;
            for (int i = 1; i < 3; ++ i) if (ab(n[i]) > ab(n[k])) k = i;
            return k;
        } else {
            P n = cross(a, b, c);
            ld v[] = {fabsl(n.x), fabsl(n.y), fabsl(n.z)};
            return int(max_element(v, v + 3) - v);
        }
    }
    // ===== 共面凸包：只在 dim == 2 时使用；可与三维算法独立裁剪 =====
    void build_planar(int k) {
        vector<int> id(p.size());
        iota(id.begin(), id.end(), 0);
        sort(id.begin(), id.end(), [&](int a, int b) {
            return project(p[a], k) < project(p[b], k);
        });
        for (int i : id) {
            while (poly.size() >= 2 &&
                   turn2(p[poly[poly.size()-2]], p[poly.back()], p[i], k) <= 0)
                poly.pop_back();
            poly.push_back(i);
        }
        int base = int(poly.size()) + 1;
        for (int j = int(id.size()) - 2; j >= 0; -- j) {
            int i = id[j];
            while (int(poly.size()) >= base &&
                   turn2(p[poly[poly.size()-2]], p[poly.back()], p[i], k) <= 0)
                poly.pop_back();
            poly.push_back(i);
        }
        poly.pop_back();
    }
    // ===== 核心：可独立构造三维凸包，O(n^2) 期望哈希复杂度 =====
    void build(vector<P> ps) {
        p = move(ps);
        sort(p.begin(), p.end());
        p.erase(unique(p.begin(), p.end()), p.end());
        f.clear(); poly.clear(); dim = -1;
        int n = int(p.size());
        if (n == 0) return;
        dim = 0;
        if (n == 1) return;
        dim = 1;
        int j = 2;
        while (j < n && collinear(p[0], p[1], p[j])) ++ j;
        if (j == n) { poly = {0, n - 1}; return; }
        swap(p[2], p[j]);
        dim = 2;
        int k = 3;
        while (k < n && side(p[0], p[1], p[2], p[k]) == 0) ++ k;
        if (k == n) { build_planar(drop_axis(p[0], p[1], p[2])); return; }
        swap(p[3], p[k]);
        dim = 3;

        f = {{{0, 1, 2}, {0, 3, 1}, {0, 2, 3}, {1, 3, 2}}};
        for (int i = 0; i < 4; ++ i) {
            auto &t = f[i];
            int other = 0;
            while (other == t[0] || other == t[1] || other == t[2]) ++ other;
            if (side(p[t[0]], p[t[1]], p[t[2]], p[other]) > 0) swap(t[1], t[2]);
        }
        for (int i = 4; i < n; ++ i) {
            vector<char> vis(f.size());
            int visible = 0;
            for (int j = 0; j < int(f.size()); ++ j) {
                auto [a, b, c] = f[j];
                vis[j] = side(p[a], p[b], p[c], p[i]) > 0;
                visible += vis[j];
            }
            if (!visible) continue;
            unordered_map<uint64_t, pair<int, int>> border;
            border.reserve(size_t(visible) * 6 + 1);
            auto add = [&](int a, int b) {
                uint64_t rev = (uint64_t(uint32_t(b)) << 32) | uint32_t(a);
                if (auto it = border.find(rev); it != border.end()) border.erase(it);
                else border.emplace((uint64_t(uint32_t(a)) << 32) | uint32_t(b), pair{a, b});
            };
            vector<array<int, 3>> nf;
            nf.reserve(f.size() + visible * 2);
            for (int j = 0; j < int(f.size()); ++ j) {
                auto [a, b, c] = f[j];
                if (vis[j]) { add(a, b); add(b, c); add(c, a); }
                else nf.push_back(f[j]);
            }
            for (auto [key, edge] : border) {
                auto [a, b] = edge;
                nf.push_back({a, b, i});
            }
            f = move(nf);
        }
    }
    // ===== 凸包判定：只适用 dim == 3 =====
    // 包含边界；strict=true 表示严格内部。dim<3 时均返回 false。
    bool contains(P q, bool strict = false) const {
        if (dim != 3) return false;
        for (auto [a, b, c] : f) {
            int t = side(p[a], p[b], p[c], q);
            if (strict ? t >= 0 : t > 0) return false;
        }
        return true;
    }
    // ===== 三维体积、表面积与体积重心：若只需面片，可整段删除 =====
    // dim!=3 返回 nullopt，不能把共面多边形当三维实体。
    optional<Wide> volume6() const {
        if (dim != 3) return nullopt;
        Wide ans = 0;
        for (auto [a, b, c] : f) ans += orient(p[0], p[a], p[b], p[c]);
        return ans;
    }
    optional<ld> volume() const {
        auto v6 = volume6();
        if (!v6) return nullopt;
        return ld(*v6) / 6;
    }
    static array<ld, 3> delta(P a, P b) {
        return {ld(a.x) - ld(b.x), ld(a.y) - ld(b.y), ld(a.z) - ld(b.z)};
    }
    static array<ld, 3> cross_ld(array<ld, 3> a, array<ld, 3> b) {
        return {a[1]*b[2]-a[2]*b[1], a[2]*b[0]-a[0]*b[2], a[0]*b[1]-a[1]*b[0]};
    }
    static ld len(array<ld, 3> a) {
        return sqrtl(a[0]*a[0] + a[1]*a[1] + a[2]*a[2]);
    }
    optional<ld> surface_area() const {
        if (dim != 3) return nullopt;
        ld ans = 0;
        for (auto [a, b, c] : f)
            ans += len(cross_ld(delta(p[b], p[a]), delta(p[c], p[a]))) / 2;
        return ans;
    }
    optional<array<ld, 3>> centroid() const {
        if (dim != 3) return nullopt;
        ld sum = 0;
        array<ld, 3> acc{};
        for (auto [a, b, c] : f) {
            ld w = ld(orient(p[0], p[a], p[b], p[c]));
            sum += w;
            auto u = delta(p[a], p[0]), v = delta(p[b], p[0]), t = delta(p[c], p[0]);
            for (int k = 0; k < 3; ++ k) acc[k] += w * (u[k] + v[k] + t[k]);
        }
        if (sum <= 0) return nullopt;
        return array<ld, 3>{ld(p[0].x) + acc[0]/(4*sum), ld(p[0].y) + acc[1]/(4*sum), ld(p[0].z) + acc[2]/(4*sum)};
    }
    // ===== 到表面距离：仅对体积重心计算 =====
    optional<ld> centroid_surface_distance() const {
        auto c = centroid();
        if (!c) return nullopt;
        ld ans = numeric_limits<ld>::infinity();
        for (auto [a, b, d] : f) {
            auto u = delta(p[b], p[a]), v = delta(p[d], p[a]);
            auto n = cross_ld(u, v);
            ld nl = len(n);
            if (nl == 0) continue;
            ld s = 0;
            for (int k = 0; k < 3; ++ k) s += n[k] * ((*c)[k] - coord(p[a], k));
            ans = min(ans, fabsl(s) / nl);
        }
        return isfinite(ans) ? optional<ld>(ans) : nullopt;
    }
    // ===== 共面情形的单侧面积与面积重心：只在 dim == 2 使用 =====
    optional<ld> planar_area() const {
        if (dim != 2) return nullopt;
        ld ans = 0;
        for (int i = 1; i + 1 < int(poly.size()); ++ i)
            ans += len(cross_ld(delta(p[poly[i]], p[poly[0]]), delta(p[poly[i+1]], p[poly[0]]))) / 2;
        return ans;
    }
    optional<array<ld, 3>> planar_centroid() const {
        if (dim != 2) return nullopt;
        ld sum = 0;
        array<ld, 3> acc{};
        for (int i = 1; i + 1 < int(poly.size()); ++ i) {
            auto u = delta(p[poly[i]], p[poly[0]]);
            auto v = delta(p[poly[i+1]], p[poly[0]]);
            ld w = len(cross_ld(u, v)) / 2;
            sum += w;
            for (int k = 0; k < 3; ++ k) acc[k] += w * (u[k] + v[k]);
        }
        if (sum == 0) return nullopt;
        return array<ld, 3>{ld(p[poly[0]].x) + acc[0]/(3*sum), ld(p[poly[0]].y) + acc[1]/(3*sum), ld(p[poly[0]].z) + acc[2]/(3*sum)};
    }
};
```
#### 成员变量
- `p`：构造后内部的点数组；**去重且可能重排**，下标不同于输入点原始下标。
- `f`：仅在 `dim == 3` 时有效，存储 `array<int,3>`，表示三维凸包的**朝外有向三角面**。相邻三角面的公共边方向相反，多个相邻三角面可能共面。
- `poly`：`dim == 1` 时为线段两端点的下标；`dim == 2` 时为平面凸多边形的循环顶点序列；其他维数下为空。
- `dim`：`-1` 空集、`0` 单点、`1` 线段、`2` 共面凸多边形、`3` 有正体积的三维凸多面体。
#### 接口

| 接口 | 作用 | 时间复杂度 |
|---|---|---|
| `Hull3<>(points)` / `build(points)` | 构造凸包、判定退化维数 | 预期 $O(n^2)$ |
| `contains(q, strict)` | 三维凸多面体内/边界判定；`strict=true` 排除边界 | $O(F)$ |
| `volume6()` | 有向三角面求和得到**六倍体积**；整数版返回 `optional<i128>`，浮点版为 `optional<ld>` | $O(F)$ |
| `volume()` | 三维凸多面体体积 `optional<ld>` | $O(F)$ |
| `surface_area()` | 三维实体**外表面**面积 `optional<ld>` | $O(F)$ |
| `centroid()` | 三维实体**均匀体积密度重心** `optional<array<ld,3>>` | $O(F)$ |
| `centroid_surface_distance()` | 三维体积重心到多面体表面的最短距离 `optional<ld>` | $O(F)$ |
| `planar_area()` | 共面凸多边形的**单侧面积** `optional<ld>` | $O(h)$ |
| `planar_centroid()` | 共面凸多边形的面积重心 `optional<array<ld,3>>` | $O(h)$ |
`F = f.size()` 是输出三角面数，`h = poly.size()` 是共面凸多边形顶点数；三维凸包中 `F = O(n)`。
#### 关键数学约定
三维有向四面体六倍体积为
$$
6V_{abcd} = \det(b-a,c-a,d-a) = (b-a)\cdot((c-a)\times(d-a)).
$$
实际计算三维凸包时选择一个凸包顶点 $o$，对每个向外三角面 $(a,b,c)$ 计算非负的有向六倍体积
$$
W_{abc}=\det(a-o,b-o,c-o),\qquad
V=\frac16\sum_{(a,b,c)\in f} W_{abc}.
$$
相同的四面体剖分用于体积重心，
$$
G=\frac{\sum_{(a,b,c)\in f}W_{abc}(o+a+b+c)/4}
{\sum_{(a,b,c)\in f}W_{abc}}.
$$
`centroid()` 在实现中先对 $o$ 做相对位移，避免大坐标参与反复加权时的部分消去误差。
#### 注意事项
- **`f` 的元素是三角形，不等于多面体的几何面数。** 例如立方体六个平面，每个平面可能被分为多个三角形；不能直接使用 `f.size()` 作为原始面数。不能保证共面三角划分唯一。
- **面片法向量朝外。** 对于凸包内点 `q`，每个面满足 `orient(a,b,c,q) <= 0`；若严格大于零则处于该面的外侧。
- **退化情形不假装有三维体积。** 当 `dim < 3`，`volume()`、`surface_area()`、`centroid()`、`contains()` 的含义与真三维实体不同：前三者返回 `nullopt`，`contains()` 返回 `false`。`dim == 2` 时使用 `planar_area()` / `planar_centroid()`，它们对应**单面面积**，不是实体“外表面积”。
- **浮点容差带长度量纲。** 通过三角面法向量长度归一化的点到平面有符号距离与 `EPS` 比较；因此同一组数据平移通常不改变尺度，但缩放坐标仍会改变判定结果。极近共面、大坐标小间距、极扁四面体可能触发数值不稳定，无法等同精确谓词。
- **整数运算要防溢出。** `orient()` 的每个中间乘积、三项累加及 `volume6()` 的所有面体积累加，均须处于 `i128` 范围。整数坐标很大时，不能仅凭输入能表示为 `i64` 就认定安全。
- **体积重心不是顶点坐标的简单平均。** 必须采用有向四面体的体积加权；面片数改变不会改变理论结果。
- **到表面距离只对体积重心定义。** 该重心严格位于非退化凸多面体内部，最近距离可由对各支持平面的垂直距离取最小得到；此函数不是“任意外部点到三角网格最短距离”。
- `contains(q, true)` 仅接受真三维内部点，处于任一面上的点都会被排除。浮点版边界由 `EPS` 决定。
- 若后续需要构建面邻接、法向量、棱边集合，直接从三角面 `f` 生成；对使用共面大面的题目，先合并同一支持平面上的三角面。
#### 常见用法
##### 构造三维凸包、求体积与表面积
```cpp
vector<Point3> p = {
    {0, 0, 0}, {1, 0, 0}, {0, 1, 0}, {0, 0, 1}
};
Hull3<> h(p);
if (h.dim == 3) {
    ld v = *h.volume();              // 1/6
    ld s = *h.surface_area();        // (3 + sqrt(3))/2
    auto g = *h.centroid();          // (1/4, 1/4, 1/4)
    ld d = *h.centroid_surface_distance();
}
```
##### 点在凸多面体内及严格内部
```cpp
if (h.dim == 3) {
    bool inside = h.contains(q);        // 边界也算在内
    bool strictly_inside = h.contains(q, true);
}
```
##### 遍历朝外三角面
```cpp
for (auto [a, b, c] : h.f) {
    Point3 A = h.p[a], B = h.p[b], C = h.p[c];
    // 三角面 A -> B -> C 的叉积法向量朝向外部
}
```
##### 全点共面的多边形
```cpp
Hull3<> h(p);
if (h.dim == 2) {
    ld s = *h.planar_area();
    auto c = *h.planar_centroid();
    for (int id : h.poly) {
        Point3 vertex = h.p[id]; // 按多边形循环顺序
    }
}
```
##### 精确六倍体积（整数版）
```cpp
Hull3<> h(p);                     // 3.1 使用整数 Point3 时
if (auto v6 = h.volume6()) {
    i128 ans6 = *v6;               // 必要时用自定义 i128 输出函数
}
```
#### 复杂度

| 操作 | 时间复杂度 | 额外空间 |
|---|---|---|
| `build()` | 预期 $O(n^2)$，包括初始排序 $O(n\log n)$ | $O(n)$ |
| 三维 `contains()` | $O(F)$ | $O(1)$ |
| 体积、表面积、重心、重心到表面距离 | $O(F)$ | $O(1)$ |
| `planar_area()` / `planar_centroid()` | $O(h)$ | $O(1)$ |
增量算法每次扫描全部当前三角面，并通过哈希容器取消可见面内部的反向边以提取轮廓边。三维凸包始终只有 $O(n)$ 个三角面，故**哈希操作摊还期望**总时间为 $O(n^2)$；对抗性哈希行为下无严格的 $O(n^2)$ 最坏时间保证
