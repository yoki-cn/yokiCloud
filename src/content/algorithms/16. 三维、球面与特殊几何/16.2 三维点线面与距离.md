### 三维点线面与距离
---
**依赖 3.1：** 仅复制对应数值版本的 `Point3`、`dot`、`cross`、`norm2`、`length`、`distance` 等基础函数。整数版与浮点版分别使用，不能混合定义。本节只处理直线/线段/平面的基础关系、正交投影及最短距离；严格的交点分类和三角形相交放在 3.3
#### 整数版：精确位置关系
---
适用于整数坐标的共线、共面、点在线上/线段上、两非退化直线平行/垂直、直线与平面平行/垂直以及平面同侧判定。避免除法及开平方。
```cpp
struct Line3 {
    Point3 p, dir;
    Line3() = default;
    Line3(Point3 p_, Point3 dir_) : p(p_), dir(dir_) {}
    // a != b，且 b-a 的每个分量必须可由 i64 表示
    static Line3 through(Point3 a, Point3 b) { return {a, b - a}; }
};
struct Segment3 {
    Point3 a, b;
    Segment3() = default;
    Segment3(Point3 a_, Point3 b_) : a(a_), b(b_) {}
};
struct Plane3 {
    Point3 a, b, c;
    Plane3() = default;
    Plane3(Point3 a_, Point3 b_, Point3 c_) : a(a_), b(b_), c(c_) {}
};

// ===== 精确位置判定 =====
bool on_line(Point3 p, Line3 l) {
    if (l.dir == Point3{}) return p == l.p;
    i128 x = i128(p.x) - l.p.x, y = i128(p.y) - l.p.y;
    i128 z = i128(p.z) - l.p.z;
    return i128(l.dir.y) * z - i128(l.dir.z) * y == 0
        && i128(l.dir.z) * x - i128(l.dir.x) * z == 0
        && i128(l.dir.x) * y - i128(l.dir.y) * x == 0;
}
bool on_seg(Point3 p, Segment3 s) {
    return collinear(s.a, s.b, p) && dot(p, s.a, s.b) <= 0;
}
// 前提：平面 a,b,c 不共线
int pt_side(Point3 p, Plane3 pl) { return sgn(orient(pl.a, pl.b, pl.c, p)); }
bool on_plane(Point3 p, Plane3 pl) { return coplanar(pl.a, pl.b, pl.c, p); }
// stct=true：严格同侧；false：把处于平面的点视为两侧皆可
bool same_sd(Point3 p, Point3 q, Plane3 pl, bool stct = true) {
    int a = pt_side(p, pl), b = pt_side(q, pl);
    return stct ? (a != 0 && a == b) : (a == 0 || b == 0 || a == b);
}
// 以下方向关系都要求输入方向非零；平面须由三非共线点定义
bool line_parallel(Line3 a, Line3 b) { return parallel(a.dir, b.dir); }
bool line_perp(Line3 a, Line3 b) { return perp(a.dir, b.dir); }
// 法向量来自宽整数叉积，避免存回 i64
bool line_plane_parallel(Line3 l, Plane3 pl) {
    auto n = cross(pl.a, pl.b, pl.c);
    return n[0] * l.dir.x + n[1] * l.dir.y + n[2] * l.dir.z == 0;
}
bool line_plane_perp(Line3 l, Plane3 pl) {
    auto n = cross(pl.a, pl.b, pl.c);
    // n × dir == 0，乘积须处于 i128 范围
    return n[1] * l.dir.z == n[2] * l.dir.y
        && n[2] * l.dir.x == n[0] * l.dir.z
        && n[0] * l.dir.y == n[1] * l.dir.x;
}
```
**整数版额外限制：** `on_line` 直接采用宽整数差分，不要求 `p-l.p` 仍落在 `i64` 范围内。`Line3::through` 构造仍需保证坐标差可由 `i64` 表示。`orient` 与宽整数叉积、点积仍要求所有中间乘积及累加值在 `i128` 范围内。`Plane3` 不负责三点共线检查，调用方应排除退化平面。
#### 浮点版：结构体与判定
复用 3.1 的浮点 `Point3`、`EPS`、`sgn`、`dot`、`cross`、`norm2`、`length`、`distance`、`unit`。本节浮点版 `Plane3` 采用**一点 + 法向量**表示，避免反复构造三个顶点
```cpp
struct Line3 {
    Point3 p, dir;
    Line3() = default;
    Line3(Point3 p_, Point3 dir_) : p(p_), dir(dir_) {}
    static Line3 through(Point3 a, Point3 b) { return {a, b - a}; }
};
struct Segment3 {
    Point3 a, b;
    Segment3() = default;
    Segment3(Point3 a_, Point3 b_) : a(a_), b(b_) {}
};
struct Plane3 {
    Point3 p, n;
    Plane3() = default;
    Plane3(Point3 p_, Point3 n_) : p(p_), n(n_) {}
    // 三点不共线时有效
    static Plane3 through(Point3 a, Point3 b, Point3 c) {
        return {a, cross(b - a, c - a)};
    }
    // ax+by+cz=d，要求法向量 (a,b,c) 非零
    static optional<Plane3> equation(ld a, ld b, ld c, ld d) {
        Point3 n{a, b, c};
        ld nn = norm2(n);
        if (nn == 0) return nullopt;
        return Plane3(n * (d / nn), n);
    }
};

// ===== 位置判定 =====
// 方向类函数均要求直线、平面非退化
bool on_line(Point3 p, Line3 l) {
    ld len = length(l.dir);
    if (len == 0) return same_pt(p, l.p);
    Point3 v = p - l.p;
    return length(cross(l.dir / len, v)) <= EPS;
}
bool on_seg(Point3 p, Segment3 s) {
    Point3 v = s.b - s.a;
    ld vv = norm2(v);
    if (vv == 0) return same_pt(p, s.a);
    ld t = dot(p - s.a, v) / vv;
    return t >= -EPS && t <= 1 + EPS && on_line(p, Line3{s.a, v});
}
// 正负侧遵循法向量 n，前提：n != 0
ld signed_dis_plane(Point3 p, Plane3 pl) {
    return dot(p - pl.p, pl.n) / length(pl.n);
}
int pt_side(Point3 p, Plane3 pl) { return sgn(signed_dis_plane(p, pl)); }
bool on_plane(Point3 p, Plane3 pl) { return pt_side(p, pl) == 0; }
bool same_sd(Point3 p, Point3 q, Plane3 pl, bool stct = true) {
    int a = pt_side(p, pl), b = pt_side(q, pl);
    return stct ? (a != 0 && a == b) : (a == 0 || b == 0 || a == b);
}
bool line_parallel(Line3 a, Line3 b) { return parallel(a.dir, b.dir); }
bool line_perp(Line3 a, Line3 b) { return perp(a.dir, b.dir); }
bool line_plane_parallel(Line3 l, Plane3 pl) { return perp(l.dir, pl.n); }
bool line_plane_perp(Line3 l, Plane3 pl) { return parallel(l.dir, pl.n); }
bool plane_parallel(Plane3 a, Plane3 b) { return parallel(a.n, b.n); }
bool plane_perp(Plane3 a, Plane3 b) { return perp(a.n, b.n); }

// ===== 投影、对称与基本距离 =====
// 退化直线按单点处理；退化线段按端点处理
Point3 proj(Point3 p, Line3 l) {
    ld dd = norm2(l.dir);
    return dd == 0 ? l.p : l.p + l.dir * (dot(p - l.p, l.dir) / dd);
}
Point3 refl(Point3 p, Line3 l) { return 2.L * proj(p, l) - p; }
ld dis_line(Point3 p, Line3 l) { return distance(p, proj(p, l)); }
Point3 proj(Point3 p, Plane3 pl) {
    return p - pl.n * (dot(p - pl.p, pl.n) / norm2(pl.n));
}
Point3 refl(Point3 p, Plane3 pl) { return 2.L * proj(p, pl) - p; }
ld dis_plane(Point3 p, Plane3 pl) { return fabsl(signed_dis_plane(p, pl)); }
Point3 closest(Point3 p, Segment3 s) {
    Point3 v = s.b - s.a;
    ld vv = norm2(v);
    if (vv == 0) return s.a;
    ld t = clamp(dot(p - s.a, v) / vv, 0.L, 1.L);
    return s.a + v * t;
}
ld dis_seg(Point3 p, Segment3 s) { return distance(p, closest(p, s)); }
```
#### 浮点版：空间直线与线段最近点
```cpp
struct Closest3 {
    Point3 a, b; // 两对象上的最近点
    ld dist;      // 两点距离
};

// ===== 两条无限直线：独立扩展；退化直线作为点处理 =====
// 平行时选第一条直线的基点 a.p 作为一端，另一端取其投影
Closest3 closest(Line3 a, Line3 b) {
    Point3 u = a.dir, v = b.dir, w = a.p - b.p;
    ld A = norm2(u), C = norm2(v);
    ld D = dot(u, w), E = dot(v, w);
    ld s = 0, t = 0;
    if (A == 0 && C == 0) return {a.p, b.p, distance(a.p, b.p)};
    if (A == 0) t = E / C;
    else if (C == 0) s = -D / A;
    else {
        Point3 n = cross(u, v);
        ld det = norm2(n);
        // 叉积形式避免近乎平行时 A*C-B*B 相减丢失有效位
        if (det > 0) {
            s = dot(cross(v, w), n) / det;
            t = dot(cross(u, w), n) / det;
        } else {
            t = E / C;
        }
    }
    Point3 p = a.p + u * s, q = b.p + v * t;
    return {p, q, distance(p, q)};
}
ld line_dis(Line3 a, Line3 b) { return closest(a, b).dist; }

// ===== 两条闭线段：独立扩展；允许一条或两条线段退化为点 =====
// 返回全局最近点对，不可只取四个端点到对方线段的距离
Closest3 closest(Segment3 x, Segment3 y) {
    Point3 u = x.b - x.a, v = y.b - y.a, w = x.a - y.a;
    ld A = norm2(u), B = dot(u, v), C = norm2(v);
    ld D = dot(u, w), E = dot(v, w);
    ld s = 0, t = 0;
    if (A == 0 && C == 0) {
        return {x.a, y.a, distance(x.a, y.a)};
    } else if (A == 0) {
        t = clamp(E / C, 0.L, 1.L);
    } else if (C == 0) {
        s = clamp(-D / A, 0.L, 1.L);
    } else {
        // 叉积形式避免近乎平行时 A*C-B*B 相减丢失有效位
        Point3 n = cross(u, v);
        ld det = norm2(n);
        s = det > 0 ? clamp(dot(cross(v, w), n) / det, 0.L, 1.L) : 0.L;
        t = (B * s + E) / C;
        if (t < 0) {
            t = 0;
            s = clamp(-D / A, 0.L, 1.L);
        } else if (t > 1) {
            t = 1;
            s = clamp((B - D) / A, 0.L, 1.L);
        }
    }
    Point3 p = x.a + u * s, q = y.a + v * t;
    return {p, q, distance(p, q)};
}
ld seg_dis(Segment3 a, Segment3 b) { return closest(a, b).dist; }
```
#### 接口说明
##### 基础结构与位置关系

| 接口 | 作用 | 时间 |
|---|---|---|
| `Line3(p,dir)` / `Line3::through(a,b)` | 直线的一点与方向 / 过两点的直线 | $O(1)$ |
| `Segment3(a,b)` | 闭线段 | $O(1)$ |
| 整数 `Plane3(a,b,c)` | 三点确定平面 | $O(1)$ |
| 浮点 `Plane3(p,n)` / `through(a,b,c)` | 点+法向量 / 三点构造平面 | $O(1)$ |
| 浮点 `Plane3::equation(a,b,c,d)` | 方程 $ax+by+cz=d$；零法向量返回 `nullopt` | $O(1)$ |
| `on_line(p,l)` / `on_seg(p,s)` | 点在直线 / 闭线段上（支持退化） | $O(1)$ |
| `pt_side(p,pl)` / `on_plane(p,pl)` | 点相对有向平面的正负侧 / 在平面上 | $O(1)$ |
| `same_sd(p,q,pl,stct)` | 同侧判定；非严格时平面上的点两侧皆可 | $O(1)$ |
| `line_parallel` / `line_perp` | 两直线方向平行 / 垂直 | $O(1)$ |
| `line_plane_parallel` / `line_plane_perp` | 直线方向与平面平行 / 垂直 | $O(1)$ |
| `plane_parallel` / `plane_perp` | 两平面平行 / 垂直（仅浮点版） | $O(1)$ |
##### 浮点距离与投影

| 接口 | 作用 | 时间 |
|---|---|---|
| `proj(p, Line3)` / `proj(p, Plane3)` | 点到无限直线 / 平面的正交投影 | $O(1)$ |
| `refl(p, Line3)` / `refl(p, Plane3)` | 点关于直线 / 平面的镜像点 | $O(1)$ |
| `dis_line(p,l)` / `dis_plane(p,pl)` | 点到直线 / 平面的距离 | $O(1)$ |
| `signed_dis_plane(p,pl)` | 点到有向平面的带符号距离 | $O(1)$ |
| `closest(p, Segment3)` / `dis_seg(p,s)` | 点在线段上的最近点 / 距离 | $O(1)$ |
| `closest(Line3,Line3)` / `line_dis(a,b)` | 两无限直线最近点对 / 距离 | $O(1)$ |
| `closest(Segment3,Segment3)` / `seg_dis(a,b)` | 两闭线段最近点对 / 距离 | $O(1)$ |
#### 注意事项
- **3.1 与 3.2 的依赖**：本文件代码必须置于对应版本的 3.1 之后；整数与浮点两套 `Line3` / `Plane3` 不可同时复制。`Line3`、`Plane3` 不做自动合法性校验。
- 平面必须非退化，直线/平面方向判断也要求方向非零；投影和距离函数允许退化直线 / 线段，视为单点。
- `pt_side` 的符号：整数版由 `orient(a,b,c,p)` 决定，浮点版由 `dot(p-pl.p,pl.n)` 决定；`Plane3::through(a,b,c)` 的法向量为 `(b-a)×(c-a)`。
- `line_plane_parallel` 仅指**方向平行**，直线也可能恰好在平面内；`line_parallel` 不区分两直线重合或平行不重合。
- 两条空间直线可能**异面**，其方向既不平行也不相交；`line_dis` 才能反映异面最短距离。求最近点对时，平行情况下最近点不唯一，返回任意有效一对。
- 空间两线段距离不能简单推广二维板子的“四端点到另一线段距离取最小”公式；异面线段的两个内部点可能是最近点。
- `EPS` 为绝对坐标距离阈值，浮点谓词不能保证极大坐标或极小夹角下的鲁棒性；高精度要求需使用自适应/精确谓词，避免绝对误差与相对误差混用。
- 对于近乎平行的直线/线段，点对参数求解可能数值不稳定；距离通常更稳定，极端条件建议提高精度或独立复核。
- 浮点 `on_seg` 对参数 `t` 采用容差，容差是**无量纲参数**，而 `on_line` 使用坐标距离的 `EPS`；数据尺度悬殊时应自行调整。
- 整数版 `line_plane_parallel` 和 `line_plane_perp` 内部继续产生额外 `i128` 乘积；必须确认这些表达式不会溢出。
#### 常见用法
##### 空间点到直线、线段、平面（浮点版）
```cpp
Point3 p{2, 3, 4};
Line3 ln = Line3::through({0, 0, 0}, {1, 0, 0});
Segment3 seg{{0, 0, 0}, {1, 0, 0}};
Plane3 pl = Plane3::through({0, 0, 0}, {1, 0, 0}, {0, 1, 0});
Point3 foot1 = proj(p, ln);    // (2,0,0)
Point3 foot2 = proj(p, pl);    // (2,3,0)
ld d1 = dis_line(p, ln);       // 5
ld d2 = dis_seg(p, seg);       // sqrt(26)
ld d3 = dis_plane(p, pl);      // 4
```
##### 空间异面直线、线段最近点对（浮点版）
```cpp
Line3 l1 = Line3::through({0, 0, 0}, {1, 0, 0});
Line3 l2 = Line3::through({0, 0, 1}, {0, 1, 1});
auto res = closest(l1, l2);    // res.dist == 1

Segment3 s1{{-1, 0, 0}, {1, 0, 0}};
Segment3 s2{{0, -1, 1}, {0, 1, 1}};
auto ans = closest(s1, s2);   // 最近点是两段内部点，距离 1
```
##### 精确平面同侧（整数版）
```cpp
Plane3 pl{{0, 0, 0}, {1, 0, 0}, {0, 1, 0}};
int side = pt_side({0, 0, 5}, pl); // +1
bool ok = same_sd({0, 0, 2}, {1, 1, 3}, pl); // true
```

所有函数：时间复杂度 $O(1)$，额外空间 $O(1)$
