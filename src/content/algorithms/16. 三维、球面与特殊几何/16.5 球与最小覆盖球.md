### 球与最小覆盖球
---
**依赖：** 复用 **3.1 浮点版**的 `Point3`、`ld`、`EPS`、`PI`、`sgn(ld)`、`dot()`、`cross()`、`triple()`、`norm2()`、`length()`、`distance()`。球与直线相交额外依赖 **3.2 浮点版 `Line3`**，球与平面相交额外依赖 **3.2 浮点版 `Plane3`（一点 `p` + 法向量 `n`）**。**不依赖 3.3、3.4**；`min_ball()` 也不依赖 3.2。整数坐标可先转为浮点 `Point3`，不能将整数版/浮点版的 `Point3` 混用
本节所有球均使用**球心 + 半径**表示。球面是边界，实心球包含内部；球面交点/交圆和实心球交体积是不同概念。合法半径为 `r >= 0`（`r == 0` 视为单点球），`r < 0` 表示无效球。
#### 基础结构与点的包含判定
---
```cpp
// ===== 球与圆：依赖 3.1 浮点 Point3 / ld / EPS / dot / cross / norm2 / length / distance =====
// 球面允许 r == 0，此时退化为一个点；r < 0 表示无效球。
struct Sphere3 {
    Point3 c{};
    ld r{ -1 };
    Sphere3() = default;
    Sphere3(Point3 c_, ld r_) : c(c_), r(r_) {}
    bool valid() const { return r >= 0; }
    ld area() const { return 4.L * PI * r * r; }
    ld volume() const { return 4.L * PI * r * r * r / 3.L; }
};
struct Circle3 {
    Point3 c, n; // 圆心、单位法向量
    ld r;        // 圆半径；切点视为 r == 0 的圆
};
// ===== 球面与实体球的点判定 =====
// -1 球内，0 球面上，1 球外；无效球不应调用
int sphere_side(Point3 p, Sphere3 s) {
    return sgn(distance(p, s.c) - s.r);
}
bool in_ball(Point3 p, Sphere3 s) { return sphere_side(p, s) <= 0; }
bool on_sphere(Point3 p, Sphere3 s) { return sphere_side(p, s) == 0; }
```
#### 直线与球面交点
---
```cpp
// ===== 直线与球面：依赖 3.2 浮点 Line3 =====
enum class SphereLineRel {
    INVALID,  // 无效：球 r < 0 或直线零方向
    DISJOINT, // 相离：没有交点
    TANGENT,  // 相切：唯一公共点
    SECANT    // 相交：两个不同公共点
};
struct SphereLineHit {
    SphereLineRel rel;
    vector<Point3> p; // 0、1 或 2 个球面交点
};
SphereLineHit sphere_line_int(Sphere3 s, Line3 l) {
    if (!s.valid() || norm2(l.dir) == 0) return {SphereLineRel::INVALID, {}};
    ld t = dot(s.c - l.p, l.dir) / norm2(l.dir);
    Point3 foot = l.p + l.dir * t;
    ld d = distance(s.c, foot);
    int rel = sgn(d - s.r);
    if (rel > 0) return {SphereLineRel::DISJOINT, {}};
    if (rel == 0) return {SphereLineRel::TANGENT, {foot}};
    ld h = sqrtl(max(0.L, (s.r - d) * (s.r + d)));
    Point3 v = l.dir * (h / length(l.dir));
    return {SphereLineRel::SECANT, {foot - v, foot + v}};
}
```
#### 平面与球面交圆
---
```cpp
// ===== 平面与球面：依赖 3.2 浮点 Plane3(p,n)；可整体删除 =====
enum class SpherePlaneRel {
    INVALID,  // 无效：球 r < 0 或平面零法向量
    DISJOINT, // 相离：无公共点
    TANGENT,  // 相切：唯一公共点
    INTERSECT // 相交：公共部分是一个圆
};
struct SpherePlaneHit {
    SpherePlaneRel rel;
    optional<Circle3> circle; // 相切时为半径 0 的退化圆
};
SpherePlaneHit sphere_plane_int(Sphere3 s, Plane3 pl) {
    ld len = length(pl.n);
    if (!s.valid() || len == 0) return {SpherePlaneRel::INVALID, nullopt};
    Point3 n = pl.n / len;
    ld h = dot(s.c - pl.p, n);
    int rel = sgn(fabsl(h) - s.r);
    if (rel > 0) return {SpherePlaneRel::DISJOINT, nullopt};
    Point3 foot = s.c - n * h;
    if (rel == 0) return {SpherePlaneRel::TANGENT, Circle3{foot, n, 0}};
    ld rr = sqrtl(max(0.L, (s.r - fabsl(h)) * (s.r + fabsl(h))));
    return {SpherePlaneRel::INTERSECT, Circle3{foot, n, rr}};
}
```
#### 两个球面的交圆与位置关系
---
```cpp
// ===== 两球面交集：独立扩展，可整体删除 =====
enum class SphereSphereRel {
    INVALID,          // 无效：存在负半径
    DISJOINT,         // 外离：球面不相交，两个实心球也不相交
    EXTERNAL_TANGENT, // 外切：球面交于一点
    INTERSECT,        // 相交：球面交于一个圆
    INTERNAL_TANGENT, // 内切：球面交于一点
    CONTAINED,        // 内含：较小球严格处于另一实心球内，球面不接触
    COINCIDENT        // 重合：球心、半径均相同
};
struct SphereSphereHit {
    SphereSphereRel rel;
    optional<Circle3> circle; // 相交圆；相切时半径 0；重合时不存在唯一交圆
};
SphereSphereHit sphere_sphere_int(Sphere3 a, Sphere3 b) {
    if (!a.valid() || !b.valid()) return {SphereSphereRel::INVALID, nullopt};
    Point3 v = b.c - a.c;
    ld d = length(v);
    if (sgn(d) == 0) {
        if (sgn(a.r - b.r) == 0) return {SphereSphereRel::COINCIDENT, nullopt};
        return {SphereSphereRel::CONTAINED, nullopt};
    }
    int outer = sgn(d - (a.r + b.r));
    if (outer > 0) return {SphereSphereRel::DISJOINT, nullopt};
    int inner = sgn(d - fabsl(a.r - b.r));
    if (inner < 0) return {SphereSphereRel::CONTAINED, nullopt};
    Point3 n = v / d;
    if (outer == 0) {
        return {SphereSphereRel::EXTERNAL_TANGENT, Circle3{a.c + n * a.r, n, 0}};
    }
    if (inner == 0) {
        // 内切时，切点位于大球朝向小球的一侧
        Point3 p = a.r >= b.r ? a.c + n * a.r : a.c - n * a.r;
        return {SphereSphereRel::INTERNAL_TANGENT, Circle3{p, n, 0}};
    }
    ld x = (d + (a.r - b.r) * ((a.r + b.r) / d)) / 2.L;
    ld rr = sqrtl(max(0.L, (a.r - x) * (a.r + x)));
    return {SphereSphereRel::INTERSECT, Circle3{a.c + n * x, n, rr}};
}
```
#### 两个实心球的交体积
---
不依赖上述相交关系接口，可以独立保留 `Sphere3` 与本代码块。
```cpp
// ===== 两实心球交体积：只需 Sphere3 和距离 =====
ld sphere_cap_volume(ld r, ld h) {
    h = clamp(h, 0.L, 2.L * r);
    return PI * h * h * (r - h / 3.L);
}
ld ball_overlap_volume(Sphere3 a, Sphere3 b) {
    // 前提：球有效
    ld d = distance(a.c, b.c);
    if (d >= a.r + b.r) return 0;
    if (d <= fabsl(a.r - b.r)) return min(a.volume(), b.volume());
    ld x = (d + (a.r - b.r) * ((a.r + b.r) / d)) / 2.L;
    return sphere_cap_volume(a.r, a.r - x) + sphere_cap_volume(b.r, b.r - (d - x));
}
```
#### 最小覆盖球
---
只依赖本节 `Sphere3` 和 **3.1 浮点 `Point3` 基础函数**
实现为随机打乱后的四层增量维护；支持空集、单点、重复点、共线、共面点集。求三点外接圆或四点外接球时采用局部缩放以减轻量级影响，退化时枚举至多四点的低维支撑球。`min_ball()` 内部先把点集平移到局部原点，并在返回前修正半径以覆盖全部输入点的计算距离。
```cpp
// ===== 最小覆盖球：只依赖 3.1 浮点 Point3、基础函数；可独立复制 =====
// 当 3 或 4 个边界点退化时，从低维候选球中寻找覆盖它们的最小球。
Sphere3 ball2(Point3 a, Point3 b) {
    Point3 c = a + (b - a) / 2.L;
    return {c, distance(c, a)};
}
optional<Sphere3> ball3(Point3 a, Point3 b, Point3 c) {
    Point3 u = b - a, v = c - a;
    ld scale = max(length(u), length(v));
    if (scale == 0) return nullopt;
    u /= scale; v /= scale;
    Point3 n = cross(u, v);
    ld den = 2.L * norm2(n);
    if (den <= 2.L * EPS * EPS * norm2(u) * norm2(v)) return nullopt;
    Point3 o = a + (cross(v, n) * norm2(u) + cross(n, u) * norm2(v)) * (scale / den);
    return Sphere3{o, distance(o, a)};
}
optional<Sphere3> ball4(Point3 a, Point3 b, Point3 c, Point3 d) {
    Point3 u = b - a, v = c - a, w = d - a;
    ld scale = max({length(u), length(v), length(w)});
    if (scale == 0) return nullopt;
    u /= scale; v /= scale; w /= scale;
    ld den = 2.L * triple(u, v, w);
    if (fabsl(den) <= 2.L * EPS * length(u) * length(v) * length(w)) return nullopt;
    Point3 o = a + (cross(v, w) * norm2(u) + cross(w, u) * norm2(v)
                    + cross(u, v) * norm2(w)) * (scale / den);
    return Sphere3{o, distance(o, a)};
}
// 处理边界点共线、共面，亦可作为 <= 4 点的暴力正确性参考
Sphere3 min_ball_small(const vector<Point3> &p) {
    Sphere3 ans;
    auto check = [&](Sphere3 s) {
        // 这里只做极小相对浮点容差，避免明显错误的候选球被接受
        for (auto q : p) {
            ld d = distance(q, s.c);
            if (d > s.r + EPS * max(s.r, d)) return;
        }
        if (!ans.valid() || s.r < ans.r) ans = s;
    };
    int n = p.size();
    for (int i = 0; i < n; ++i) check({p[i], 0});
    for (int i = 0; i < n; ++i) for (int j = 0; j < i; ++j)
        check(ball2(p[i], p[j]));
    for (int i = 0; i < n; ++i) for (int j = 0; j < i; ++j)
        for (int k = 0; k < j; ++k)
            if (auto s = ball3(p[i], p[j], p[k])) check(*s);
    if (n == 4) if (auto s = ball4(p[0], p[1], p[2], p[3])) check(*s);
    return ans;
}
// 随机增量，固定维数期望 O(n)；强退化/病态数据可能影响数值稳定性
optional<Sphere3> min_ball(vector<Point3> p,
                          uint64_t seed = chrono::steady_clock::now().time_since_epoch().count()) {
    if (p.empty()) return nullopt;
    // 平移到局部坐标系，减少大坐标导致的球心精度损失
    Point3 origin = p[0];
    for (auto &q : p) q -= origin;
    mt19937_64 rng(seed);
    shuffle(p.begin(), p.end(), rng);
    auto inside = [](Point3 q, Sphere3 s) {
        ld d = distance(q, s.c);
        return d <= s.r + EPS * max(s.r, d);
    };
    Sphere3 s{p[0], 0};
    for (int i = 1; i < int(p.size()); ++i) {
        if (inside(p[i], s)) continue;
        s = {p[i], 0};
        for (int j = 0; j < i; ++j) {
            if (inside(p[j], s)) continue;
            s = ball2(p[i], p[j]);
            for (int k = 0; k < j; ++k) {
                if (inside(p[k], s)) continue;
                if (auto t = ball3(p[i], p[j], p[k])) s = *t;
                else s = min_ball_small({p[i], p[j], p[k]});
                for (int l = 0; l < k; ++l) {
                    if (inside(p[l], s)) continue;
                    if (auto t = ball4(p[i], p[j], p[k], p[l])) s = *t;
                    else s = min_ball_small({p[i], p[j], p[k], p[l]});
                }
            }
        }
    }
    // 修补最后的微小舍入：返回球保证覆盖所有输入点的计算距离
    for (auto q : p) s.r = max(s.r, distance(q, s.c));
    s.c += origin;
    return s;
}
```
#### 接口说明
##### 基础接口

| 接口 | 含义 | 复杂度 |
|---|---|---|
| `Sphere3(c,r)` | 球心 `c`、半径 `r`，默认构造是无效球 | $O(1)$ |
| `s.valid()` | `r >= 0` | $O(1)$ |
| `s.area()` | 球面面积 $4\pi r^2$，要求有效球 | $O(1)$ |
| `s.volume()` | 实心球体积 $\frac43\pi r^3$，要求有效球 | $O(1)$ |
| `sphere_side(p,s)` | `-1/0/1`：球内 / 球面上 / 球外 | $O(1)$ |
| `in_ball(p,s)` | 点在闭实心球内（包括球面） | $O(1)$ |
| `on_sphere(p,s)` | 点在球面上 | $O(1)$ |
| `Circle3{c,n,r}` | 空间圆：圆心、**单位**法向量、圆半径 | $O(1)$ |
##### 相交接口

| 接口                         | 返回类型              | 含义                      | 复杂度    |
| -------------------------- | ----------------- | ----------------------- | ------ |
| `sphere_line_int(s,l)`     | `SphereLineHit`   | 无限直线与球面交点，返回 `0～2` 个点   | $O(1)$ |
| `sphere_plane_int(s,pl)`   | `SpherePlaneHit`  | 平面与球面交圆，切点用半径 `0` 的圆表示  | $O(1)$ |
| `sphere_sphere_int(a,b)`   | `SphereSphereHit` | 两球面位置关系及唯一交圆；重合时不返回唯一交圆 | $O(1)$ |
| `sphere_cap_volume(r,h)`   | `ld`              | 球冠体积，`h` 限制在 $[0,2r]$   | $O(1)$ |
| `ball_overlap_volume(a,b)` | `ld`              | 两**实心球**的公共部分体积         | $O(1)$ |
##### 枚举值与中文含义

| 枚举 | 值 | 中文含义 | 交集 |
|---|---|---|---|
| `SphereLineRel` | `INVALID` | 无效：球半径为负或直线退化 | 无法分类 |
| | `DISJOINT` | 相离 | 空集 |
| | `TANGENT` | 相切 | 一个点 |
| | `SECANT` | 割线相交 | 两个不同的点 |
| `SpherePlaneRel` | `INVALID` | 无效：球半径为负或平面法向量为零 | 无法分类 |
| | `DISJOINT` | 相离 | 空集 |
| | `TANGENT` | 相切 | 一个点 |
| | `INTERSECT` | 相交 | 一个非退化空间圆 |
| `SphereSphereRel` | `INVALID` | 无效：至少一个球半径为负 | 无法分类 |
| | `DISJOINT` | 外离 | 球面与实体球均无交集 |
| | `EXTERNAL_TANGENT` | 外切 | 球面交于一个点 |
| | `INTERSECT` | 相交 | 球面交于一个空间圆 |
| | `INTERNAL_TANGENT` | 内切 | 球面交于一个点 |
| | `CONTAINED` | 内含、不相切 | 两球面无交集，小实心球包含于大实心球 |
| | `COINCIDENT` | 两球面重合 | 球面完全相同，没有唯一的交圆 |
`SphereLineHit::p` 保存交点，空集/切线/割线分别有 `0/1/2` 个元素；`SpherePlaneHit::circle`、`SphereSphereHit::circle` 为 `optional<Circle3>`，只有存在**唯一交圆（含退化切点）**时才有值。**
##### 最小覆盖球接口

| 接口 | 含义 | 复杂度 |
|---|---|---|
| `ball2(a,b)` | 两点作为边界得到的直径球 | $O(1)$ |
| `ball3(a,b,c)` | 非共线三点的外接圆所在球（球心处于三点平面），近共线返回 `nullopt` | $O(1)$ |
| `ball4(a,b,c,d)` | 非共面四点的外接球，近共面返回 `nullopt` | $O(1)$ |
| `min_ball_small(p)` | `p.size() <= 4` 时枚举低维候选球，选择可包含所有点的最小者 | $O(1)$ |
| `min_ball(p,seed)` | 求点集最小覆盖球；空集返回 `nullopt`，`seed` 控制随机打乱 | **期望** $O(n)$，最坏 $O(n^4)$ |
最小覆盖球的 `ball3`/`ball4` 是**外接球构造**，不是任意三点/四点的最小覆盖球。三点形成钝角三角形时，其最小覆盖球可能由最长边的两端决定；单独求 `<=4` 点最小覆盖球请调用 `min_ball_small()` 或 `min_ball()`。
#### 注意事项
- **数值版本**：本节统一使用浮点 `Point3`；`Line3`、`Plane3` 也必须是 3.2 的浮点版。整数版三维几何中的 `Plane3(a,b,c)` 与这里的 `Plane3(p,n)` 构造约定不同，不能混用。
- **球面 vs 实心球**：`sphere_sphere_int()` 判断的是**两个球面的交集**。`CONTAINED` 时两球面无交集，但实心球交集体积通常是较小球的体积。
- **退化**：半径 `0` 的球视为单点。此时外切与内切在几何语义上可能不唯一，代码优先按外切分支分类；两点球同心同半径视为 `COINCIDENT`，`circle == nullopt`。
- **球面相切**：浮点 `EPS` 分支可把非常接近相切的状态按相切处理；返回的切点只在相应容差内满足球面方程。
- **零方向**：零方向直线或零法向量平面返回 `INVALID`；不尝试做点代替直线或任意平面代替非法平面。
- **交圆法向量**：返回的 `Circle3::n` 已单位化；两球交圆的 `n` 朝向第一个球心到第二个球心，改变两球顺序会反向。
- **最小覆盖球精度**：增量判定使用相对球尺度的容差，不直接使用 `sgn(distance-r)` 的固定绝对阈值，以免微小尺度数据被误判。近共线/近共面时的分支会引入近似，极端病态数据并无严格鲁棒性保证。
- **坐标范围**：`min_ball()` 内部已做平移，但返回绝对球心时仍受 `long double` 表示精度限制；特别大的坐标平移可能损失末位。建议预先按题目容差合理缩放输入。
- **面积与体积**：球冠体积公式适用于球半径 `r >= 0`；`ball_overlap_volume()` 要求两球有效。输出精度由题目要求决定，通常使用 `fixed << setprecision(12)`。
- **最坏复杂度**：随机增量法的期望线性复杂度建立在随机打乱的前提下，固定极端顺序或浮点近退化可能影响运行时间。
#### 常见用法
##### 直线与球面、平面与球面相交
```cpp
Sphere3 s{{0, 0, 0}, 2};
Line3 l = Line3::through({-3, 0, 0}, {3, 0, 0});
auto a = sphere_line_int(s, l);
if (a.rel == SphereLineRel::SECANT) {
    // a.p[0]、a.p[1] 分别为两个球面交点
}
Plane3 pl{{0, 0, 1}, {0, 0, 1}}; // z = 1
if (auto b = sphere_plane_int(s, pl); b.circle) {
    // b.circle->c == (0,0,1), b.circle->r == sqrt(3)
}
```
##### 两球相交与体积
```cpp
Sphere3 a{{0, 0, 0}, 2}, b{{2, 0, 0}, 2};
auto hit = sphere_sphere_int(a, b);
if (hit.rel == SphereSphereRel::INTERSECT) {
    Point3 c = hit.circle->c; // (1,0,0)
    ld r = hit.circle->r;    // sqrt(3)
}
ld vol = ball_overlap_volume(a, b);
```
##### 最小覆盖球
```cpp
vector<Point3> p{{0,0,0}, {2,0,0}, {0,2,0}, {0,0,2}};
auto ans = min_ball(p); // 可传第二参数 seed 复现实验
if (ans) {
    cout << fixed << setprecision(12)
         << ans->c.x << ' ' << ans->c.y << ' ' << ans->c.z << ' '
         << ans->r << '\n';
}
```
#### 复杂度
除 `min_ball()` 外所有操作均为 $O(1)$ 时间、$O(1)$ 额外空间（相交点数最多 `2`）；`min_ball()` 因复制、随机打乱输入而使用 $O(n)$ 额外空间，**期望** $O(n)$ 时间，最坏 $O(n^4)$
