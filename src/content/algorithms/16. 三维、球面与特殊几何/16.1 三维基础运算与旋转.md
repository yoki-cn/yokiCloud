### 三维基础运算与旋转
---
#### 数值与版本约定
与二维 `Point` 一致：**整数版负责精确的向量运算与方向判定；浮点版负责长度、投影、角度及任意轴旋转**。两版同名代码是替换关系，不应同时粘贴
- 整数版：输入坐标、加减、数乘与取反结果必须能由 `i64` 表示；叉积三个分量、点积、混合积与距离平方的**每个中间结果及总和**均须在有符号 `i128` 范围内。使用 `i128` 计算不能保证任意满范围 `i64` 坐标安全
- 浮点版：使用 `ld = long double`、`EPS = 1E-12L`；`EPS` 用于几何分支判断，不是题目输出误差
- 三维叉积是**向量**，混合积是**标量**；三维不存在与二维完全等价的全局逆时针极角顺序
- 与二维板子相同，浮点结构的 `==` 和 `<=>` 比较原始值，几何意义上的重合使用 `same_pt()`
#### 整数结构体
---
```cpp
int sgn(i128 x) { return (x > 0) - (x < 0); }
struct Point3 {
    i64 x{ 0 }, y{ 0 }, z{ 0 };
    Point3() = default;
    Point3(i64 x_, i64 y_, i64 z_) : x(x_), y(y_), z(z_) {}
    Point3 operator+() const { return *this; }
    Point3 operator-() const { return {-x, -y, -z}; }
    Point3 &operator+=(Point3 p) {
        x += p.x; y += p.y; z += p.z;
        return *this;
    }
    Point3 &operator-=(Point3 p) {
        x -= p.x; y -= p.y; z -= p.z;
        return *this;
    }
    friend Point3 operator+(Point3 a, Point3 b) { return a += b; }
    friend Point3 operator-(Point3 a, Point3 b) { return a -= b; }
    Point3 &operator*=(i128 k) {
        x *= k; y *= k; z *= k;
        return *this;
    }
    friend Point3 operator*(Point3 p, i128 k) { return p *= k; }
    friend Point3 operator*(i128 k, Point3 p) { return p *= k; }
    friend bool operator==(Point3 a, Point3 b) {
        return a.x == b.x && a.y == b.y && a.z == b.z;
    }
    friend auto operator<=>(Point3 a, Point3 b) = default;
};
bool same_pt(Point3 a, Point3 b) { return a == b; }
```
#### 整数基础函数
---
```cpp
// ===== 点积、叉积、混合积：其他空间判定依赖 =====
i128 dot(Point3 a, Point3 b) {
    return i128(a.x) * b.x + i128(a.y) * b.y + i128(a.z) * b.z;
}
// 返回 i128 三元组，避免把叉积误收窄到 Point3(i64)
array<i128, 3> cross(Point3 a, Point3 b) {
    return {
        i128(a.y) * b.z - i128(a.z) * b.y,
        i128(a.z) * b.x - i128(a.x) * b.z,
        i128(a.x) * b.y - i128(a.y) * b.x
    };
}
i128 triple(Point3 a, Point3 b, Point3 c) {
    auto t = cross(b, c);
    return i128(a.x) * t[0] + i128(a.y) * t[1] + i128(a.z) * t[2];
}
// 用宽整数差分，避免计算两个位置差时提前发生 i64 溢出
i128 dot(Point3 o, Point3 a, Point3 b) {
    return (i128(a.x) - o.x) * (i128(b.x) - o.x)
         + (i128(a.y) - o.y) * (i128(b.y) - o.y)
         + (i128(a.z) - o.z) * (i128(b.z) - o.z);
}
array<i128, 3> cross(Point3 o, Point3 a, Point3 b) {
    i128 x1 = i128(a.x) - o.x, y1 = i128(a.y) - o.y, z1 = i128(a.z) - o.z;
    i128 x2 = i128(b.x) - o.x, y2 = i128(b.y) - o.y, z2 = i128(b.z) - o.z;
    return {y1 * z2 - z1 * y2, z1 * x2 - x1 * z2, x1 * y2 - y1 * x2};
}
// det(b-a,c-a,d-a)，返回有向六倍四面体体积
i128 orient(Point3 a, Point3 b, Point3 c, Point3 d) {
    auto n = cross(a, b, c);
    return n[0] * (i128(d.x) - a.x)
         + n[1] * (i128(d.y) - a.y)
         + n[2] * (i128(d.z) - a.z);
}

i128 norm2(Point3 p) { return dot(p, p); }
i128 dist2(Point3 a, Point3 b) {
    i128 x = i128(a.x) - b.x, y = i128(a.y) - b.y, z = i128(a.z) - b.z;
    return x * x + y * y + z * z;
}

// ===== 共线、共面、垂直 =====
bool collinear(Point3 a, Point3 b, Point3 c) {
    auto t = cross(a, b, c);
    return t[0] == 0 && t[1] == 0 && t[2] == 0;
}
bool coplanar(Point3 a, Point3 b, Point3 c, Point3 d) {
    return orient(a, b, c, d) == 0;
}
bool parallel(Point3 a, Point3 b) {
    auto t = cross(a, b);
    return t[0] == 0 && t[1] == 0 && t[2] == 0;
}
bool perp(Point3 a, Point3 b) { return dot(a, b) == 0; }

// ===== 90 度坐标轴旋转 =====
// 均为沿坐标轴正向观察、按右手法则旋转 +90°
Point3 rotate90x(Point3 p) { return {p.x, -p.z, p.y}; }
Point3 rotate90y(Point3 p) { return {p.z, p.y, -p.x}; }
Point3 rotate90z(Point3 p) { return {-p.y, p.x, p.z}; }
```
#### 浮点结构体
---
以下代码**整体替换整数结构体及整数基础函数**；不必再复制其 `sgn(i128)`、`dot()`、`cross()` 等函数
```cpp
using ld = long double;
constexpr ld EPS{ 1E-12L };
const ld PI = acosl(-1.L);
int sgn(ld x, ld eps = EPS) { return (x > eps) - (x < -eps); }

struct Point3 {
    ld x{ 0 }, y{ 0 }, z{ 0 };
    Point3() = default;
    Point3(ld x_, ld y_, ld z_) : x(x_), y(y_), z(z_) {}
    Point3 operator+() const { return *this; }
    Point3 operator-() const { return {-x, -y, -z}; }
    Point3 &operator+=(Point3 p) {
        x += p.x; y += p.y; z += p.z;
        return *this;
    }
    Point3 &operator-=(Point3 p) {
        x -= p.x; y -= p.y; z -= p.z;
        return *this;
    }
    friend Point3 operator+(Point3 a, Point3 b) { return a += b; }
    friend Point3 operator-(Point3 a, Point3 b) { return a -= b; }
    Point3 &operator*=(ld k) {
        x *= k; y *= k; z *= k;
        return *this;
    }
    Point3 &operator/=(ld k) {
        x /= k; y /= k; z /= k;
        return *this;
    }
    friend Point3 operator*(Point3 p, ld k) { return p *= k; }
    friend Point3 operator*(ld k, Point3 p) { return p *= k; }
    friend Point3 operator/(Point3 p, ld k) { return p /= k; }
    friend bool operator==(Point3 a, Point3 b) {
        return a.x == b.x && a.y == b.y && a.z == b.z;
    }
    friend auto operator<=>(Point3 a, Point3 b) = default;
};

bool same_pt(Point3 a, Point3 b) {
    return sgn(a.x - b.x) == 0
        && sgn(a.y - b.y) == 0 && sgn(a.z - b.z) == 0;
}
```
### 浮点基础函数
---
```cpp
// ===== 点积、叉积、混合积 =====
ld dot(Point3 a, Point3 b) { return a.x*b.x + a.y*b.y + a.z*b.z; }
Point3 cross(Point3 a, Point3 b) {
    return {a.y*b.z - a.z*b.y, a.z*b.x - a.x*b.z, a.x*b.y - a.y*b.x};
}
ld triple(Point3 a, Point3 b, Point3 c) { return dot(a, cross(b, c)); }
ld dot(Point3 o, Point3 a, Point3 b) { return dot(a - o, b - o); }
Point3 cross(Point3 o, Point3 a, Point3 b) { return cross(a - o, b - o); }
ld orient(Point3 a, Point3 b, Point3 c, Point3 d) {
    return triple(b - a, c - a, d - a);
}
ld norm2(Point3 p) { return dot(p, p); }
ld dist2(Point3 a, Point3 b) { return norm2(a - b); }

// ===== 长度、角度与单位化 =====
ld length(Point3 p) { return sqrtl(norm2(p)); }
ld distance(Point3 a, Point3 b) { return length(a - b); }
optional<Point3> unit(Point3 p) {
    ld len = length(p);
    if (len == 0) return nullopt;
    return p / len;
}
// 两个非零向量的无向夹角 [0, PI]
ld vec_ang(Point3 a, Point3 b) {
    return atan2l(length(cross(a, b)), dot(a, b));
}

// ===== 向量投影与对称 =====
// 以下函数的 dir / normal 必须非零
Point3 proj_vec(Point3 p, Point3 dir) {
    return dir * (dot(p, dir) / norm2(dir));
}
Point3 refl_vec(Point3 p, Point3 normal) {
    return p - 2.L * proj_vec(p, normal);
}

// ===== 旋转：只需基础运算与长度，可独立删除 =====
// 右手法则绕坐标轴旋转 ang 弧度
Point3 rotate_x(Point3 p, ld ang) {
    ld c = cosl(ang), s = sinl(ang);
    return {p.x, p.y*c - p.z*s, p.y*s + p.z*c};
}
Point3 rotate_y(Point3 p, ld ang) {
    ld c = cosl(ang), s = sinl(ang);
    return {p.x*c + p.z*s, p.y, -p.x*s + p.z*c};
}
Point3 rotate_z(Point3 p, ld ang) {
    ld c = cosl(ang), s = sinl(ang);
    return {p.x*c - p.y*s, p.x*s + p.y*c, p.z};
}
// Rodrigues：绕经过原点、方向为 axis 的轴旋转
Point3 rotate_axis(Point3 p, Point3 axis, ld ang) {
    ld len = length(axis);
    // 前置条件：axis != 0
    Point3 u = axis / len;
    ld c = cosl(ang), s = sinl(ang);
    return p*c + cross(u, p)*s + u*(dot(u, p)*(1.L - c));
}
// 绕经过 o、方向为 axis 的空间直线旋转
Point3 rotate_about(Point3 p, Point3 o, Point3 axis, ld ang) {
    return o + rotate_axis(p - o, axis, ang);
}

// ===== 空间姿态判定 =====
// 此处仅判定数学上的零向量；大规模数值下应自定相对容差
bool parallel(Point3 a, Point3 b) {
    ld la = length(a), lb = length(b);
    if (la == 0 || lb == 0) return false;
    return length(cross(a / la, b / lb)) <= EPS;
}
bool perp(Point3 a, Point3 b) {
    ld la = length(a), lb = length(b);
    if (la == 0 || lb == 0) return false;
    return fabsl(dot(a / la, b / lb)) <= EPS;
}
bool collinear(Point3 a, Point3 b, Point3 c) {
    Point3 u = b - a, v = c - a;
    if (u == Point3{} || v == Point3{}) return true;
    return parallel(u, v);
}
bool coplanar(Point3 a, Point3 b, Point3 c, Point3 d) {
    Point3 u = b-a, v = c-a, w = d-a;
    ld lu = length(u), lv = length(v), lw = length(w);
    if (lu == 0 || lv == 0 || lw == 0) return true;
    return fabsl(triple(u / lu, v / lv, w / lw)) <= EPS;
}
```
#### 成员与类型
- `Point3(x,y,z)`：构造空间点或向量；原始字典序比较
- 整数版 `cross()` 返回 `array<i128,3>`；浮点版返回 `Point3`
- 浮点版 `unit()` 返回 `optional<Point3>`，零向量返回 `nullopt`
#### 基础接口

| 接口 | 返回值或用途 | 时间复杂度 |
|---|---|---|
| `dot(a,b)` | 点积 | $O(1)$ |
| `cross(a,b)` | 向量叉积，方向满足右手法则 | $O(1)$ |
| `triple(a,b,c)` | 标量三重积 $\mathbf a\cdot(\mathbf b\times\mathbf c)$ | $O(1)$ |
| `dot(o,a,b)` | $\overrightarrow{oa}\cdot\overrightarrow{ob}$ | $O(1)$ |
| `cross(o,a,b)` | $\overrightarrow{oa}\times\overrightarrow{ob}$ | $O(1)$ |
| `orient(a,b,c,d)` | $\det(b-a,c-a,d-a)$，有向六倍四面体体积 | $O(1)$ |
| `norm2(p)` / `dist2(a,b)` | 长度平方 / 距离平方 | $O(1)$ |
| `collinear(a,b,c)` | 三点共线（允许重合） | $O(1)$ |
| `coplanar(a,b,c,d)` | 四点共面（允许退化） | $O(1)$ |
| `parallel(a,b)` / `perp(a,b)` | 非零向量的平行 / 垂直判断 | $O(1)$ |
#### 浮点扩展接口

| 接口 | 用途 | 时间复杂度 |
|---|---|---|
| `length(p)` / `distance(a,b)` | 长度 / 欧氏距离 | $O(1)$ |
| `unit(p)` | 单位化；零向量返回 `nullopt` | $O(1)$ |
| `vec_ang(a,b)` | 两非零向量无向夹角，范围 $[0,\pi]$ | $O(1)$ |
| `proj_vec(p,dir)` | 向量 `p` 在非零 `dir` 上的投影 | $O(1)$ |
| `refl_vec(p,normal)` | 向量 `p` 关于过原点、法向量为 `normal` 的平面的对称向量 | $O(1)$ |
| `rotate_x/y/z(p,ang)` | 绕原点处坐标轴旋转 | $O(1)$ |
| `rotate_axis(p,axis,ang)` | 绕过原点、方向 `axis` 的轴旋转 | $O(1)$ |
| `rotate_about(p,o,axis,ang)` | 绕过 `o`、方向 `axis` 的空间直线旋转 | $O(1)$ |
#### 注意事项
- `orient(a,b,c,d)` 的符号由向量顺序决定；交换 `b,c,d` 中任意两个点，符号反转。`orient / 6` 才是有向四面体体积。
- 三维叉积不应强行用 `Point3<i64>` 储存，结果即使输入坐标适合 `i64`，仍可能超出 `i64`。
- 整数 `Point3` 的 `operator-` 是便捷接口，若坐标差可能超过 `i64`，必须使用直接提升到 `i128` 的 `dot(o,a,b)`、`cross(o,a,b)`、`orient()`、`dist2()` 等接口；即使使用这些接口，`i128` 中间结果仍不可溢出。
- `parallel()` 和 `perp()` 整数版将零向量分别判作与所有向量平行及垂直（代数约定），在几何问题中需自行排除零向量；浮点版遇到零向量返回 `false`。
- 浮点排序必须使用原始坐标比较，不可在比较器中引入 `EPS`。
- `vec_ang()`、`proj_vec()`、`refl_vec()`、`rotate_axis()`、`rotate_about()` 的相应方向向量必须非零；`unit()` 则会明确报告零向量。
- `parallel()` / `coplanar()` 的浮点判定基于归一化方向，适合有限、正常范围的数值；接近退化或坐标量级很大时仍可能受到消去误差影响，不是精确谓词。
- 绕轴旋转角度均为**弧度**；沿旋转轴正向观察时，正角满足右手法则。`rotate_about` 中 `axis` 是方向向量，不是轴上第二个点。
- 旋转通常改变浮点数的末位，不应使用原始 `operator==` 检查旋转后的几何相等。
#### 常见用法
##### 整数叉积与四点共面
```cpp
Point3 a{0, 0, 0}, b{1, 0, 0}, c{0, 1, 0}, d{0, 0, 1};
auto n = cross(a, b, c);    // {0, 0, 1}，array<i128, 3>
i128 vol6 = orient(a, b, c, d); // 1
bool same_plane = coplanar(a, b, c, d); // false
```
##### 浮点角度、单位向量与投影
```cpp
Point3 a{1, 0, 0}, b{0, 1, 0};
ld ang = vec_ang(a, b);         // PI / 2
auto u = unit(Point3{2, 0, 0}); // (1, 0, 0)
Point3 foot = proj_vec(Point3{1, 2, 3}, Point3{0, 0, 1}); // (0,0,3)
```
##### 绕任意空间直线旋转
```cpp
Point3 o{1, 2, 3};
Point3 axis{0, 0, 1};
Point3 p{2, 2, 3};
Point3 q = rotate_about(p, o, axis, PI / 2); // (1,3,3)
cout << fixed << setprecision(12)
     << q.x << ' ' << q.y << ' ' << q.z << '\n';
```
#### 复杂度
所有公开操作均为 $O(1)$ 时间、$O(1)$ 额外空间
