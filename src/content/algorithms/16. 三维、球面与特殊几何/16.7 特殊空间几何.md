### 特殊空间几何
---
**依赖：** 长方体表面最短路仅使用 **3.1 浮点版**的 `Point3`、`distance()`。对角顶点特判仅依赖 `ld`；圆柱和圆锥侧面最短路另外使用全局 `PI`。
这里的**表面最短路**指沿物体外表面运动的最短长度，**不是穿过内部的三维直线距离**。优先使用长方体一般版本；若题目只询问两个空间对角顶点，用后面的短公式即可。
#### 长方体表面任意两点最短路
---
设长方体为 $[0,L]\times[0,W]\times[0,H]$，要求 $L,W,H>0$。输入的两点必须位于**闭合长方体外表面**（包括棱和顶点）
核心做法：枚举 **6 个面的不重复相邻面序列**，沿公共棱依次展开为平面矩形。对展开后的两端点连线，使用凸多边形裁剪检查它是否按顺序经过相应面片；只有整个线段在合法面片上时才计入答案
在凸多面体上，若一条最短路径重复进入同一个凸面，则可以用该面内的直线段替换中间路径，不会更长。因此，只需枚举**不重复面的路径**；本题只有 6 个面，属于常数规模
**面编号：** `0: x=0`，`1: x=L`，`2: y=0`，`3: y=W`，`4: z=0`，`5: z=H`
```cpp
// ===== 长方体表面最短路：仅依赖 3.1 浮点 Point3 =====
struct BoxSurface3 {
    struct P2 {
        ld x{}, y{};
        P2 operator+(P2 b) const { return {x + b.x, y + b.y}; }
        P2 operator-(P2 b) const { return {x - b.x, y - b.y}; }
        P2 operator*(ld k) const { return {x * k, y * k}; }
    };
    struct NetFace { // 一个面的展开图；v[i] 对应该面的第 i 个顶点
        array<P2, 4> v{};
    };
    struct Result {
        ld dist{};
        vector<int> faces; // 2*axis+side：0 x=0, 1 x=L, ..., 5 z=H
    };
    array<ld, 3> d;
    BoxSurface3(ld L, ld W, ld H) : d{L, W, H} {}
    static ld cross2(P2 a, P2 b) { return a.x*b.y - a.y*b.x; }
    static ld dot2(P2 a, P2 b) { return a.x*b.x + a.y*b.y; }
    static ld len2(P2 a) { return sqrtl(dot2(a, a)); }
    static ld coord(Point3 p, int a) {
        return a == 0 ? p.x : a == 1 ? p.y : p.z;
    }
    // 面 2*a+s 固定第 a 坐标为 s*d[a]；另外两轴按循环顺序变化
    static array<int, 2> free_axes(int f) {
        int a = f / 2;
        return { (a + 1) % 3, (a + 2) % 3 };
    }
    array<int, 4> vertices(int f) const {
        int a = f/2, b = (a+1)%3, c = (a+2)%3;
        int o = (f&1) << a;
        return {o, o | (1<<b), o | (1<<b) | (1<<c), o | (1<<c)};
    }
    NetFace initial(int f) const {
        auto ax = free_axes(f);
        return {{{{0, 0}, {d[ax[0]], 0}, {d[ax[0]], d[ax[1]]}, {0, d[ax[1]]}}}};
    }
    bool contains(Point3 p) const {
        for (int a = 0; a < 3; ++ a) {
            ld tol = 1E-12L * d[a];
            if (coord(p,a) < -tol || coord(p,a) > d[a]+tol) return false;
        }
        for (int a = 0; a < 3; ++ a) {
            ld tol = 1E-12L * d[a];
            if (fabsl(coord(p,a)) <= tol || fabsl(coord(p,a)-d[a]) <= tol) return true;
        }
        return false;
    }
    int faces_of(Point3 p) const {
        int mask = 0;
        for (int a = 0; a < 3; ++ a) {
            ld tol = 1E-12L * d[a];
            if (fabsl(coord(p,a)) <= tol) mask |= 1 << (2*a);
            if (fabsl(coord(p,a)-d[a]) <= tol) mask |= 1 << (2*a+1);
        }
        return mask;
    }
    P2 locate(Point3 p, int f, const NetFace &net) const {
        auto [b,c] = free_axes(f);
        return net.v[0] + (net.v[1] - net.v[0]) * (coord(p,b)/d[b])
                        + (net.v[3] - net.v[0]) * (coord(p,c)/d[c]);
    }
    // 从当前面 cur 沿公共棱展开到相邻面 nxt；要求 cur/2 != nxt/2
    NetFace unfold(int cur, int nxt, const NetFace &nf) const {
        auto u = vertices(cur), v = vertices(nxt);
        int j = -1;
        for (int i = 0; i < 4; ++ i) {
            int a = u[i], b = u[(i+1)%4];
            if (find(v.begin(), v.end(), a) != v.end() &&
                find(v.begin(), v.end(), b) != v.end()) { j = i; break; }
        }
        assert(j != -1);
        int id1 = u[j], id2 = u[(j+1)%4];
        P2 A = nf.v[j], B = nf.v[(j+1)%4];
        P2 e = B - A;
        P2 out{ -e.y / len2(e), e.x / len2(e) };
        P2 center{};
        for (auto p : nf.v) center = center + p * 0.25L;
        if (cross2(e, center - A) > 0) out = out * -1.L;
        P2 C = A + out * d[cur/2], D = B + out * d[cur/2];
        NetFace ans;
        for (int i = 0; i < 4; ++ i) {
            if (v[i] == id1) ans.v[i] = A;
            if (v[i] == id2) ans.v[i] = B;
            if (v[i] == (id1 ^ (1 << (cur/2)))) ans.v[i] = C;
            if (v[i] == (id2 ^ (1 << (cur/2)))) ans.v[i] = D;
        }
        return ans;
    }
    // 裁剪直线段 P -> Q 落在展开面上的参数区间 [l, r]
    optional<pair<ld,ld>> clip(P2 P, P2 Q, const NetFace &nf) const {
        ld l = 0, r = 1;
        ld shortest = numeric_limits<ld>::infinity();
        for (int i = 0; i < 4; ++ i)
            shortest = min(shortest, len2(nf.v[(i+1)%4] - nf.v[i]));
        ld tol = 1E-13L * shortest;
        ld ori = cross2(nf.v[1]-nf.v[0], nf.v[3]-nf.v[0]) > 0 ? 1.L : -1.L;
        for (int i = 0; i < 4; ++ i) {
            P2 A = nf.v[i], e = nf.v[(i+1)%4] - A;
            ld el = len2(e);
            ld a = ori * cross2(e, P-A) / el;
            ld b = ori * cross2(e, Q-P) / el;
            if (b == 0) {
                if (a < -tol) return nullopt;
            } else if (b > 0) {
                l = max(l, (-tol - a) / b);
            } else {
                r = min(r, (-tol - a) / b);
            }
        }
        if (l > r + 1E-12L) return nullopt;
        return pair{clamp(l, 0.L, 1.L), clamp(r, 0.L, 1.L)};
    }
    bool valid(Point3 p, Point3 q, const vector<int> &seq,
               const array<NetFace,6> &net) const {
        P2 P = locate(p, seq.front(), net[seq.front()]);
        P2 Q = locate(q, seq.back(), net[seq.back()]);
        ld reach = 0;
        for (int f : seq) {
            auto seg = clip(P, Q, net[f]);
            if (!seg || seg->first > reach + 1E-12L) return false;
            reach = max(reach, seg->second);
        }
        return reach >= 1.L - 1E-12L;
    }
    // 输入 p、q 均在闭合长方体表面上；尺寸 L,W,H 均严格正
    optional<Result> shortest(Point3 p, Point3 q) const {
        if (min({d[0],d[1],d[2]}) <= 0 || !contains(p) || !contains(q))
            return nullopt;
        int sm = faces_of(p), tm = faces_of(q);
        if (sm & tm) {
            int f = countr_zero((unsigned)(sm & tm));
            return Result{distance(p, q), {f}};
        }
        Result best{numeric_limits<ld>::infinity(), {}};
        array<NetFace,6> net;
        vector<int> seq;
        function<void(int,int)> dfs = [&](int cur,int used) {
            if (tm & (1 << cur)) {
                if (valid(p, q, seq, net)) {
                    P2 P = locate(p, seq.front(), net[seq.front()]);
                    P2 Q = locate(q, cur, net[cur]);
                    ld len = len2(Q - P);
                    if (len < best.dist) best = {len, seq};
                }
                // 末点也可能在其他面，但不可在最短简单面序列中再次经过当前面
            }
            for (int nxt = 0; nxt < 6; ++ nxt) {
                if (cur/2 == nxt/2 || (used & (1 << nxt))) continue;
                net[nxt] = unfold(cur, nxt, net[cur]);
                seq.push_back(nxt);
                dfs(nxt, used | (1 << nxt));
                seq.pop_back();
            }
        };
        for (int f = 0; f < 6; ++ f) if (sm & (1 << f)) {
            net[f] = initial(f);
            seq = {f};
            dfs(f, 1 << f);
        }
        if (!isfinite(best.dist)) return nullopt;
        return best;
    }
};
```
#### 对角顶点、圆柱侧面和圆锥侧面
---
这些是**互不依赖**的短板子，可以只复制需要的函数。圆柱和圆锥函数使用**弧度**表示方位角；它们只允许在**侧面**行走，**不允许穿过底面或长方体内部**。
```cpp
// ===== 对角顶点特判：可独立删除，不依赖 BoxSurface3 =====
// 长方体两个空间对角顶点在表面上的最短距离
ld box_opposite_vertices_dist(ld L, ld W, ld H) {
    return min({hypotl(L + W, H), hypotl(L + H, W), hypotl(W + H, L)});
}

// ===== 圆柱侧面最短路：可独立删除，不允许经过上下底面 =====
// θ 单位为弧度；输入 r > 0，侧面两点的 z 可为任意高度
ld cylinder_side_dist(ld r, ld z1, ld t1, ld z2, ld t2) {
    ld dy = r * remainderl(t2 - t1, 2.L * PI);
    return hypotl(z2 - z1, dy);
}

// ===== 正圆锥侧面最短路：可独立删除，不允许经过底面 =====
// 圆锥底面 z=0，顶点 z=h；0 <= z1,z2 <= h，r,h > 0
ld cone_side_dist(ld r, ld h, ld z1, ld t1, ld z2, ld t2) {
    ld s = hypotl(r, h);
    ld a = s * (1.L - z1/h), b = s * (1.L - z2/h);
    ld ang = (r/s) * fabsl(remainderl(t2 - t1, 2.L * PI));
    return hypotl(a - b, 2.L * sqrtl(a*b) * sinl(ang/2.L));
}
```
#### 接口说明

| 接口 | 功能与返回值 | 复杂度 |
|---|---|---|
| `BoxSurface3(L,W,H)` | 构造轴对齐长方体，三个尺寸都必须大于零 | $O(1)$ |
| `box.shortest(p,q)` | 求闭表面任意两点最短路；无效尺寸或点不在表面返回 `nullopt` | $O(1)$ |
| `Result::dist` | 最短**表面路径长度**，`ld` | $O(1)$ |
| `Result::faces` | 对应的一条合法最短路径经过的面编号序列 | 最多 6 项 |
| `box_opposite_vertices_dist(L,W,H)` | 两个空间对角顶点的表面最短距离 | $O(1)$ |
| `cylinder_side_dist(r,z1,t1,z2,t2)` | 半径 `r` 的**无限圆柱侧面**两点最短路；`z` 为高度，`t` 为弧度 | $O(1)$ |
| `cone_side_dist(r,h,z1,t1,z2,t2)` | 底面半径 `r`、高度 `h` 的**有限正圆锥侧面**两点最短路；圆锥顶点在 `z=h`，底面在 `z=0` | $O(1)$ |
`Result::faces` 描述**经过哪些面**，并非空间最短路的折点坐标；不要把不同面的展开直线当作三维空间中的直线段。
#### 公式速查
**长方体空间对角顶点：** 设三条棱长为 $L,W,H$，有
$$
D=\min\left\{
\sqrt{(L+W)^2+H^2},\;
\sqrt{(L+H)^2+W^2},\;
\sqrt{(W+H)^2+L^2}
\right\}.
$$
**圆柱侧面：** 将侧面展开为矩形，周向距离取较短绕行方向。记 $\Delta\theta=\operatorname{remainder}(t_2-t_1,2\pi)\in[-\pi,\pi]$，则
$$
D=\sqrt{(z_2-z_1)^2+(r\Delta\theta)^2}.
$$
**正圆锥侧面：** 母线长度 $s=\sqrt{r^2+h^2}$；点的顶点沿母线距离 $a=s(1-z_1/h)$、$b=s(1-z_2/h)$。把圆锥展开为半径 $s$、圆心角 $2\pi r/s$ 的扇形，记
$$
\phi=\frac rs\left|\operatorname{remainder}(t_2-t_1,2\pi)\right|\in[0,\pi],
\qquad D=\sqrt{(a-b)^2+4ab\sin^2(\phi/2)}.
$$
该式含圆锥顶点：某点位于顶点时，对应母线距离为零，答案为另一点的母线距离。
#### 注意事项
- **输入对象约定：** `BoxSurface3` 只支持**轴对齐**的非退化长方体，坐标范围为 `[0,L]×[0,W]×[0,H]`。若长方体只是平移，先将两点减去同一个起点；若有旋转，先转至局部坐标系。
- **边与顶点：** 一点可能同时属于 2 或 3 个面，代码会枚举全部起点面、终点面。若存在一个同时包含两点的面，表面最短距离直接等于三维欧氏距离。
- **不可省略裁剪：** 某个面序列的展开直线可能穿出面片、绕过不应穿越的棱。必须验证它按该序列依次留在所有矩形内，不能仅取各展开图中的直线距离最小值。
- **不支持穿过物体内部：** 长方体、圆柱、圆锥三个表面距离都不是空间中的最短欧氏距离。圆柱/圆锥额外限定**只走侧面**；若允许走上下底面或圆锥底面，需要另行考虑其它路径。
- **浮点判定：** 长方体以各轴尺寸为尺度做表面容差判定，并用面裁剪容差处理棱上交点。几何规模极度悬殊、坐标接近机器精度极限时，浮点算法不保证严格正确。`shortest()` 的 `nullopt` 首先表示输入不满足板子约定；极端数值退化时也可能无法找到有效展开。
- **方位角：** `t1`、`t2` 单位是**弧度**，允许超出 `[0,2π)`；`remainderl` 自动选取等价的短周向差。圆柱要求 `r>0`，圆锥要求 `r>0,h>0,0≤z1,z2≤h`。
- **圆柱侧面边界：** 两点可以在有限圆柱的上下边缘；因为侧面展开域是矩形，连接两点的展开直线仍在侧面内。但是**经过圆柱端盖**的路线不包含在本接口中。
#### 常见用法
##### 长方体任意表面点
```cpp
BoxSurface3 box(2, 3, 5);
Point3 p{0, 1, 2}, q{2, 2, 4};
auto res = box.shortest(p, q);
if (res) {
    cout << fixed << setprecision(12) << res->dist << '\n';
    // res->faces 是一条最短路径依次经过的面编号
}
```
##### 单位正方体两个空间对角顶点
```cpp
BoxSurface3 box(1, 1, 1);
auto res = box.shortest({0, 0, 0}, {1, 1, 1});
// res->dist == sqrtl(5.L)
ld d = box_opposite_vertices_dist(1, 1, 1); // sqrtl(5.L)
```
##### 圆柱、圆锥的侧面最短路
```cpp
ld a = cylinder_side_dist(2, 0, 0, 3, PI);  // sqrtl(9 + 4*PI*PI)
ld b = cone_side_dist(1, 1, 1, 0, 0, PI);  // sqrtl(2)，顶点到任意底圆点
```
#### 复杂度
- `BoxSurface3::shortest`：最多枚举 6 个面的**不重复路径**，6 是常量，因此单次查询 $O(1)$ 时间、$O(1)$ 额外空间；如果把面数作为变量，一般情形的枚举是指数级，**不能当作任意多面体表面最短路算法**。
- 对角顶点、圆柱侧面、圆锥侧面距离函数：均为 $O(1)$ 时间和空间。
