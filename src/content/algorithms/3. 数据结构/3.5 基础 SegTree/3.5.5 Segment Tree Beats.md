### Segment Tree Beats
---
**解决问题：** 维护区间 `chmin/chmax/add` 与区间和、最小值、最大值，通过记录严格次大值、严格次小值和极值出现次数，使部分非线性修改能够直接停在内部节点
#### 原理
考虑修改：
$$a_i\leftarrow\min(a_i,x)$$
若节点最大值为 `mx1`，严格次大值为 `mx2`
当：
$$mx_1\le x$$
时整段都不会变化
当：
$$mx_2<x<mx_1$$
时，只有所有等于 `mx1` 的元素会被修改为 `x`
若最大值出现 `mxc` 次，则：
$$sum\leftarrow sum+(x-mx_1)\cdot mxc$$
因此无需递归到儿子
只有：
$$x\le mx_2$$
时，受影响的值不止最大值这一层，才继续向下递归
`chmax` 完全对称，因此需要同时维护：
```text
mx1 / mx2 / mxc    最大值 / 严格次大值 / 最大值次数
mn1 / mn2 / mnc    最小值 / 严格次小值 / 最小值次数
```

```cpp
struct SegTreeBeats {
    static constexpr i64 INF = 4'000'000'000'000'000'000LL;
    struct Node {
        i64 sum{0}, mx1{-INF}, mx2{-INF};
        i64 mn1{INF}, mn2{INF}, add{0};
        int mxc{0}, mnc{0};
    };
    int n;
    vector<Node> tr;
    explicit SegTreeBeats(const vector<i64> &a) : n(a.size()), tr(4 * n) {
        assert(n > 0);
        build(1, 0, n, a);
    }
    Node merge_node(const Node &a, const Node &b) const {
        Node c;
        c.sum = a.sum + b.sum;
        c.mx1 = max(a.mx1, b.mx1);
        c.mxc = (a.mx1 == c.mx1 ? a.mxc : 0) + (b.mx1 == c.mx1 ? b.mxc : 0);
        c.mx2 = max(a.mx1 == c.mx1 ? a.mx2 : a.mx1, b.mx1 == c.mx1 ? b.mx2 : b.mx1);
        c.mn1 = min(a.mn1, b.mn1);
        c.mnc = (a.mn1 == c.mn1 ? a.mnc : 0) + (b.mn1 == c.mn1 ? b.mnc : 0);
        c.mn2 = min(a.mn1 == c.mn1 ? a.mn2 : a.mn1, b.mn1 == c.mn1 ? b.mn2 : b.mn1);
        return c;
    }
    void pull(int p) { tr[p] = merge_node(tr[p << 1], tr[p << 1 | 1]); }
    void apply_add(int p, i64 v, int len) {
        Node &x{tr[p]};
        x.sum += v * len;
        x.mx1 += v;
        if (x.mx2 != -INF) x.mx2 += v;
        x.mn1 += v;
        if (x.mn2 != INF) x.mn2 += v;
        x.add += v;
    }
    // ===== chmin：若完全不需要区间取 min，可删除本函数及 range_chmin() =====
    void apply_chmin(int p, i64 v) {
        Node &x{tr[p]};
        if (x.mx1 <= v) return;
        x.sum += (v - x.mx1) * x.mxc;
        if (x.mn1 == x.mx1) x.mn1 = v;
        else if (x.mn2 == x.mx1) x.mn2 = v;
        x.mx1 = v;
    }
    // ===== chmax：若完全不需要区间取 max，可删除本函数及 range_chmax() =====
    void apply_chmax(int p, i64 v) {
        Node &x{tr[p]};
        if (x.mn1 >= v) return;
        x.sum += (v - x.mn1) * x.mnc;
        if (x.mx1 == x.mn1) x.mx1 = v;
        else if (x.mx2 == x.mn1) x.mx2 = v;
        x.mn1 = v;
    }
    void push(int p, int l, int r) {
        if (r - l == 1) return;
        int m{(l + r) / 2};
        // 必须先下传区间加。
        if (tr[p].add != 0) {
            apply_add(p << 1, tr[p].add, m - l);
            apply_add(p << 1 | 1, tr[p].add, r - m);
            tr[p].add = 0;
        }
        // 再让儿子的极值满足父节点当前的上下界。
        for (int c : {p << 1, p << 1 | 1}) {
            if (tr[c].mx1 > tr[p].mx1) apply_chmin(c, tr[p].mx1);
            if (tr[c].mn1 < tr[p].mn1) apply_chmax(c, tr[p].mn1);
        }
    }
    void build(int p, int l, int r, const vector<i64> &a) {
        if (r - l == 1) {
            tr[p].sum = tr[p].mx1 = tr[p].mn1 = a[l];
            tr[p].mxc = tr[p].mnc = 1;
            return;
        }
        int m{(l + r) / 2};
        build(p << 1, l, m, a);
        build(p << 1 | 1, m, r, a);
        pull(p);
    }
    // ===== 区间加：若不需要，可删除 range_add() / add() / Node::add =====
    void range_add(int p, int l, int r, int ql, int qr, i64 v) {
        if (qr <= l || r <= ql) return;
        if (ql <= l && r <= qr) {
            apply_add(p, v, r - l);
            return;
        }
        push(p, l, r);
        int m{(l + r) / 2};
        range_add(p << 1, l, m, ql, qr, v);
        range_add(p << 1 | 1, m, r, ql, qr, v);
        pull(p);
    }
    // ===== 区间 chmin =====
    void range_chmin(int p, int l, int r, int ql, int qr, i64 v) {
        if (qr <= l || r <= ql || tr[p].mx1 <= v) return;
        if (ql <= l && r <= qr && tr[p].mx2 < v) {
            apply_chmin(p, v);
            return;
        }
        push(p, l, r);
        int m{(l + r) / 2};
        range_chmin(p << 1, l, m, ql, qr, v);
        range_chmin(p << 1 | 1, m, r, ql, qr, v);
        pull(p);
    }
    // ===== 区间 chmax =====
    void range_chmax(int p, int l, int r, int ql, int qr, i64 v) {
        if (qr <= l || r <= ql || tr[p].mn1 >= v) return;
        if (ql <= l && r <= qr && tr[p].mn2 > v) {
            apply_chmax(p, v);
            return;
        }
        push(p, l, r);
        int m{(l + r) / 2};
        range_chmax(p << 1, l, m, ql, qr, v);
        range_chmax(p << 1 | 1, m, r, ql, qr, v);
        pull(p);
    }
    Node query(int p, int l, int r, int ql, int qr) {
        if (qr <= l || r <= ql) return {};
        if (ql <= l && r <= qr) return tr[p];
        push(p, l, r);
        int m{(l + r) / 2};
        return merge_node(query(p << 1, l, m, ql, qr), query(p << 1 | 1, m, r, ql, qr));
    }
    void add(int l, int r, i64 v) { assert(0 <= l && l <= r && r <= n); range_add(1, 0, n, l, r, v); }
    void chmin(int l, int r, i64 v) { assert(0 <= l && l <= r && r <= n); range_chmin(1, 0, n, l, r, v); }
    void chmax(int l, int r, i64 v) { assert(0 <= l && l <= r && r <= n); range_chmax(1, 0, n, l, r, v); }
    Node query(int l, int r) { assert(0 <= l && l < r && r <= n); return query(1, 0, n, l, r); }
};
```
#### 接口说明
* `add(l,r,v)`：区间加 $v$
* `chmin(l,r,v)`：逐点执行 $a_i\leftarrow\min(a_i,v)$
* `chmax(l,r,v)`：逐点执行 $a_i\leftarrow\max(a_i,v)$
* `query(l,r).sum`：区间和
* `query(l,r).mx1`：区间最大值
* `query(l,r).mn1`：区间最小值
* `mx2/mn2`：严格次大值 / 严格次小值，不存在时为哨兵
* `mxc/mnc`：最大值 / 最小值出现次数
#### 可裁剪版本
若只需要：
```text
add + chmin + sum / max
```
可以删除：
```text
mn1 / mn2 / mnc
apply_chmax()
range_chmax()
chmax()
push() 中向儿子执行 chmax 的部分
```
只保留：
```text
sum / mx1 / mx2 / mxc / add
```
若只需要：
```text
add + chmax + sum / min
```
则完全对称地删除最大值一侧
#### 注意事项
* `chmin` 能整段执行的关键条件是 $mx_2<v<mx_1$
* `chmax` 能整段执行的关键条件是 $mn_1<v<mn_2$
* 若 $mx_1\le v$ 或 $mn_1\ge v$，修改完全无效，应立即剪枝
* `mx2/mn2` 必须是严格次大 / 次小值，而不是允许和第一极值相等的第二个元素
* 下传必须先处理 `add`，再使用父节点当前的 `mx1/mn1` 限制儿子
* `Node{}` 同时作为查询空段单位元，依赖 `mx1=-INF`、`mn1=INF` 等哨兵设计
* Beats 的均摊证明与允许的操作集合紧密相关，不要随意加入新的非线性懒标记
* 所有区间和与 `(v - mx1) * mxc` 等运算必须保证不溢出
#### 常见用法
```cpp
SegTreeBeats seg(a);
seg.chmin(l, r, x);
seg.chmax(l, r, y);
seg.add(l, r, d);
auto info = seg.query(l, r);
cout << info.sum << ' ' << info.mn1 << ' ' << info.mx1 << '\n';
```
#### 复杂度
区间加和普通查询为 $O(\log n)$
`chmin/chmax` 单次最坏可能访问大量节点，但在标准 Beats 操作序列中具有均摊保证，常用保守总复杂度写为：
$$O((n+q)\log^2 n)$$
空间复杂度为 $O(n)$
