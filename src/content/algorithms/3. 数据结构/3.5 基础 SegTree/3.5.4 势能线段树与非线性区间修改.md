### 势能线段树与非线性区间修改
---
**解决问题：** 区间修改无法像普通 Lazy 一样直接作用于节点，但修改会使元素快速进入稳定状态，或在某些节点上能退化成统一的简单修改时，利用剪枝与势能分析获得均摊复杂度。
#### 基本思想
普通懒标记要求区间修改能够根据节点信息直接在 $O(1)$ 内更新节点
对于：
$$a_i\leftarrow f(a_i)$$
如果无法直接更新，可以尝试：
```text
当前节点完全不会变化    -> 直接 return
当前节点能整段等价处理  -> 直接修改节点并 return
否则                    -> 递归左右儿子
```
只要能够证明每次真正向下递归都会显著降低某种势能，就不需要保证每次修改严格 $O(\log n)$，仍然可以获得较好的总复杂度
典型操作：
* 区间开方
* 区间取模
* 区间整除
* 其他会快速收缩值域的非线性修改
---
#### 区间开方、区间和
对于非负整数：
$$a_i\leftarrow\lfloor\sqrt{a_i}\rfloor$$
当区间最大值满足：
$$mx\le 1$$
整段已经稳定，以后继续开方都不会发生变化
```cpp
struct SqrtSegTree {
    struct Node { i64 sum{0}, mx{0}; };
    int n;
    vector<Node> tr;
    explicit SqrtSegTree(const vector<i64> &a) : n(a.size()), tr(4 * n) {
        assert(n > 0);
        build(1, 0, n, a);
    }
    static i64 floor_sqrt(i64 x) {
        i64 y{static_cast<i64>(sqrtl(long double(x)))};
        while ((__int128_t)(y + 1) * (y + 1) <= x) ++ y;
        while ((__int128_t)y * y > x) -- y;
        return y;
    }
    void pull(int p) {
        tr[p].sum = tr[p << 1].sum + tr[p << 1 | 1].sum;
        tr[p].mx = max(tr[p << 1].mx, tr[p << 1 | 1].mx);
    }
    void build(int p, int l, int r, const vector<i64> &a) {
        if (r - l == 1) {
            tr[p] = {a[l], a[l]};
            return;
        }
        int m{(l + r) / 2};
        build(p << 1, l, m, a);
        build(p << 1 | 1, m, r, a);
        pull(p);
    }
    void apply_sqrt(int p, int l, int r, int ql, int qr) {
        if (qr <= l || r <= ql || tr[p].mx <= 1) return;
        if (r - l == 1) {
            tr[p].sum = tr[p].mx = floor_sqrt(tr[p].mx);
            return;
        }
        int m{(l + r) / 2};
        apply_sqrt(p << 1, l, m, ql, qr);
        apply_sqrt(p << 1 | 1, m, r, ql, qr);
        pull(p);
    }
    i64 query(int p, int l, int r, int ql, int qr) const {
        if (qr <= l || r <= ql) return 0;
        if (ql <= l && r <= qr) return tr[p].sum;
        int m{(l + r) / 2};
        return query(p << 1, l, m, ql, qr) + query(p << 1 | 1, m, r, ql, qr);
    }
    void apply_sqrt(int l, int r) { assert(0 <= l && l <= r && r <= n); apply_sqrt(1, 0, n, l, r); }
    i64 query(int l, int r) const { assert(0 <= l && l <= r && r <= n); return query(1, 0, n, l, r); }
};
```
非负整数连续开方后会很快进入 `0/1`，单个元素只有 $O(\log\log A)$ 次有效变化

#### 区间取模、区间和
修改：
$$a_i\leftarrow a_i\bmod v$$
若节点：
$$mx<v$$
则整个节点都不会变化：
```cpp
if (tr[p].mx < v) return;
```
若取模真正改变一个非负整数 $x$，则：
$$x\bmod v<\frac{x}{2}$$
因此一个元素在没有重新赋大值时，最多发生 $O(\log A)$ 次有效取模
```cpp
struct ModSegTree {
    struct Node { i64 sum{0}, mx{0}; };
    int n;
    vector<Node> tr;
    explicit ModSegTree(const vector<i64> &a) : n(a.size()), tr(4 * n) {
        assert(n > 0);
        build(1, 0, n, a);
    }
    void pull(int p) {
        tr[p].sum = tr[p << 1].sum + tr[p << 1 | 1].sum;
        tr[p].mx = max(tr[p << 1].mx, tr[p << 1 | 1].mx);
    }
    void build(int p, int l, int r, const vector<i64> &a) {
        if (r - l == 1) {
            tr[p] = {a[l], a[l]};
            return;
        }
        int m{(l + r) / 2};
        build(p << 1, l, m, a);
        build(p << 1 | 1, m, r, a);
        pull(p);
    }
    void mod(int p, int l, int r, int ql, int qr, i64 v) {
        if (qr <= l || r <= ql || tr[p].mx < v) return;
        if (r - l == 1) {
            tr[p].sum %= v;
            tr[p].mx = tr[p].sum;
            return;
        }
        int m{(l + r) / 2};
        mod(p << 1, l, m, ql, qr, v);
        mod(p << 1 | 1, m, r, ql, qr, v);
        pull(p);
    }
    void set(int p, int l, int r, int x, i64 v) {
        if (r - l == 1) {
            tr[p] = {v, v};
            return;
        }
        int m{(l + r) / 2};
        if (x < m) set(p << 1, l, m, x, v);
        else set(p << 1 | 1, m, r, x, v);
        pull(p);
    }
    i64 query(int p, int l, int r, int ql, int qr) const {
        if (qr <= l || r <= ql) return 0;
        if (ql <= l && r <= qr) return tr[p].sum;
        int m{(l + r) / 2};
        return query(p << 1, l, m, ql, qr) + query(p << 1 | 1, m, r, ql, qr);
    }
    void mod(int l, int r, i64 v) { assert(0 <= l && l <= r && r <= n && v > 0); mod(1, 0, n, l, r, v); }
    void set(int x, i64 v) { assert(0 <= x && x < n && v >= 0); set(1, 0, n, x, v); }
    i64 query(int l, int r) const { assert(0 <= l && l <= r && r <= n); return query(1, 0, n, l, r); }
};
```
`set()` 会重新提高某个位置的势能，但一次重新赋值最多重新产生 $O(\log A)$ 次有效取模机会

#### 非线性修改退化为统一增量
对于：
$$a_i\leftarrow f(a_i)$$
定义：
$$d(x)=f(x)-x$$
如果能够证明当前节点内所有元素的 $d(x)$ 相同，那么整个非线性修改等价于：
$$a_i\leftarrow a_i+d$$
因此可以直接使用普通区间加懒标记
如果 $d(x)$ 在当前值域上单调，只需要检查：
```cpp
i64 dl = f(mn) - mn;
i64 dr = f(mx) - mx;
if (dl == dr) {
    apply_add(p, dl, r - l);
    return;
}
```
因为此时 `[mn,mx]` 中所有值的变化量都相同
##### 开方
$$d(x)=\lfloor\sqrt{x}\rfloor-x$$
可以使用：
```cpp
i64 dl = floor_sqrt(mn) - mn;
i64 dr = floor_sqrt(mx) - mx;
```
##### 整除
对于非负整数和 $v\ge2$：
$$d(x)=\left\lfloor\frac{x}{v}\right\rfloor-x$$
可以使用：
```cpp
i64 dl = mn / v - mn;
i64 dr = mx / v - mx;
```
---
#### 区间加、区间整除
这是“统一变化量”技巧的典型高级应用
支持：
* 区间加
* 区间向下整除
* 区间和、最小值、最大值
要求所有元素始终非负，除数满足 $v\ge2$
```cpp
struct DivSegTree {
    static constexpr i64 INF = 4'000'000'000'000'000'000LL;
    struct Node { i64 sum{0}, mn{INF}, mx{-INF}, add{0}; };
    int n;
    vector<Node> tr;
    explicit DivSegTree(const vector<i64> &a) : n(a.size()), tr(4 * n) {
        assert(n > 0);
        build(1, 0, n, a);
    }
    void pull(int p) {
        tr[p].sum = tr[p << 1].sum + tr[p << 1 | 1].sum;
        tr[p].mn = min(tr[p << 1].mn, tr[p << 1 | 1].mn);
        tr[p].mx = max(tr[p << 1].mx, tr[p << 1 | 1].mx);
    }
    void apply_add(int p, i64 v, int len) {
        tr[p].sum += v * len;
        tr[p].mn += v;
        tr[p].mx += v;
        tr[p].add += v;
    }
    void push(int p, int l, int r) {
        if (tr[p].add == 0 || r - l == 1) return;
        int m{(l + r) / 2};
        apply_add(p << 1, tr[p].add, m - l);
        apply_add(p << 1 | 1, tr[p].add, r - m);
        tr[p].add = 0;
    }
    void build(int p, int l, int r, const vector<i64> &a) {
        if (r - l == 1) {
            tr[p] = {a[l], a[l], a[l], 0};
            return;
        }
        int m{(l + r) / 2};
        build(p << 1, l, m, a);
        build(p << 1 | 1, m, r, a);
        pull(p);
    }
    void add(int p, int l, int r, int ql, int qr, i64 v) {
        if (qr <= l || r <= ql) return;
        if (ql <= l && r <= qr) {
            apply_add(p, v, r - l);
            return;
        }
        push(p, l, r);
        int m{(l + r) / 2};
        add(p << 1, l, m, ql, qr, v);
        add(p << 1 | 1, m, r, ql, qr, v);
        pull(p);
    }
    void divide(int p, int l, int r, int ql, int qr, i64 v) {
        if (qr <= l || r <= ql || tr[p].mx == 0) return;
        if (ql <= l && r <= qr) {
            i64 dl{tr[p].mn / v - tr[p].mn}, dr{tr[p].mx / v - tr[p].mx};
            if (dl == dr) {
                apply_add(p, dl, r - l);
                return;
            }
        }
        push(p, l, r);
        int m{(l + r) / 2};
        divide(p << 1, l, m, ql, qr, v);
        divide(p << 1 | 1, m, r, ql, qr, v);
        pull(p);
    }
    Node query(int p, int l, int r, int ql, int qr) {
        if (qr <= l || r <= ql) return {};
        if (ql <= l && r <= qr) return tr[p];
        push(p, l, r);
        int m{(l + r) / 2};
        Node a{query(p << 1, l, m, ql, qr)};
        Node b{query(p << 1 | 1, m, r, ql, qr)};
        return { a.sum + b.sum, min(a.mn, b.mn), max(a.mx, b.mx), 0 };
    }
    void add(int l, int r, i64 v) { assert(0 <= l && l <= r && r <= n); add(1, 0, n, l, r, v); }
    void divide(int l, int r, i64 v) { assert(0 <= l && l <= r && r <= n && v >= 2); divide(1, 0, n, l, r, v); }
    Node query(int l, int r) { assert(0 <= l && l < r && r <= n); return query(1, 0, n, l, r); }
};
```
#### 接口说明
* `SqrtSegTree::apply_sqrt(l,r)`：逐点执行 $a_i\leftarrow\lfloor\sqrt{a_i}\rfloor$
* `SqrtSegTree::query(l,r)`：区间和
* `ModSegTree::mod(l,r,v)`：逐点执行 `a[i] %= v`
* `ModSegTree::set(x,v)`：单点赋值
* `ModSegTree::query(l,r)`：区间和
* `DivSegTree::add(l,r,v)`：区间加
* `DivSegTree::divide(l,r,v)`：逐点执行 `a[i] /= v`
* `DivSegTree::query(l,r)`：返回区间 `sum/mn/mx`
#### 注意事项
* 三类技巧都依赖具体修改的数学性质，不能把“递归到不能递归为止”当成通用均摊保证
* `sqrt/mod/divide` 这里均按非负整数处理
* `ModSegTree` 要求 `v > 0`
* `DivSegTree` 要求 `v >= 2`，并要求所有区间加之后数组仍保持非负
* C++ 负数整数除法是向零截断，不等于数学意义上的向下取整
* `floor_sqrt()` 使用浮点数初估后再用 `__int128_t` 校正，避免大整数舍入误差
* “稳定节点剪枝”和“统一变化量整段处理”可以同时使用，后者属于进一步优化
* 所有 `sum`、`v * len` 等运算仍需保证不会超出 `i64`
#### 复杂度
区间查询为 $O(\log n)$
区间开方中，单个非负整数最多经历 $O(\log\log A)$ 次有效变化，因此总复杂度可按：
$$O((n\log\log A+q)\log n)$$
估计
区间取模中，每次真正修改都会使元素至少减半；包含单点重新赋值时，常用宽松总复杂度为：
$$O((n+q)\log n\log A)$$
`DivSegTree` 单次最坏可能访问大量节点；它依赖统一变化量剪枝与整除造成的值域收缩获得均摊效率，不应视为普通 Lazy 的严格 $O(\log n)$ 修改
