### 区间加、区间和
```cpp
struct SegAddSum {
    int n;
    vector<i64> tr, lz;
    explicit SegAddSum(int n_) : n(n_), tr(4 * n + 4), lz(4 * n + 4) {}
    explicit SegAddSum(const vector<i64> &a) : SegAddSum(int(a.size())) {
        if (n) build(1, 0, n, a);
    }
private:
    void pull(int p) { tr[p] = tr[p << 1] + tr[p << 1 | 1]; }
    void apply_node(int p, int l, int r, i64 v) {
        tr[p] += v * (r - l);
        lz[p] += v;
    }
    void push(int p, int l, int r) {
        if (lz[p] == 0 || r - l == 1) return;
        int m{(l + r) / 2};
        apply_node(p << 1, l, m, lz[p]);
        apply_node(p << 1 | 1, m, r, lz[p]);
        lz[p] = 0;
    }
    void build(int p, int l, int r, const vector<i64> &a) {
        if (r - l == 1) return void(tr[p] = a[l]);
        int m{(l + r) / 2};
        build(p << 1, l, m, a);
        build(p << 1 | 1, m, r, a);
        pull(p);
    }
    void set(int p, int l, int r, int x, i64 v) {
        if (r - l == 1) {
            tr[p] = v;
            lz[p] = 0;
            return;
        }
        push(p, l, r);
        int m{(l + r) / 2};
        if (x < m) set(p << 1, l, m, x, v);
        else set(p << 1 | 1, m, r, x, v);
        pull(p);
    }
    void add(int p, int l, int r, int ql, int qr, i64 v) {
        if (qr <= l || r <= ql) return;
        if (ql <= l && r <= qr) return apply_node(p, l, r, v);
        push(p, l, r);
        int m{(l + r) / 2};
        add(p << 1, l, m, ql, qr, v);
        add(p << 1 | 1, m, r, ql, qr, v);
        pull(p);
    }
    i64 query(int p, int l, int r, int ql, int qr) {
        if (qr <= l || r <= ql) return 0;
        if (ql <= l && r <= qr) return tr[p];
        push(p, l, r);
        int m{(l + r) / 2};
        return query(p << 1, l, m, ql, qr) + query(p << 1 | 1, m, r, ql, qr);
    }
public:
    void set(int x, i64 v) { set(1, 0, n, x, v); }
    void add(int l, int r, i64 v) { if (l < r) add(1, 0, n, l, r, v); }
    i64 query(int l, int r) { return l < r ? query(1, 0, n, l, r) : 0; }
    i64 all() const { return n ? tr[1] : 0; }
};
```
### 通用线段树
---
默认给出**区间加、区间乘、区间赋值、区间和**。标记统一表示仿射变换
$$x\mapsto mul\cdot x+add$$
因此：
- 区间加 $v$：`{1, v}`
- 区间乘 $v$：`{v, 0}`
- 区间赋值 $v$：`{0, v}`
```cpp
struct Tag {
    i64 mul{1}, add{0};
    static Tag plus(i64 v) { return {1, v}; }
    static Tag times(i64 v) { return {v, 0}; }
    static Tag assign(i64 v) { return {0, v}; }
    bool empty() const { return mul == 1 && add == 0; }
    // 已有变换之后再执行 t
    void apply(const Tag &t) {
        mul *= t.mul;
        add = add * t.mul + t.add;
    }
};
struct Info {
    i64 sum{0};
    Info(i64 x = 0) : sum(x) {}
    void apply(const Tag &t, int l, int r) {
        sum = sum * t.mul + t.add * (r - l);
    }
};
Info operator+(const Info &a, const Info &b) { return Info(a.sum + b.sum); }

struct NoTag {};

template<class Info, class Tag = NoTag>
struct LazySegTree {
    static constexpr bool has_lazy = !is_same_v<Tag, NoTag>;
    int n;
    vector<Info> tr;
    vector<Tag> lz;
    LazySegTree(int n_, const Info &value = Info()) : LazySegTree(vector<Info>(n_, value)) {}
    template<class T>
    explicit LazySegTree(const vector<T> &a) : n(a.size()), tr(4 * n + 4), lz(4 * n + 4) {
        if (n) build(1, 0, n, a);
    }
private:
    template<class T>
    void build(int p, int l, int r, const vector<T> &a) {
        if (r - l == 1) return void(tr[p] = Info(a[l]));
        int mid{ l + r >> 1 };
        build(p << 1, l, mid, a);
        build(p << 1 | 1, mid, r, a);
        pull(p);
    }
    void pull(int p) { tr[p] = tr[p << 1] + tr[p << 1 | 1]; }
    void apply_node(int p, int l, int r, const Tag &tag) {
        if constexpr (has_lazy) {
            tr[p].apply(tag, l, r);
            lz[p].apply(tag);
        }
    }
    void push(int p, int l, int r) {
        if constexpr (has_lazy) {
            if (lz[p].empty() || r - l == 1) return;
            int mid{ l + r >> 1 };
            apply_node(p << 1, l, mid, lz[p]);
            apply_node(p << 1 | 1, mid, r, lz[p]);
            lz[p] = Tag();
        }
    }
    void set(int p, int l, int r, int x, const Info &value) {
        if (r - l == 1) {
            tr[p] = value;
            if constexpr (has_lazy) lz[p] = Tag();
            return;
        }
        push(p, l, r);
        int mid{ l + r >> 1 };
        if (x < mid) set(p << 1, l, mid, x, value);
        else set(p << 1 | 1, mid, r, x, value);
        pull(p);
    }
    void apply(int p, int l, int r, int ql, int qr, const Tag &tag) {
        if (qr <= l || r <= ql) return;
        if (ql <= l && r <= qr) return apply_node(p, l, r, tag);
        push(p, l, r);
        int mid{ l + r >> 1 };
        apply(p << 1, l, mid, ql, qr, tag);
        apply(p << 1 | 1, mid, r, ql, qr, tag);
        pull(p);
    }
    Info query(int p, int l, int r, int ql, int qr) {
        if (qr <= l || r <= ql) return Info();
        if (ql <= l && r <= qr) return tr[p];
        push(p, l, r);
        int mid{ l + r >> 1 };
        return query(p << 1, l, mid, ql, qr) + query(p << 1 | 1, mid, r, ql, qr);
    }
    template<class F>
    int find_first(int p, int l, int r, int ql, int qr, F &pred) {
        if (qr <= l || r <= ql) return -1;
        if (ql <= l && r <= qr && !pred(tr[p])) return -1;
        if (r - l == 1) return l;
        push(p, l, r);
        int mid{ l + r >> 1 };
        int res = find_first(p << 1, l, mid, ql, qr, pred);
        if (res == -1) res = find_first(p << 1 | 1, mid, r, ql, qr, pred);
        return res;
    }
    template<class F>
    int find_last(int p, int l, int r, int ql, int qr, F &pred) {
        if (qr <= l || r <= ql) return -1;
        if (ql <= l && r <= qr && !pred(tr[p])) return -1;
        if (r - l == 1) return l;
        push(p, l, r);
        int mid{ l + r >> 1 };
        int res = find_last(p << 1 | 1, mid, r, ql, qr, pred);
        if (res == -1) res = find_last(p << 1, l, mid, ql, qr, pred);
        return res;
    }
public:
    void set(int x, const Info &value) { set(1, 0, n, x, value); }
    template<class U = Tag>
    enable_if_t<!is_same_v<U, NoTag>> apply(int l, int r, const U &tag) {
        if (l < r) apply(1, 0, n, l, r, tag);
    }
    Info query(int l, int r) { return l < r ? query(1, 0, n, l, r) : Info(); }
    Info all() const { return n ? tr[1] : Info(); }
    template<class F>
    int find_first(int l, int r, F pred) { return l < r ? find_first(1, 0, n, l, r, pred) : -1; }
    template<class F>
    int find_last(int l, int r, F pred) { return l < r ? find_last(1, 0, n, l, r, pred) : -1; }
};
```
#### 短版常见修改
如果题目仍然是**区间加**，通常无需抄泛型版，只修改 `SegAddSum` 的 `tr`、`pull()`、`apply_node()` 和 `query()` 即可

| 维护信息               | `pull()`                   | 区间加如何作用                     | 能否直接修改短版 |
| ------------------ | -------------------------- | --------------------------- | -------- |
| 区间和                | 两儿子求和                      | `sum += v * len`            | 是        |
| 区间最大值              | `max`                      | `mx += v`                   | 是        |
| 区间最小值              | `min`                      | `mn += v`                   | 是        |
| 和 + 最大/最小          | 分别合并                       | `sum += v*len`，`mx/mn += v` | 是        |
| 最大值及出现次数           | 最大值合并，相等次数相加               | `mx += v`，次数不变              | 是        |
| 平方和                | 分别维护 $\sum a_i,\sum a_i^2$ | 用二项式展开                      | 是        |
| 下标加权和 $\sum i a_i$ | 两儿子求和                      | 加上 $v\sum i$                | 是        |
| GCD                | `gcd`                      | 不能由原 GCD 推出                 | **否**    |
| 最大子段和              | 四元组区间合并                    | 区间加后最优子段长度未知                | **否**    |
| AND / OR / XOR     | 对应位运算                      | 普通加法会产生进位                   | **否**    |
| 大于某值的个数            | 次数合并                       | 加法会改变值域分布                   | **否**    |
##### 区间加、区间最大值
只改三处：
```cpp
void pull(int p) { tr[p] = max(tr[p << 1], tr[p << 1 | 1]); }
void apply_node(int p, int l, int r, i64 v) {
    tr[p] += v;
    lz[p] += v;
}
i64 query(int p, int l, int r, int ql, int qr) {
    if (qr <= l || r <= ql) return -INF;
    if (ql <= l && r <= qr) return tr[p];
    push(p, l, r);
    int m{(l + r) / 2};
    return max(query(p << 1, l, m, ql, qr), query(p << 1 | 1, m, r, ql, qr));
}
```
区间最小值同理，把 `max` / `-INF` 改为 `min` / `INF`
##### 区间加，同时维护和、最小值、最大值
把 `tr` 改成：
```cpp
struct Node {
    i64 sum{0}, mn{INF}, mx{-INF};
    Node() = default;
    Node(i64 x) : sum(x), mn(x), mx(x) {}
};
vector<Node> tr;
```
修改：
```cpp
Node merge(const Node &a, const Node &b) {
    return { a.sum + b.sum, min(a.mn, b.mn), max(a.mx, b.mx) };
}
void pull(int p) { tr[p] = merge(tr[p << 1], tr[p << 1 | 1]); }
void apply_node(int p, int l, int r, i64 v) {
    tr[p].sum += v * (r - l);
    tr[p].mn += v;
    tr[p].mx += v;
    lz[p] += v;
}
```
叶子改为：
```cpp
tr[p] = Node(a[l]);
```
`query()` 返回 `Node`，区间外返回 `Node()`，合并改为 `merge()`
##### 区间加、区间最大值及最大值出现次数
```cpp
struct Node {
    i64 mx{-INF};
    int cnt{0};
    Node() = default;
    Node(i64 x) : mx(x), cnt(1) {}
};
Node merge(const Node &a, const Node &b) {
    if (a.mx > b.mx) return a;
    if (a.mx < b.mx) return b;
    return {a.mx, a.cnt + b.cnt};
}
```
若使用聚合初始化，需要允许：
```cpp
Node(i64 mx_, int cnt_) : mx(mx_), cnt(cnt_) {}
```
此时：
```cpp
void pull(int p) { tr[p] = merge(tr[p << 1], tr[p << 1 | 1]); }
void apply_node(int p, int l, int r, i64 v) {
    tr[p].mx += v;
    lz[p] += v;
}
```
区间加不会改变同一区间内各元素的大小关系，因此最大值出现次数不变
##### 区间加、区间和与平方和
维护
$$S_1=\sum a_i,\quad S_2=\sum a_i^2$$
所有数加上 $v$ 后：
$$S_2'=S_2+2vS_1+nv^2$$
```cpp
struct Node { i64 sum{0}, sq{0}; };
void pull(int p) {
    tr[p].sum = tr[p << 1].sum + tr[p << 1 | 1].sum;
    tr[p].sq = tr[p << 1].sq + tr[p << 1 | 1].sq;
}
void apply_node(int p, int l, int r, i64 v) {
    i64 len{r - l};
    tr[p].sq += 2 * v * tr[p].sum + v * v * len;
    tr[p].sum += v * len;
    lz[p] += v;
}
```
注意必须先更新 `sq`，再更新 `sum`
更高次幂和同理使用
$$(a+v)^k=\sum_{j=0}^{k}\binom{k}{j}a^jv^{k-j}$$
##### 区间加、下标加权和
维护
$$S=\sum a_i,\qquad W=\sum i\cdot a_i$$
由于内部下标为 `0base`，区间 `[l,r)` 的下标和为
$$\sum_{i=l}^{r-1}i=\frac{(l+r-1)(r-l)}2$$
```cpp
struct Node { i64 sum{0}, weighted{0}; };
void apply_node(int p, int l, int r, i64 v) {
    i64 len{r - l};
    i64 index_sum{i64(l + r - 1) * len / 2};
    tr[p].sum += v * len;
    tr[p].weighted += v * index_sum;
    lz[p] += v;
}
```
这也是通用版将 `Info::apply()` 传入 `l,r` 而不只传 `len` 的原因

##### 区间 GCD

##### 最大子段和
参考 [[3.5.2 非递归线段树（ZKW 风格）]] 中的 *区间最大子段和* 
##### AND / OR / XOR
普通“区间加”会发生二进制进位，因此仅保存区间 `AND/OR/XOR` 无法 $O(1)$ 推出加法后的结果
但如果修改本身也是 XOR，则是另一套懒标记。例如维护区间 XOR：
```cpp
void apply_node(int p, int l, int r, i64 v) {
    if ((r - l) & 1) tr[p] ^= v;
    lz[p] ^= v;
}
```
因为偶数个相同的 `v` 异或后抵消
#### 接口说明
- `LazySegTree(a)`：由能转换为 `Info` 的数组建树
- `LazySegTree(n,value)`：建立长度为 `n`、每个位置均为 `value` 的树
- `set(x,value)`：单点赋值
- `apply(l,r,tag)`：对 `[l,r)` 追加标记
- `query(l,r)`：查询 `[l,r)` 的聚合信息
- `all()`：查询整个数组的聚合信息
- `find_first(l,r,pred)`：从左向右寻找第一个可能满足 `pred` 的位置
- `find_last(l,r,pred)`：从右向左寻找第一个可能满足 `pred` 的位置
### 注意事项
- 全部下标为 `0base`，区间统一为 `[l,r)`
- `Info()` 必须是区间合并的单位元
- `Tag()` 必须表示空标记
- `operator+` 必须按**左区间、右区间**的顺序合并；最大子段和等信息不可交换顺序
- `Tag::apply(t)` 表示“已有标记之后再执行 `t`”，复合标记时顺序不能写反
- 默认仿射 `Tag` 只保证当前 `sum` 信息正确；若同时维护 `min/max`，特别是允许负数乘法时，需要重新推导 `Info::apply()`
- 若题目取模，应在 `Tag::apply()`、`Info::apply()` 和 `operator+` 中统一处理
- 先估算 `sum * mul`、`v * v * len` 等中间结果是否溢出；必要时使用 `i128` 或取模类
- `find_first/find_last` 中，`pred(info)==false` 必须保证该整段不存在答案，例如维护 `mx` 时寻找第一个 `a[i]>=x` 可使用 `info.mx>=x`
- 不要因为“线段树能维护某信息”就认为它也能支持任意懒修改；关键是必须能够仅根据当前节点信息和标记，在 $O(1)$ 内得到修改后的节点信息
#### 常见用法
##### 查询第一个不小于 `x` 的位置
需要让 `Info` 维护区间最大值 `mx`：
```cpp
int p = seg.find_first(l, r, [&](const Info &info) {
    return info.mx >= x;
});
```
若不存在返回 `-1`
