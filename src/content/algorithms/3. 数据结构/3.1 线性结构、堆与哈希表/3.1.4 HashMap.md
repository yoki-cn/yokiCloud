### 开放寻址高性能 HashMap
---
```cpp
template<class Key, class Value, class Hash = CustomHash>
struct FlatHashMap {
private:
    struct Slot {
        Key key{};
        Value val{};
        unsigned char state{ 0 }; // 0：空，1：有效，2：墓碑
    };
    Hash hasher;
    vector<Slot> t;
    int siz{ 0 }, used{ 0 }; // used = 有效元素 + 墓碑
    // ===== 定位与扩容：以下代码被所有接口依赖，不建议删除 =====
    int locate(const Key &key) const {
        int mask{ int(t.size()) - 1 }, p{ int(hasher(key) & size_t(mask)) };
        while (t[p].state != 0) {
            if (t[p].state == 1 && t[p].key == key) return p;
            p = (p + 1) & mask;
        }
        return -1;
    }
    int insertion_pos(const Key &key) const {
        int mask{ int(t.size()) - 1 }, p{ int(hasher(key) & size_t(mask)) }, dead{ -1 };
        while (t[p].state != 0) {
            if (t[p].state == 1 && t[p].key == key) return p;
            if (t[p].state == 2 && dead == -1) dead = p;
            p = (p + 1) & mask;
        }
        return dead == -1 ? p : dead;
    }
    // 仅供 rehash 使用，新表不存在重复键和墓碑
    void place(Key &&key, Value &&val) {
        int mask{ int(t.size()) - 1 }, p{ int(hasher(key) & size_t(mask)) };
        while (t[p].state == 1) p = (p + 1) & mask;
        t[p].key = move(key);
        t[p].val = move(val);
        t[p].state = 1;
        ++ siz;
        ++ used;
    }
    void rehash(int cap) {
        vector<Slot> old{ move(t) };
        t = vector<Slot>(cap);
        siz = used = 0;
        for (auto &s : old) if (s.state == 1) {
            place(move(s.key), move(s.val));
        }
    }
    // used 达到约 70% 时：墓碑较多则原地清理，否则扩容
    bool prepare_insert() {
        int cap{ int(t.size()) };
        if (1LL * (used + 1) * 10 < 1LL * cap * 7) return false;
        if (1LL * (siz + 1) * 10 < 1LL * cap * 7) rehash(cap);
        else rehash(cap << 1);
        return true;
    }
public:
    explicit FlatHashMap(int expected = 0) {
        int cap{ 8 };
        while (1LL * cap < 2LL * expected) cap <<= 1;
        t.resize(cap);
    }
    // ===== 基础接口：查询 / 插入 / 下标访问 =====
    Value *find(const Key &key) {
        int p{ locate(key) };
        return p == -1 ? nullptr : &t[p].val;
    }
    const Value *find(const Key &key) const {
        int p{ locate(key) };
        return p == -1 ? nullptr : &t[p].val;
    }
    bool contains(const Key &key) const { return locate(key) != -1; }
    pair<Value *, bool> insert(Key key, Value val = Value()) {
        int p{ insertion_pos(key) };
        if (t[p].state == 1) return { &t[p].val, false };
        if (prepare_insert()) p = insertion_pos(key);
        if (t[p].state == 0) ++ used;
        t[p].key = move(key);
        t[p].val = move(val);
        t[p].state = 1;
        ++ siz;
        return { &t[p].val, true };
    }
    Value &operator[](const Key &key) { return *insert(key).first; }
    int size() const { return siz; }
    bool empty() const { return siz == 0; }
    // ===== 删除接口：若不需要删除，以下函数可直接删除 =====
    bool erase(const Key &key) {
        int p{ locate(key) };
        if (p == -1) return false;
        t[p].state = 2;
        -- siz;
        // 墓碑达到容量的 1/4 时原地重建，避免查询链过长
        if (1LL * (used - siz) * 4 >= int(t.size())) rehash(int(t.size()));
        return true;
    }
};
```
#### 接口说明
- `FlatHashMap(expected)`：构造哈希表；初始容量取不小于 `2 * expected` 的二次幂，最小为 `8`，构造 $O(C)$
- `find(key)`：存在时返回对应值的指针，否则返回 `nullptr`，期望 $O(1)$
- `contains(key)`：判断键是否存在，期望 $O(1)$
- `insert(key, val)`：键不存在时插入，返回 `{值指针, true}`；已存在时不修改原值并返回 `{原值指针, false}`，期望均摊 $O(1)$
- `operator[](key)`：返回对应值的引用；键不存在时插入 `Value{}`，期望均摊 $O(1)$
- `erase(key)`：删除键，存在时返回 `true`，否则返回 `false`，期望均摊 $O(1)$
- `size()` / `empty()`：元素数量与判空，$O(1)$
#### 注意事项
- 容量始终为二的幂，通过 `hash & (capacity - 1)` 定位，因此 `Hash` 的**低位必须具有良好的分布**；默认使用前文的 `CustomHash`
- `Key` 必须支持 `operator==`；当前实现还要求 `Key`、`Value` 可默认构造并可移动赋值，最适合整数、编码状态和其他轻量类型
- `Slot` 会为整个容量预先构造 `Key` 和 `Value`，因此不建议用很大的 `string`、`vector` 或其他重对象作为键值
- `state = 0/1/2` 分别表示空槽、有效元素、墓碑；查询遇到墓碑不能停止
- `used` 同时统计有效元素和墓碑。`used` 达到约 $70%$ 时扩容或清理墓碑，以保证线性探测不会形成过长的查询链
    
- `erase()` 产生墓碑；墓碑达到容量的 $1/4$ 时会按原容量重新哈希，不会缩小容量
- `insert()` 和 `erase()` 都可能触发 `rehash()`，此前通过 `find()`、`insert()` 或 `operator[]` 获得的值指针、引用会失效
- `insert(key, val)` 不会覆盖已经存在的值；需要修改时使用返回的指针或 `operator[]`
- 已知最大元素数量时应尽量传入 `expected`，通常可以避免运行过程中扩容
- 哈希表只保证期望复杂度；若哈希函数产生大量冲突，最坏仍会退化为线性时间
#### 常见用法
##### 计数
```cpp
FlatHashMap<u64, int> mp(n);
for (u64 x : a) ++ mp[x];
if (int *p = mp.find(x)) {
    cout << *p << '\n';
}
```
##### 判断是否存在并插入
```cpp
auto [p, inserted] = mp.insert(x, value);
if (!inserted) {
    // x 原本已经存在，*p 为原值
}
```
##### 自定义结构体键
```cpp
struct GridKey {
    int x{}, y{};
    friend bool operator==(const GridKey &a, const GridKey &b) {
        return a.x == b.x && a.y == b.y;
    }
};
struct GridKeyHash {
    size_t operator()(const GridKey &p) const {
        u64 x = u64(uint32_t(p.x)) << 32 | uint32_t(p.y);
        return CustomHash{}(x);
    }
};
FlatHashMap<GridKey, int, GridKeyHash> mp(n);
++ mp[{x, y}];
```
