### SWAG（滑动窗口聚合）
---
```cpp
template<class T>
struct GcdMerge {
    T operator()(const T &a, const T &b) const { return gcd(a, b); }
};
template<class T = int, class Merge = GcdMerge<T>>
struct SWAG {
    struct Node { T value, prod; };
    T identity;
    Merge merge;
    vector<Node> left, right;
    explicit SWAG(T identity_ = T{}, Merge merge_ = Merge()) : identity(identity_), merge(merge_) {}
    void push(const T &value) {
        T prod = right.empty() ? value : merge(right.back().prod, value);
        right.push_back({value, prod});
    }
    void pop() {
        if (left.empty()) {
            while (!right.empty()) {
                T value = right.back().value, prod = left.empty() ? value : merge(value, left.back().prod);
                right.pop_back();
                left.push_back({ value, prod });
            }
        }
        left.pop_back();
    }
    T query() const {
        T result = identity;
        if (!left.empty()) result = merge(result, left.back().prod);
        if (!right.empty()) result = merge(result, right.back().prod);
        return result;
    }
    bool empty() const { return left.empty() && right.empty(); }
    int size() const { return int(left.size() + right.size()); }
};
```
#### 修改为矩阵乘法
设队列从前到后为
$$A1,A2,…,Ak,A_1,A_2,\ldots,A_k$$
则 `query()` 返回
$$A1A2⋯Ak.A_1A_2\cdots A_k$$
只需要提供矩阵乘法 `Merge`，并将单位元设为单位矩阵：
```cpp
template<class T>
struct Multiply {
    T operator()(const T &a, const T &b) const { return a * b; }
};
Matrix I = Matrix::identity(n);
SWAG<Matrix, Multiply<Matrix>> q(I);
q.push(A);
q.push(B);
q.push(C);
Matrix prod = q.query(); // A * B * C
q.pop();
prod = q.query();        // B * C
```
