### 基础 BIT2D：单点加、矩形和
---
```cpp
template<class T>
struct BIT2D {
    int n, m;
    vector<vector<T>> t;
    BIT2D(int n_ = 0, int m_ = 0) : n(n_), m(m_), t(n + 1, vector<T>(m + 1)) {}
    explicit BIT2D(const vector<vector<T>> &a)
        : n(a.size()), m(n ? a[0].size() : 0), t(n + 1, vector<T>(m + 1)) {
        for (int i = 1; i <= n; ++ i) {
            for (int j = 1; j <= m; ++ j) t[i][j] = a[i - 1][j - 1];
        }
        for (int i = 1; i <= n; ++ i) {
            for (int j = 1; j <= m; ++ j) {
                int k{ j + (j & -j) };
                if (k <= m) t[i][k] += t[i][j];
            }
        }
        for (int i = 1; i <= n; ++ i) {
            int k{ i + (i & -i) };
            if (k <= n) {
                for (int j = 1; j <= m; ++ j) t[k][j] += t[i][j];
            }
        }
    }
    void add(int x, int y, const T &v) {
        for (++ x; x <= n; x += x & -x) {
            for (int j = y + 1; j <= m; j += j & -j) t[x][j] += v;
        }
    }
    T prefix(int x, int y) const {
        T res{};
        for (; x > 0; x -= x & -x) {
            for (int j = y; j > 0; j -= j & -j) res += t[x][j];
        }
        return res;
    }
    T query(int x1, int y1, int x2, int y2) const {
        return prefix(x2, y2) - prefix(x1, y2)
             - prefix(x2, y1) + prefix(x1, y1);
    }
};
```
### 前缀最大值 BIT2D
---
```cpp
template<class T>
struct MaxBIT2D {
    int n, m;
    T e;
    vector<vector<T>> t;
    MaxBIT2D(int n_ = 0, int m_ = 0, T e_ = numeric_limits<T>::lowest())
        : n(n_), m(m_), e(e_), t(n + 1, vector<T>(m + 1, e)) {}
    void apply(int x, int y, const T &v) {
        for (++ x; x <= n; x += x & -x) {
            for (int j = y + 1; j <= m; j += j & -j) t[x][j] = max(t[x][j], v);
        }
    }
    T prefix(int x, int y) const {
        T res{ e };
        for (; x > 0; x -= x & -x) {
            for (int j = y; j > 0; j -= j & -j) res = max(res, t[x][j]);
        }
        return res;
    }
};
```
#### 修改为其他前缀运算
只需修改单位元 `e` 和两处聚合运算
`e` 必须满足：
$$\operatorname{op}(e,a)=a$$
常见修改：

| 运算  | `e`                           | 聚合          |
| --- | ----------------------------- | ----------- |
| 最大值 | `numeric_limits<T>::lowest()` | `max(a, b)` |
| 最小值 | `numeric_limits<T>::max()`    | `min(a, b)` |
| GCD | `T{}`，即 `0`                   | `gcd(a, b)` |
| 按位或 | `T{}`，即 `0`                   | `a \| b`    |
| 按位与 | `~T{}`                        | `a & b`     |
### 矩形加、单点查询
---
使用二维差分。为了允许在 `x = n` 或 `y = m` 处加入右边界/下边界差分，内部 BIT 开到 `(n + 1) × (m + 1)`
```cpp
template<class T>
struct RectAddPoint {
    BIT2D<T> bit;
private:
    void corner(int x, int y, const T &v) { bit.add(x, y, v); }
public:
    RectAddPoint(int n, int m) : bit(n + 1, m + 1) {}
    void add(int x1, int y1, int x2, int y2, const T &v) {
        corner(x1, y1, v);
        corner(x1, y2, T{} - v);
        corner(x2, y1, T{} - v);
        corner(x2, y2, v);
    }
    T get(int x, int y) const { return bit.prefix(x + 1, y + 1); }
};
```
### 矩形加、矩形求和
---
二维差分中，前缀矩形和为：
$$\sum_{i<x,\ j<y}d_{i,j}(x-i)(y-j)$$
展开后分别维护：
$$d,\quad i d,\quad j d,\quad ijd$$
因此需要四棵二维 BIT
```cpp
template<class T>
struct RectAddRectSum {
    BIT2D<T> b00, b10, b01, b11;
private:
    void corner(int x, int y, const T &v) {
        b00.add(x, y, v);
        b10.add(x, y, v * T(x));
        b01.add(x, y, v * T(y));
        b11.add(x, y, v * T(x) * T(y));
    }
public:
    RectAddRectSum(int n, int m) : b00(n + 1, m + 1), b10(n + 1, m + 1),
          b01(n + 1, m + 1), b11(n + 1, m + 1) {}
    void add(int x1, int y1, int x2, int y2, const T &v) {
        corner(x1, y1, v);
        corner(x1, y2, T{} - v);
        corner(x2, y1, T{} - v);
        corner(x2, y2, v);
    }
    T prefix(int x, int y) const {
        return b00.prefix(x, y) * T(x) * T(y) - b10.prefix(x, y) * T(y)
             - b01.prefix(x, y) * T(x) + b11.prefix(x, y);
    }
    T query(int x1, int y1, int x2, int y2) const {
        return prefix(x2, y2) - prefix(x1, y2) - prefix(x2, y1) + prefix(x1, y1);
    }
};
```
### 注意事项
- 所有坐标均为 `0base`
- 所有矩形均使用半开区间 `[x1, x2) × [y1, y2)`
- `prefix(x, y)` 中的 `x, y` 表示矩形长度，不包含坐标 `x` 和 `y`
- 所有非初始化操作复杂度均为 $\log{n}\log{m}$
