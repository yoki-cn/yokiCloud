### 基础 BIT
---
```cpp
template<class T>
struct BIT {
    int n;
    vector<T> t;
    explicit BIT(int n_ = 0) : n(n_), t(n + 1) {}
    explicit BIT(const vector<T> &a) : BIT(int(a.size())) {
        for (int i = 1; i <= n; ++ i) {
            t[i] += a[i - 1];
            int j{ i + (i & -i) };
            if (j <= n) t[j] += t[i];
        }
    }
    void add(int x, const T &v) { for (++ x; x <= n; x += x & -x) t[x] += v; }
    T sum(int r) const {
        T res{};
        for (; r > 0; r -= r & -r) res += t[r];
        return res;
    }
    T query(int l, int r) const { return sum(r) - sum(l); }
    // 返回最大的 r，满足 sum(r) <= limit，O(log_n)
    // 要求所有权值非负
    int max_right(const T &limit) const {
        int x{ 0 }, step{ 1 };
        T cur{};
        while (step <= n / 2) step <<= 1;
        for (; step; step >>= 1) if (x + step <= n && cur + t[x + step] <= limit) {
            cur += t[x += step];
        }
        return x;
    }
    // 权值 BIT 中返回第 k(从 1 开始) 个元素所在的 0base 下标，O(log_n)
    int kth(const T &k) const { return max_right(k - T(1)); }
};
```
### 区间加、区间求和 BIT
---
使用差分数组和两棵基础 `BIT`
```cpp
template<class T>
struct RangeBIT {
    int n;
    BIT<T> b1, b2;
    explicit RangeBIT(int n_ = 0) : n(n_), b1(n + 1), b2(n + 1) {}
    explicit RangeBIT(const vector<T> &a) : n(int(a.size())) {
        vector<T> d(n + 1), w(n + 1);
        for (int i = 0; i < n; ++ i) d[i] = a[i] - (i ? a[i - 1] : T{});
        if (n) d[n] = T{} - a.back();
        for (int i = 0; i <= n; ++ i) w[i] = d[i] * T(i);
        b1 = BIT<T>(d);
        b2 = BIT<T>(w);
    }
private:
    void add_point(int x, const T &v) {
        b1.add(x, v);
        b2.add(x, v * T(x));
    }
public:
    void add(int l, int r, const T &v) {
        add_point(l, v);
        add_point(r, T{} - v);
    }
    T sum(int r) const { return b1.sum(r) * T(r) - b2.sum(r); }
    T query(int l, int r) const { return sum(r) - sum(l); }
};
```
### 前缀最大值 BIT
---
```cpp
template<class T>
struct MaxBIT {
    int n;
    T e; // 初始聚合值
    vector<T> t;
    explicit MaxBIT(int n_ = 0, T e_ = numeric_limits<T>::lowest()) : n(n_), e(e_), t(n + 1, e) {}
    // 单点执行 chmax
    void apply(int x, const T &v) { for (++ x; x <= n; x += x & -x) t[x] = max(t[x], v); }
    // 查询 [0, r) 最大值
    T prefix(int r) const {
        T res{ e };
        for (; r > 0; r -= r & -r) res = max(res, t[r]);
        return res;
    }
};
```
#### 修改为其他前缀运算
只需修改：
1. 单位元 `e`
2. `apply()` 和 `prefix()` 中的运算
其中 `e` 必须满足：
$$\operatorname{op}(e,x)=x$$
常见运算如下：

| 运算  | `e`                           | 合并方式        |
| --- | ----------------------------- | ----------- |
| 最大值 | `numeric_limits<T>::lowest()` | `max(x, y)` |
| 最小值 | `numeric_limits<T>::max()`    | `min(x, y)` |
| GCD | `T{}`，即 `0`                   | `gcd(x, y)` |
| 按位或 | `T{}`，即 `0`                   | `x \| y`    |
| 按位与 | 全 `1`                         | `x & y`     |
