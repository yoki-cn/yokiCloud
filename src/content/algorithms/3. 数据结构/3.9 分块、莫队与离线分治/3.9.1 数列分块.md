### 数列分块
---
#### 基础分块：区间加与区间和
---
```cpp
template<class T = i64>
struct BlockSum {
    int n, B, m;
    vector<T> a, s, t;
    explicit BlockSum(const vector<T> &v, int bs = 0)
        : n(v.size()), B(bs ? min(bs, max(1, n)) : max(1, int(sqrt(max(1, n))))),
          m(n / B + (n % B != 0)), a(v), s(m), t(m) {
        for (int i = 0; i < n; ++ i) s[i / B] += a[i];
    }
    void add(int l, int r, T x) {
        if (l >= r) return;
        int p{ l / B }, q{ (r - 1) / B };
        if (p == q) {
            for (int i = l; i < r; ++ i) a[i] += x;
            s[p] += x * (r - l);
            return;
        }
        int pr{ min(n, (p + 1) * B) };
        for (int i = l; i < pr; ++ i) a[i] += x;
        s[p] += x * (pr - l);
        for (int b = p + 1; b < q; ++ b) t[b] += x;
        int ql{ q * B };
        for (int i = ql; i < r; ++ i) a[i] += x;
        s[q] += x * (r - ql);
    }
    T sum(int l, int r) const {
        if (l >= r) return T{};
        int p{ l / B }, q{ (r - 1) / B };
        T res{};
        if (p == q) {
            for (int i = l; i < r; ++ i) res += a[i] + t[p];
            return res;
        }
        int pr{ min(n, (p + 1) * B) };
        for (int i = l; i < pr; ++ i) res += a[i] + t[p];
        for (int b = p + 1; b < q; ++ b) {
            int len{ min(n, (b + 1) * B) - b * B };
            res += s[b] + t[b] * len;
        }
        for (int i = q * B; i < r; ++ i) res += a[i] + t[q];
        return res;
    }
};
```
##### 成员变量
- `n`：序列长度
- `B`：块长
- `m`：块数
- `a[i]`：去掉块标记后的值
- `s[b]`：第 `b` 块的 `a[i]` 之和
- `t[b]`：第 `b` 块整体加法标记
- 实际值为 `a[i] + t[i / B]`
##### 接口
- `BlockSum(a, bs)`：建块，`bs = 0` 时自动取约 $\sqrt n$ 的块长，$O(n)$
- `add(l, r, x)`：区间 `[l,r)` 加 `x`，$O(B+n/B)$
- `sum(l, r)`：查询 `[l,r)` 区间和，$O(B+n/B)$
##### 注意事项
- 下标为 `0base`，区间为 `[l,r)`
- 完整块修改只修改 `t[b]`
- 边角块修改 `a[i]`，同时修正 `s[b]`
- `s[b]` 不包含 `t[b]`
- 这份板子主要作为数列分块的基础骨架，若题目只有区间加和区间和，线段树或树状数组通常更优
##### 常见用法
```cpp
BlockSum<> b(a);
b.add(l, r, x);
i64 ans = b.sum(l, r);
```
##### 复杂度
| 操作 | 时间复杂度 |
|---|---|
| 构造 | $O(n)$ |
| `add` | $O(B+n/B)$ |
| `sum` | $O(B+n/B)$ |
空间复杂度 $O(n)$
取 $B=\Theta(\sqrt n)$ 时，区间操作为 $O(\sqrt n)$
#### 块内有序分块
---
```cpp
template<class T = i64>
struct BlockOrd {
    using W = conditional_t<is_integral_v<T>, i128, T>;
    int n, B, m;
    vector<T> a, t;
    vector<vector<T>> v;
    explicit BlockOrd(const vector<T> &a_, int bs = 0)
        : n(a_.size()), B(bs ? min(bs, max(1, n)) : max(1, int(sqrt(max(1, n))))),
          m(n / B + (n % B != 0)), a(a_), t(m), v(m) {
        for (int b = 0; b < m; ++ b) pull(b);
    }
    // ===== 块内重构 =====
    void pull(int b) {
        int l{ b * B }, r{ min(n, (b + 1) * B) };
        v[b].assign(a.begin() + l, a.begin() + r);
        sort(v[b].begin(), v[b].end());
    }
    // ===== 基础接口 =====
    T get(int i) const { return a[i] + t[i / B]; }
    void set(int i, T x) {
        int b{ i / B };
        a[i] = x - t[b];
        pull(b);
    }
    void add(int l, int r, T x) {
        if (l >= r) return;
        int p{ l / B }, q{ (r - 1) / B };
        if (p == q) {
            for (int i = l; i < r; ++ i) a[i] += x;
            pull(p);
            return;
        }
        int pr{ min(n, (p + 1) * B) };
        for (int i = l; i < pr; ++ i) a[i] += x;
        pull(p);
        for (int b = p + 1; b < q; ++ b) t[b] += x;
        for (int i = q * B; i < r; ++ i) a[i] += x;
        pull(q);
    }
    int lt(int l, int r, T x) const {
        if (l >= r) return 0;
        int p{ l / B }, q{ (r - 1) / B }, res{};
        if (p == q) {
            for (int i = l; i < r; ++ i) res += a[i] + t[p] < x;
            return res;
        }
        int pr{ min(n, (p + 1) * B) };
        for (int i = l; i < pr; ++ i) res += a[i] + t[p] < x;
        for (int b = p + 1; b < q; ++ b) {
            res += lower_bound(v[b].begin(), v[b].end(), W(x) - t[b]) - v[b].begin();
        }
        for (int i = q * B; i < r; ++ i) res += a[i] + t[q] < x;
        return res;
    }
    // ===== 前驱后继 =====
    optional<T> pre(int l, int r, T x) const {
        if (l >= r) return nullopt;
        int p{ l / B }, q{ (r - 1) / B };
        optional<T> res;
        auto upd = [&](T y) {
            if (y < x && (!res || y > *res)) res = y;
        };
        if (p == q) {
            for (int i = l; i < r; ++ i) upd(a[i] + t[p]);
            return res;
        }
        int pr{ min(n, (p + 1) * B) };
        for (int i = l; i < pr; ++ i) upd(a[i] + t[p]);
        for (int b = p + 1; b < q; ++ b) {
            auto it{ lower_bound(v[b].begin(), v[b].end(), W(x) - t[b]) };
            if (it != v[b].begin()) upd(*prev(it) + t[b]);
        }
        for (int i = q * B; i < r; ++ i) upd(a[i] + t[q]);
        return res;
    }
    optional<T> suc(int l, int r, T x) const {
        if (l >= r) return nullopt;
        int p{ l / B }, q{ (r - 1) / B };
        optional<T> res;
        auto upd = [&](T y) {
            if (y > x && (!res || y < *res)) res = y;
        };
        if (p == q) {
            for (int i = l; i < r; ++ i) upd(a[i] + t[p]);
            return res;
        }
        int pr{ min(n, (p + 1) * B) };
        for (int i = l; i < pr; ++ i) upd(a[i] + t[p]);
        for (int b = p + 1; b < q; ++ b) {
            auto it{ upper_bound(v[b].begin(), v[b].end(), W(x) - t[b]) };
            if (it != v[b].end()) upd(*it + t[b]);
        }
        for (int i = q * B; i < r; ++ i) upd(a[i] + t[q]);
        return res;
    }
};
```
##### 成员变量
- `a[i]`：去掉块标记后的值
- `t[b]`：第 `b` 块整体加法标记
- `v[b]`：第 `b` 块所有 `a[i]` 排序后的结果
##### 接口
- `BlockOrd(a, bs)`：建块并排序，$O(n\log B)$
- `pull(b)`：重建第 `b` 块，$O(B\log B)$
- `get(i)`：查询实际值，$O(1)$
- `set(i, x)`：单点赋值，$O(B\log B)$
- `add(l, r, x)`：区间 `[l,r)` 加 `x`，$O(B\log B+n/B)$
- `lt(l, r, x)`：查询 `[l,r)` 中严格小于 `x` 的数量，$O(B+(n/B)\log B)$
- `pre(l, r, x)`：查询严格小于 `x` 的最大值，不存在返回 `nullopt`
- `suc(l, r, x)`：查询严格大于 `x` 的最小值，不存在返回 `nullopt`
##### 注意事项
- `v[b]` 保存的是裸值 `a[i]`，不包含 `t[b]`
- 完整块二分使用宽类型计算 `x - t[b]`，避免查询阈值在 T 极值处溢出；实际值、裸值、标记及修改运算仍须可由 T 表示
- 完整块加法不改变相对顺序，因此无需 `pull()`
- 边角修改会改变块内顺序，修改后必须 `pull()`
- `lt()` 统计 `< x`，不是 `<= x`
- 若不需要前驱后继，可直接删除 `pre()`、`suc()`
- 整数值域下可以对答案二分，并用 `lt()` 求区间第 `k` 小，不单独封装 `kth()`
##### 常见用法
###### 区间加
```cpp
BlockOrd<> b(a);
b.add(l, r, x);
```
###### 查询小于 `x` 的数量
```cpp
int cnt = b.lt(l, r, x);
```
###### 查询前驱后继
```cpp
auto p = b.pre(l, r, x);
auto s = b.suc(l, r, x);
if (p) {
    // *p
}
if (s) {
    // *s
}
```
##### 复杂度
| 操作 | 时间复杂度 |
|---|---|
| 构造 | $O(n\log B)$ |
| `pull` | $O(B\log B)$ |
| `get` | $O(1)$ |
| `set` | $O(B\log B)$ |
| `add` | $O(B\log B+n/B)$ |
| `lt` | $O(B+(n/B)\log B)$ |
| `pre` / `suc` | $O(B+(n/B)\log B)$ |
空间复杂度 $O(n)$
取 $B=\Theta(\sqrt n)$ 时，修改与顺序统计通常为 $O(\sqrt n\log n)$
#### 重构型分块
---
当完整块不能只靠一个标记维护时，可以将块内真实值下放，暴力修改，再重新计算块信息
常见于区间开方、区间取模、`chmin`、`chmax` 等特殊修改
##### 改造骨架
```cpp
void push(int b) {
    if (!t[b]) return;
    int l{ b * B }, r{ min(n, (b + 1) * B) };
    for (int i = l; i < r; ++ i) a[i] += t[b];
    t[b] = 0;
}

void pull(int b) {
    int l{ b * B }, r{ min(n, (b + 1) * B) };
    // 根据 a[l..r) 重建 sum / mn / mx / v 等块信息
}
```
修改一个无法整体处理的块时通常写成
```cpp
push(b);
for (int i = l; i < r; ++ i) {
    a[i] = f(a[i]);
}
pull(b);
```
##### 注意事项
- `push()` 与 `pull()` 只是改造模式，不是独立通用数据结构
- 若块信息能判断本次操作对整块无影响，可直接跳过
- 若完整块能整体修改，应优先整体处理
- 若必须频繁暴力重构，需要证明总重构次数或势能下降有均摊上界
- 不能因为使用分块就默认复杂度是 $O(\sqrt n)$
#### 块长选择
设块长为 $B$，块数约为 $n/B$，基础分块单次操作通常为
$$
O\left(B+\frac nB\right)
$$
因此通常取
$$
B=\Theta(\sqrt n)
$$
若块内还需要排序、重构或其他较重操作，最优块长会受修改与查询比例影响，可通过构造函数参数 `bs` 手动调整
