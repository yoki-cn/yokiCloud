### 离散浮点横坐标 Discrete Real Li Chao
---
```cpp
template<bool MIN = true>
struct DiscreteRealLiChao {
    static constexpr ld E = MIN ? numeric_limits<ld>::infinity() : -numeric_limits<ld>::infinity();
    struct Line {
        ld k{}, b{};
        ld get(ld x) const { return k * x + b; }
    };
    int n;
    vector<ld> xs;
    vector<Line> line{{}};
    vector<int> tr;
    explicit DiscreteRealLiChao(vector<ld> x, int max_lines = 0) : xs(move(x)) {
        for (ld x : xs) assert(isfinite(x));
        ranges::sort(xs);
        xs.erase(ranges::unique(xs).begin(), xs.end());
        n = int(xs.size());
        assert(n > 0);
        tr.assign(4 * n + 4, 0);
        if (max_lines > 0) line.reserve(max_lines + 1);
    }
private:
    // ===== 基础比较：插入与查询共同依赖 =====
    ld calc(int id, int x) const { return id ? line[id].get(xs[x]) : E; }
    ld diff(int u, int v, int x) const {
        return (line[u].k - line[v].k) * xs[x] + (line[u].b - line[v].b);
    }
    bool better(int u, int v, int x) const {
        if (!u) return false;
        if (!v) return true;
        ld d{ diff(u, v, x) };
        if (d != 0) return MIN ? d < 0 : d > 0;
        return u < v;
    }
    int best(int u, int v, int x) const { return better(u, v, x) ? u : v; }
    int index(ld x) const {
        auto it = ranges::lower_bound(xs, x);
        assert(it != xs.end() && *it == x);
        return int(it - xs.begin());
    }
    pair<int, int> range(ld l, ld r) const {
        if (l > r) swap(l, r);
        int ql{ int(ranges::lower_bound(xs, l) - xs.begin()) };
        int qr{ int(ranges::upper_bound(xs, r) - xs.begin()) };
        return { ql, qr };
    }
    // ===== 整线插入核心：完整直线与区间直线共同依赖 =====
    void insert(int p, int l, int r, int u) {
        int v{ tr[p] };
        if (!v) return void(tr[p] = u);
        if (line[u].k == line[v].k) {
            if (line[u].b != line[v].b) {
                bool take{ MIN ? line[u].b < line[v].b : line[u].b > line[v].b };
                if (take) tr[p] = u;
            } else if (u < v) {
                tr[p] = u;
            }
            return;
        }
        int m{ (l + r) >> 1 };
        if (better(u, v, m)) swap(u, v);
        tr[p] = v;
        if (r - l == 1) return;
        if (better(u, v, l)) insert(p << 1, l, m, u);
        else insert(p << 1 | 1, m, r, u);
    }
    // ===== 区间插线：若只插完整直线，可删除 update()、range() 与对应公开接口 =====
    void update(int p, int l, int r, int ql, int qr, int u) {
        if (qr <= l || r <= ql) return;
        if (ql <= l && r <= qr) return insert(p, l, r, u);
        int m{ (l + r) >> 1 };
        update(p << 1, l, m, ql, qr, u);
        update(p << 1 | 1, m, r, ql, qr, u);
    }
    int query_index(int x) const {
        int p{ 1 }, l{}, r{ n }, ans{};
        while (true) {
            ans = best(ans, tr[p], x);
            if (r - l == 1) return ans;
            int m{ (l + r) >> 1 };
            if (x < m) p <<= 1, r = m;
            else p = p << 1 | 1, l = m;
        }
    }
public:
    // ===== 基础接口 =====
    int add_line(ld k, ld b) {
        assert(isfinite(k) && isfinite(b));
        line.push_back({ k, b });
        int id{ int(line.size()) - 1 };
        insert(1, 0, n, id);
        return id;
    }
    // ===== 有限有效域：若只插完整直线，可删除以下两个插入接口 =====
    int add_line(ld k, ld b, ld l, ld r) {
        assert(isfinite(k) && isfinite(b) && isfinite(l) && isfinite(r));
        line.push_back({ k, b });
        int id{ int(line.size()) - 1 };
        auto [ql, qr] = range(l, r);
        if (ql < qr) update(1, 0, n, ql, qr, id);
        return id;
    }
    int add_seg(ld x0, ld y0, ld x1, ld y1) {
        assert(isfinite(x0) && isfinite(y0) && isfinite(x1) && isfinite(y1));
        if (x0 > x1) swap(x0, x1), swap(y0, y1);
        if (x0 == x1) {
            ld y{ MIN ? min(y0, y1) : max(y0, y1) };
            return add_line(0, y, x0, x0);
        }
        ld k{ (y1 - y0) / (x1 - x0) };
        return add_line(k, y0 - k * x0, x0, x1);
    }
    // ===== 查询 =====
    int query_id(ld x) const { return query_index(index(x)); }
    ld query(ld x) const {
        int p{ index(x) };
        return calc(query_index(p), p);
    }
    pair<ld, int> query_info(ld x) const {
        int p{ index(x) }, id{ query_index(p) };
        return { calc(id, p), id };
    }
    // ===== 重置：对象只使用一次时可删除 =====
    void clear() {
        ranges::fill(tr, 0);
        line.resize(1);
    }
};
```
#### 模板参数
- `MIN = true`：维护最小值
- `MIN = false`：维护最大值
#### 接口
- `DiscreteRealLiChao(xs, max_lines = 0)`：收集、排序、去重所有允许查询的浮点横坐标，$O(n \log n)$
- `add_line(k, b)`：在所有离散横坐标插入 `y = kx + b`，$O(\log n)$
- `add_line(k, b, l, r)`：只在满足 `l <= xs[i] <= r` 的离散点上生效，$O(\log^2 n)$
- `add_seg(x0, y0, x1, y1)`：插入两端点之间的有限线段，仅覆盖端点横坐标范围内的离散点，$O(\log^2 n)$
- `query(x)`：返回 `x` 处最优值，要求 `x` 精确存在于 `xs`，$O(\log n)$
- `query_id(x)`：返回最优插入对象编号；无覆盖时返回 `0`，$O(\log n)$
- `query_info(x)`：返回 `{最优值, 编号}`，$O(\log n)$
- `clear()`：清空所有已插入对象，保留离散横坐标，$O(n)$
#### 注意事项
- `query(x)` 中的 `x` 必须与构造时某个 `xs[i]` 的 `long double` 实际值完全相同；不要重新通过另一套浮点计算构造一个“数学上应该相等”的值
- 内部区间树维护的是离散下标 `[0, n)`；`add_line(k, b, l, r)` 的用户语义仍是闭区间 `[l, r]`，通过 `lower_bound / upper_bound` 映射到离散下标半开区间
- 浮点 Li Chao 内部不使用 `EPS`。等斜率只在 `k` 实际相等时处理；直线比较使用实际 `long double` 结果
- `xs`、直线参数、线段端点均应为有限值。极端量级仍可能使浮点乘加上溢
- 多条候选值完全相同时返回编号较小者
##### 竖直线段
若 `x0 == x1`，线段不是函数。模板把它解释为：
- 最小值版只在离散点 `x0` 上加入 `min(y0, y1)`
- 最大值版只在离散点 `x0` 上加入 `max(y0, y1)`
如果 `x0` 不在 `xs` 中，这条竖直线段不会影响任何查询
##### 与其他版本的关系
- 只插完整直线时，直接使用泛型 `LiChao<ld, MIN, ld>` 更短
- 浮点查询横坐标无法提前收集且只插完整直线时，使用 `RealLiChao`
#### 常见用法
##### 区间有效直线
```cpp
DiscreteRealLiChao<> seg(xs, m);
seg.add_line(k, b, l, r);
ld ans = seg.query(xs[i]);
```
##### 两端点有限线段
```cpp
int id = seg.add_seg(x0, y0, x1, y1);
auto [ans, best_id] = seg.query_info(xs[i]);
```
##### 最大值
```cpp
DiscreteRealLiChao<false> seg(xs);
seg.add_line(k, b, l, r);
```
#### 复杂度
设离散横坐标数量为 `n`，插入对象数为 `L`

| 操作                                   | 时间复杂度         |
| ------------------------------------ | ------------- |
| 构造                                   | $O(n \log n)$ |
| `add_line(k, b)` / 查询                | $O(\log n)$   |
| `add_line(k, b, l, r)` / `add_seg()` | $O(\log^2 n)$ |
| `clear()`                            | $O(n)$        |
空间复杂度：$O(n + L)$
