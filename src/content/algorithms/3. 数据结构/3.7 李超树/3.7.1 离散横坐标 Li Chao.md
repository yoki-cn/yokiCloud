### 离散横坐标 Li Chao
---
```cpp
template<class T = i64, bool MIN = true, class V = conditional_t<is_integral_v<T>, i128, T>>
struct LiChao {
    struct Line {
        T k{}, b{};
        V get(T x) const { return V(k) * V(x) + V(b); }
    };
    int n;
    vector<T> xs;
    vector<Line> line{{}};
    vector<int> tr;
    explicit LiChao(vector<T> x, int max_lines = 0) : xs(move(x)) {
        if constexpr (is_floating_point_v<T>) for (T x : xs) assert(isfinite(x));
        ranges::sort(xs);
        xs.erase(ranges::unique(xs).begin(), xs.end());
        n = int(xs.size());
        assert(n > 0);
        tr.assign(4 * n + 4, 0);
        if (max_lines > 0) line.reserve(max_lines + 1);
    }
private:
    // ===== 基础比较：插入与查询共同依赖，不建议删除 =====
    static V empty_value() {
        if constexpr (numeric_limits<V>::has_infinity) {
            return MIN ? numeric_limits<V>::infinity() : -numeric_limits<V>::infinity();
        } else {
            return MIN ? numeric_limits<V>::max() : numeric_limits<V>::lowest();
        }
    }
    V calc(int id, int x) const { return id ? line[id].get(xs[x]) : empty_value(); }
    V diff(int u, int v, int x) const {
        return (V(line[u].k) - V(line[v].k)) * V(xs[x]) + (V(line[u].b) - V(line[v].b));
    }
    bool better(int u, int v, int x) const {
        if (!u) return false;
        if (!v) return true;
        V d{ diff(u, v, x) };
        if (d != V{}) return MIN ? d < V{} : d > V{};
        return u < v;
    }
    int best(int u, int v, int x) const { return better(u, v, x) ? u : v; }
    int index(T x) const {
        auto it = ranges::lower_bound(xs, x);
        assert(it != xs.end() && *it == x);
        return int(it - xs.begin());
    }
    // ===== Li Chao 核心：只插完整直线 =====
    void insert(int p, int l, int r, int u) {
        int v{ tr[p] };
        if (!v) return void(tr[p] = u);
        if (line[u].k == line[v].k) {
            if (line[u].b != line[v].b) {
                bool take{ MIN ? line[u].b < line[v].b : line[u].b > line[v].b };
                if (take) tr[p] = u;
            } else if (u < v) {
                tr[p] = u;
            }
            return;
        }
        int m{ (l + r) >> 1 };
        if (better(u, v, m)) swap(u, v);
        tr[p] = v;
        if (r - l == 1) return;
        if (better(u, v, l)) insert(p << 1, l, m, u);
        else insert(p << 1 | 1, m, r, u);
    }
    int query_index(int x) const {
        int p{ 1 }, l{}, r{ n }, ans{};
        while (true) {
            ans = best(ans, tr[p], x);
            if (r - l == 1) return ans;
            int m{ (l + r) >> 1 };
            if (x < m) p <<= 1, r = m;
            else p = p << 1 | 1, l = m;
        }
    }
public:
    // ===== 基础接口 =====
    int add_line(T k, T b) {
        if constexpr (is_floating_point_v<T>) assert(isfinite(k) && isfinite(b));
        line.push_back({ k, b });
        int id{ int(line.size()) - 1 };
        insert(1, 0, n, id);
        return id;
    }
    int query_id(T x) const { return query_index(index(x)); }
    V query(T x) const {
        int p{ index(x) };
        return calc(query_index(p), p);
    }
    pair<V, int> query_info(T x) const {
        int p{ index(x) }, id{ query_index(p) };
        return { calc(id, p), id };
    }
    // ===== 重置：对象只使用一次时可删除 =====
    void clear() {
        ranges::fill(tr, 0);
        line.resize(1);
    }
};
```
#### 模板参数
- `T`：横坐标、斜率、截距类型，默认 `i64`
- `MIN = true`：维护最小值；`MIN = false`：维护最大值
- `V`：函数值计算类型。整数 `T` 默认使用 `i128`，避免常见的 `k * x + b` 溢出
#### 接口
- `LiChao(xs, max_lines = 0)`：对查询横坐标排序去重并建树，$O(n \log n)$
- `add_line(k, b)`：插入完整直线 `y = kx + b`，返回从 `1` 开始的直线编号，$O(\log n)$
- `query(x)`：返回 `x` 处最优值，要求 `x` 存在于构造时的 `xs` 中，$O(\log n)$
- `query_id(x)`：返回 `x` 处最优直线编号；没有直线时返回 `0`，$O(\log n)$
- `query_info(x)`：返回 `{最优值, 直线编号}`，只进行一次定位和树上查询，$O(\log n)$
- `clear()`：删除所有直线并保留离散横坐标，$O(n)$
#### 注意事项
- 只支持**完整直线**。若需要区间有效直线，整数连续值域使用 `SegmentLiChao`，浮点离散横坐标使用 `DiscreteRealLiChao`
- 多条直线计算结果完全相同时返回编号较小者
- 内部直接比较两条直线之差 `(k1 - k2) * x + (b1 - b2)`；等斜率时直接比较截距
##### 浮点版本
```cpp
using ld = long double;
LiChao<ld, true, ld> seg(xs);   // 最小值
LiChao<ld, false, ld> seg2(xs); // 最大值
```
浮点时额外注意：
- `query(x)` 要求 `x` 与构造时某个 `xs[i]` 的 `long double` **实际值完全相同**。最稳妥的是保存全部查询横坐标后，之后继续用原来的 `xs[i]` 查询
- Li Chao 内部不要使用 `EPS` 判断斜率相同或函数值相同。非常接近但不相等的两条直线仍可能交换优劣
- `xs`、`k`、`b` 应为有限值，不要传入 `NaN` 或无穷
- 若所有浮点查询点已知但还需要区间插线，应改用 `DiscreteRealLiChao`，而不是给本结构增加区间接口
#### 常见用法
##### 整数最小值
```cpp
vector<i64> xs = { 1, 3, 7, 10 };
LiChao seg(xs, m);
seg.add_line(k, b);
i128 ans = seg.query(7);
```
##### 最大值
```cpp
LiChao<i64, false> seg(xs);
seg.add_line(k, b);
auto [ans, id] = seg.query_info(x);
```
##### 浮点横坐标
```cpp
vector<ld> xs(q);
LiChao<ld, true, ld> seg(xs, m);
seg.add_line(k, b);
ld ans = seg.query(xs[i]);
```
#### 复杂度
设离散横坐标数量为 `n`，插入直线数为 `L`

| 操作                                                       | 时间复杂度         |
| -------------------------------------------------------- | ------------- |
| 构造                                                       | $O(n \log n)$ |
| `add_line()` / `query()` / `query_id()` / `query_info()` | $O(\log n)$   |
| `clear()`                                                | $O(n)$        |
空间复杂度：$O(n + L)$
