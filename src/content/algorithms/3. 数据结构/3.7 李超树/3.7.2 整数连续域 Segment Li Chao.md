### 整数连续域 Segment Li Chao
---
```cpp
template<class T = i64, bool MIN = true, class V = conditional_t<is_integral_v<T>, i128, T>>
struct SegmentLiChao {
    static_assert(is_integral_v<T>);
    struct Line {
        T k{}, b{};
        V get(T x) const { return V(k) * V(x) + V(b); }
    };
    struct Node {
        int ch[2]{};
        int line{};
    };
    T xl, xr;
    int root{ 1 };
    vector<Line> line{{}};
    vector<Node> tr{{}, {}};
    explicit SegmentLiChao(T xl_, T xr_, int max_lines = 0, int max_nodes = 0) : xl(xl_), xr(xr_) {
        assert(xl < xr);
        if (max_lines > 0) line.reserve(max_lines + 1);
        if (max_nodes > 0) tr.reserve(max_nodes + 2);
    }
private:
    // ===== 基础比较与节点创建 =====
    static V empty_value() {
        if constexpr (numeric_limits<V>::has_infinity) {
            return MIN ? numeric_limits<V>::infinity() : -numeric_limits<V>::infinity();
        } else {
            return MIN ? numeric_limits<V>::max() : numeric_limits<V>::lowest();
        }
    }
    int new_node() {
        tr.push_back({});
        return int(tr.size()) - 1;
    }
    V calc(int id, T x) const { return id ? line[id].get(x) : empty_value(); }
    V diff(int u, int v, T x) const {
        return (V(line[u].k) - V(line[v].k)) * V(x) + (V(line[u].b) - V(line[v].b));
    }
    bool better(int u, int v, T x) const {
        if (!u) return false;
        if (!v) return true;
        V d{ diff(u, v, x) };
        if (d != V{}) return MIN ? d < V{} : d > V{};
        return u < v;
    }
    int best(int u, int v, T x) const { return better(u, v, x) ? u : v; }
    // ===== 整线插入核心：add_line() 与区间插线共同依赖 =====
    void insert(int p, T l, T r, int u) {
        int v{ tr[p].line };
        if (!v) return void(tr[p].line = u);
        if (line[u].k == line[v].k) {
            if (line[u].b != line[v].b) {
                bool take{ MIN ? line[u].b < line[v].b : line[u].b > line[v].b };
                if (take) tr[p].line = u;
            } else if (u < v) {
                tr[p].line = u;
            }
            return;
        }
        T m{ midpoint(l, r) };
        if (better(u, v, m)) swap(u, v);
        tr[p].line = v;
        if (r == l + 1) return;
        int d{ better(u, v, l) ? 0 : 1 };
        T ql{ d ? m : l }, qr{ d ? r : m };
        int c{ tr[p].ch[d] };
        if (!c) tr[p].ch[d] = c = new_node();
        insert(c, ql, qr, u);
    }
    // ===== 区间插线：若只插完整直线，可删除 update() 与三参数区间接口 =====
    int update(int p, T l, T r, T ql, T qr, int u) {
        if (qr <= l || r <= ql) return p;
        if (!p) p = new_node();
        if (ql <= l && r <= qr) {
            insert(p, l, r, u);
            return p;
        }
        T m{ midpoint(l, r) };
        int lc{ tr[p].ch[0] }, rc{ tr[p].ch[1] };
        lc = update(lc, l, m, ql, qr, u);
        rc = update(rc, m, r, ql, qr, u);
        tr[p].ch[0] = lc;
        tr[p].ch[1] = rc;
        return p;
    }
public:
    // ===== 基础接口 =====
    int add_line(T k, T b) {
        line.push_back({ k, b });
        int id{ int(line.size()) - 1 };
        insert(root, xl, xr, id);
        return id;
    }
    int add_line(T k, T b, T l, T r) {
        if (l > r) swap(l, r);
        line.push_back({ k, b });
        int id{ int(line.size()) - 1 };
        l = max(l, xl);
        r = min(r, xr);
        if (l < r) root = update(root, xl, xr, l, r, id);
        return id;
    }
    int query_id(T x) const {
        assert(xl <= x && x < xr);
        int p{ root }, ans{};
        T l{ xl }, r{ xr };
        while (p) {
            ans = best(ans, tr[p].line, x);
            if (r == l + 1) break;
            T m{ midpoint(l, r) };
            if (x < m) p = tr[p].ch[0], r = m;
            else p = tr[p].ch[1], l = m;
        }
        return ans;
    }
    V query(T x) const { return calc(query_id(x), x); }
    pair<V, int> query_info(T x) const {
        int id{ query_id(x) };
        return { calc(id, x), id };
    }
    // ===== 重置：对象只使用一次时可删除 =====
    void clear() {
        line.resize(1);
        tr.assign(2, {});
        root = 1;
    }
};
```
#### 模板参数
- `T`：整数横坐标、斜率和截距类型，默认 `i64`；该版本要求 `T` 为整数类型
- `MIN = true`：维护最小值；`MIN = false`：维护最大值
- `V`：函数值计算类型，默认整数时使用 `i128`
#### 接口
- `SegmentLiChao(xl, xr, max_lines = 0, max_nodes = 0)`：维护全部整数横坐标 `x ∈ [xl, xr)`，构造 $O(1)$
- `add_line(k, b)`：在整个值域插入 `y = kx + b`，返回直线编号，$O(\log C)$
- `add_line(k, b, l, r)`：只在整数横坐标 `[l, r)` 上生效；自动与总值域求交，返回直线编号，$O(\log^2 C)$
- `query(x)`：查询 `x` 处最优值，$O(\log C)$
- `query_id(x)`：查询最优直线编号；无覆盖时返回 `0`，$O(\log C)$
- `query_info(x)`：返回 `{最优值, 直线编号}`，$O(\log C)$
- `clear()`：清空所有动态节点和直线，保留值域，复杂度与已分配节点数线性相关
其中 `C = xr - xl`
#### 注意事项
- 总值域和区间插线都使用半开区间 `[l, r)`
- 若题目给出闭区间 `[l, r]` 且 `r + 1` 不会溢出，可写 `add_line(k, b, l, r + 1)`
- 采用动态开点，不需要为整个值域申请 `4 * C` 个节点，适合非常大的整数值域
- `new_node()` 会 `push_back()`，因此不要在调用它时长期保存 `tr[p]` 内元素的引用；`vector` 扩容后引用会失效
- 不提供“两整数端点自动生成线段”的接口。两整数端点确定的斜率未必是整数，直接用 `T` 相除会截断
- 多条直线值完全相同时返回编号较小者
#### 常见用法
##### 整个整数值域插线
```cpp
SegmentLiChao seg(-1000000000LL, 1000000001LL, m);
seg.add_line(k, b);
i128 ans = seg.query(x);
```
##### 闭区间 `[l, r]` 生效
```cpp
seg.add_line(k, b, l, r + 1);
```
##### 最大值
```cpp
SegmentLiChao<i64, false> seg(L, R);
seg.add_line(k, b, l, r);
auto [ans, id] = seg.query_info(x);
```
#### 复杂度
设整数值域长度为 `C = xr - xl`

| 操作                                                           | 时间复杂度         |
| ------------------------------------------------------------ | ------------- |
| `add_line(k, b)` / `query()` / `query_id()` / `query_info()` | $O(\log C)$   |
| `add_line(k, b, l, r)`                                       | $O(\log^2 C)$ |

动态节点空间最坏约为：
- 只插完整直线：$O(L \log C)$
- 含区间插线：$O(L \log^2 C)$
