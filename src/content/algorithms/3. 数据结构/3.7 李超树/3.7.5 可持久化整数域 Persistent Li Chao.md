### 可持久化整数域 Persistent Li Chao
---
```cpp
template<class T = i64, bool MIN = true, class V = conditional_t<is_integral_v<T>, i128, T>>
struct PersistentLiChao {
    static_assert(is_integral_v<T>);
    struct Line {
        T k{}, b{};
        V get(T x) const { return V(k) * V(x) + V(b); }
    };
    struct Node {
        int ch[2]{};
        int line{};
    };
    T xl, xr;
    vector<Line> line{{}};
    vector<Node> tr{{}};
    vector<int> root{ 0 };
    explicit PersistentLiChao(T xl_, T xr_, int max_lines = 0, int max_nodes = 0) : xl(xl_), xr(xr_) {
        assert(xl < xr);
        if (max_lines > 0) {
            line.reserve(max_lines + 1);
            root.reserve(max_lines + 1);
        }
        if (max_nodes > 0) tr.reserve(max_nodes + 1);
    }
private:
    // ===== 基础比较与路径复制 =====
    static V empty_value() {
        if constexpr (numeric_limits<V>::has_infinity) {
            return MIN ? numeric_limits<V>::infinity() : -numeric_limits<V>::infinity();
        } else {
            return MIN ? numeric_limits<V>::max() : numeric_limits<V>::lowest();
        }
    }
    int clone(int p) {
        tr.push_back(p ? tr[p] : Node{});
        return int(tr.size()) - 1;
    }
    V calc(int id, T x) const { return id ? line[id].get(x) : empty_value(); }
    V diff(int u, int v, T x) const {
        return (V(line[u].k) - V(line[v].k)) * V(x) + (V(line[u].b) - V(line[v].b));
    }
    bool better(int u, int v, T x) const {
        if (!u) return false;
        if (!v) return true;
        V d{ diff(u, v, x) };
        if (d != V{}) return MIN ? d < V{} : d > V{};
        return u < v;
    }
    int best(int u, int v, T x) const { return better(u, v, x) ? u : v; }
    // ===== 持久化整线插入 =====
    int insert(int p, T l, T r, int u) {
        int q{ clone(p) }, v{ tr[q].line };
        if (!v) {
            tr[q].line = u;
            return q;
        }
        if (line[u].k == line[v].k) {
            if (line[u].b != line[v].b) {
                bool take{ MIN ? line[u].b < line[v].b : line[u].b > line[v].b };
                if (take) tr[q].line = u;
            } else if (u < v) {
                tr[q].line = u;
            }
            return q;
        }
        T m{ midpoint(l, r) };
        if (better(u, v, m)) swap(u, v);
        tr[q].line = v;
        if (r == l + 1) return q;
        int d{ better(u, v, l) ? 0 : 1 };
        T ql{ d ? m : l }, qr{ d ? r : m };
        tr[q].ch[d] = insert(tr[q].ch[d], ql, qr, u);
        return q;
    }
    // ===== 持久化区间插线：若只插完整直线，可删除 update() 与对应公开接口 =====
    int update(int p, T l, T r, T ql, T qr, int u) {
        if (qr <= l || r <= ql) return p;
        if (ql <= l && r <= qr) return insert(p, l, r, u);
        int q{ clone(p) };
        T m{ midpoint(l, r) };
        tr[q].ch[0] = update(tr[q].ch[0], l, m, ql, qr, u);
        tr[q].ch[1] = update(tr[q].ch[1], m, r, ql, qr, u);
        return q;
    }
    int query_root(int p, T x) const {
        int ans{};
        T l{ xl }, r{ xr };
        while (p) {
            ans = best(ans, tr[p].line, x);
            if (r == l + 1) break;
            T m{ midpoint(l, r) };
            if (x < m) p = tr[p].ch[0], r = m;
            else p = tr[p].ch[1], l = m;
        }
        return ans;
    }
public:
    // ===== 版本接口：每次插入创建一个新版本 =====
    int add_line(int ver, T k, T b) {
        assert(0 <= ver && ver < int(root.size()));
        line.push_back({ k, b });
        int id{ int(line.size()) - 1 };
        root.push_back(insert(root[ver], xl, xr, id));
        assert(id == int(root.size()) - 1);
        return id;
    }
    int add_line(int ver, T k, T b, T l, T r) {
        assert(0 <= ver && ver < int(root.size()));
        if (l > r) swap(l, r);
        line.push_back({ k, b });
        int id{ int(line.size()) - 1 };
        l = max(l, xl);
        r = min(r, xr);
        root.push_back(l < r ? update(root[ver], xl, xr, l, r, id) : root[ver]);
        assert(id == int(root.size()) - 1);
        return id;
    }
    int query_id(int ver, T x) const {
        assert(0 <= ver && ver < int(root.size()));
        assert(xl <= x && x < xr);
        return query_root(root[ver], x);
    }
    V query(int ver, T x) const { return calc(query_id(ver, x), x); }
    pair<V, int> query_info(int ver, T x) const {
        int id{ query_id(ver, x) };
        return { calc(id, x), id };
    }
    int versions() const { return int(root.size()); }
    // ===== 重置：对象只使用一次时可删除 =====
    void clear() {
        line.resize(1);
        tr.resize(1);
        root.assign(1, 0);
    }
};
```
#### 接口说明
##### 模板参数
- `T`：整数横坐标、斜率和截距类型，默认 `i64`
- `MIN = true`：维护最小值；`MIN = false`：维护最大值
- `V`：函数值类型，整数默认使用 `i128`
##### 版本语义
- 版本 `0` 是空树
- 每次 `add_line()` 都创建一个新版本，不修改原版本
- 为了减少现场管理，本模板保持：**新版本编号 = 本次插入对象编号**
- 可以从任意历史版本继续插入，因此版本可以分叉
##### 接口
- `PersistentLiChao(xl, xr, max_lines = 0, max_nodes = 0)`：维护整数值域 `[xl, xr)`
- `add_line(ver, k, b)`：从版本 `ver` 创建新版本并插入完整直线，返回新版本编号，$O(\log C)$
- `add_line(ver, k, b, l, r)`：从版本 `ver` 创建新版本，并令直线只在 `[l, r)` 生效，返回新版本编号，$O(\log^2 C)$
- `query(ver, x)`：查询版本 `ver` 在 `x` 处的最优值，$O(\log C)$
- `query_id(ver, x)`：返回该版本最优插入对象编号；无覆盖时为 `0`，$O(\log C)$
- `query_info(ver, x)`：返回 `{最优值, 对象编号}`，$O(\log C)$
- `versions()`：返回当前版本数量，包含版本 `0`，$O(1)$
- `clear()`：恢复为只有空版本 `0` 的状态
其中 `C = xr - xl`
#### 注意事项
- 总值域和区间插线均使用半开区间 `[l, r)`
- 从旧版本分叉完全合法：
```cpp
int v1 = seg.add_line(0, k1, b1);
int v2 = seg.add_line(v1, k2, b2);
int v3 = seg.add_line(v1, k3, b3);
```
此时 `v2` 与 `v3` 分别继承 `v1`，互不影响
- 即使区间插线与总值域没有交集，也仍然创建一个新版本，只是新版本根与旧版本相同；这样可以继续保持“版本编号 = 插入对象编号”
- 路径复制仅复制发生修改的节点，旧版本节点始终不修改
- 多条直线值完全相同时返回编号较小者
#### 浮点版本
这个版本与浮点的差异较大，**不能只把 `T` 改成 `ld`**。整数实现依赖：
```cpp
r == l + 1
```
作为精确叶子，并依赖 `[l, r)` 的整数区间分解
##### 浮点 `x` 可以提前收集
这是最推荐的持久化浮点方案：
```text
收集全部会查询的 x
→ 排序去重得到 xs
→ 树上维护下标区间 [0, n)
→ 比较位置使用 xs[l] / xs[mid]
```
具体修改原则：
- `xl, xr` 删除，增加 `vector<ld> xs` 与 `n`
- `insert(p, l, r, u)` 中的 `l / m` 仍是整数下标，但 `better()` 使用 `xs[x]` 计算函数值
- 叶子条件改为 `r - l == 1`，这是离散下标的精确叶子
- 浮点区间 `[L, R]` 通过 `lower_bound(xs, L)` 与 `upper_bound(xs, R)` 映射成下标半开区间 `[ql, qr)`，再复用当前持久化 `update()`
- 查询时必须要求 `x` 精确存在于 `xs`，与 `DiscreteRealLiChao` 相同
这样可以同时得到：
```text
浮点横坐标
+ 持久化
+ 精确的离散区间有效域
```
这是需要区间插线时的首选
##### 浮点 `x` 无法提前收集，只插完整直线
可以按 `RealLiChao` 修改成连续实数持久化：
```text
T xl, xr       -> ld xl, xr
T k, b         -> ld k, b
V               -> ld
整数叶子        -> DEPTH / midpoint 停止条件
insert()        -> 增加 dep 参数
```
停止条件改为：
```cpp
ld m = midpoint(l, r);
if (dep == DEPTH || m == l || m == r) return q;
```
路径复制逻辑本身无需改变，因此完整直线仍可持久化
#### 常见用法
##### 链式建立历史版本
```cpp
PersistentLiChao seg(L, R);
int v1 = seg.add_line(0, k1, b1);
int v2 = seg.add_line(v1, k2, b2);
i128 ans1 = seg.query(v1, x);
i128 ans2 = seg.query(v2, x);
```
##### 从历史版本分叉
```cpp
int v1 = seg.add_line(0, k1, b1);
int v2 = seg.add_line(v1, k2, b2);
int v3 = seg.add_line(v1, k3, b3);
```
##### 持久化区间直线
```cpp
int nv = seg.add_line(ver, k, b, l, r);
auto [ans, id] = seg.query_info(nv, x);
```
#### 复杂度
设整数值域长度为 `C = xr - xl`

| 操作   | 时间复杂度         | 新增节点          |
| ---- | ------------- | ------------- |
| 完整直线 | $O(\log C)$   | $O(\log C)$   |
| 区间直线 | $O(\log^2 C)$ | $O(\log^2 C)$ |
| 查询   | $O(\log C)$   | $O(1)$        |
