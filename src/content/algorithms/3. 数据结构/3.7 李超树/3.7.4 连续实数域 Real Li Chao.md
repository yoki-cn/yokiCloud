### 连续实数域 Real Li Chao
---
```cpp
template<bool MIN = true, int DEPTH = 80>
struct RealLiChao {
    static constexpr ld E = MIN ? numeric_limits<ld>::infinity() : -numeric_limits<ld>::infinity();
    struct Line {
        ld k{}, b{};
        ld get(ld x) const { return k * x + b; }
    };
    struct Node {
        int ch[2]{};
        int line{};
    };
    ld xl, xr;
    int root{ 1 };
    vector<Line> line{{}};
    vector<Node> tr{{}, {}};
    explicit RealLiChao(ld xl_, ld xr_, int max_lines = 0, int max_nodes = 0) : xl(xl_), xr(xr_) {
        assert(isfinite(xl) && isfinite(xr) && xl < xr);
        if (max_lines > 0) line.reserve(max_lines + 1);
        if (max_nodes > 0) tr.reserve(max_nodes + 2);
    }
private:
    // ===== 基础比较与节点创建：以下代码被全部操作依赖，不建议删除 =====
    int new_node() {
        tr.push_back({});
        return int(tr.size()) - 1;
    }
    ld calc(int id, ld x) const { return id ? line[id].get(x) : E; }
    ld diff(int u, int v, ld x) const {
        return (line[u].k - line[v].k) * x + (line[u].b - line[v].b);
    }
    bool better(int u, int v, ld x) const {
        if (!u) return false;
        if (!v) return true;
        ld d{ diff(u, v, x) };
        if (d != 0) return MIN ? d < 0 : d > 0;
        return u < v;
    }
    int best(int u, int v, ld x) const { return better(u, v, x) ? u : v; }
    // ===== 连续实数 Li Chao 核心：只插完整直线 =====
    void insert(int p, ld l, ld r, int dep, int u) {
        int v{ tr[p].line };
        if (!v) return void(tr[p].line = u);
        if (line[u].k == line[v].k) {
            if (line[u].b != line[v].b) {
                bool take{ MIN ? line[u].b < line[v].b : line[u].b > line[v].b };
                if (take) tr[p].line = u;
            } else if (u < v) {
                tr[p].line = u;
            }
            return;
        }
        ld m{ midpoint(l, r) };
        if (better(u, v, m)) swap(u, v);
        tr[p].line = v;
        if (dep == DEPTH || m == l || m == r) return;
        int d{ better(u, v, l) ? 0 : 1 };
        ld ql{ d ? m : l }, qr{ d ? r : m };
        int c{ tr[p].ch[d] };
        if (!c) tr[p].ch[d] = c = new_node();
        insert(c, ql, qr, dep + 1, u);
    }
public:
    // ===== 基础接口 =====
    int add_line(ld k, ld b) {
        assert(isfinite(k) && isfinite(b));
        line.push_back({ k, b });
        int id{ int(line.size()) - 1 };
        insert(root, xl, xr, 0, id);
        return id;
    }
    int query_id(ld x) const {
        assert(isfinite(x) && xl <= x && x <= xr);
        int p{ root }, ans{}, dep{};
        ld l{ xl }, r{ xr };
        while (p) {
            ans = best(ans, tr[p].line, x);
            ld m{ midpoint(l, r) };
            if (dep == DEPTH || m == l || m == r) break;
            if (x <= m) p = tr[p].ch[0], r = m;
            else p = tr[p].ch[1], l = m;
            ++ dep;
        }
        return ans;
    }
    ld query(ld x) const { return calc(query_id(x), x); }
    pair<ld, int> query_info(ld x) const {
        int id{ query_id(x) };
        return { calc(id, x), id };
    }
    // ===== 重置：对象只使用一次时可删除 =====
    void clear() {
        line.resize(1);
        tr.assign(2, {});
        root = 1;
    }
};
```
#### 接口说明
##### 模板参数
- `MIN = true`：维护最小值；`MIN = false`：维护最大值
- `DEPTH`：最多二分层数，默认 `80`
##### 接口
- `RealLiChao(xl, xr, max_lines = 0, max_nodes = 0)`：建立连续实数值域 `[xl, xr]`，$O(1)$
- `add_line(k, b)`：在整个实数值域插入 `y = kx + b`，返回直线编号，$O(DEPTH)$
- `query(x)`：查询任意 `x ∈ [xl, xr]` 的最优值，$O(DEPTH)$
- `query_id(x)`：返回最优直线编号；无直线时返回 `0`，$O(DEPTH)$
- `query_info(x)`：返回 `{最优值, 直线编号}`，$O(DEPTH)$
- `clear()`：清空所有直线和动态节点，保留值域
#### 注意事项
- 连续实数没有天然叶子，因此使用 `DEPTH` 控制递归深度，并在 `midpoint(l, r)` 已经等于端点时提前停止
- 默认 `DEPTH = 80` 对常见 `ld` 数据范围通常足够；如果值域跨度极大且需要区分非常接近的查询位置，可以适当增大
- 内部不使用 `EPS` 判断斜率、交点或函数值相等
- `xl`、`xr`、查询横坐标、斜率、截距都应是有限值；极端数值仍可能使浮点乘加上溢
- 多条候选值实际计算完全相同时返回编号较小者
#### 常见用法
##### 连续实数最小值
```cpp
RealLiChao<> seg(-1E9L, 1E9L, m);
seg.add_line(k, b);
ld ans = seg.query(x);
```
##### 最大值
```cpp
RealLiChao<false> seg(L, R);
seg.add_line(k, b);
auto [ans, id] = seg.query_info(x);
```
##### 增加二分深度
```cpp
RealLiChao<true, 120> seg(L, R);
```
#### 复杂度

| 操作                                        | 时间复杂度      |
| ----------------------------------------- | ---------- |
| `add_line()`                              | $O(DEPTH)$ |
| `query()` / `query_id()` / `query_info()` | $O(DEPTH)$ |
最坏空间复杂度：$O(L * DEPTH)$
