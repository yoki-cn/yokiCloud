### 划分树
---
只需要静态区间第 `k` 小时使用
```cpp
template<class T>
struct PartitionTree {
private:
    int n, lg;
    vector<vector<T>> tr;
    vector<vector<int>> pref;
    void build(int d, int l, int r, const vector<T> &sorted) {
        if (r - l <= 1) return;
        int m{ (l + r) / 2 };
        const T &pivot{ sorted[m - 1] };
        // 左半部分固定需要 m-l 个元素，重复 pivot 时精确控制放左边的数量
        int same{ m - l };
        for (int i = l; i < r; ++ i) {
            if (tr[d][i] < pivot) -- same;
        }
        int x{ l }, y{ m };
        for (int i = l; i < r; ++ i) {
            bool left{
                tr[d][i] < pivot ||
                (!(pivot < tr[d][i]) && same -- > 0)
            };
            pref[d][i + 1] = pref[d][i] + left;
            tr[d + 1][left ? x ++ : y ++] = tr[d][i];
        }
        build(d + 1, l, m, sorted);
        build(d + 1, m, r, sorted);
    }
    T kth(int d, int l, int r, int ql, int qr, int k) const {
        if (r - l == 1) return tr[d][l];
        int m{ (l + r) / 2 };
        int before{ pref[d][ql] - pref[d][l] };
        int cnt{ pref[d][qr] - pref[d][ql] };
        if (k < cnt) {
            int nl{ l + before };
            return kth(d + 1, l, m, nl, nl + cnt, k);
        }
        int nl{ m + (ql - l) - before };
        return kth(d + 1, m, r, nl, nl + (qr - ql - cnt), k - cnt);
    }
public:
    explicit PartitionTree(const vector<T> &a)
        : n(int(a.size())), lg(n <= 1 ? 0 : __lg(n - 1) + 1),
          tr(lg + 1, vector<T>(n)), pref(lg, vector<int>(n + 1)) {
        tr[0] = a;
        // sorted 只在建树时需要，不作为成员长期保存
        vector<T> sorted(a);
        ranges::sort(sorted);
        build(0, 0, n, sorted);
    }
    T kth(int l, int r, int k) const {
        assert(0 <= l && l < r && r <= n);
        assert(0 <= k && k < r - l);
        return kth(0, 0, n, l, r, k);
    }
};
```

- `PartitionTree(a)`：预处理静态数组，$O(n\log n)$
- `kth(l,r,k)`：返回 `[l,r)` 中第 `k` 小的值，`k` 为 `0base`，$O(\log n)$
划分树只保留最核心的区间第 `k` 小接口；需要复杂值域统计时优先使用下面的小波矩阵
### 小波矩阵（离散值 + 压位）
---
```cpp
template<class T>
struct WaveletMatrix {
private:
    struct BitVector {
        int words{};
        vector<u64> bit;
        vector<int> sum;
        BitVector() = default;
        explicit BitVector(int n) : words((n + 63) >> 6), bit(words), sum(words + 1) {}
        void set(int i) { bit[i >> 6] |= 1ULL << (i & 63); }
        void build() {
            for (int i = 0; i < words; ++ i) {
                sum[i + 1] = sum[i] + popcount(bit[i]);
            }
        }
        // [0,i) 中 1 的数量
        int rank1(int i) const {
            int w{ i >> 6 }, b{ i & 63 };
            return sum[w] + (b ? popcount(bit[w] & ((1ULL << b) - 1)) : 0);
        }
    };
    int n, lg;
    vector<T> xs;
    vector<int> zero;
    vector<BitVector> mat;
    int kth_id(int l, int r, int k) const {
        int id{};
        for (int d = 0; d < lg; ++ d) {
            int bit{ lg - 1 - d };
            int one_l{ mat[d].rank1(l) };
            int one_r{ mat[d].rank1(r) };
            int zeros{ (r - l) - (one_r - one_l) };
            if (k < zeros) {
                l -= one_l;
                r -= one_r;
            } else {
                k -= zeros;
                id |= 1 << bit;
                l = zero[d] + one_l;
                r = zero[d] + one_r;
            }
        }
        return id;
    }
    // 统计离散编号 < upper 的元素个数
    int count_id_less(int l, int r, int upper) const {
        assert(0 <= l && l <= r && r <= n);
        if (upper <= 0) return 0;
        if (upper >= int(xs.size())) return r - l;
        int ans{};
        for (int d = 0; d < lg; ++ d) {
            int bit{ lg - 1 - d }, one_l{ mat[d].rank1(l) };
            int one_r{ mat[d].rank1(r) }, zeros{ (r - l) - (one_r - one_l) };
            if (upper >> bit & 1) {
                ans += zeros;
                l = zero[d] + one_l;
                r = zero[d] + one_r;
            } else {
                l -= one_l;
                r -= one_r;
            }
        }
        return ans;
    }
public:
    explicit WaveletMatrix(const vector<T> &a) : n(int(a.size())), xs(a) {
        ranges::sort(xs);
        // 只要求 T 定义严格弱序 <，不额外要求 ==
        xs.erase(ranges::unique(xs, [](const T &a, const T &b) {
            return !(a < b) && !(b < a);
        }).begin(), xs.end());
        int sigma{ int(xs.size()) };
        lg = sigma <= 1 ? 0 : __lg(sigma - 1) + 1;
        zero.resize(lg);
        mat.reserve(lg);
        vector<int> cur(n), nxt(n);
        for (int i = 0; i < n; ++ i) {
            cur[i] = ranges::lower_bound(xs, a[i]) - xs.begin();
        }
        // 从高位到低位稳定划分
        for (int d = 0; d < lg; ++ d) {
            int bit{ lg - 1 - d };
            mat.emplace_back(n);
            for (int i = 0; i < n; ++ i) {
                if (cur[i] >> bit & 1) mat.back().set(i);
            }
            mat.back().build();
            zero[d] = n - mat.back().rank1(n);
            int p0{ 0 }, p1{ zero[d] };
            for (int x : cur) {
                if (x >> bit & 1) nxt[p1 ++] = x;
                else nxt[p0 ++] = x;
            }
            cur.swap(nxt);
        }
    }
    // ===== 顺序统计 =====
    T kth(int l, int r, int k) const {
        assert(0 <= l && l < r && r <= n);
        assert(0 <= k && k < r - l);
        return xs[kth_id(l, r, k)];
    }
    // ===== 频次统计 =====
    int count_less(int l, int r, const T &x) const {
        int upper{ int(ranges::lower_bound(xs, x) - xs.begin()) };
        return count_id_less(l, r, upper);
    }
    int count_leq(int l, int r, const T &x) const {
        int upper{ int(ranges::upper_bound(xs, x) - xs.begin()) };
        return count_id_less(l, r, upper);
    }
    int count(int l, int r, const T &x) const {
        return count_leq(l, r, x) - count_less(l, r, x);
    }
    int count_range(int l, int r, const T &lo, const T &hi) const {
        assert(!(hi < lo));
        return count_less(l, r, hi) - count_less(l, r, lo);
    }
    // ===== 前驱后继：若不需要可以整段删除 =====
    optional<T> predecessor(int l, int r, const T &x) const {
        int k{ count_less(l, r, x) };
        if (k == 0) return nullopt;
        return kth(l, r, k - 1);
    }
    optional<T> successor(int l, int r, const T &x) const {
        int k{ count_leq(l, r, x) };
        if (k == r - l) return nullopt;
        return kth(l, r, k);
    }
    optional<T> next_ge(int l, int r, const T &x) const {
        int k{ count_less(l, r, x) };
        if (k == r - l) return nullopt;
        return kth(l, r, k);
    }
};
```
#### 小波矩阵
- `WaveletMatrix(a)`：离散化并建立小波矩阵
- `kth(l,r,k)`：返回 `[l,r)` 中第 `k` 小，`k` 为 `0base`
- `count_less(l,r,x)`：统计 `[l,r)` 中 `< x` 的数量
- `count_leq(l,r,x)`：统计 `[l,r)` 中 `<= x` 的数量
- `count(l,r,x)`：统计 `[l,r)` 中等于 `x` 的数量
- `count_range(l,r,lo,hi)`：统计位置 `[l,r)`、值域 `[lo,hi)` 中的元素数量
- `predecessor(l,r,x)`：返回 `< x` 的最大值，不存在返回 `nullopt`
- `successor(l,r,x)`：返回 `> x` 的最小值，不存在返回 `nullopt`
- `next_ge(l,r,x)`：返回 `>= x` 的最小值，不存在返回 `nullopt`
除构造外，上述查询均为 $O(\log\sigma)$，其中 $\sigma$ 为不同值数量
#### 注意事项
- 两个结构均为静态结构；原数组修改后必须重新构造
- 所有位置区间均采用 `0base` 左闭右开 `[l,r)`
- `kth(l,r,k)` 中 `k` 为 `0base`，合法范围为 `0 <= k < r-l`
- 划分树中不能简单按 `<= pivot` 分割。出现重复值时必须通过 `same` 精确控制进入左半部分的 `pivot` 数量，否则左右子树大小会错误
- 划分树的 `pref[d][i]` 是第 `d` 层位置前缀中进入左儿子的元素数量，用它把原查询区间映射到下一层
- 小波矩阵先把值离散为 `[0,\sigma)`，因此可以直接处理负数以及任意可排序类型
- 本版 `T` 只要求能够通过 `<` 建立严格弱序，不要求定义 `==`
- `count_range(l,r,lo,hi)` 的值域也是半开区间 `[lo,hi)`
- `predecessor()` / `successor()` 均为严格前驱、严格后继；如果需要 `>= x` 的非严格后继使用 `next_ge()`
- 小波矩阵本身只维护离散编号和位向量，因此当前版本只能查询**数量与值**，不维护区间元素和。若需要“前 `k` 小之和 / 值域和”，需要在每层额外维护前缀和
- 小波矩阵中的 `BitVector` 使用 `u64` 压位，并每 `64` 位保存一次前缀 popcount；相比每层保存 `n+1` 个 `int` 前缀数组，内存常数显著降低
- 两个结构都可以构造空数组，但所有 `kth()` 查询必须保证区间非空
#### 常见用法
##### 区间第 k 小
```cpp
WaveletMatrix<int> wm(a);
int x = wm.kth(l, r, k);
```
##### 区间中位数
下中位数：
```cpp
int median = wm.kth(l, r, (r - l - 1) / 2);
```
上中位数：
```cpp
int median = wm.kth(l, r, (r - l) / 2);
```
##### 值域频次
统计 `[l,r)` 中满足 `low <= a[i] < high` 的元素：
```cpp
int cnt = wm.count_range(l, r, low, high);
```
统计某个值出现次数：
```cpp
int cnt = wm.count(l, r, x);
```
##### 排名
区间中严格小于 `x` 的元素数量就是 `x` 的 `0base` lower-bound 排名：
```cpp
int rank = wm.count_less(l, r, x);
```
##### 前驱与后继
```cpp
auto pre = wm.predecessor(l, r, x); // 最大的 < x
auto nxt = wm.successor(l, r, x);   // 最小的 > x

if (pre) cout << *pre << '\n';
if (nxt) cout << *nxt << '\n';
```
如果题目需要最小的 `>= x`：
```cpp
auto nxt = wm.next_ge(l, r, x);
```
##### 自定义可排序结构体
```cpp
struct WaveletKey {
    i64 value{};
    int id{};
    friend bool operator<(const WaveletKey &a, const WaveletKey &b) {
        return tie(a.value, a.id) < tie(b.value, b.id);
    }
};

vector<WaveletKey> a(n);
WaveletMatrix<WaveletKey> wm(a);

WaveletKey x = wm.kth(l, r, k);
```
如果只希望按 `value` 查询而不区分 `id`，不要把 `id` 加入排序键
#### 复杂度
设不同值数量为 $\sigma$

|结构 / 操作|时间复杂度|
|---|--:|
|划分树构造|$O(n\log n)$|
|划分树 `kth`|$O(\log n)$|
|小波矩阵离散化与构造|$O(n\log n)$|
|小波矩阵 `kth`|$O(\log\sigma)$|
|小波矩阵计数查询|$O(\log\sigma)$|
|小波矩阵前驱 / 后继|$O(\log\sigma)$|
划分树空间复杂度为 $O(n\log n)$
小波矩阵的位向量部分约占 $O(n\log\sigma)$ bits，并额外维护每 `64` 位一个前缀计数；离散值表占 $O(\sigma)$ 个 `T`
