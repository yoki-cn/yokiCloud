### 二维 ST 表
---
```cpp
template<class T, class Merge>
struct SparseTable2D {
    int n, m, kn, km;
    Merge merge;
    vector<vector<T>> st;
    explicit SparseTable2D(const vector<vector<T>> &a, Merge merge_ = Merge())
        : n(int(a.size())), m(n ? int(a[0].size()) : 0), merge(move(merge_)) {
        assert(n > 0 && m > 0);
        for (const auto &row : a) assert(int(row.size()) == m);
        kn = __lg(n) + 1;
        km = __lg(m) + 1;
        st.resize(kn * km);
        // st[x * km + y] 连续存储所有 2^x * 2^y 子矩形
        st[0].resize(n * m);
        for (int i = 0; i < n; ++ i) {
            copy(a[i].begin(), a[i].end(), st[0].begin() + i * m);
        }
        // 先沿列方向倍增
        for (int y = 1; y < km; ++ y) {
            int len{ 1 << y }, half{ len >> 1 };
            int w{ m - len + 1 }, pw{ m - half + 1 };
            auto &cur{ st[y] };
            const auto &pre{ st[y - 1] };
            cur.resize(n * w);
            for (int i = 0; i < n; ++ i) {
                for (int j = 0; j < w; ++ j) {
                    cur[i * w + j] = merge(pre[i * pw + j], pre[i * pw + j + half]);
                }
            }
        }
        // 再沿行方向倍增
        for (int x = 1; x < kn; ++ x) {
            int len{ 1 << x }, half{ len >> 1 }, h{ n - len + 1 };
            for (int y = 0; y < km; ++ y) {
                int w{ m - (1 << y) + 1 };
                auto &cur{ st[x * km + y] };
                const auto &pre{ st[(x - 1) * km + y] };
                cur.resize(h * w);
                for (int i = 0; i < h; ++ i) {
                    for (int j = 0; j < w; ++ j) {
                        cur[i * w + j] = merge(pre[i * w + j], pre[(i + half) * w + j]);
                    }
                }
            }
        }
    }
    T query(int x1, int y1, int x2, int y2) const {
        assert(0 <= x1 && x1 < x2 && x2 <= n);
        assert(0 <= y1 && y1 < y2 && y2 <= m);
        int kx{ __lg(x2 - x1) }, ky{ __lg(y2 - y1) }, hx{ 1 << kx }, wy{ 1 << ky };
        int xa{ x2 - hx }, ya{ y2 - wy }, w{ m - wy + 1 };
        const auto &v{ st[kx * km + ky] };
        auto get = [&](int x, int y) -> const T & {
            return v[x * w + y];
        };
        return merge(
            merge(get(x1, y1), get(x1, ya)),
            merge(get(xa, y1), get(xa, ya))
        );
    }
};
```
- `SparseTable2D<T, Merge>(a, merge)`：对非空矩阵 `a` 建立二维 ST 表，$O(nm\log n\log m)$
- `query(x1, y1, x2, y2)`：查询半开矩形 `[x1,x2) × [y1,y2)`，$O(1)$
- `Merge`：通常使用 `min`、`max`、`gcd`、按位与、按位或等幂等运算
#### 注意事项
- 二维 ST 的空间常数很大，实际存储元素数为
$$\sum_{x=0}^{\lfloor\log_2 n\rfloor} \sum_{y=0}^{\lfloor\log_2 m\rfloor} (n-2^x+1)(m-2^y+1)$$
大规模矩阵使用前应先估算内存
- 若 `T` 是较大的结构体，二维 ST 的内存会进一步放大；只需要位置时可以考虑仅保存下标
#### 二维最小值
```cpp
struct Min {
    int operator()(int a, int b) const { return min(a, b); }
};
SparseTable2D<int, Min> st(a);
int ans = st.query(x1, y1, x2, y2);
```
#### 同时返回二维最小值及位置
```cpp
struct GridInfo {
    i64 value{};
    int x{ -1 }, y{ -1 };
};

struct MinGridInfo {
    GridInfo operator()(const GridInfo &a, const GridInfo &b) const {
        return tie(a.value, a.x, a.y) < tie(b.value, b.x, b.y) ? a : b;
    }
};

vector<vector<GridInfo>> a(n, vector<GridInfo>(m));
for (int i = 0; i < n; ++ i) {
    for (int j = 0; j < m; ++ j) {
        a[i][j] = { value[i][j], i, j };
    }
}

SparseTable2D<GridInfo, MinGridInfo> st(a);
GridInfo ans = st.query(x1, y1, x2, y2);
```
### 猫树（Disjoint Sparse Table）
---
```cpp
template<class T, class Merge>
struct DisjointSparseTable {
    int n, lg;
    Merge merge;
    vector<T> a;
    vector<vector<T>> st;
    explicit DisjointSparseTable(vector<T> a_, Merge merge_ = Merge())
        : n(int(a_.size())), lg(n <= 1 ? 0 : __lg(n - 1) + 1),
          merge(move(merge_)), a(move(a_)), st(lg, vector<T>(n)) {
        assert(n > 0);
        // 第 k 层每块长度为 2^(k+1)，分别维护中点向左、向右的区间积
        for (int k = 0; k < lg; ++ k) {
            int half{ 1 << k }, len{ half << 1 };
            for (int l = 0; l < n; l += len) {
                int mid{ min(l + half, n) }, r{ min(l + len, n) };
                st[k][mid - 1] = a[mid - 1];
                for (int i = mid - 2; i >= l; -- i) {
                    st[k][i] = merge(a[i], st[k][i + 1]);
                }
                if (mid < r) {
                    st[k][mid] = a[mid];
                    for (int i = mid + 1; i < r; ++ i) {
                        st[k][i] = merge(st[k][i - 1], a[i]);
                    }
                }
            }
        }
    }
    T query(int l, int r) const {
        assert(0 <= l && l < r && r <= n);
        if (r - l == 1) return a[l];
        int k{ __lg(l ^ (r - 1)) };
        return merge(st[k][l], st[k][r - 1]);
    }
};
```
- `DisjointSparseTable<T, Merge>(a, merge)`：对非空静态数组建立猫树，$O(n\log n)$
- `query(l, r)`：查询半开区间 `[l,r)`，$O(1)$
- `Merge`：只要求满足结合律，不要求幂等，也不要求交换律
#### 注意事项
- 猫树查询时 `merge(st[k][l], st[k][r-1])` 的顺序不能交换，因此可以正确处理字符串连接、矩阵乘法等非交换结合运算
- 猫树单元素区间直接返回 `a[l]`；否则 `l ^ (r - 1)` 一定非零，可以安全调用 `__lg()`
- `Merge::operator()` 应能够在 `const` 查询中调用，通常写成：
```cpp
T operator()(const T &a, const T &b) const;
```
#### 猫树求区间和
```cpp
DisjointSparseTable<i64, plus<i64>> st(a);
i64 ans = st.query(l, r);
```
#### 猫树处理非交换运算
例如字符串拼接：
```cpp
struct Merge {
    string operator()(const string &a, const string &b) const {
        return a + b;
    }
};

DisjointSparseTable<string, Merge> st(a);
string ans = st.query(l, r);
```
矩阵乘法同理，只要 `merge(a, b)` 表示严格的 `a * b` 即可
