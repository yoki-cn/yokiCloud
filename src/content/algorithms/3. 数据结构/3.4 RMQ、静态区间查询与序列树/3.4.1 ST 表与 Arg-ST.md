### ST 表
---
```cpp
struct OR {
    template<class T>
    T operator()(const T &a, const T &b) const { return a | b; }
};
struct AND {
    template<class T>
    T operator()(const T &a, const T &b) const { return a & b; }
};
struct MAX {
    template<class T>
    T operator()(const T &a, const T &b) const { return max(a, b); }
};
struct MIN {
    template<class T>
    T operator()(const T &a, const T &b) const { return min(a, b); }
};
struct GCD {
    template<class T>
    T operator()(T a, T b) const { return gcd(a, b); }
};
template<class T, class Merge>
struct ST {
    int n{};
    Merge merge;
    vector<vector<T>> st;
    explicit ST(vector<T> a, Merge m = Merge()) : merge(m) { build(move(a)); }
    void build(vector<T> a) {
        n = int(a.size());
        int k{ __lg(n) + 1 };
        st.assign(k, {});
        st[0] = move(a);
        for (int i = 1; i < k; ++ i) {
            int len{ 1 << i }, half{ len >> 1 };
            st[i].resize(n - len + 1);
            for (int j = 0; j + len <= n; ++ j) st[i][j] = merge(st[i - 1][j], st[i - 1][j + half]);
        }
    }
    T query(int l, int r) const {
        if (l > r) swap(l, r);
        int k{ __lg(r - l + 1) };
        return merge(st[k][l], st[k][r - (1 << k) + 1]);
    }
};
```
### Argmin / Argmax
修改比较器即可切换 `Argmin` / `Argmax`
```cpp
template<class T, class Cmp = less<T>>
struct ArgST {
    int n{};
    bool prefer_left{ true };
    Cmp cmp;
    vector<T> a;
    vector<vector<int>> st;
    explicit ArgST(vector<T> a_, Cmp c = Cmp(), bool preferLeft = true) : prefer_left(preferLeft), cmp(c) { build(move(a_)); }
    void build(vector<T> a_) {
        a = move(a_);
        n = int(a.size());
        int k{ __lg(n) + 1 };
        st.assign(k, {});
        st[0].resize(n);
        iota(st[0].begin(), st[0].end(), 0);
        for (int j = 1; j < k; ++ j) {
            int len{ 1 << j }, half{ len >> 1 };
            st[j].resize(n - len + 1);
            for (int i = 0; i + len <= n; ++ i) st[j][i] = better(st[j - 1][i], st[j - 1][i + half]);
        }
    }
    int queryIndex(int l, int r) const {
        if (l > r) swap(l, r);
        int k{ __lg(r - l + 1) };
        return better(st[k][l], st[k][r - (1 << k) + 1]);
    }
    const T &query(int l, int r) const { return a[queryIndex(l, r)]; }
    const T &at(int i) const { return a[i]; }
private:
    int better(int i, int j) const {
        if (i < 0) return j;
        if (j < 0) return i;
        if (cmp(a[i], a[j])) return i;
        if (cmp(a[j], a[i])) return j;
        return prefer_left ? min(i, j) : max(i, j);
    }
};
```
### 接口说明
#### 成员变量
- `n`：数组长度
- `merge`：区间合并函数
- `st[k][i]`：长度为 `2^k`、左端点为 `i` 的区间信息
- `ArgST::a`：保存原数组，避免外部数组失效
- `ArgST::prefer_left`：最优值相同时是否返回更靠左的位置
#### 接口
- `ST(vector<T> a, Merge m = Merge())`：用数组 `a` 建表，$O(n\log n)$
- `build(vector<T> a)`：重新建表，$O(n\log n)$
- `query(l, r)`：查询闭区间 `[l, r]` 的合并值，$O(1)$
- `ArgST(vector<T> a, Cmp c = Cmp(), bool preferLeft = true)`：用比较器建位置表，$O(n\log n)$
- `queryIndex(l, r)`：返回闭区间 `[l, r]` 内最优元素下标，$O(1)$
- `ArgST::query(l, r)`：返回闭区间 `[l, r]` 内最优元素引用，$O(1)$
- `at(i)`：返回原数组第 `i` 个元素引用，$O(1)$
### 注意事项
- 下标为 `0base`，查询区间为闭区间 `[l, r]`
- `ST` 只能用于可重叠合并的幂等运算，例如 `min`、`max`、`gcd`、按位或、按位与
- `sum`、矩阵乘法这类非幂等运算不能直接用这个 `query(l, r)`，因为左右两段可能重叠
- `ArgST` 默认 `less<T>`，即返回最小值位置；若要最大值，使用 `greater<T>`
### 常见用法
#### 区间极值位置
```cpp
ArgST<int> st(a); // 最大值改为 ArgST<int, greater<int>> st(a);
int p{ st.queryIndex(l, r) }, val{ st.query(l, r) };
```

##### ST
```cpp
struct Info {
    i64 key = 0;
    int id = -1;
};
struct MinKey {
    Info operator()(const Info &a, const Info &b) const {
        if (a.key != b.key) return a.key < b.key ? a : b;
        return a.id < b.id ? a : b;
    }
};
vector<Info> a(n);
for (int i = 0; i < n; ++ i) {
    cin >> a[i].key;
    a[i].id = i;
}
ST<Info, MinKey> st(a);
Info res{ st.query(l, r) };
int pos{ res.id };
```
##### ArgST
```cpp
struct Info {  
    i64 key = 0;  
    int id = -1;  
};
struct CmpKey {
    bool operator()(const Info &a, const Info &b) const {
        if (a.key != b.key) return a.key < b.key;  
        return a.id < b.id;
    }
};
vector<Info> a(n);
for (int i = 0; i < n; ++ i) cin >> a[i].key;
ArgST<Info, CmpKey> st(a);
int pos{ st.queryIndex(l, r) };
Info res{ st.query(l, r) };
```
