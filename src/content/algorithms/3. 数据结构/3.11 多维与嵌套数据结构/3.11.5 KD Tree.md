### KD Tree（静态与动态）
---
二维空间最近邻和矩形计数；动态版支持插入、按编号删除、自动重构
#### 静态二维 KD Tree
---
```cpp
struct KDTree {
    using i64 = long long;
    using i128 = __int128_t;
    static constexpr i128 INF = numeric_limits<i128>::max();
    struct Point {
        i64 x[2];
        int id;
    };
    struct Node {
        Point p{};
        int ch[2]{};
        i64 mn[2]{}, mx[2]{};
        int size{1};
    };
    vector<Node> tr{{}};
    int root{0};
    explicit KDTree(vector<Point> p) { root = build(p, 0, int(p.size()), 0); }
    static i128 sq(i128 x) { return x * x; }
    static i128 distance(const Point &a, const Point &b) { return sq(i128(a.x[0]) - b.x[0]) + sq(i128(a.x[1]) - b.x[1]); }
    int new_node(const Point &p) {
        int u = int(tr.size());
        tr.push_back({p});
        for (int d = 0; d < 2; ++d) tr[u].mn[d] = tr[u].mx[d] = p.x[d];
        return u;
    }
    void pull(int u) {
        tr[u].size = 1;
        for (int c : tr[u].ch) if (c) tr[u].size += tr[c].size;
        for (int d = 0; d < 2; ++d) {
            tr[u].mn[d] = tr[u].mx[d] = tr[u].p.x[d];
            for (int c : tr[u].ch) if (c) {
                tr[u].mn[d] = min(tr[u].mn[d], tr[c].mn[d]);
                tr[u].mx[d] = max(tr[u].mx[d], tr[c].mx[d]);
            }
        }
    }
    int build(vector<Point> &p, int l, int r, int depth) {
        if (l == r) return 0;
        int d = depth & 1, m = (l + r) / 2;
        nth_element(p.begin() + l, p.begin() + m, p.begin() + r,
            [d](const Point &a, const Point &b) {
                return pair{a.x[d], a.id} < pair{b.x[d], b.id};
            });
        int u = new_node(p[m]);
        tr[u].ch[0] = build(p, l, m, depth + 1);
        tr[u].ch[1] = build(p, m + 1, r, depth + 1);
        pull(u);
        return u;
    }
    i128 min_box(int u, const Point &q) const {
        i128 ans = 0;
        for (int d = 0; d < 2; ++d) {
            if (q.x[d] < tr[u].mn[d]) ans += sq(i128(tr[u].mn[d]) - q.x[d]);
            if (q.x[d] > tr[u].mx[d]) ans += sq(i128(q.x[d]) - tr[u].mx[d]);
        }
        return ans;
    }
    i128 max_box(int u, const Point &q) const {
        i128 ans = 0;
        for (int d = 0; d < 2; ++d) {
            ans += max(sq(i128(q.x[d]) - tr[u].mn[d]), sq(i128(q.x[d]) - tr[u].mx[d]));
        }
        return ans;
    }
    void nearest(int u, const Point &q, int ban, pair<i128, int> &best) const {
        if (!u) return;
        if (tr[u].p.id != ban) {
            best = min(best, pair{distance(tr[u].p, q), tr[u].p.id});
        }
        int a = tr[u].ch[0], b = tr[u].ch[1];
        if (a && b && min_box(a, q) > min_box(b, q)) swap(a, b);
        if (a && min_box(a, q) <= best.first) nearest(a, q, ban, best);
        if (b && min_box(b, q) <= best.first) nearest(b, q, ban, best);
    }
    void farthest(int u, const Point &q, int ban, pair<i128, int> &best) const {
        if (!u) return;
        if (tr[u].p.id != ban) {
            best = max(best, pair{distance(tr[u].p, q), -tr[u].p.id});
        }
        int a = tr[u].ch[0], b = tr[u].ch[1];
        if (a && b && max_box(a, q) < max_box(b, q)) swap(a, b);
        if (a && max_box(a, q) >= best.first) farthest(a, q, ban, best);
        if (b && max_box(b, q) >= best.first) farthest(b, q, ban, best);
    }
    void k_nearest(int u, const Point &q, int ban, int k,
                   priority_queue<pair<i128, int>> &heap) const {
        if (!u) return;
        if (tr[u].p.id != ban) {
            pair<i128, int> cand{distance(tr[u].p, q), tr[u].p.id};
            if (int(heap.size()) < k) heap.push(cand);
            else if (cand < heap.top()) {
                heap.pop();
                heap.push(cand);
            }
        }
        int a = tr[u].ch[0], b = tr[u].ch[1];
        if (a && b && min_box(a, q) > min_box(b, q)) swap(a, b);
        auto possible = [&](int v) {
            return v && (int(heap.size()) < k || min_box(v, q) <= heap.top().first);
        };
        if (possible(a)) k_nearest(a, q, ban, k, heap);
        if (possible(b)) k_nearest(b, q, ban, k, heap);
    }
    // ===== 静态点集矩形计数 =====
    int count_rect(int u, i64 x1, i64 y1, i64 x2, i64 y2) const {
        if (!u || tr[u].mx[0] < x1 || tr[u].mn[0] >= x2
               || tr[u].mx[1] < y1 || tr[u].mn[1] >= y2) return 0;
        if (x1 <= tr[u].mn[0] && tr[u].mx[0] < x2
         && y1 <= tr[u].mn[1] && tr[u].mx[1] < y2) return tr[u].size;
        const auto &p = tr[u].p;
        int ans = (x1 <= p.x[0] && p.x[0] < x2 && y1 <= p.x[1] && p.x[1] < y2);
        for (int c : tr[u].ch) ans += count_rect(c, x1, y1, x2, y2);
        return ans;
    }
    int count_rect(i64 x1, i64 y1, i64 x2, i64 y2) const {
        return count_rect(root, x1, y1, x2, y2);
    }
    pair<i128, int> nearest(const Point &q, int ban = -1) const {
        pair<i128, int> ans{INF, numeric_limits<int>::max()};
        nearest(root, q, ban, ans);
        return ans;
    }
    pair<i128, int> farthest(const Point &q, int ban = -1) const {
        pair<i128, int> ans{-1, numeric_limits<int>::min()};
        farthest(root, q, ban, ans);
        if (ans.first >= 0) ans.second = -ans.second;
        return ans;
    }
    vector<pair<i128, int>> k_nearest(const Point &q, int k, int ban = -1) const {
        if (k <= 0) return {};
        priority_queue<pair<i128, int>> heap;
        k_nearest(root, q, ban, k, heap);
        vector<pair<i128, int>> ans;
        while (!heap.empty()) ans.push_back(heap.top()), heap.pop();
        reverse(ans.begin(), ans.end());
        return ans;
    }
};
```
##### 动态二维 KD Tree
---
```cpp
struct DynamicKDTree {
    using i64 = long long;
    using i128 = __int128_t;
    static constexpr i128 INF = numeric_limits<i128>::max();
    struct Point { i64 x[2]; int id; };
    struct Node {
        Point p{};
        int ch[2]{}, size{1}, dim{0};
        i64 mn[2]{}, mx[2]{};
        bool alive{true};
    };
    vector<Node> tr{{}};
    vector<int> bin;
    unordered_map<int, int> where;
    int root{0}, live_count{0};
    static i128 sq(i128 x) { return x * x; }
    static i128 distance(const Point &a, const Point &b) { return sq(i128(a.x[0]) - b.x[0]) + sq(i128(a.x[1]) - b.x[1]); }
    int size(int u) const { return u ? tr[u].size : 0; }
    int new_node(Point p, int dim) {
        int u;
        if (bin.empty()) {
            u = int(tr.size());
            tr.push_back({});
        } else {
            u = bin.back();
            bin.pop_back();
        }
        tr[u] = Node{p, {}, 1, dim};
        for (int d = 0; d < 2; ++d) tr[u].mn[d] = tr[u].mx[d] = p.x[d];
        where[p.id] = u;
        ++live_count;
        return u;
    }
    void pull(int u) {
        tr[u].size = 1;
        for (int d = 0; d < 2; ++d) tr[u].mn[d] = tr[u].mx[d] = tr[u].p.x[d];
        for (int c : tr[u].ch) if (c) {
            tr[u].size += tr[c].size;
            for (int d = 0; d < 2; ++d) {
                tr[u].mn[d] = min(tr[u].mn[d], tr[c].mn[d]);
                tr[u].mx[d] = max(tr[u].mx[d], tr[c].mx[d]);
            }
        }
    }
    void collect(int u, vector<int> &ids) {
        if (!u) return;
        collect(tr[u].ch[0], ids);
        if (tr[u].alive) ids.push_back(u);
        else bin.push_back(u);
        collect(tr[u].ch[1], ids);
    }
    int rebuild(vector<int> &ids, int l, int r, int depth) {
        if (l == r) return 0;
        int d = depth & 1, m = (l + r) / 2;
        nth_element(ids.begin() + l, ids.begin() + m, ids.begin() + r,
            [&](int a, int b) {
                return pair{tr[a].p.x[d], tr[a].p.id}
                     < pair{tr[b].p.x[d], tr[b].p.id};
            });
        int u = ids[m];
        tr[u].dim = d;
        tr[u].ch[0] = rebuild(ids, l, m, depth + 1);
        tr[u].ch[1] = rebuild(ids, m + 1, r, depth + 1);
        pull(u);
        return u;
    }
    int rebuild_subtree(int u, int depth) {
        vector<int> ids;
        collect(u, ids);
        return rebuild(ids, 0, int(ids.size()), depth);
    }
    bool unbalanced(int u) const { return 4LL * max(size(tr[u].ch[0]), size(tr[u].ch[1])) > 3LL * tr[u].size; }
    int insert(int u, Point p, int depth) {
        if (!u) return new_node(p, depth & 1);
        int d = tr[u].dim;
        int side = pair{p.x[d], p.id} > pair{tr[u].p.x[d], tr[u].p.id};
        tr[u].ch[side] = insert(tr[u].ch[side], p, depth + 1);
        pull(u);
        if (unbalanced(u)) u = rebuild_subtree(u, depth);
        return u;
    }
    void insert(Point p) {
        assert(!where.contains(p.id));
        root = insert(root, p, 0);
    }
    bool erase(int id) {
        auto it = where.find(id);
        if (it == where.end()) return false;
        tr[it->second].alive = false;
        where.erase(it);
        --live_count;
        if (root && 2LL * live_count < size(root)) root = rebuild_subtree(root, 0);
        return true;
    }
    i128 min_box(int u, const Point &q) const {
        i128 ans = 0;
        for (int d = 0; d < 2; ++d) {
            if (q.x[d] < tr[u].mn[d]) ans += sq(i128(tr[u].mn[d]) - q.x[d]);
            if (q.x[d] > tr[u].mx[d]) ans += sq(i128(q.x[d]) - tr[u].mx[d]);
        }
        return ans;
    }
    void nearest(int u, const Point &q, int ban, pair<i128, int> &best) const {
        if (!u) return;
        if (tr[u].alive && tr[u].p.id != ban) {
            best = min(best, pair{distance(tr[u].p, q), tr[u].p.id});
        }
        int a = tr[u].ch[0], b = tr[u].ch[1];
        if (a && b && min_box(a, q) > min_box(b, q)) swap(a, b);
        if (a && min_box(a, q) <= best.first) nearest(a, q, ban, best);
        if (b && min_box(b, q) <= best.first) nearest(b, q, ban, best);
    }
    pair<i128, int> nearest(Point q, int ban = -1) const {
        pair<i128, int> ans{INF, numeric_limits<int>::max()};
        nearest(root, q, ban, ans);
        return ans;
    }
};
```
#### 接口说明
- 静态 `KDTree(points)`：构造，通常 $O(n\log n)$；`nearest(q,ban)`：最近点，`farthest(q,ban)`：最远点，返回 `{距离平方,id}`。
- 静态 `k_nearest(q,k,ban)`：按 `{距离平方,id}` 升序输出至多 `k` 个点；`count_rect(x1,y1,x2,y2)`：统计半开矩形中的点数。
- 动态 `DynamicKDTree::insert({{x,y},id})`：插入新点；`erase(id)`：成功删除返回 `true`，未找到返回 `false`；`nearest(q,ban)`：当前活跃点中最近点。
- 两版最近邻不存在候选时均返回 `{INF,INT_MAX}`；静态最远点不存在时返回 `{-1,INT_MIN}`；`ban` 默认 `-1`，用于排除指定 `id`。
#### 注意事项
- **必须在坐标差与乘方前转换为 `i128`**；已修复 `i64` 先做差导致的溢出。仍须保证最终两维平方和不超过 `i128` 上限。
- `id` 应使用非负整数，不能与默认排除哨兵 `-1` 冲突；同一活动点集中不允许重复 `id`。
- 距离相同按较小 `id` 决胜，因此矩形剪枝必须保留相等边界。
- 动态版的删除为惰性删除：节点失衡或死亡节点过多时重构；被重构丢弃的节点编号进入 `bin` 以供复用。
- 动态版仅封装最近邻，矩形查询是静态版提供的可选功能；不要误认为动态版也实现了 `count_rect`。
- KD-Tree 的近邻查询最坏 $O(n)$；均衡构造和包围盒剪枝仅改善通常情形，不能保证每次对数复杂度。
#### 常见用法
```cpp
KDTree kd(points);
auto [dist2, id] = kd.nearest(query, ban);
auto near = kd.k_nearest(query, k);
int cnt = kd.count_rect(x1, y1, x2, y2);

DynamicKDTree dyn;
dyn.insert({{x, y}, id});
bool ok = dyn.erase(id);
auto [d2, who] = dyn.nearest(query);
```
#### 复杂度
静态构造通常 $O(n\log n)$、空间 $O(n)$。静态最近邻/最远邻/k 近邻与矩形查询均依赖剪枝效果，最坏可达 $O(n)$（`k_nearest` 另有结果堆操作成本）。动态插入包含偶发重构，无法保证对数最坏时间；全局重构 $O(n\log n)$，节点池按历史峰值分配。
