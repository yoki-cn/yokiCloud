### BIT 套动态权值线段树
---
外层按位置维护 Fenwick 树，每棵内部动态权值树支持计数与共同下降求第 k 小。
```cpp
struct BITValueTree {
    struct Node {
        int ch[2]{};
        int sum{};
    };
    int n;
    i64 vl, vr;
    vector<int> root;
    vector<Node> tr{{}};
    vector<int> bin;
    BITValueTree(int n_, i64 vl_, i64 vr_)
        : n(n_), vl(vl_), vr(vr_), root(n + 1) {
        assert(0 <= n && vl < vr);
    }
    BITValueTree(const vector<i64> &a, i64 vl_, i64 vr_)
        : BITValueTree(int(a.size()), vl_, vr_) {
        for (int i = 0; i < n; ++ i) add(i, a[i], 1);
    }
    void reserve(size_t nodes) { tr.reserve(nodes + 1); }
private:
    // ===== 节点复用 =====
    int new_node() {
        if (bin.empty()) {
            tr.push_back({});
            return int(tr.size()) - 1;
        }
        int p = bin.back();
        bin.pop_back();
        tr[p] = {};
        return p;
    }
    int add_node(int p, i64 l, i64 r, i64 x, int delta) {
        if (!p) {
            assert(delta > 0);
            p = new_node();
        }
        tr[p].sum += delta;
        assert(tr[p].sum >= 0);
        if (l < r - 1) {
            i64 m = midpoint(l, r);
            int d = x >= m;
            tr[p].ch[d] = add_node(tr[p].ch[d], d ? m : l,
                                   d ? r : m, x, delta);
        }
        if (tr[p].sum == 0) {
            assert(!tr[p].ch[0] && !tr[p].ch[1]);
            tr[p] = {};
            bin.push_back(p);
            return 0;
        }
        return p;
    }
    int prefix_node(int p, i64 l, i64 r, i64 x) const {
        if (!p || x <= l) return 0;
        if (r <= x) return tr[p].sum;
        i64 m = midpoint(l, r);
        return prefix_node(tr[p].ch[0], l, m, x)
             + prefix_node(tr[p].ch[1], m, r, x);
    }
public:
    // ===== 基础修改与值域计数 =====
    void add(int pos, i64 x, int delta) {
        assert(0 <= pos && pos < n && vl <= x && x < vr);
        for (++ pos; pos <= n; pos += pos & -pos) {
            root[pos] = add_node(root[pos], vl, vr, x, delta);
        }
    }
    void modify(int pos, i64 old_value, i64 new_value) {
        if (old_value == new_value) return;
        add(pos, old_value, -1);
        add(pos, new_value, 1);
    }
    int prefix_count(int pos, i64 value) const {
        int res = 0;
        for (; pos > 0; pos -= pos & -pos) {
            res += prefix_node(root[pos], vl, vr, value);
        }
        return res;
    }
    int count(int l, int r, i64 x) const {
        return prefix_count(r, x) - prefix_count(l, x);
    }
    int count_range(int l, int r, i64 lo, i64 hi) const {
        return count(l, r, hi) - count(l, r, lo);
    }
    int size(int l, int r) const {
        auto pref = [&](int x) {
            int s = 0;
            for (; x > 0; x -= x & -x) s += tr[root[x]].sum;
            return s;
        };
        return pref(r) - pref(l);
    }
    int rank(int l, int r, i64 x) const { return count(l, r, x) + 1; }
    // ===== 第 k 小：若仅需要计数，可删除此函数 =====
    i64 kth(int l, int r, int k) const {
        assert(1 <= k && k <= size(l, r));
        array<int, 32> a{}, b{};
        int na = 0, nb = 0;
        for (int x = r; x > 0; x -= x & -x) a[na ++] = root[x];
        for (int x = l; x > 0; x -= x & -x) b[nb ++] = root[x];
        i64 L = vl, R = vr;
        while (L < R - 1) {
            int cnt = 0;
            for (int i = 0; i < na; ++ i) cnt += tr[tr[a[i]].ch[0]].sum;
            for (int i = 0; i < nb; ++ i) cnt -= tr[tr[b[i]].ch[0]].sum;
            i64 M = midpoint(L, R);
            int d = k > cnt;
            if (d) k -= cnt, L = M;
            else R = M;
            for (int i = 0; i < na; ++ i) a[i] = tr[a[i]].ch[d];
            for (int i = 0; i < nb; ++ i) b[i] = tr[b[i]].ch[d];
        }
        return L;
    }
};
```
#### 接口说明
- `BITValueTree(n, vl, vr)` / `BITValueTree(a, vl, vr)`：空建或从数组建树，值域 `[vl, vr)`。
- `add(pos, x, delta)`：在位置 `pos` 增删权值 `x` 的出现次数，$O(\log n\log V)$。
- `modify(pos, old, now)`：单个位置换值，$O(\log n\log V)$。
- `prefix_count(pos, x)` / `count(l,r,x)`：统计位置前缀或 `[l,r)` 中权值 `< x` 的个数，$O(\log n\log V)$。
- `count_range(l,r,lo,hi)`：位置 `[l,r)` 且权值 `[lo,hi)` 的个数，$O(\log n\log V)$。
- `size(l,r)`：元素个数，$O(\log n)$；`rank(l,r,x)`：`< x` 的个数加一，$O(\log n\log V)$。
- `kth(l,r,k)`：区间第 `k` 小，`k` 为 **1base**，$O(\log n\log V)$。
- `reserve(nodes)`：预分配节点池容量，避免大规模增删时反复扩容。
#### 注意事项
- 数组位置采用 `0base`，区间统一 `[l,r)`；`add` 的频率必须始终非负，删除不存在的值违反前提。
- 权值域必须满足 `vl <= x < vr`。`midpoint(l,r)` 无有符号加法溢出风险；须包含 `<numeric>`（通用 `bits/stdc++.h` 已包含）。
- 单点换值时自行维护原数组。空子树会回收到 `bin`，但容器 `tr` 的容量不会自动缩小。
- 无法预先列举权值且需要 **动态第 k 小** 时优先本结构；能够离线登记所有修改值时也可以选择 `3.11.3`。
- 节点 `sum` 使用 `int`，总元素数量不得超过 `INT_MAX`；满 64 位有符号整数域的右开端点无法表示 `LLONG_MAX+1`。
#### 常见用法
```cpp
BITValueTree bit(a, -1000000000LL, 1000000001LL);
bit.modify(pos, a[pos], x);
a[pos] = x;
int k = bit.count(l, r, x);            // < x 的数量
if (k) i64 pre = bit.kth(l, r, k);    // 严格前驱
if (k < bit.size(l,r)) {
    i64 ge = bit.kth(l, r, k + 1);   // >= x 的最小值
}
```
#### 复杂度
所有增删与计数/第 k 小操作为 $O(\log n\log V)$，`size` 为 $O(\log n)$。令 $U$ 为历史更新次数，节点池的最坏上界 $O(U\log n\log V)$；回收后活跃节点随当前使用的权值路径数变化，但 `tr` 已分配容量按历史峰值保留。
