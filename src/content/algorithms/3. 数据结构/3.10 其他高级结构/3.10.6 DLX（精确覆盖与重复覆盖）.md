### DLX（精确覆盖与重复覆盖）
---
```cpp
struct DLX {
    struct Node {
        int l, r, u, d, c, row;
    };
    int m, rows{};
    vector<Node> tr;
    vector<int> sz, path, ans;
    explicit DLX(int columns, int ones_hint = 0) : m(columns), tr(m + 1), sz(m + 1) {
        assert(m >= 0 && ones_hint >= 0);
        tr.reserve(size_t(m) + 1 + ones_hint);
        for (int i = 0; i <= m; ++ i) {
            tr[i] = {i - 1, i + 1, i, i, i, -1};
        }
        tr[0].l = m;
        tr[m].r = 0;
    }
    // ===== 建立稀疏 01 矩阵：列号 0base，行内列号不得重复 =====
    void add_row(const vector<int> &cols, int row_id = -1) {
        int id = row_id < 0 ? rows : row_id;
        ++ rows;
        int first = 0;
        for (int x : cols) {
            assert(0 <= x && x < m);
            int c = x + 1, p = int(tr.size());
            tr.push_back({p, p, tr[c].u, c, c, id});
            tr[tr[c].u].d = p;
            tr[c].u = p;
            ++ sz[c];
            if (!first) {
                first = p;
            } else {
                int last = tr[first].l;
                tr[p].l = last;
                tr[p].r = first;
                tr[last].r = p;
                tr[first].l = p;
            }
        }
    }
    int choose_column() const {
        int chosen = tr[0].r;
        for (int c = tr[chosen].r; c != 0; c = tr[c].r) {
            if (sz[c] < sz[chosen]) chosen = c;
        }
        return chosen;
    }
    // ===== 精确覆盖：以下部分可与重复覆盖部分二选一 =====
    void cover(int c) {
        tr[tr[c].r].l = tr[c].l;
        tr[tr[c].l].r = tr[c].r;
        for (int i = tr[c].d; i != c; i = tr[i].d) {
            for (int j = tr[i].r; j != i; j = tr[j].r) {
                tr[tr[j].d].u = tr[j].u;
                tr[tr[j].u].d = tr[j].d;
                -- sz[tr[j].c];
            }
        }
    }
    void uncover(int c) {
        for (int i = tr[c].u; i != c; i = tr[i].u) {
            for (int j = tr[i].l; j != i; j = tr[j].l) {
                ++ sz[tr[j].c];
                tr[tr[j].d].u = j;
                tr[tr[j].u].d = j;
            }
        }
        tr[tr[c].r].l = c;
        tr[tr[c].l].r = c;
    }
    bool dfs_exact() {
        if (tr[0].r == 0) {
            ans = path;
            return true;
        }
        int c = choose_column();
        if (!sz[c]) return false;
        cover(c);
        for (int i = tr[c].d; i != c; i = tr[i].d) {
            path.push_back(tr[i].row);
            for (int j = tr[i].r; j != i; j = tr[j].r) cover(tr[j].c);
            bool ok = dfs_exact();
            for (int j = tr[i].l; j != i; j = tr[j].l) uncover(tr[j].c);
            path.pop_back();
            if (ok) {
                uncover(c);
                return true;
            }
        }
        uncover(c);
        return false;
    }
    bool solve_exact() {
        ans.clear();
        path.clear();
        return dfs_exact();
    }
    // ===== 重复覆盖：只删除列头，行的环形链表不变；不用时可整段删除 =====
    vector<unsigned char> active;
    vector<int> changed, seen;
    int stamp{}, best{};
    void remove_column(int c) {
        active[c] = false;
        tr[tr[c].r].l = tr[c].l;
        tr[tr[c].l].r = tr[c].r;
        changed.push_back(c);
    }
    void restore_columns(int checkpoint) {
        while (int(changed.size()) > checkpoint) {
            int c = changed.back();
            changed.pop_back();
            tr[tr[c].r].l = c;
            tr[tr[c].l].r = c;
            active[c] = true;
        }
    }
    // 贪心找两两不能被同一行覆盖的列，作为还需选择行数的下界
    int lower_bound(int limit) {
        if (stamp == INT_MAX) {
            fill(seen.begin(), seen.end(), 0);
            stamp = 0;
        }
        ++ stamp;
        int need = 0;
        for (int c = tr[0].r; c != 0; c = tr[c].r) {
            if (seen[c] == stamp) continue;
            if (++ need >= limit) return need;
            for (int i = tr[c].d; i != c; i = tr[i].d) {
                int j = i;
                do {
                    seen[tr[j].c] = stamp;
                    j = tr[j].r;
                } while (j != i);
            }
        }
        return need;
    }
    bool dfs_repeat(bool first_only) {
        if (tr[0].r == 0) {
            best = int(path.size());
            ans = path;
            return first_only;
        }
        if (lower_bound(best - int(path.size())) >= best - int(path.size())) return false;
        int c = choose_column();
        if (!sz[c]) return false;
        for (int i = tr[c].d; i != c; i = tr[i].d) {
            int checkpoint = changed.size();
            path.push_back(tr[i].row);
            int j = i;
            do {
                int col = tr[j].c;
                if (active[col]) remove_column(col);
                j = tr[j].r;
            } while (j != i);
            bool ok = dfs_repeat(first_only);
            restore_columns(checkpoint);
            path.pop_back();
            if (ok) return true;
        }
        return false;
    }
    void prepare_repeat(int limit) {
        ans.clear();
        path.clear();
        changed.clear();
        active.assign(m + 1, true);
        seen.assign(m + 1, 0);
        stamp = 0;
        best = min(rows, limit < 0 ? rows : limit) + 1;
    }
    // 返回所需的最少行数；无解或超过 limit 返回 -1，答案保存在 ans
    int min_cover(int limit = -1) {
        prepare_repeat(limit);
        dfs_repeat(false);
        return ans.empty() && m != 0 ? -1 : int(ans.size());
    }
    // 只判断是否能用不超过 limit 行覆盖；找到一组就结束
    bool cover_at_most(int limit) {
        assert(limit >= 0);
        prepare_repeat(limit);
        return dfs_repeat(true);
    }
};
```

## 接口说明

### 成员变量

- `m`：约束列数，外部列号为 `[0, m)`，内部列头为 `[1, m]`；`0` 为总表头。
- `tr`：十字链表节点池，`Node::{l,r,u,d}` 为上下左右四向链接，`c` 为所属列头编号，`row` 为原始行编号。
- `sz[c]`：当前列的候选行数。**精确覆盖**中随删列更新；**重复覆盖**中保留原始行数，便于选择分支列。
- `rows`：调用 `add_row()` 的次数（包括空行），`ans`：成功时选中行编号组成的一个解，按搜索顺序排列。
- `path`：当前搜索路径；`active/changed/seen/stamp/best` 仅供重复覆盖搜索使用。

### 构造与公开接口

| 接口 | 作用 | 复杂度 |
|---|---|---|
| `DLX(m, ones_hint = 0)` | 建立 `m` 列空矩阵，按需预留数据节点 | `O(m)` |
| `add_row(cols, row_id = -1)` | 加入一行，列号 `0base`；默认自动编号 | `O(|cols|)`，不计扩容 |
| `solve_exact()` | 查找**每列恰好出现一次**的任意解；成功返回 `true`，答案写入 `ans` | 指数级搜索 |
| `min_cover(limit = -1)` | 选最少行，使**每列至少出现一次**；若无解或最优解超过 `limit`，返回 `-1`；成功返回最小行数，答案写入 `ans` | 指数级搜索 |
| `cover_at_most(limit)` | 只判断能否用至多 `limit` 行覆盖所有列；找到一组即退出，成功时 `ans` 给出方案 | 指数级搜索 |

- `ones_hint` 是矩阵中 **1 的总个数**的容量预估，不是行数；设置不足不影响正确性。
- 精确覆盖只能选择互不冲突的行；重复覆盖允许已覆盖的列被其他选中行再次覆盖。
- `limit = -1` 仅对 `min_cover()` 表示不限制行数；`cover_at_most(limit)` 要求 `limit >= 0`。

## 注意事项

- **建图语义**：每行表示一个候选方案，每列表示一个约束；每次 `add_row(cols)` 的 `cols` 必须是**互不相同**的列编号（无需排序）。重复列会破坏十字链表的搜索逻辑，应在外部去重。
- `row_id` 若省略，采用该行加入时的 `0base` 编号；指定后原样写入 `ans`。希望方案中行号唯一时，不要传入重复的自定义 `row_id`。
- 空行不会帮助覆盖任何列，可以加入但不会被选中；若有一列从未被任何行包含，则两类问题均无解。
- `m = 0` 视为空约束：精确覆盖成功、最少重复覆盖数为 `0`、限制非负时可以覆盖，`ans` 为空。
- **两种覆盖的删链过程不同，不能混用**：
  - 精确覆盖的 `cover/uncover` 会删掉冲突行在其他列中的垂直节点，同时维护 `sz`，恢复时必须逆序。
  - 重复覆盖的 `remove_column/restore_columns` **仅摘除/恢复列头**，保留各行的水平环与各列的垂直链，所以允许行交叠。`lower_bound()` 通过列间可能共享的行构造仍需选行数的下界。
- 两类搜索都会在返回前**完整恢复链表**；同一对象可重复查询，甚至先 `solve_exact()` 再 `min_cover()`，但不要在递归搜索过程中并发调用其它搜索或修改矩阵。
- 只支持**无权**覆盖：`min_cover()` 最小化的是选中行数。带权集合覆盖需要更换代价与剪枝下界，不可直接将 `best` 当权重使用。
- `ans` 只提供一组解；需要枚举全部精确覆盖解时，应修改 `dfs_exact()` 的成功分支与返回方式，不能直接复用当前的提前返回逻辑。
- 该实现默认**所有列都是必选约束**。部分列“至多选一次但可以不选”（secondary columns）属于扩展模型，需要修改表头初始化和精确覆盖建模，不是重复覆盖。
- 使用 C++20 环境（本实现自身的语法兼容 C++17），默认外部已有 `<bits/stdc++.h>` 和 `using namespace std;`。

### 功能裁剪

- **仅精确覆盖**：删除 `active/changed/seen/stamp/best` 和从 `remove_column()` 到 `cover_at_most()` 的整段重复覆盖实现；`Node/tr/sz/add_row/choose_column` 不变。
- **仅重复覆盖**：删除 `cover()`、`uncover()`、`dfs_exact()`、`solve_exact()`；共用建图、`choose_column()` 和重复覆盖部分保留。
- **只需最少行数**：可删除 `cover_at_most()`；保留 `min_cover()`、`prepare_repeat()`、`dfs_repeat()` 与 `lower_bound()`。
- **只需限制行数可行性**：可删除 `min_cover()`；保留 `cover_at_most()` 和重复覆盖辅助函数。
- 若只想获得解是否存在，`ans` / `path` 仍需要用于搜索路径，不能直接删除；可以仅取消成功分支对 `ans` 的拷贝，并相应修改接口。

## 常见用法

### 精确覆盖：每列恰好覆盖一次

```cpp
DLX dlx(3, 6); // 3 个约束，预计 6 个 1

dlx.add_row({0, 2}, 10);
dlx.add_row({1}, 20);
dlx.add_row({0, 1}, 30);
dlx.add_row({2}, 40);

if (dlx.solve_exact()) {
    for (int id : dlx.ans) {
        // 一组可行方案：{10,20} 或 {30,40}，顺序不保证
    }
}
```

### 重复覆盖：最少选多少行

```cpp
DLX dlx(3);
dlx.add_row({0, 1});
dlx.add_row({1, 2});

bool exact = dlx.solve_exact();   // false：列 1 会重叠
int k = dlx.min_cover();          // 2：选两行覆盖三列
bool ok = dlx.cover_at_most(1);   // false：1 行不够
```

### 二分答案的判定函数

当题目在某个参数下建立可覆盖关系，且参数越大可选择的行覆盖范围越大，可以每次重新建矩阵，用 `cover_at_most(k)` 判断能否用不超过 `k` 行覆盖全部约束：

```cpp
DLX dlx(m, ones_hint);
for (int i = 0; i < n; ++ i) {
    vector<int> cols;
    // 收集第 i 个方案能够覆盖的列
    dlx.add_row(cols, i);
}
bool ok = dlx.cover_at_most(k);
```

`dlx` 需要在矩阵关系变化后重新构建，不能仅通过修改 `sz` 代替增删矩阵中的 1。

### 数独：把填数方案转为四类精确覆盖约束

普通 9×9 数独有 `324` 列约束（81 个格子、81 个行数字、81 个列数字、81 个宫数字），每个填数候选方案恰好关联四列：

```cpp
DLX dlx(324, 729 * 4);
for (int r = 0; r < 9; ++ r) {
    for (int c = 0; c < 9; ++ c) {
        for (int d = 0; d < 9; ++ d) {
            // 若 (r,c) 是已填格且不等于 d+1，跳过
            int b = (r / 3) * 3 + c / 3;
            dlx.add_row({r * 9 + c, 81 + r * 9 + d,
                         162 + c * 9 + d, 243 + b * 9 + d},
                        (r * 9 + c) * 9 + d);
        }
    }
}
if (dlx.solve_exact()) {
    for (int id : dlx.ans) {
        int d = id % 9, c = (id / 9) % 9, r = id / 81;
        // a[r][c] = d + 1;
    }
}
```

## 复杂度

记列数为 `m`、行数为 `n`、稀疏矩阵中 1 的个数为 `E`，当前分支所剩约束数为 `m'`。

| 操作 | 时间复杂度 |
|---|---|
| 构造 | `O(m)` |
| 加入一行 | `O(行内 1 的个数)`（均摊） |
| 选分支列 | `O(m')`；采用当前候选行数最小列启发式 |
| 精确覆盖 `cover/uncover` | 按涉及的十字链表节点数线性增量处理 |
| 重复覆盖一次列摘除/恢复 | `O(1)`；一整行的覆盖操作为 `O(行内 1 的个数)` |
| 重复覆盖贪心下界 `lower_bound()` | 最坏 `O(m E)` 每个搜索节点，通常更少 |
| 两类求解总体 | **最坏指数级**；分支数量取决于约束稀疏程度与剪枝 |

- **空间**：`O(m + E + n)`（节点池、列大小、候选路径与重复覆盖工作空间）。
- 每个矩阵 1 使用一个 `Node`（通常 24 字节），外加 `m + 1` 个表头；预留 `ones_hint` 减少扩容。
- 在重复覆盖中，`lower_bound()` 是用于优化搜索的正确下界；在极少分支的小问题中它也会引入额外扫描成本，可按题型决定是否简化剪枝。
