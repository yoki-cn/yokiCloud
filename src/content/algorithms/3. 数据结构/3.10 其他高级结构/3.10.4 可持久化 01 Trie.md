### 可持久化 01 Trie
---
```cpp
template<class U = uint32_t, int B = numeric_limits<U>::digits>
struct Persistent01Trie {
    static_assert(is_integral_v<U> && is_unsigned_v<U> && !is_same_v<U, bool>);
    static_assert(1 <= B && B <= numeric_limits<U>::digits);
    struct Node {
        int ch[2]{};
        int cnt{};
    };
    vector<Node> tr{ Node{} };
    vector<int> root{ 0 };
    Persistent01Trie() = default;
    explicit Persistent01Trie(const vector<U> &a) {
        reserve_updates(a.size());
        for (U x : a) append(x);
    }
private:
    static void check(U x) {
        if constexpr (B < numeric_limits<U>::digits) assert((x >> B) == 0);
    }
    int clone(int p) {
        tr.push_back(tr[p]);
        return int(tr.size()) - 1;
    }
    U xor_best_root(int a, int b, U x, bool mx) const {
        check(x);
        assert(size_root(a, b) > 0);
        U ans{};
        for (int i = B - 1; i >= 0; -- i) {
            int v = int((x >> i) & U(1));
            int d = v ^ int(mx);
            if (tr[tr[b].ch[d]].cnt == tr[tr[a].ch[d]].cnt) d ^= 1;
            if (d != v) ans |= U(1) << i;
            a = tr[a].ch[d];
            b = tr[b].ch[d];
        }
        return ans;
    }
public:
    // ===== 路径复制与版本管理 =====
    void reserve_updates(size_t q) { tr.reserve(tr.size() + q * (B + 1)); }
    int add_root(int p, U x, int delta = 1) {
        check(x);
        if (delta == 0) return p;
        int q = clone(p), u = q;
        assert(tr[u].cnt + delta >= 0);
        tr[u].cnt += delta;
        for (int i = B - 1; i >= 0; -- i) {
            int d = int((x >> i) & U(1));
            int v = tr[p].ch[d], w = clone(v);
            tr[u].ch[d] = w;
            assert(tr[w].cnt + delta >= 0);
            tr[w].cnt += delta;
            p = v;
            u = w;
        }
        return q;
    }
    void append(U x) { root.push_back(add_root(root.back(), x)); }
    int size_root(int a, int b) const { return tr[b].cnt - tr[a].cnt; }
    int size(int l, int r) const {
        assert(0 <= l && l <= r && r < int(root.size()));
        return size_root(root[l], root[r]);
    }
    // ===== 最大 / 最小异或 =====
    U max_xor_root(int a, int b, U x) const { return xor_best_root(a, b, x, true); }
    U min_xor_root(int a, int b, U x) const { return xor_best_root(a, b, x, false); }
    U max_xor(int l, int r, U x) const {
        assert(size(l, r) > 0);
        return max_xor_root(root[l], root[r], x);
    }
    U min_xor(int l, int r, U x) const {
        assert(size(l, r) > 0);
        return min_xor_root(root[l], root[r], x);
    }
    // ===== 按异或值排序第 k 小 =====
    U kth_xor_root(int a, int b, U x, int k) const {
        check(x);
        assert(0 <= k && k < size_root(a, b));
        U ans{};
        for (int i = B - 1; i >= 0; -- i) {
            int d = int((x >> i) & U(1));
            int ca = tr[a].ch[d], cb = tr[b].ch[d];
            int cnt = tr[cb].cnt - tr[ca].cnt;
            if (k >= cnt) {
                k -= cnt;
                d ^= 1;
                ans |= U(1) << i;
            }
            a = tr[a].ch[d];
            b = tr[b].ch[d];
        }
        return ans;
    }
    U kth_xor(int l, int r, U x, int k) const {
        assert(0 <= k && k < size(l, r));
        return kth_xor_root(root[l], root[r], x, k);
    }
    // ===== 异或值小于 limit 的个数 =====
    int count_xor_less_root(int a, int b, U x, U limit) const {
        check(x);
        if constexpr (B < numeric_limits<U>::digits) {
            if (limit >> B) return size_root(a, b);
        }
        int ans = 0;
        for (int i = B - 1; i >= 0; -- i) {
            int d = int((x >> i) & U(1));
            if ((limit >> i) & U(1)) {
                ans += tr[tr[b].ch[d]].cnt - tr[tr[a].ch[d]].cnt;
                d ^= 1;
            }
            a = tr[a].ch[d];
            b = tr[b].ch[d];
        }
        return ans;
    }
    int count_xor_less(int l, int r, U x, U limit) const {
        assert(0 <= l && l <= r && r < int(root.size()));
        return count_xor_less_root(root[l], root[r], x, limit);
    }
    // ===== 树上路径四版本最大异或 =====
    // a + b - c - d，四个参数是版本根，不是序列下标
    U max_xor_path(int a, int b, int c, int d, U x) const {
        check(x);
        assert(tr[a].cnt + tr[b].cnt - tr[c].cnt - tr[d].cnt > 0);
        U ans{};
        for (int i = B - 1; i >= 0; -- i) {
            int v = int((x >> i) & U(1)), t = v ^ 1;
            int ca = tr[a].ch[t], cb = tr[b].ch[t];
            int cc = tr[c].ch[t], cd = tr[d].ch[t];
            int cnt = tr[ca].cnt + tr[cb].cnt - tr[cc].cnt - tr[cd].cnt;
            if (cnt == 0) t ^= 1;
            if (t != v) ans |= U(1) << i;
            a = tr[a].ch[t];
            b = tr[b].ch[t];
            c = tr[c].ch[t];
            d = tr[d].ch[t];
        }
        return ans;
    }
};
```
#### 成员变量
- `tr`：持久化节点池；`tr[0]` 为空节点，`Node::ch[0/1]` 为子节点，`Node::cnt` 为经过该节点的元素出现次数。
- `root[i]`：使用 `append()` 依次插入时，保存前缀 `[0, i)` 的版本根；`root[0] = 0`。
- `U`：无符号整数类型，默认 `uint32_t`；`B`：维护的有效低位数，默认 `numeric_limits<U>::digits`。
#### 接口

| 接口 | 用途 | 时间复杂度 |
|---|---|---|
| `Persistent01Trie(a)` | 从 `vector<U>` 建立全部前缀版本 | $O(nB)$ |
| `Persistent01Trie()` | 建立空结构 | $O(1)$ |
| `reserve_updates(q)` | 为后续 `q` 次非零增量的版本更新预留节点容量 | 重新分配时 $O(\text{当前节点数})$ |
| `add_root(p, x, delta=1)` | 在版本根 `p` 中改变 `x` 的出现次数，返回新根；不自动加入 `root` | $O(B)$ |
| `append(x)` | 在最新前缀版本后插入 `x`，追加一个版本根 | $O(B)$ |
| `size_root(a, b)` | 两个版本差分的元素总数 `cnt(b)-cnt(a)` | $O(1)$ |
| `size(l, r)` | 前缀版本之差 `[l,r)` 的元素总数 | $O(1)$ |
| `max_xor_root(a,b,x)` / `min_xor_root(a,b,x)` | 两版本差分集合中最大 / 最小的 `y ^ x` | $O(B)$ |
| `max_xor(l,r,x)` / `min_xor(l,r,x)` | `[l,r)` 中最大 / 最小的 `a[i] ^ x` | $O(B)$ |
| `kth_xor_root(a,b,x,k)` | 两版本差分中按 `y ^ x` 升序的第 `k` 个**异或值** | $O(B)$ |
| `kth_xor(l,r,x,k)` | `[l,r)` 中按 `a[i] ^ x` 升序的第 `k` 个**异或值** | $O(B)$ |
| `count_xor_less_root(a,b,x,limit)` | 两版本差分中满足 `(y ^ x) < limit` 的元素个数 | $O(B)$ |
| `count_xor_less(l,r,x,limit)` | `[l,r)` 中满足 `(a[i] ^ x) < limit` 的元素个数 | $O(B)$ |
| `max_xor_path(a,b,c,d,x)` | 四版本组合 `a+b-c-d` 对应集合中的最大异或值 | $O(B)$ |
其中 `k` 从 **0** 开始；带 `_root` 的接口传版本根编号，其余 `[l,r)` 接口传数组下标。
#### 注意事项
- **值域限制**：只接受无符号整数 `U`；当 `B` 小于类型位宽时，插入的 `x`、异或查询参数 `x` 都必须小于 `2^B`。`limit` 允许超出 `B` 位范围，此时计数直接返回区间总数。
- **返回异或值**：`max_xor`、`min_xor`、`kth_xor` 返回 `y ^ x`，不是元素 `y`。若要取得对应元素，使用 `y = result ^ x`；有重复值时返回的元素值仍唯一确定，但不能区分原始下标。
- **合法版本差分**：`root[r] - root[l]` 仅在 `l <= r` 时表示非负频数集合。对任意两个自定义版本根，须保证差分后**每个值的频数均非负**，不能简单对两个无关版本相减。
- **版本删除**：`add_root(p, x, delta < 0)` 仅在旧版本中 `x` 的数量至少为 `-delta` 时合法；必须保证所有版本的频数始终非负。`delta=0` 原样返回 `p`。代码在启用断言时会检查沿路径各节点的计数不会减成负数。
- **树上路径**：路径点权前缀根 `rt[u]` 包含根至 `u`；应传入 `rt[u], rt[v], rt[lca], rt[parent(lca)]`，根节点无父节点时使用 `0` 号空根。
- **内存**：每次非零 `add_root` 恰好增加 `B+1` 个节点；`Node` 通常是 12 字节。`B=32`、`n=2e5` 时约 660 万节点，仅节点池就约 79 MB（不计 `vector` 容量冗余），内存紧张时设小有效位宽，且优先 `reserve_updates()`。
- **裁剪**：只需要最大异或时，可删除整个 `kth_xor`、`count_xor_less`、`max_xor_path` 功能分区以及 `min_xor_root()` / `min_xor()`；基础建树及 `xor_best_root()` 必须保留。
#### 常见用法
##### 静态区间最大 / 最小异或
```cpp
vector<uint32_t> a = {3, 8, 2, 7, 5};
Persistent01Trie<uint32_t, 4> trie(a);
auto mx = trie.max_xor(1, 5, 6);  // max(8^6, 2^6, 7^6, 5^6) = 14
auto mn = trie.min_xor(1, 5, 6);  // 7^6 = 1
auto y = mx ^ 6U;                // 获得产生最大异或的原值 8
```
##### 异或第 k 小与异或值域计数
```cpp
auto kth = trie.kth_xor(1, 5, 6, 2);          // 异或结果升序第 3 个（0base）
int cnt = trie.count_xor_less(1, 5, 6, 8U);  // 异或结果 < 8 的个数
```
##### 手动建立历史版本
```cpp
Persistent01Trie<uint32_t, 20> trie;
int v1 = trie.add_root(0, 7);
int v2 = trie.add_root(v1, 11);
int v3 = trie.add_root(v2, 7, -1);  // 删除一个 7，v2 仍可用
uint32_t ans = trie.max_xor_root(v1, v2, 2); // 只查询新增的 11
```
##### 树上路径最大异或
```cpp
Persistent01Trie<uint32_t, 31> trie;
trie.reserve_updates(n);
vector<int> rt(n);
// DFS 遍历树：rt[u] = trie.add_root(fa == -1 ? 0 : rt[fa], w[u]);
int p = lca(u, v);
int rp = parent[p] == -1 ? 0 : rt[parent[p]];
auto ans = trie.max_xor_path(rt[u], rt[v], rt[p], rp, x);
```
#### 复杂度
令 $n$ 为插入次数，$B$ 为维护的二进制位数。

| 操作 | 时间复杂度 | 新增节点 |
|---|---|---|
| 初始化空结构 | $O(1)$ | 1 个空节点 |
| 建立 `n` 个前缀版本 | $O(nB)$ | $n(B+1)$ |
| 非零增量的 `add_root` / `append` | $O(B)$ | $B+1$ |
| 两版本差分求元素总数 | $O(1)$ | 0 |
| 最大 / 最小异或、异或第 k 小、异或值小于阈值的数量 | $O(B)$ | 0 |
| 四版本树上路径最大异或 | $O(B)$ | 0 |
空间复杂度：$O(nB)$；每个版本根仅需一个整数下标。`B=32` 或 `64` 时，查询只沿一条从最高位到最低位的路径下降。
