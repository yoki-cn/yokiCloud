### 01 Trie 多重集
---
```cpp
template<int B = 31>
struct BinaryTrie {
    static_assert(1 <= B && B <= 64);
    struct Node {
        int ch[2]{ -1, -1 };
        int siz{};
    };
    vector<Node> tr;
    vector<int> bin;
    explicit BinaryTrie(int want = 0) {
        tr.reserve(1LL + 1LL * want * B);
        tr.push_back({});
    }
    int size() const { return tr[0].siz; }
    bool empty() const { return size() == 0; }
    // ===== 节点创建：以下代码被 add() 依赖，不建议删除 =====
    int new_node() {
        if (!bin.empty()) {
            int p = bin.back();
            bin.pop_back();
            tr[p] = Node{};
            return p;
        }
        tr.push_back({});
        return int(tr.size()) - 1;
    }
    // ===== 基础修改与计数 =====
    void add(u64 x) {
        int p{ 0 };
        ++ tr[p].siz;
        for (int bit = B - 1; bit >= 0; -- bit) {
            int b{ int(x >> bit & 1ULL) };
            if (tr[p].ch[b] == -1) tr[p].ch[b] = new_node();
            p = tr[p].ch[b];
            ++ tr[p].siz;
        }
    }
    int count(u64 x) const {
        int p{ 0 };
        for (int bit = B - 1; bit >= 0; -- bit) {
            int b{ int(x >> bit & 1ULL) };
            p = tr[p].ch[b];
            if (p == -1 || tr[p].siz == 0) return 0;
        }
        return tr[p].siz;
    }
    bool erase(u64 x) {
        if (count(x) == 0) return false;
        array<int, B + 1> path{};
        path[0] = 0;
        -- tr[0].siz;
        for (int i = 0; i < B; ++ i) {
            int bit = B - 1 - i;
            path[i + 1] = tr[path[i]].ch[x >> bit & 1ULL];
            -- tr[path[i + 1]].siz;
        }
        for (int i = B; i > 0 && tr[path[i]].siz == 0; -- i) {
            int bit = B - i;
            tr[path[i - 1]].ch[x >> bit & 1ULL] = -1;
            tr[path[i]] = Node{};
            bin.push_back(path[i]);
        }
        return true;
    }
    // ===== 异或序统计：若只需要插入 / 删除 / 计数，以下代码可直接删除 =====
    // 统计满足 (x ^ y) < k 的 y 的个数
    int count_xor_less(u64 x, u64 k) const {
        if constexpr (B < 64) { if (k >> B) return size(); }
        int p{ 0 }, ans{ 0 };
        for (int bit = B - 1; bit >= 0 && p != -1; -- bit) {
            int xb{ int(x >> bit & 1ULL) }, kb{ int(k >> bit & 1ULL) };
            if (kb) {
                int q{ tr[p].ch[xb] };
                if (q != -1) ans += tr[q].siz;
                p = tr[p].ch[xb ^ 1];
            } else {
                p = tr[p].ch[xb];
            }
        }
        return ans;
    }
    // 返回第 k 小的 x ^ y，k 为 0base
    optional<u64> kth_xor(u64 x, int k) const {
        if (k < 0 || k >= size()) return nullopt;
        int p{ 0 };
        u64 ans{};
        for (int bit = B - 1; bit >= 0; -- bit) {
            int b{ int(x >> bit & 1ULL) }, q{ tr[p].ch[b] };
            int cnt{ q == -1 ? 0 : tr[q].siz };
            if (k < cnt) {
                p = q;
            } else {
                k -= cnt;
                ans |= u64(1) << bit;
                p = tr[p].ch[b ^ 1];
            }
        }
        return ans;
    }
    optional<u64> min_xor(u64 x) const { return kth_xor(x, 0); }
    optional<u64> max_xor(u64 x) const { return kth_xor(x, size() - 1); }
};
```

* `BinaryTrie<B>(want)`：构造空 Trie；`B` 为参与比较的二进制位数，`want` 为预计插入次数，仅用于预留空间
* `size()`：当前元素总数，包含重复元素，$O(1)$
* `empty()`：判断是否为空，$O(1)$
* `add(x)`：插入一个 `x`，$O(B)$
* `count(x)`：返回 `x` 的出现次数，$O(B)$
* `erase(x)`：删除一个 `x`；不存在时返回 `false`，$O(B)$
* `count_xor_less(x,k)`：返回满足 `x ^ y < k` 的元素 `y` 的个数，$O(B)$
* `kth_xor(x,k)`：返回第 `k` 小的 `x ^ y`，`k` 为 `0base`；不存在返回 `nullopt`，$O(B)$
* `min_xor(x)`：返回最小的 `x ^ y`；空集合返回 `nullopt`，$O(B)$
* `max_xor(x)`：返回最大的 `x ^ y`；空集合返回 `nullopt`，$O(B)$
令 `x = 0` 后，异或序统计就退化成普通整数序统计：
* `count_xor_less(0,k)`：集合中 `< k` 的元素数量
* `kth_xor(0,k)`：集合第 `k` 小元素
### 可持久化 01 Trie
---
按顺序执行 `add(a[i])` 后，`root[r] - root[l]` 表示插入序列区间 `[l,r)` 中的多重集
```cpp
template<int B = 31>
struct PersistentBinaryTrie {
    static_assert(1 <= B && B <= 64);
    struct Node {
        int ch[2]{};
        int siz{};
    };
    vector<Node> tr{ {} };
    vector<int> root{ 0 };
    explicit PersistentBinaryTrie(int want = 0) { tr.reserve(1LL + 1LL * want * (B + 1)); }
    int size() const { return int(root.size()) - 1; }
    // ===== 节点复制与插入：可持久化核心，不建议删除 =====
    int clone(int p) {
        tr.push_back(tr[p]);
        return int(tr.size()) - 1;
    }
    // 从版本 pre 插入 x，返回新根；不会自动加入 root
    int insert(int pre, u64 x) {
        int rt{ clone(pre) }, p{ rt };
        ++ tr[p].siz;
        for (int bit = B - 1; bit >= 0; -- bit) {
            int b{ int(x >> bit & 1ULL) }, q{ clone(tr[pre].ch[b]) };
            tr[p].ch[b] = q;
            p = q;
            pre = tr[pre].ch[b];
            ++ tr[p].siz;
        }
        return rt;
    }
    // 在最后一个前缀版本上插入 x
    void add(u64 x) { root.push_back(insert(root.back(), x)); }
    // ===== 两版本作差查询：以下函数均作用于 (left_root, right_root] 表示的多重集 =====
    int count(int left_root, int right_root, u64 x) const {
        for (int bit = B - 1; bit >= 0; -- bit) {
            int b{ int(x >> bit & 1ULL) };
            left_root = tr[left_root].ch[b];
            right_root = tr[right_root].ch[b];
        }
        return tr[right_root].siz - tr[left_root].siz;
    }
    int count_xor_less(int left_root, int right_root, u64 x, u64 k) const {
        int total{ tr[right_root].siz - tr[left_root].siz };
        if constexpr (B < 64) { if (k >> B) return total; }
        int ans{ 0 };
        for (int bit = B - 1; bit >= 0; -- bit) {
            int xb{ int(x >> bit & 1ULL) }, kb{ int(k >> bit & 1ULL) };
            if (kb) {
                int l{ tr[left_root].ch[xb] }, r{ tr[right_root].ch[xb] };
                ans += tr[r].siz - tr[l].siz;
                left_root = tr[left_root].ch[xb ^ 1];
                right_root = tr[right_root].ch[xb ^ 1];
            } else {
                left_root = tr[left_root].ch[xb];
                right_root = tr[right_root].ch[xb];
            }
        }
        return ans;
    }
    // 返回第 k 小的 x ^ y，k 为 0base
    optional<u64> kth_xor(int left_root, int right_root, u64 x, int k) const {
        int total{ tr[right_root].siz - tr[left_root].siz };
        if (k < 0 || k >= total) return nullopt;
        u64 ans{};
        for (int bit = B - 1; bit >= 0; -- bit) {
            int b{ int(x >> bit & 1ULL) }, l{ tr[left_root].ch[b] };
            int r{ tr[right_root].ch[b] }, cnt{ tr[r].siz - tr[l].siz };
            if (k < cnt) {
                left_root = l;
                right_root = r;
            } else {
                k -= cnt;
                ans |= u64(1) << bit;
                left_root = tr[left_root].ch[b ^ 1];
                right_root = tr[right_root].ch[b ^ 1];
            }
        }
        return ans;
    }
    optional<u64> min_xor(int left_root, int right_root, u64 x) const {
        return kth_xor(left_root, right_root, x, 0);
    }
    optional<u64> max_xor(int left_root, int right_root, u64 x) const {
        int siz{ tr[right_root].siz - tr[left_root].siz };
        return kth_xor(left_root, right_root, x, siz - 1);
    }
    // ===== 前缀版本区间接口：若只手动管理版本根，以下代码可直接删除 =====
    int range_count(int l, int r, u64 x) const { return count(root[l], root[r], x); }
    int range_count_xor_less(int l, int r, u64 x, u64 k) const {
        return count_xor_less(root[l], root[r], x, k);
    }
    optional<u64> range_kth_xor(int l, int r, u64 x, int k) const {
        return kth_xor(root[l], root[r], x, k);
    }
    optional<u64> range_min_xor(int l, int r, u64 x) const {
        return min_xor(root[l], root[r], x);
    }
    optional<u64> range_max_xor(int l, int r, u64 x) const {
        return max_xor(root[l], root[r], x);
    }
};
```
* `PersistentBinaryTrie<B>(want)`：构造空结构，`root[0]` 为空版本
* `size()`：使用 `add()` 建立的前缀版本个数，即已插入元素数，$O(1)$
* `insert(pre,x)`：从根 `pre` 复制路径并插入一个 `x`，返回新版本根；新增 `B+1` 个节点，$O(B)$
* `add(x)`：在 `root.back()` 上插入 `x`，并把新根加入 `root`，$O(B)$
* `count(left_root,right_root,x)`：查询两版本作差后的多重集中 `x` 的出现次数，$O(B)$
* `count_xor_less(left_root,right_root,x,k)`：两版本作差后统计 `x ^ y < k` 的元素数量，$O(B)$
* `kth_xor(left_root,right_root,x,k)`：两版本作差后查询第 `k` 小异或值，`k` 为 `0base`，$O(B)$
* `min_xor/max_xor(left_root,right_root,x)`：两版本作差后的最小 / 最大异或值，$O(B)$
* `range_count(l,r,x)`：查询插入序列 `[l,r)` 中 `x` 的出现次数，$O(B)$
* `range_count_xor_less(l,r,x,k)`：查询 `[l,r)` 中满足 `x ^ y < k` 的元素数量，$O(B)$
* `range_kth_xor(l,r,x,k)`：查询 `[l,r)` 中第 `k` 小的 `x ^ y`，$O(B)$
* `range_min_xor/range_max_xor(l,r,x)`：查询 `[l,r)` 中最小 / 最大的 `x ^ y`，$O(B)$
### 注意事项
* `B` 表示参与 Trie 的**位数**，遍历范围为 `[B-1,0]`。例如所有数均满足 $0\le x<2^{31}$ 时使用 `BinaryTrie<31>`
* 当 `B < 64` 时，插入值和查询值应满足 $x<2^B$；高于第 `B-1` 位的部分不会被 Trie 保存
* 默认使用 `u64` 按无符号整数处理。若题目中的数可能为负数，需要先明确题目要求的是二进制异或还是有符号整数顺序
* `min_xor/max_xor/kth_xor` 返回的是**异或结果 `x ^ y`**，不是被选择的原数 `y`；若需要原数，可由 `y = x ^ result` 得到
* `kth_xor()` 的 `k` 为 `0base`，因此最小值对应 `k = 0`
* 普通版本是多重集，同一个值可以插入多次；`erase(x)` 每次只删除一次
* 普通 `BinaryTrie::erase()` 在计数减至 0 后自底向上回收空路径节点，后续插入优先复用空闲节点；可持久化版本不回收历史版本节点
* 可持久化版本的 `root[i]` 表示前 `i` 次 `add()` 后的状态，因此 `root[r] - root[l]` 正好对应 `[l,r)`
* 只有沿同一插入序列得到的前缀版本才能直接解释为区间差；若使用 `insert(pre,x)` 从任意历史版本分叉，新根仍然合法，但两个无祖先关系的版本之差通常没有“集合差”的意义
* 可持久化版本使用 `0` 号空节点作为所有空儿子，因此查询过程中可以直接访问 `tr[0]`
### 常见用法
#### 动态集合最大异或
```cpp
BinaryTrie<31> trie(n);
for (u64 x : a) trie.add(x);
if (auto ans = trie.max_xor(x)) {
    cout << *ans << '\n';
}
```
#### 删除与重复元素
```cpp
trie.add(x);
trie.add(x);
cout << trie.count(x) << '\n'; // 2
trie.erase(x);
cout << trie.count(x) << '\n'; // 1
```
#### 第 k 小异或
```cpp
int k; // 0base
cin >> k;
if (auto ans = trie.kth_xor(x, k)) {
    cout << *ans << '\n';
}
```
#### 统计异或值小于 k 的元素数量
```cpp
int cnt = trie.count_xor_less(x, k);
```
因此还可以统计半开异或值域 `[L,R)`：
```cpp
int cnt = trie.count_xor_less(x, R) - trie.count_xor_less(x, L);
```
#### 插入序列区间最大异或
```cpp
PersistentBinaryTrie<31> trie(n);
for (u64 x : a) trie.add(x);
if (auto ans = trie.range_max_xor(l, r, x)) {
    cout << *ans << '\n';
}
```
其中查询范围为 `[l,r)`
#### 插入序列区间第 k 小异或
```cpp
if (auto ans = trie.range_kth_xor(l, r, x, k)) {
    cout << *ans << '\n';
}
```
`k` 为 `0base`
### 复杂度
设 Trie 位数为 `B`
#### 01 Trie 多重集

| 操作                                                                               |  时间复杂度 |
| -------------------------------------------------------------------------------- | -----: |
| `size / empty`                                                                   | $O(1)$ |
| `add` / `count` / `erase` / `count_xor_less` / `kth_xor` / `min_xor` / `max_xor` | $O(B)$ |
同时存在 $n$ 个不同路径时最坏需要 $O(nB)$ 个有效节点；普通版回收空路径，累计申请的内存由历史峰值决定，而不是历史插入次数之和
#### 可持久化 01 Trie

| 操作                                                                              |  时间复杂度 |    新增节点 |
| ------------------------------------------------------------------------------- | -----: | ------: |
| `insert / add`                                                                  | $O(B)$ | `B + 1` |
| `count` / `count_xor_less` / `kth_xor` / `min_xor` / `max_xor` / 各 `range_*` 查询 | $O(B)$ |     $0$ |
插入 $n$ 次后的空间复杂度为 $O(nB)$
