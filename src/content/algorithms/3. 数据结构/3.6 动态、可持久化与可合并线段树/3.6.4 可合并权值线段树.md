### 可合并权值线段树
---
```cpp
template<integral Coord = i64>
struct MergeableValueSegTree {
    struct Node {
        int ch[2]{};
        int cnt{};
    };
    static_assert(!same_as<Coord, bool>);
    Coord lo, hi;
    vector<Node> tr{ Node{} };
    vector<int> free_nodes;
    explicit MergeableValueSegTree(Coord lo_, Coord hi_, size_t reserve_nodes = 0) : lo(lo_), hi(hi_) {
        assert(lo < hi);
        if (reserve_nodes) tr.reserve(reserve_nodes + 1);
    }
    // ===== 节点池与维护：所有操作依赖，不建议删除 =====
    int size(int p) const { return p ? tr[p].cnt : 0; }
    int new_node() {
        if (free_nodes.empty()) {
            tr.emplace_back();
            return int(tr.size()) - 1;
        }
        int p = free_nodes.back();
        free_nodes.pop_back();
        tr[p] = {};
        return p;
    }
    void recycle_node(int p) {
        assert(p);
        tr[p] = {};
        free_nodes.push_back(p);
    }
    int normalize(int p) {
        if (p && tr[p].cnt == 0) {
            recycle_node(p);
            return 0;
        }
        return p;
    }
    void pull(int p) { tr[p].cnt = size(tr[p].ch[0]) + size(tr[p].ch[1]); }
    // ===== 单点增删：权值多重集基础，不建议删除 =====
    int add_node(int p, Coord l, Coord r, Coord x, int delta) {
        if (!p) p = new_node();
        if (l + 1 == r) {
            tr[p].cnt += delta;
            assert(tr[p].cnt >= 0);
            return normalize(p);
        }
        Coord m = midpoint(l, r);
        int d = x >= m;
        int q = add_node(tr[p].ch[d], d ? m : l, d ? r : m, x, delta);
        tr[p].ch[d] = q;
        pull(p);
        return normalize(p);
    }
    void add(int &p, Coord x, int delta = 1) {
        assert(lo <= x && x < hi);
        p = add_node(p, lo, hi, x, delta);
    }
    template<ranges::input_range R>
    int build(R &&a) {
        int p = 0;
        for (auto &&x : a) add(p, static_cast<Coord>(x));
        return p;
    }
    // ===== 顺序统计：若只需要 merge / split，可按需删除部分接口 =====
    int count_node(int p, Coord l, Coord r, Coord ql, Coord qr) const {
        if (!p || qr <= l || r <= ql) return 0;
        if (ql <= l && r <= qr) return tr[p].cnt;
        Coord m = midpoint(l, r);
        return count_node(tr[p].ch[0], l, m, ql, qr) + count_node(tr[p].ch[1], m, r, ql, qr);
    }
    int count(int p, Coord l, Coord r) const {
        l = max(l, lo);
        r = min(r, hi);
        return l < r ? count_node(p, lo, hi, l, r) : 0;
    }
    int count_less(int p, Coord x) const {
        if (x <= lo) return 0;
        if (hi <= x) return size(p);
        return count_node(p, lo, hi, lo, x);
    }
    int count_leq(int p, Coord x) const {
        if (x < lo) return 0;
        if (x >= hi - 1) return size(p);
        return count_less(p, x + 1);
    }
    int frequency(int p, Coord x) const {
        if (x < lo || hi <= x) return 0;
        return count(p, x, x + 1);
    }
    Coord kth(int p, int k) const {
        assert(0 <= k && k < size(p));
        Coord l = lo, r = hi;
        while (l + 1 < r) {
            Coord m = midpoint(l, r);
            int q = tr[p].ch[0];
            int left = size(q);
            if (k < left) {
                p = q;
                r = m;
            } else {
                k -= left;
                p = tr[p].ch[1];
                l = m;
            }
        }
        return l;
    }
    Coord kth_largest(int p, int k) const {
        assert(0 <= k && k < size(p));
        return kth(p, size(p) - 1 - k);
    }
    int rank(int p, Coord x) const { return count_less(p, x) + 1; }
    optional<Coord> predecessor(int p, Coord x) const {
        int k = count_less(p, x);
        if (!k) return nullopt;
        return kth(p, k - 1);
    }
    optional<Coord> successor(int p, Coord x) const {
        int k = count_leq(p, x);
        if (k == size(p)) return nullopt;
        return kth(p, k);
    }
    // ===== 合并：破坏性地将 b 合入 a，结束后 b = 0 =====
    int merge_node(int a, int b, Coord l, Coord r) {
        if (!a) return b;
        if (!b) return a;
        assert(a != b);
        if (l + 1 == r) {
            tr[a].cnt += tr[b].cnt;
            recycle_node(b);
            return a;
        }
        Coord m = midpoint(l, r);
        int lc = merge_node(tr[a].ch[0], tr[b].ch[0], l, m);
        int rc = merge_node(tr[a].ch[1], tr[b].ch[1], m, r);
        tr[a].ch[0] = lc;
        tr[a].ch[1] = rc;
        pull(a);
        recycle_node(b);
        return a;
    }
    void merge(int &a, int &b) {
        assert(!a || a != b);
        a = merge_node(a, b, lo, hi);
        b = 0;
    }
    // ===== 按值分裂：返回 { < x, >= x }，原根被消费并置 0 =====
    pair<int, int> split_value_node(int p, Coord l, Coord r, Coord x) {
        if (!p) return {};
        if (r <= x) return { p, 0 };
        if (x <= l) return { 0, p };
        Coord m = midpoint(l, r);
        int lc = tr[p].ch[0], rc = tr[p].ch[1];
        auto [a0, b0] = split_value_node(lc, l, m, x);
        auto [a1, b1] = split_value_node(rc, m, r, x);
        int q = new_node();
        tr[p].ch[0] = a0;
        tr[p].ch[1] = a1;
        pull(p);
        tr[q].ch[0] = b0;
        tr[q].ch[1] = b1;
        pull(q);
        p = normalize(p);
        q = normalize(q);
        return { p, q };
    }
    pair<int, int> split_by_value(int &p, Coord x) {
        auto res = split_value_node(p, lo, hi, x);
        p = 0;
        return res;
    }
    // ===== 按排名分裂：前 k 小 / 其余，k 表示元素数量 =====
    pair<int, int> split_rank_node(int p, Coord l, Coord r, int k) {
        if (!p) return {};
        if (k <= 0) return { 0, p };
        if (k >= size(p)) return { p, 0 };
        if (l + 1 == r) {
            int q = new_node();
            tr[q].cnt = tr[p].cnt - k;
            tr[p].cnt = k;
            return { p, q };
        }
        Coord m = midpoint(l, r);
        int lc = tr[p].ch[0], rc = tr[p].ch[1];
        int left = size(lc);
        int q = new_node();
        if (k < left) {
            auto [a, b] = split_rank_node(lc, l, m, k);
            tr[p].ch[0] = a;
            tr[p].ch[1] = 0;
            pull(p);
            tr[q].ch[0] = b;
            tr[q].ch[1] = rc;
            pull(q);
        } else if (k == left) {
            tr[p].ch[0] = lc;
            tr[p].ch[1] = 0;
            pull(p);
            tr[q].ch[0] = 0;
            tr[q].ch[1] = rc;
            pull(q);
        } else {
            auto [a, b] = split_rank_node(rc, m, r, k - left);
            tr[p].ch[0] = lc;
            tr[p].ch[1] = a;
            pull(p);
            tr[q].ch[0] = 0;
            tr[q].ch[1] = b;
            pull(q);
        }
        return { p, q };
    }
    pair<int, int> split_by_rank(int &p, int k) {
        assert(0 <= k && k <= size(p));
        auto res = split_rank_node(p, lo, hi, k);
        p = 0;
        return res;
    }
    // ===== 整棵删除：若不会主动丢弃集合，可删除 =====
    void clear_node(int p) {
        if (!p) return;
        int lc = tr[p].ch[0], rc = tr[p].ch[1];
        clear_node(lc);
        clear_node(rc);
        recycle_node(p);
    }
    void clear(int &p) {
        clear_node(p);
        p = 0;
    }
    int nodes() const { return int(tr.size()) - 1 - int(free_nodes.size()); }
};
```
#### 成员变量
- `[lo, hi)`：整个整数权值域
- `tr`：共享节点池，`tr[0]` 为永久空节点
- `free_nodes`：已经释放、可以复用的节点编号
- `Node::cnt`：当前节点对应值域中的元素总数
- 每一个 `int root` 表示一个独立的**权值多重集**
例如：
```cpp
MergeableValueSegTree seg(0, 1'000'000'001LL);
vector<int> root(n);
```
此时 `root[u] == 0` 表示第 `u` 个集合为空
#### 基础修改
单点加入一个元素：
```cpp
seg.add(root, x);
```
加入多个：
```cpp
seg.add(root, x, cnt);
```
删除一个：
```cpp
seg.add(root, x, -1);
```
要求删除后频数不能为负
从一个 range / view 建立一棵树：
```cpp
int root = seg.build(a);
```
也可以：
```cpp
int root = seg.build(a | views::filter([](int x) {
    return x >= 0;
}));
```
#### 顺序统计
集合大小：
```cpp
int n = seg.size(root);
```
第 `k` 小，`k` 为 `0base`：
```cpp
auto x = seg.kth(root, k);
```
第 `k` 大：
```cpp
auto x = seg.kth_largest(root, k);
```
严格小于 `x` 的数量：
```cpp
int cnt = seg.count_less(root, x);
```
不大于 `x` 的数量：
```cpp
int cnt = seg.count_leq(root, x);
```
值域 `[l, r)` 中的数量：
```cpp
int cnt = seg.count(root, l, r);
```
某个值出现次数：
```cpp
int cnt = seg.frequency(root, x);
```
排名采用竞赛中常见定义：
```cpp
int rk = seg.rank(root, x);
```
其含义为：
$$\#\{v\mid v<x\}$$
前驱、后继：
```cpp
auto pre = seg.predecessor(root, x);
auto nxt = seg.successor(root, x);
```
不存在时返回 `nullopt`
#### 合并
```cpp
seg.merge(a, b);
```
执行后：
```text
a = 原 a ∪ 原 b
b = 0
```
这里的并是**多重集相加**：
如果：
```text
a: {1, 1, 3}
b: {1, 2}
```
合并后：
```text
a: {1, 1, 1, 2, 3}
b: {}
```
这是破坏性合并，不是持久化操作
#### 按值分裂
```cpp
auto [a, b] = seg.split_by_value(root, x);
```
得到：
```text
a：所有 < x 的元素
b：所有 >= x 的元素
root：0
```
例如：
```text
root = {1, 2, 2, 4, 7}
split_by_value(root, 4)
a = {1, 2, 2}
b = {4, 7}
```
这里 `x` 不要求当前集合中存在
#### 按排名分裂
```cpp
auto [a, b] = seg.split_by_rank(root, k);
```
得到：
```text
a：最小的 k 个元素
b：剩余元素
root：0
```
其中 `k` 表示**数量**，不是 `0base` 下标
例如：
```text
root = {1, 2, 2, 2, 5}
k = 3
```
得到：
```text
a = {1, 2, 2}
b = {2, 5}
```
因此如果切分点落在重复值内部，**同一个权值可以同时存在于两棵结果树中**
这也是按排名分裂与按值分裂最重要的区别
#### 注意事项
##### 所有权是这一份板子最重要的规则
每个非零节点只能被**一棵当前有效的树**拥有
因此不要写：
```cpp
int b = a;
```
然后分别修改：
```cpp
seg.add(a, x);
seg.add(b, y); // 错误
```
因为这只是复制了根编号，两棵“树”实际上共享同一批节点。之后 `merge / split / clear` 可能把另一棵树仍在使用的节点回收
如果需要：
```text
一个版本复制成两个版本
两边以后互不影响
```
应该使用上一份**可持久化线段树**，而不是这一份
##### `merge()` 会消费第二棵树
```cpp
seg.merge(a, b);
```
之后：
```cpp
assert(b == 0);
```
不要保存并继续使用旧的 `b`
这种接口故意模拟了“move”语义，可以有效避免节点所有权混乱
##### `split()` 会消费原根
```cpp
auto [a, b] = seg.split_by_value(root, x);
```
之后：
```cpp
assert(root == 0);
```
同理：
```cpp
auto [a, b] = seg.split_by_rank(root, k);
```
也会消费 `root`
##### 删除到空时会自动回收整条无用路径
例如：
```cpp
seg.add(root, x);
seg.add(root, x, -1);
```
如果这棵树原本只有一个 `x`，最终：
```cpp
root == 0
```
沿途节点都会进入 `free_nodes`
因此不需要额外手动清理空节点
##### 值域必须统一
同一个 `MergeableValueSegTree` 中的所有根天然拥有相同 `[lo, hi)`，因此可以直接合并
不要把不同对象中的根节点编号拿来混用
##### 不支持 lazy
这里维护的是权值多重集，不是普通位置线段树
典型操作是：
```text
插入 / 删除一个值
统计
kth
merge
split
```
如果需要区间加、区间赋值等懒标记操作，那已经不是这份结构最合适的用途
#### 常见用法
##### 树上子树权值集合合并
这是这份结构最经典的应用之一
每个节点先建立只包含自己权值的树：
```cpp
MergeableValueSegTree seg(0, V);
vector<int> root(n);
for (int u : views::iota(0, n)) {
    seg.add(root[u], a[u]);
}
```
后序 DFS：
```cpp
auto dfs = Y([&](auto self, int u, int fa) -> void {
    for (int v : g[u]) if (v != fa) {
        self(v, u);
        seg.merge(root[u], root[v]);
    }
    // 此时 root[u] = u 整棵子树的权值多重集
    if (seg.size(root[u]) >= k) {
        ans[u] = seg.kth_largest(root[u], k - 1);
    }
});
```
注意：
```cpp
seg.merge(root[u], root[v]);
```
之后 `root[v] == 0`
因此必须在子节点答案已经统计完成以后再合并，正好符合树形 DP 的后序过程
##### 提取一个值域
想把 `[L, R)` 整段从 `root` 中拿出来：
```cpp
auto [left, ge_l] = seg.split_by_value(root, L);
auto [mid, right] = seg.split_by_value(ge_l, R);
```
此时：
```text
left  = (-∞, L)
mid   = [L, R)
right = [R, +∞)
```
处理 `mid`：
```cpp
// ...
```
重新拼回：
```cpp
seg.merge(left, mid);
seg.merge(left, right);
root = left;
```
这其实把 `split_by_value()` 变成了一种很方便的**权值区间剪切工具**
#### 从一个集合搬走最小的 `k` 个
```cpp
auto [take, rest] = seg.split_by_rank(a, k);
a = rest;
seg.merge(b, take);
```
执行后，原来 `a` 中最小的 `k` 个元素被整体搬入 `b`
##### 搬走最大的 `k` 个
设：
```cpp
int n = seg.size(a);
```
那么：
```cpp
auto [rest, take] = seg.split_by_rank(a, n - k);
a = rest;
seg.merge(b, take);
```
即可把最大的 `k` 个元素搬走
##### 搬走某个值域
例如把 `a` 中 `[L, R)` 的所有元素移动到 `b`：
```cpp
auto [left, ge_l] = seg.split_by_value(a, L);
auto [mid, right] = seg.split_by_value(ge_l, R);
a = left;
seg.merge(a, right);
seg.merge(b, mid);
```
这个套路在“集合之间批量转移满足某权值条件的元素”时非常方便
##### 离散化后使用
如果真实权值范围很大，但所有可能值都能提前获得，可以先压成 `[0, m)`，缩短树高
```cpp
vector<i64> xs = a;
ranges::sort(xs);
xs.erase(ranges::unique(xs).begin(), xs.end());
auto id = [&](i64 x) {
    return int(ranges::lower_bound(xs, x) - xs.begin());
};
MergeableValueSegTree<int> seg(0, xs.size());
```
插入：
```cpp
seg.add(root, id(x));
```
查询第 `k` 小真实值：
```cpp
i64 x = xs[seg.kth(root, k)];
```
如果需要提前收集修改值，也可以直接组合 view：
```cpp
vector<i64> xs;
ranges::copy(
    queries
    | views::filter([](const auto &q) {
        return q.type == 1;
    })
    | views::transform([](const auto &q) {
        return q.x;
    }),
    back_inserter(xs)
);
```
然后排序去重即可
#### 常见维护扩展：同时维护权值和
很多题不仅需要数量，还需要：
```text
值域元素和
前 k 小元素和
前 k 大元素和
```
这时非常值得给 `Node` 再加一个 `sum`
例如：
```cpp
struct Node {
    int ch[2]{};
    int cnt{};
    i64 sum{};
};
```
##### 修改 `pull()`
```cpp
i64 sum(int p) const {
    return p ? tr[p].sum : 0;
}
void pull(int p) {
    tr[p].cnt = size(tr[p].ch[0]) + size(tr[p].ch[1]);
    tr[p].sum = sum(tr[p].ch[0]) + sum(tr[p].ch[1]);
}
```
##### 修改单点叶子
将：
```cpp
tr[p].cnt += delta;
```
改为：
```cpp
tr[p].cnt += delta;
tr[p].sum += i64(x) * delta;
```
##### 修改 `merge()` 的叶子
```cpp
tr[a].cnt += tr[b].cnt;
tr[a].sum += tr[b].sum;
```
内部节点仍然统一 `pull()` 即可
##### 按排名分裂时叶子要额外处理
原来：
```cpp
int q = new_node();
tr[q].cnt = tr[p].cnt - k;
tr[p].cnt = k;
```
需要改成：
```cpp
int q = new_node();
int c = tr[p].cnt;
tr[p].cnt = k;
tr[p].sum = i64(l) * k;
tr[q].cnt = c - k;
tr[q].sum = i64(l) * (c - k);
```
因为两个结果树都可能在同一个权值叶子上保留若干个元素
##### 前 `k` 小元素和
增加：
```cpp
i64 sum_k_smallest(int p, int k) const {
    assert(0 <= k && k <= size(p));
    if (!k) return 0;
    i64 ans = 0;
    Coord l = lo, r = hi;
    while (l + 1 < r) {
        Coord m = midpoint(l, r);
        int q = tr[p].ch[0];
        int left = size(q);
        if (k <= left) {
            p = q;
            r = m;
        } else {
            ans += sum(q);
            k -= left;
            p = tr[p].ch[1];
            l = m;
        }
    }
    return ans + i64(l) * k;
}
```
于是前 `k` 大元素和可以直接写成：
```cpp
i64 sum_k_largest(int p, int k) const {
    assert(0 <= k && k <= size(p));
    return sum(p) - sum_k_smallest(p, size(p) - k);
}
```
#### 另一种常见维护：不同权值个数
如果题目需要：
```text
当前集合有多少种不同权值
```
可以增加：
```cpp
int distinct{};
```
内部节点：
```cpp
tr[p].distinct = (tr[p].ch[0] ? tr[tr[p].ch[0]].distinct : 0) + (tr[p].ch[1] ? tr[tr[p].ch[1]].distinct : 0);
```
叶子：
```cpp
tr[p].distinct = tr[p].cnt > 0;
```
`merge()` 的叶子累加频数后：
```cpp
tr[a].distinct = 1;
```
`split_by_rank()` 如果把一个重复值拆到两边，则两个叶子的 `distinct` 都应为 `1`
#### 复杂度
设整个整数权值域宽度为 $W$
树高为：
$$H=O(\log W)$$
若提前离散化后共有 $m$ 个不同坐标，则：
$$H=O(\log m)$$
##### 单点插入 / 删除
$$O(H)$$
最多访问一条根到叶路径
##### 区间计数 / 第 $k$ 小 / 第 $k$ 大 / 排名 / 前驱 / 后继
$$O(H)$$
##### 按值分裂
只需复制分界值经过的一条路径，因此：
$$O(H)$$
并最多额外产生：
$$O(H)$$
个节点
##### 按排名分裂
同样只沿着第 kk 个元素所在路径递归：
$$O(H)$$
并最多额外产生：
$$O(H)$$
个节点
##### 合并
设两棵树当前分别拥有 $S_a,S_b$ 个有效节点
只有在两边对应节点都非空时才继续递归，因此单次最坏可以写为：
$$O(\min(S_a,S_b))$$
而且每个真正重叠的 `b` 节点在这次破坏性合并后都会被回收
在树形 DP 一类“子树只向父亲合并、源树之后不再使用”的场景中，这也是线段树合并能够非常高效的关键
##### 清空一棵树
设该树当前有 $S$ 个节点：
$$O(S)$$
##### 空间
若进行了 $q$ 次彼此独立的单点插入而没有任何回收，最多产生：
$$O(qH)$$
个节点

每次分裂额外产生：
$$O(H)$$
个节点
但 `merge()`、删除到空以及 `clear()` 都会把废弃节点加入 `free_nodes`，之后 `new_node()` 会优先复用，因此实际节点池大小由**历史最大同时存活节点数**主导，而不是所有操作累计产生的节点数
