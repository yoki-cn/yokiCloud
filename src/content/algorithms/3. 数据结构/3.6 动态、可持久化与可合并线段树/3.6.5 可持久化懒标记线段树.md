### 可持久化懒标记线段树
---
```cpp
template<class Info, class Tag>
struct PersistentLazySegTree {
    struct Node {
        array<int, 2> ch{};
        Info info{};
        Tag tag{};
        bool has_tag{};
    };
    int n{};
    vector<Node> tr{ Node{} };
    vector<int> roots;
    explicit PersistentLazySegTree(int n_, const Info &v = Info{}, size_t reserve_nodes = 0) : n(n_) {
        assert(n > 0);
        if (reserve_nodes) tr.reserve(reserve_nodes + 1);
        roots.push_back(build_same(0, n, v));
    }
    template<ranges::input_range R>
    requires constructible_from<Info, ranges::range_reference_t<R>>
    explicit PersistentLazySegTree(R &&a, size_t reserve_nodes = 0) {
        vector<Info> init;
        if constexpr (ranges::sized_range<R>) init.reserve(ranges::size(a));
        for (auto &&x : a) init.emplace_back(forward<decltype(x)>(x));
        n = init.size();
        assert(n > 0);
        if (reserve_nodes) tr.reserve(reserve_nodes + 1);
        roots.push_back(build(0, n, init));
    }
    // ===== 节点创建与维护：所有修改依赖，不建议删除 =====
    int new_node(Node x = {}) {
        tr.push_back(move(x));
        return int(tr.size()) - 1;
    }
    int clone(int p) { return new_node(tr[p]); }
    void pull(int p) { tr[p].info = tr[tr[p].ch[0]].info + tr[tr[p].ch[1]].info; }
    // 当前 Tag 后再执行 v；Tag::apply(v) 必须遵守同一顺序
    static void append_tag(Tag &cur, bool &has, const Tag &v) {
        if (has) {
            cur.apply(v);
        } else {
            cur = v;
            has = true;
        }
    }
    // ===== 懒标记维护：push() 下传前必须复制共享儿子 =====
    void apply(int p, int l, int r, const Tag &v) {
        tr[p].info.apply(v, l, r);
        append_tag(tr[p].tag, tr[p].has_tag, v);
    }
    // 返回是否发生过下传；若返回 true，则两个儿子都已经是当前版本独占节点
    bool push(int p, int l, int r) {
        if (!tr[p].has_tag || r - l == 1) return false;
        int m = midpoint(l, r);
        Tag t = tr[p].tag;
        int lc = clone(tr[p].ch[0]), rc = clone(tr[p].ch[1]);
        tr[p].ch = { lc, rc };
        apply(lc, l, m, t);
        apply(rc, m, r, t);
        tr[p].tag = Tag{};
        tr[p].has_tag = false;
        return true;
    }
    // ===== 建树：若只保留一种初始化方式，可删除另一组 build =====
    int build_same(int l, int r, const Info &v) {
        int p = new_node();
        if (r - l == 1) {
            tr[p].info = v;
            return p;
        }
        int m = midpoint(l, r);
        int lc = build_same(l, m, v), rc = build_same(m, r, v);
        tr[p].ch = { lc, rc };
        pull(p);
        return p;
    }
    int build(int l, int r, const vector<Info> &a) {
        int p = new_node();
        if (r - l == 1) {
            tr[p].info = a[l];
            return p;
        }
        int m = midpoint(l, r);
        int lc = build(l, m, a), rc = build(m, r, a);
        tr[p].ch = { lc, rc };
        pull(p);
        return p;
    }
    // ===== 区间修改：可持久化 lazy 核心，不建议删除 =====
    void range_apply_owned(int p, int l, int r, int ql, int qr, const Tag &v) {
        if (ql <= l && r <= qr) {
            apply(p, l, r, v);
            return;
        }
        bool pushed = push(p, l, r);
        int m = midpoint(l, r);
        if (ql < m) {
            int c = tr[p].ch[0];
            if (!pushed) c = clone(c);
            range_apply_owned(c, l, m, ql, qr, v);
            tr[p].ch[0] = c;
        }
        if (m < qr) {
            int c = tr[p].ch[1];
            if (!pushed) c = clone(c);
            range_apply_owned(c, m, r, ql, qr, v);
            tr[p].ch[1] = c;
        }
        pull(p);
    }
    int range_apply(int ver, int l, int r, const Tag &v) {
        assert(0 <= ver && ver < versions());
        assert(0 <= l && l <= r && r <= n);
        if (l == r) return copy_version(ver);
        int q = clone(roots[ver]);
        range_apply_owned(q, 0, n, l, r, v);
        roots.push_back(q);
        return versions() - 1;
    }
    // ===== 单点修改：若不需要 set / 任意叶子修改，可整段删除 =====
    template<class F>
    void modify_owned(int p, int l, int r, int x, F &f) {
        if (r - l == 1) {
            tr[p].tag = Tag{};
            tr[p].has_tag = false;
            invoke(f, tr[p].info);
            return;
        }
        bool pushed = push(p, l, r);
        int m = midpoint(l, r);
        int d = x >= m;
        int c = tr[p].ch[d];
        if (!pushed) c = clone(c);
        modify_owned(c, d ? m : l, d ? r : m, x, f);
        tr[p].ch[d] = c;
        pull(p);
    }
    template<class F>
    int modify(int ver, int x, F &&f) {
        assert(0 <= ver && ver < versions());
        assert(0 <= x && x < n);
        int q = clone(roots[ver]);
        modify_owned(q, 0, n, x, f);
        roots.push_back(q);
        return versions() - 1;
    }
    int set(int ver, int x, const Info &v) {
        return modify(ver, x, [&](Info &a) {
            a = v;
        });
    }
    // ===== 只读查询：不 push、不复制节点 =====
    // up 表示尚未作用到当前节点的祖先标记。
    // 当前节点 tag 一定早于祖先尚未下传的 up，因此组合顺序为 tag -> up。
    static pair<Tag, bool> next_tag(const Node &node, Tag up, bool has_up) {
        Tag next{};
        bool has_next = false;
        if (node.has_tag) append_tag(next, has_next, node.tag);
        if (has_up) append_tag(next, has_next, up);
        return { next, has_next };
    }
    Info query_node(int p, int l, int r, int ql, int qr, Tag up = {} bool has_up = false) const {
        if (qr <= l || r <= ql) return Info{};
        if (ql <= l && r <= qr) {
            Info res = tr[p].info;
            if (has_up) res.apply(up, l, r);
            return res;
        }
        auto [next, has_next] = next_tag(tr[p], up, has_up);
        int m = midpoint(l, r);
        return query_node(tr[p].ch[0], l, m, ql, qr, next, has_next)
             + query_node(tr[p].ch[1], m, r, ql, qr, next, has_next);
    }
    Info query(int ver, int l, int r) const {
        assert(0 <= ver && ver < versions());
        assert(0 <= l && l <= r && r <= n);
        if (l == r) return Info{};
        return query_node(roots[ver], 0, n, l, r);
    }
    Info get(int ver, int x) const {
        assert(0 <= x && x < n);
        return query(ver, x, x + 1);
    }
    const Info &all(int ver) const {
        assert(0 <= ver && ver < versions());
        return tr[roots[ver]].info;
    }
    // ===== 查找接口：若不需要 find_first / find_last，可整段删除 =====
    // pred 必须满足：pred(整段信息) == false 时，该段一定不存在答案。
    template<class F>
    int find_first_node(int p, int l, int r, int ql, int qr, Tag up, bool has_up, F &pred) const {
        if (qr <= l || r <= ql) return -1;
        if (ql <= l && r <= qr) {
            Info cur = tr[p].info;
            if (has_up) cur.apply(up, l, r);
            if (!invoke(pred, cur)) return -1;
        }
        if (r - l == 1) return l;
        auto [next, has_next] = next_tag(tr[p], up, has_up);
        int m = midpoint(l, r);
        int res = find_first_node(tr[p].ch[0], l, m, ql, qr, next, has_next, pred);
        if (res == -1) {
            res = find_first_node(tr[p].ch[1], m, r, ql, qr, next, has_next, pred);
        }
        return res;
    }
    template<class F>
    int find_first(int ver, int l, int r, F &&pred) const {
        assert(0 <= ver && ver < versions());
        assert(0 <= l && l <= r && r <= n);
        if (l == r) return -1;
        return find_first_node(roots[ver], 0, n, l, r, Tag{}, false, pred);
    }
    template<class F>
    int find_last_node(int p, int l, int r, int ql, int qr, Tag up, bool has_up, F &pred) const {
        if (qr <= l || r <= ql) return -1;
        if (ql <= l && r <= qr) {
            Info cur = tr[p].info;
            if (has_up) cur.apply(up, l, r);
            if (!invoke(pred, cur)) return -1;
        }
        if (r - l == 1) return l;
        auto [next, has_next] = next_tag(tr[p], up, has_up);
        int m = midpoint(l, r);
        int res = find_last_node(tr[p].ch[1], m, r, ql, qr, next, has_next, pred);
        if (res == -1) {
            res = find_last_node(tr[p].ch[0], l, m, ql, qr, next, has_next, pred);
        }
        return res;
    }
    template<class F>
    int find_last(int ver, int l, int r, F &&pred) const {
        assert(0 <= ver && ver < versions());
        assert(0 <= l && l <= r && r <= n);
        if (l == r) return -1;
        return find_last_node(roots[ver], 0, n, l, r, Tag{}, false, pred);
    }
    // ===== 版本与空间管理 =====
    int copy_version(int ver) {
        assert(0 <= ver && ver < versions());
        roots.push_back(roots[ver]);
        return versions() - 1;
    }
    void reserve_nodes(size_t extra) { tr.reserve(tr.size() + extra); }
    int versions() const { return int(roots.size()); }
    int nodes() const { return int(tr.size()) - 1; }
};
```
#### 接口说明
##### `Info / Tag` 约定
`Info` 需要支持：
```cpp
Info operator+(const Info &a, const Info &b);
void Info::apply(const Tag &v, int l, int r);
```
其中 `Info{}` 必须是合并单位元
`Tag` 需要支持：
```cpp
void Tag::apply(const Tag &v);
```
而且**组合顺序非常重要**：
```text
tag.apply(v)
```
表示：
> 已经存在的 `tag` 先执行，再执行新的 `v`

也就是：
$$v\circ tag$$
这一约定不能反过来，否则赋值、仿射变换等非交换标记都会错
##### 版本修改
`range_apply(ver, l, r, tag)`：从版本 `ver` 出发，对 `[l,r)` 施加标记并生成新版本
时间复杂度：
$$O(\log n)$$
新增节点数量：
$$O(\log n)$$
`modify(ver, x, f)`：从历史版本 `ver` 出发修改一个叶子并生成新版本。
$$O(\log n)$$
`set(ver, x, info)` / `get(ver, x)`
$$O(\log n)$$
`copy_version(ver)`：生成内容完全相同的新版本，不复制任何树节点。
$$O(1)$$
##### 查询
`query(ver, l, r)`：查询历史版本 `ver` 的 `[l,r)`
$$O(\log n)$$
并且：
$$\text{新增节点数}=0$$
查询单点
$$O(\log n)$$
`all(ver)`：查询整个版本的根信息
$$O(1)$$
##### 查找
`find_first(ver, l, r, pred)`：寻找 `[l,r)` 中最靠左、其所在路径没有被 `pred` 剪掉的位置
`find_last()` 同理寻找最靠右位置
在常见能够有效剪枝的维护信息下：
$$O(\log n)$$
若 `pred` 不具有正确的区间剪枝性质，则不仅可能复杂度退化，逻辑本身也可能不成立
#### 注意事
- 区间统一使用 `[l,r)`，下标为 `0ba生新分支
- `push()` 绝不能直接修改旧儿子。旧儿子可能仍被其他历史版本引用，所以必须先 `clone()`
- 查询不能使用普通 Lazy Segment Tree 那种破坏性的 `push()`。本板子通过携带“祖先尚未下传的标记”完成只读查询，因此查询不会产生节点
- `has_tag` 独立于 `Tag` 保存，避免强迫 `Tag{}` 必须能够判断“是否为空”。`Tag{}` 只要求可以默认构造
- `Info{}` 必须是 `operator+` 的单位元
- `set()` 和 `modify()` 到达叶子时会清除叶子的旧懒标记，因为叶子已经没有需要继续下传的子树
- 不建议加入节点回收：多个版本可能共同引用同一节点，普通 free list 无法判断一个节点是否仍被其他版本使用
- 如果题目只有单点修改而没有区间修改，应使用前一份普通**可持久化线段树**，代码和常数都会更小
#### 常见用法
##### 区间加 + 区间和
这是最简洁的一组 `Info / Tag`：
```cpp
struct Tag {
    i64 add{};
    void apply(const Tag &t) {
        add += t.add;
    }
};
struct Info {
    i64 sum{};
    Info() = default;
    Info(i64 x) : sum(x) {}
    void apply(const Tag &t, int l, int r) {
        sum += t.add * (r - l);
    }
};
Info operator+(const Info &a, const Info &b) {
    return { a.sum + b.sum };
}
```
直接从数组或 view 建树：
```cpp
PersistentLazySegTree<Info, Tag> seg(
    a | views::transform([](i64 x) {
        return Info(x);
    })
);
```
版本 `0` 加：
```cpp
int v1 = seg.range_apply(0, l, r, Tag{ x });
```
继续从 `v1` 修改：
```cpp
int v2 = seg.range_apply(v1, ql, qr, Tag{ y });
```
旧版本依旧可查：
```cpp
i64 a0 = seg.query(0, l, r).sum;
i64 a1 = seg.query(v1, l, r).sum;
i64 a2 = seg.query(v2, l, r).sum;
```
##### 同时维护 `sum / min / max`
```cpp
struct Tag {
    i64 add{};
    void apply(const Tag &t) {
        add += t.add;
    }
};
struct Info {
    i64 sum{};
    i64 mn{ INF };
    i64 mx{ -INF };
    void apply(const Tag &t, int l, int r) {
        sum += t.add * (r - l);
        mn += t.add;
        mx += t.add;
    }
};
Info operator+(const Info &a, const Info &b) {
    return {
        a.sum + b.sum,
        min(a.mn, b.mn),
        max(a.mx, b.mx)
    };
}
```
初始化：
```cpp
PersistentLazySegTree<Info, Tag> seg(
    a | views::transform([](i64 x) {
        return Info{ x, x, x };
    })
);
```
查询：
```cpp
auto res = seg.query(ver, l, r);
i64 sum = res.sum;
i64 mn = res.mn;
i64 mx = res.mx;
```
##### 找第一个不小于 `x` 的位置
维护 `mx` 后：
```cpp
int p = seg.find_first(ver, l, r, [&](const Info &a) {
    return a.mx >= x;
});
```
最后一个不小于 `x`：
```cpp
int p = seg.find_last(ver, l, r, [&](const Info &a) {
    return a.mx >= x;
});
```
这里 `mx < x` 能保证整段没有答案，所以满足剪枝要求。
##### 区间赋值 + 区间加
这一组很适合检查 `Tag::apply()` 的组合方向
```cpp
struct Tag {
    optional<i64> set;
    i64 add{};
    void apply(const Tag &t) {
        if (t.set) {
            set = t.set;
            add = t.add;
        } else {
            add += t.add;
        }
    }
};
struct Info {
    i64 sum{}, mn{ INF }, mx{ -INF };
    void apply(const Tag &t, int l, int r) {
        if (t.set) {
            i64 x = *t.set;
            sum = x * (r - l);
            mn = mx = x;
        }
        sum += t.add * (r - l);
        mn += t.add;
        mx += t.add;
    }
};
Info operator+(const Info &a, const Info &b) {
    return {
        a.sum + b.sum,
        min(a.mn, b.mn),
        max(a.mx, b.mx)
    };
}
```
区间赋值：
```cpp
int nv = seg.range_apply(ver, l, r, Tag{ x, 0 });
```
区间加：
```cpp
int nv = seg.range_apply(ver, l, r, Tag{ nullopt, x });
```
若已有：
```text
先赋值 5
再加 3
```
最终必须等价于：
```text
赋值 8
```
而：
```text
先加 3
再赋值 5
```
最终必须等价于：
```text
赋值 5
```
这就是为什么 `Tag::apply()` 的先后顺序不能随便写
##### 仿射修改
若已经有取模类 `Z`，可以维护：
$$x\leftarrow ax+b$$
```cpp
struct Tag {
    Z mul{ 1 }, add{};
    void apply(const Tag &t) {
        add = add * t.mul + t.add;
        mul *= t.mul;
    }
};
struct Info {
    Z sum{};
    void apply(const Tag &t, int l, int r) {
        sum = sum * t.mul + t.add * (r - l);
    }
};
Info operator+(const Info &a, const Info &b) {
    return { a.sum + b.sum };
}
```
这里若当前标记为：
`f(x)=ax+b`
新标记为：
$g(x)=cx+d$
那么组合后应该是：
$$g(f(x)) = acx+bc+d$$
所以对应代码正是：
```cpp
add = add * t.mul + t.add;
mul *= t.mul;
```
这也是验证 `Tag::apply()` 是否写反的最好方法之一
##### 分叉版本
```cpp
int v1 = seg.range_apply(0, 0, 5, Tag{ 3 });
int v2 = seg.range_apply(v1, 2, 8, Tag{ 4 });
int v3 = seg.range_apply(v1, 1, 6, Tag{ 10 });
```
版本关系：
```text
0
└── v1
    ├── v2
    └── v3
```
`v2` 和 `v3` 共享没有被修改的绝大多数节点，但之后彼此完全独立
##### 每次操作都要求产生版本
修改：
```cpp
seg.range_apply(ver, l, r, tag);
```
查询操作如果同样占一个版本：
```cpp
auto ans = seg.query(ver, l, r);
seg.copy_version(ver);
```
于是始终可以保持：
```text
操作编号 = 新版本编号
```
##### `reserve`
与普通可持久化线段树相比，这一版一次区间修改的节点常数会更
- 边界路径需要复制
- 遇到尚未下传的 Tag 时，两个儿子都必须复制
所以若操作很多，建议提前：
```cpp
seg.reserve_nodes(estimated_nodes);
```
不要机械地按普通主席树“一次修改只开一条链”的常数预估。
##### 复杂度
设序列长度为 $n$
初始完整建树：
$$O(n)$$
初始节点数：
$$2n-1$$
单次区间修改 / 单点修改：
$$O(\log n)$$
单次区间修改新增节点数：
$$O(\log n)$$
查询整棵树 / 复制一个版本：
$$O(1)$$
若初始建树后进行了 $q$ 次区间修改，总空间复杂度为：
$$O(n+q\log n)$$
版本根数组额外占用：
$$
O(V)$$

其中$v$ 为版本数量
对于满足正确区间剪枝条件的 `find_first / find_last`，通常为
$(\log n)$
若谓词不能有效剪枝，最坏可能退化到：
$$O(n)$$


> **修改时：要改共享儿子，先 clone**
> **查询时：不要 push，把祖先 Tag 带下去**
