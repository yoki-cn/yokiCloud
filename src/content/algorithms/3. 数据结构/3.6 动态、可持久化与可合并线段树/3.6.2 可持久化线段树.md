### 可持久化线段树
---
```cpp
template<class Info>
struct PersistentSegTree {
    struct Node {
        int ch[2]{};
        Info info{};
    };
    int n{};
    vector<Node> tr{ Node{} };
    vector<int> roots;
    // 空初始版本：所有位置均为 Info{}
    explicit PersistentSegTree(int n_, size_t reserve_nodes = 0) : n(n_), roots{ 0 } {
        assert(n > 0);
        if (reserve_nodes) tr.reserve(reserve_nodes + 1);
    }
    // 从任意随机访问 range / view 建立第 0 个版本
    template<ranges::random_access_range R>
    requires ranges::sized_range<R>
          && convertible_to<ranges::range_reference_t<R>, Info>
    explicit PersistentSegTree(R &&a, size_t reserve_nodes = 0)
        : n(int(ranges::size(a))) {
        assert(n > 0);
        if (reserve_nodes) tr.reserve(reserve_nodes + 1);
        roots.push_back(build(0, n, a));
    }
    // ===== 节点创建与维护：build / modify 依赖，不建议删除 =====
    int new_node(Node x = {}) {
        tr.push_back(move(x));
        return int(tr.size()) - 1;
    }
    int clone(int p) { return new_node(tr[p]); }
    void pull(int p) {
        tr[p].info = tr[tr[p].ch[0]].info + tr[tr[p].ch[1]].info;
    }
    // ===== 初始建树：使用空初始版本时可删除 build() 与 range 构造函数 =====
    template<ranges::random_access_range R>
    int build(int l, int r, R &a) {
        int p = new_node();
        if (r - l == 1) {
            tr[p].info = *(ranges::begin(a) + l);
            return p;
        }
        int m = midpoint(l, r);
        tr[p].ch[0] = build(l, m, a);
        tr[p].ch[1] = build(m, r, a);
        pull(p);
        return p;
    }
    // ===== 路径复制修改：可持久化核心，不建议删除 =====
    template<class F>
    int modify(int p, int l, int r, int x, F &f) {
        int q = clone(p);
        if (r - l == 1) {
            invoke(f, tr[q].info);
            return q;
        }
        int m = midpoint(l, r);
        int d = x >= m;
        int c = modify(tr[q].ch[d], d ? m : l, d ? r : m, x, f);
        tr[q].ch[d] = c;
        pull(q);
        return q;
    }
    template<class F>
    int modify(int ver, int x, F &&f) {
        assert(0 <= ver && ver < versions());
        assert(0 <= x && x < n);
        roots.push_back(modify(roots[ver], 0, n, x, f));
        return versions() - 1;
    }
    int set(int ver, int x, const Info &v) {
        return modify(ver, x, [&](Info &a) { a = v; });
    }
    // ===== 查询：只读任意历史版本，不创建节点 =====
    Info query(int p, int l, int r, int ql, int qr) const {
        if (!p || qr <= l || r <= ql) return Info{};
        if (ql <= l && r <= qr) return tr[p].info;
        int m = midpoint(l, r);
        return query(tr[p].ch[0], l, m, ql, qr) + query(tr[p].ch[1], m, r, ql, qr);
    }
    Info query(int ver, int l, int r) const {
        assert(0 <= ver && ver < versions());
        assert(0 <= l && l <= r && r <= n);
        return query(roots[ver], 0, n, l, r);
    }
    Info get(int ver, int x) const {
        assert(0 <= x && x < n);
        return query(ver, x, x + 1);
    }
    const Info &all(int ver) const {
        assert(0 <= ver && ver < versions());
        return tr[roots[ver]].info;
    }
    // ===== 版本管理：需要“查询操作也生成一个版本”时很方便 =====
    int copy_version(int ver) {
        assert(0 <= ver && ver < versions());
        roots.push_back(roots[ver]);
        return versions() - 1;
    }
    // 按预计单点修改次数预留空间
    void reserve_updates(size_t q) {
        size_t per = bit_width(static_cast<unsigned>(n - 1)) + 1;
        tr.reserve(tr.size() + q * per);
        roots.reserve(roots.size() + q);
    }
    int versions() const { return int(roots.size()); }
    int nodes() const { return int(tr.size()) - 1; }
};
```
#### 接口说明
##### 模板参数与成员变量
* `Info`：线段树节点维护的信息
* `Info{}`：必须是合并运算的单位元
* `tr`：所有历史版本共享的节点池，`tr[0]` 是空节点
* `roots[v]`：版本 `v` 的根节点
* `n`：维护下标范围 `[0, n)`
其中每个版本实际上只保存一个根：
```text
roots[0] ──┐
           ├── 大量共享旧节点
roots[1] ──┤
           ├── 仅修改路径上的节点不同
roots[2] ──┘
```
##### 构造
空初始版本：
```cpp
PersistentSegTree<Info> seg(n);
```
此时第 `0` 个版本的所有位置都等价于：
```cpp
Info{}
```
从初始序列建树：
```cpp
PersistentSegTree<Info> seg(a);
```
也支持 `view`，因此不必为了转换成 `Info` 再额外构造数组：
```cpp
PersistentSegTree<Info> seg(
    a | views::transform([](i64 x) {
        return Info{ x };
    })
);
```
##### 修改与版本
`modify(ver, x, f)`：以版本 `ver` 为基础修改位置 `x`，产生一个**新版本**，返回新版本编号
```cpp
int nv = seg.modify(ver, x, [&](Info &a) { a.sum += d; });
```
原来的 `ver` 不受影响
`set(ver, x, v)`：覆盖位置 `x`：
```cpp
int nv = seg.set(ver, x, Info{ v });
```
`copy_version(ver)`：创建一个完全相同的新版本，但不复制任何树节点：
```cpp
int nv = seg.copy_version(ver);
```
这个接口对“**每次操作都必须产生一个编号版本**”的问题尤其方便
##### 查询
`query(ver, l, r)`：查询版本 `ver` 的 `[l, r)`
```cpp
auto ans = seg.query(ver, l, r);
```
`get(ver, x)`：查询某版本单点
```cpp
auto value = seg.get(ver, x);
```
`all(ver)`：查询某版本整棵树维护的信息
```cpp
auto info = seg.all(ver);
```
`versions()`：当前版本数
`nodes()`：节点池实际创建的节点数量
`reserve_updates(q)`：已知接下来大约有 `q` 次单点修改时预留节点空间，避免节点池频繁扩容
#### 注意事项
* 区间统一为 `[l, r)`，下标为 `0base`
* 当前版本只做**单点修改**；不加入 `Tag / push()`。可持久化区间修改需要处理共享节点上的 lazy，下一个“可持久化懒标记线段树”再单独整理
* `modify(ver, ...)` 可以从**任意历史版本**继续生成新版本，因此不仅能形成版本链，也能形成版本树：
```text
0
├── 1
│   ├── 2
│   └── 3
└── 4
```
* `Info{}` 必须满足合并单位元要求：
$$e+x=x+e=x$$
否则空初始版本以及查询越界部分都会产生错误信息
* 未发生修改的子树会在多个版本之间共享，因此**绝对不能原地修改旧节点**。所有会改变节点内容的操作必须先经过 `clone()`
* 也正因为节点可能同时被很多历史版本引用，本板子**不能使用简单 free list 回收节点**。节点回收留给不存在共享所有权问题的可合并权值线段树
* `copy_version()` 只复制根编号，不创建树节点
* `query()` 完全只读，不会创建节点
#### 常见用法
##### 1. 可持久化数组：单点赋值 + 历史查询
最简单的 `Info`：
```cpp
struct Info {
    i64 sum{};
};
Info operator+(const Info &a, const Info &b) {
    return { a.sum + b.sum };
}
```
从原数组建立版本 `0`：
```cpp
vector<i64> a(n);
PersistentSegTree<Info> seg(
    a | views::transform([](i64 x) {
        return Info{ x };
    })
);
```
从版本 `v` 修改：
```cpp
int nv = seg.set(v, p, { x });
```
查询历史版本：
```cpp
i64 x = seg.get(v, p).sum;
i64 sum = seg.query(v, l, r).sum;
```
##### 2. 在任意历史版本上分叉
例如：
```cpp
int v1 = seg.set(0, 3, { 10 });
int v2 = seg.set(v1, 5, { 20 });
int v3 = seg.set(v1, 7, { 30 });
```
版本关系是：
```text
0
└── v1
    ├── v2
    └── v3
```
其中：
```cpp
seg.get(v2, 7);
seg.get(v3, 5);
```
互不影响
##### 3. 按操作编号建立版本
如果每个修改操作恰好生成一个版本，可以直接利用版本编号：
```cpp
for (int i : views::iota(0, q)) {
    int p;
    i64 x;
    cin >> p >> x;
    seg.set(i, p, { x });
}
```
于是：
```text
版本 0 = 初始状态
版本 1 = 第 1 次修改后
版本 2 = 第 2 次修改后
...
```
若某个操作只是查询，但题意仍要求产生一个新版本：
```cpp
seg.copy_version(old_ver);
```
#### 常见 `Info`
##### 区间和
```cpp
struct Info {
    i64 sum{};
};
Info operator+(const Info &a, const Info &b) {
    return { a.sum + b.sum };
}
```
单点加：
```cpp
int nv = seg.modify(ver, p, [&](Info &a) {
    a.sum += x;
});
```
单点赋值：
```cpp
int nv = seg.set(ver, p, { x });
```
##### 同时维护和、最小值、最大值
空节点必须是单位元，所以最小值与最大值不能默认设成 `0`：
```cpp
struct Info {
    i64 sum{};
    i64 mn{ INF };
    i64 mx{ -INF };
};
Info operator+(const Info &a, const Info &b) {
    return {
        a.sum + b.sum,
        min(a.mn, b.mn),
        max(a.mx, b.mx)
    };
}
```
初始化一个实际元素：
```cpp
auto make_info = [](i64 x) {
    return Info{ x, x, x };
};
```
可以直接用 `views::transform` 建树：
```cpp
PersistentSegTree<Info> seg(
    a | views::transform([&](i64 x) {
        return make_info(x);
    })
);
```
查询：
```cpp
auto res = seg.query(ver, l, r);
i64 sum = res.sum;
i64 mn = res.mn;
i64 mx = res.mx;
```
##### 维护最大公约数
`0` 正好是 `gcd` 的单位元：
```cpp
struct Info {
    i64 g{};
};
Info operator+(const Info &a, const Info &b) {
    return { gcd(a.g, b.g) };
}
```
因此：
```cpp
auto g = seg.query(ver, l, r).g;
```
可以直接查询任意历史版本的区间 gcd
##### 维护最大子段和
这种 `Info` 最容易犯错，因为空区间不能简单当成全 `0` 节点。可以显式维护 `empty`：
```cpp
struct Info {
    i64 sum{};
    i64 pre{}, suf{}, ans{};
    bool empty{ true };
    Info() = default;
    explicit Info(i64 x) : sum(x), pre(x), suf(x), ans(x), empty(false) {}
};
Info operator+(const Info &a, const Info &b) {
    if (a.empty) return b;
    if (b.empty) return a;
    Info c;
    c.empty = false;
    c.sum = a.sum + b.sum;
    c.pre = max(a.pre, a.sum + b.pre);
    c.suf = max(b.suf, b.sum + a.suf);
    c.ans = max({ a.ans, b.ans, a.suf + b.pre });
    return c;
}
```
于是初始序列仍然可以漂亮地写成：
```cpp
PersistentSegTree<Info> seg(
    a | views::transform([](i64 x) {
        return Info(x);
    })
);
```
之后：
```cpp
i64 ans = seg.query(ver, l, r).ans;
```
就是某个历史版本 `[l, r)` 的最大子段和
这个例子其实也说明了一个原则：
> 可持久化只改变“节点如何产生”，**不会改变普通线段树的 `Info` 合并逻辑**
##### 维护频数与权值和
有时某个坐标代表一个值，希望同时记录出现次数与权值和：
```cpp
struct Info {
    int cnt{};
    i64 sum{};
};
Info operator+(const Info &a, const Info &b) {
    return {
        a.cnt + b.cnt,
        a.sum + b.sum
    };
}
```
加入一个值：
```cpp
int nv = seg.modify(ver, x, [&](Info &a) {
    ++ a.cnt;
    a.sum += x;
});
```
删除：
```cpp
int nv = seg.modify(ver, x, [&](Info &a) {
    -- a.cnt;
    a.sum -= x;
});
```
这样已经能处理“某个历史版本的值域计数 / 值域和”
不过一旦要进一步写：
```text
两版本作差
kth
rank
前驱
后继
```

就应该切换到**主席树**
#### 常见套路
##### 历史版本回退
其实完全不需要真正“撤销”：
```cpp
int current = old_version;
```
之后继续：
```cpp
current = seg.set(current, p, { x });
```
即可从那个历史状态重新分叉
##### 查询操作也产生版本
某些题目的第 `i` 次操作对应版本 `i`，即使本次没有修改：
```cpp
if (op == 1) {
    seg.set(ver, p, { x });
} else {
    cout << seg.query(ver, l, r).sum << '\n';
    seg.copy_version(ver);
}
```
这样始终保持：
```text
操作编号 = 新版本编号
```
会比在外面维护复杂映射干净很多
##### 预留节点
假设之后大约有 `q` 次单点修改：
```cpp
seg.reserve_updates(q);
```
一次修改只复制根到叶的一条路径
若维护长度为 $n$ 的序列，每次修改创建的节点数至多约为：
$$\lceil \log_2 n\rceil+1$$
因此节点数通常可以预估为：
$$N_{\text{initial}} + q\left(\lceil\log_2 n\rceil+1\right)$$
其中从完整初始数组建树时：
$$N_{\text{initial}}=2n-1$$
如果初始版本为空树：
$$N_{\text{initial}}=0$$
`reserve_updates()` 内部利用 `bit_width()` 自动估计这部分空间
#### 复杂度
设序列长度为 $n$
从完整初始数组建树：
$$O(n)$$
单次 `modify()` 或 `set()`：
$$O(\log n)$$
并新建：
$$O(\log n)$$
个节点
单点查询：
$$O(\log n)$$
区间查询：
$$O(\log n)$$
`all()`：
$$O(1)$$
`copy_version()`：
$$O(1)$$
且不创建任何树节点
若初始建树后进行了 \(q\) 次单点修改，总空间复杂度为：
$$O(n+q\log n)$$
若初始版本为空、之后仅通过单点修改动态产生节点，则为：
$$O(q\log n)$$
版本根数组本身额外占用：
$$(V)$$
其中 $V$ 为版本数量
