### 主席树
---
```cpp
template<class T = i64>
struct ChairmanTree {
    struct Node {
        int ch[2]{};
        int cnt{};
    };
    vector<T> val;
    vector<Node> tr{ Node{} };
    vector<int> root{ 0 };
    bool prefix_built{ false };
    // build_prefix = true：a 既作为待离散化值域，也建立前缀版本
    // build_prefix = false：只离散化 a，常用于树上路径等自定义版本关系
    template<ranges::input_range R>
    explicit ChairmanTree(R &&a, bool build_prefix = true, size_t reserve_nodes = 0) {
        vector<T> seq;
        if constexpr (ranges::sized_range<R>) seq.reserve(ranges::size(a));
        for (auto &&x : a) seq.emplace_back(x);
        val = seq;
        ranges::sort(val);
        val.erase(ranges::unique(val).begin(), val.end());
        assert(!val.empty());
        if (reserve_nodes) tr.reserve(reserve_nodes + 1);
        prefix_built = build_prefix;
        if (build_prefix) {
            root.resize(seq.size() + 1);
            if (!reserve_nodes) reserve_updates(seq.size());
            for (int i : views::iota(0, int(seq.size()))) {
                root[i + 1] = add_root(root[i], seq[i]);
            }
        }
    }
    // ===== 节点创建与单点路径复制：主席树核心，不建议删除 =====
    int new_node(Node x = {}) {
        tr.push_back(x);
        return int(tr.size()) - 1;
    }
    int clone(int p) { return new_node(tr[p]); }
    int id(const T &x) const {
        return int(ranges::lower_bound(val, x) - val.begin());
    }
    int add(int p, int l, int r, int x, int delta) {
        int q = clone(p);
        tr[q].cnt += delta;
        if (r - l == 1) return q;
        int m = midpoint(l, r);
        int d = x >= m;
        tr[q].ch[d] = add(tr[q].ch[d], d ? m : l, d ? r : m, x, delta);
        return q;
    }
    // 从任意 root 产生新 root；x 必须已存在于离散值域中
    int add_root(int p, const T &x, int delta = 1) {
        int k = id(x);
        assert(k < int(val.size()) && val[k] == x);
        return add(p, 0, val.size(), k, delta);
    }
    // ===== 两版本差分查询：前缀区间 [l, r) 的基础 =====
    int prefix_count(int a, int b, int l, int r, int x) const {
        if (x <= l) return 0;
        if (r <= x) return tr[b].cnt - tr[a].cnt;
        int m = midpoint(l, r);
        return prefix_count(tr[a].ch[0], tr[b].ch[0], l, m, x) + prefix_count(tr[a].ch[1], tr[b].ch[1], m, r, x);
    }
    int prefix_count(int a, int b, int x) const { return prefix_count(a, b, 0, val.size(), x); }
    int kth_index(int a, int b, int k) const {
        int l = 0, r = val.size();
        while (r - l > 1) {
            int m = midpoint(l, r);
            int ca = tr[a].ch[0], cb = tr[b].ch[0];
            int left = tr[cb].cnt - tr[ca].cnt;
            if (k < left) {
                a = ca;
                b = cb;
                r = m;
            } else {
                k -= left;
                a = tr[a].ch[1];
                b = tr[b].ch[1];
                l = m;
            }
        }
        return l;
    }
    // ===== 静态区间接口：[l, r)，k 为 0base =====
    int size(int l, int r) const {
        assert_prefix(l, r);
        return tr[root[r]].cnt - tr[root[l]].cnt;
    }
    T kth(int l, int r, int k) const {
        assert_prefix(l, r);
        assert(0 <= k && k < size(l, r));
        return val[kth_index(root[l], root[r], k)];
    }
    T kth_largest(int l, int r, int k) const { return kth(l, r, size(l, r) - 1 - k); }
    int count_less(int l, int r, const T &x) const {
        assert_prefix(l, r);
        int p = ranges::lower_bound(val, x) - val.begin();
        return prefix_count(root[l], root[r], p);
    }
    int count_leq(int l, int r, const T &x) const {
        assert_prefix(l, r);
        int p = int(ranges::upper_bound(val, x) - val.begin());
        return prefix_count(root[l], root[r], p);
    }
    int count(int l, int r, const T &lo, const T &hi) const {
        assert_prefix(l, r);
        if (!(lo < hi)) return 0;
        int L = ranges::lower_bound(val, lo) - val.begin();
        int R = ranges::lower_bound(val, hi) - val.begin();
        return prefix_count(root[l], root[r], R) - prefix_count(root[l], root[r], L);
    }
    int frequency(int l, int r, const T &x) const { return count_leq(l, r, x) - count_less(l, r, x); }
    // 排名按竞赛中常见定义：小于 x 的元素数 + 1
    int rank(int l, int r, const T &x) const { return count_less(l, r, x) + 1; }
    optional<T> predecessor(int l, int r, const T &x) const {
        int k = count_less(l, r, x);
        if (!k) return nullopt;
        return kth(l, r, k - 1);
    }
    optional<T> successor(int l, int r, const T &x) const {
        int k = count_leq(l, r, x);
        if (k == size(l, r)) return nullopt;
        return kth(l, r, k);
    }
    // ===== 树上路径四版本 kth：不需要时可删除 =====
    // root[u] + root[v] - root[lca] - root[parent(lca)]
    T kth_path(int a, int b, int c, int d, int k) const {
        int total = tr[a].cnt + tr[b].cnt - tr[c].cnt - tr[d].cnt;
        assert(0 <= k && k < total);
        int l = 0, r = val.size();
        while (r - l > 1) {
            int m = midpoint(l, r);
            int al = tr[a].ch[0], bl = tr[b].ch[0];
            int cl = tr[c].ch[0], dl = tr[d].ch[0];
            int left = tr[al].cnt + tr[bl].cnt - tr[cl].cnt - tr[dl].cnt;
            if (k < left) {
                a = al;
                b = bl;
                c = cl;
                d = dl;
                r = m;
            } else {
                k -= left;
                a = tr[a].ch[1];
                b = tr[b].ch[1];
                c = tr[c].ch[1];
                d = tr[d].ch[1];
                l = m;
            }
        }
        return val[l];
    }
    // ===== 辅助接口 =====
    void reserve_updates(size_t q) {
        size_t h = bit_width(unsigned(val.size() - 1)) + 1;
        tr.reserve(tr.size() + q * h);
    }
    int nodes() const { return int(tr.size()) - 1; }
private:
    void assert_prefix(int l, int r) const {
        assert(prefix_built);
        assert(0 <= l && l <= r && r < int(root.size()));
    }
};
```
这版和一般可持久化线段树相比做了两个重要特化：
1. 不再泛化 `Info`，只维护 `cnt`，因为主席树的核心就是**权值域频数版本差分**
2. 不建立完整零树，`0` 号节点直接代表整棵空树；每插入一个元素仅复制一条路径
#### 接口说明
##### 成员变量
- `val`：排序去重后的权值数组
- `tr`：所有版本共享的节点池，`tr[0]` 表示空节点
- `root[i]`：前缀 `[0, i)` 对应的版本根
- `prefix_built`：是否建立了标准前缀版本
标准构造：
```cpp
ChairmanTree seg(a);
```
版本含义为：
```text
root[0]：空集
root[1]：a[0]
root[2]：a[0..1]
...
root[i]：a[0..i)
```
所以区间 `[l, r)` 的权值集合就是：
```text
root[r] - root[l]
```
##### 查询接口
- `kth(l, r, k)`：返回 `[l, r)` 中第 `k` 小，`k` 为 **0base**
- `kth_largest(l, r, k)`：返回第 `k` 大，`k` 为 **0base**
- `count_less(l, r, x)`：统计 `< x`
- `count_leq(l, r, x)`：统计 `<= x`
- `count(l, r, lo, hi)`：统计值落在 `[lo, hi)` 的元素数
- `frequency(l, r, x)`：统计 `x` 出现次数
- `rank(l, r, x)`：返回 `x` 的排名，即“小于 `x` 的数量 + 1”
- `predecessor(l, r, x)`：严格小于 `x` 的最大值，不存在返回 `nullopt`
- `successor(l, r, x)`：严格大于 `x` 的最小值，不存在返回 `nullopt`
- `size(l, r)`：区间元素数量
上述查询的时间复杂度均为：
$$O(\log m)$$
其中 $m$ 是不同权值数量
##### 自定义版本接口
- `add_root(p, x, delta)`：从根 `p` 出发，将 `x` 的频数增加 `delta`，返回新根
- `kth_path(a, b, c, d, k)`：对四个根的线性组合
$$a+b-c-d$$
求第 `k` 小
单次 `add_root()` 的复杂度为：
$$O(\log m)$$
并新增：
$$O(\log m)$$
个节点
#### 注意事项
- 数组区间统一使用 `[l, r)`，数组下标为 `0base`
- `kth()` 的 `k` 也是 `0base`：
```cpp
seg.kth(l, r, 0);          // 最小值
seg.kth(l, r, r - l - 1);  // 最大值
```
- `rank()` 则沿用常见题面定义，从 `1` 开始
- `count(l, r, lo, hi)` 的值域区间也是 `[lo, hi)`
- 主席树查询依赖：
$$\operatorname{cnt}(root_r)-\operatorname{cnt}(root_l)$$
因此标准前缀版本只能用于**静态序列**；原数组发生修改后，后面的所有前缀版本都会受到影响
- `add_root()` 的 `x` 必须已经存在于初始化时的 `val` 中。若后续可能插入其他值，构造时必须把所有可能出现的值提前加入离散值域
- 如果只需要静态区间第 `k` 小，核心真正必需的只有：
```text
Node
add()
add_root()
kth_index()
kth()
```
`count / rank / predecessor / successor / kth_path` 都可以按题目裁掉
- 动态单点修改 + 区间第 `k` 小不是普通主席树直接擅长的模型；传统做法通常转为 BIT 套动态权值线段树等结构。
#### 常见用法
##### 静态区间第 `k` 小
```cpp
vector<i64> a(n);
ChairmanTree seg(a);
i64 ans = seg.kth(l, r, k);
```
例如第 `3` 小：
```cpp
i64 ans = seg.kth(l, r, 2);
```
##### 区间排名
求 `x` 在 `[l, r)` 中的排名：
```cpp
int rk = seg.rank(l, r, x);
```
其含义为：
$$1+\#\{a_i<x\}$$
因此即使 `x` 本身不存在，也能得到它插入后的排名
##### 前驱与后继
```cpp
if (auto x = seg.predecessor(l, r, v)) {
    // *x 为严格小于 v 的最大值
}
if (auto x = seg.successor(l, r, v)) {
    // *x 为严格大于 v 的最小值
}
```
如果题目保证一定存在，可以直接：
```cpp
i64 pre = *seg.predecessor(l, r, v);
```
##### 值域计数
查询 `[l, r)` 中：
###### 小于 `x`
```cpp
int cnt = seg.count_less(l, r, x);
```
###### 不大于 `x`
```cpp
int cnt = seg.count_leq(l, r, x);
```
###### 落在 `[lo, hi)`
```cpp
int cnt = seg.count(l, r, lo, hi);
```
这套接口经常可以直接替代额外二分：
```text
第 k 小
排名
频数
值域计数
前驱后继
```
本质都只是对同一份版本差分频数树的不同下降方式
#### 树上路径第 `k` 小
这是主席树非常经典的另一套版本组织方式
“LCA + 主席树”的树上路径点权第 $k$ 大
先只离散化所有点权，不建立数组前缀：
```cpp
ChairmanTree seg(w, false);
```
令：
```text
root[u] = 从树根到 u 的路径上所有点权形成的主席树
```
DFS 时：
```cpp
root[u] = seg.add_root(root[fa], w[u]);
```
例如：
```cpp
vector<int> root(n);
auto dfs = Y([&](auto self, int u, int fa) -> void {
    root[u] = seg.add_root(fa == -1 ? 0 : root[fa], w[u]);
    for (int v : g[u]) if (v != fa) {
        self(v, u);
    }
});
```
设：
```cpp
int p = lca(u, v);
int pp = parent[p];
```
则路径 `u -> v` 上的点权集合为：
$$root_u+root_v-root_p-root_{\operatorname{parent}(p)}$$
于是：
```cpp
int rp = parent[p] == -1 ? 0 : root[parent[p]];
i64 ans = seg.kth_path(root[u], root[v], root[p], rp, k);
```
这里的 `k` 同样为 `0base`
为什么是四个版本很好记：
```text
root[u] + root[v]
```
会把 LCA 及其祖先算两次，因此减去：
```text
root[lca] + root[parent(lca)]
```
恰好保留路径上的所有节点一次
#### 常见维护扩展：同时维护权值和
这个功能很有价值，但**不建议默认塞进核心主席树**
原因很现实：节点从：
```cpp
struct Node {
    int ch[2];
    int cnt;
};
```
扩成：
```cpp
struct Node {
    int ch[2]{};
    int cnt{};
    i64 sum{};
};
```
大量节点下会明显增加内存
如果题目需要：
- 前 `k` 小之和
- 后 `k` 大之和
- 小于某值的元素和
- 区间值域和
再增加即可
##### 修改节点
在 `add()` 中加入：
```cpp
tr[q].sum += i64(val[x]) * delta;
```
即：
```cpp
int add(int p, int l, int r, int x, int delta) {
    int q = clone(p);
    tr[q].cnt += delta;
    tr[q].sum += i64(val[x]) * delta;
    if (r - l == 1) return q;
    int m = midpoint(l, r);
    int d = x >= m;
    tr[q].ch[d] = add(tr[q].ch[d], d ? m : l, d ? r : m, x, delta);
    return q;
}
```
##### 前 `k` 小元素和
这里的 `k` 表示**取多少个元素**，范围为：
$$0\le k\le r-l$$
```cpp
i64 sum_k_smallest_root(int a, int b, int k) const {
    i64 ans = 0;
    int l = 0, r = val.size();
    while (r - l > 1) {
        int m = midpoint(l, r);
        int al = tr[a].ch[0];
        int bl = tr[b].ch[0];
        int left = tr[bl].cnt - tr[al].cnt;
        if (k <= left) {
            a = al;
            b = bl;
            r = m;
        } else {
            ans += tr[bl].sum - tr[al].sum;
            k -= left;
            a = tr[a].ch[1];
            b = tr[b].ch[1];
            l = m;
        }
    }
    return ans + i64(val[l]) * k;
}
i64 sum_k_smallest(int l, int r, int k) const {
    assert_prefix(l, r);
    assert(0 <= k && k <= size(l, r));
    return sum_k_smallest_root(root[l], root[r], k);
}
```
查询前 `k` 大之和也无需重新写下降逻辑：
```cpp
i64 sum_all(int l, int r) const {
    return tr[root[r]].sum - tr[root[l]].sum;
}
i64 sum_k_largest(int l, int r, int k) const {
    return sum_all(l, r) - sum_k_smallest(l, r, size(l, r) - k);
}
```
单次仍然只下降一条值域路径：
$$O(\log m)$$
这一套路在“选若干最小/最大值求代价”类问题里非常实用
#### 常见套路：区间不同数个数
这个也是经典“主席树”应用，但它和上面的**权值主席树不是同一种版本含义**
从右向左建立版本：对于每个后缀，只让每一种值的**最左一次出现位置**为 `1`。如果当前值之前已经在更右侧位置 `p` 被标记，则先把 `p` 改成 `0`，再把当前位置改成 `1`。这样版本 `root[l]` 查询位置区间 `[l,r]` 的和，就是原数组 `[l,r]` 中不同数的个数
例如：
```text
a = [1, 2, 1, 3]
```
从位置 `0` 看：
```text
1 1 0 1
```
只保留：
```text
1 的第一次出现
2 的第一次出现
3 的第一次出现
```
于是 `[0,4)` 的和就是 `3`
这个套路本质是：
> **值决定“更新哪个旧位置”，而线段树下标维护的是“数组位置”**

因此不建议硬塞进当前 `ChairmanTree<T>` 类；直接用上一份通用的**可持久化线段树**实现更自然
#### 复杂度
设：
- 原序列长度为 $n$
- 不同权值数量为 $m$
离散化：
$$O(n\log n)$$
建立全部前缀版本：
$$O(n\log m)$$
每插入一个元素新增节点数：
$$O(\log m)$$
因此总节点数 / 总空间复杂度：
$$O(n\log m)$$
单次第 $k$ 小(大) / 单次排名查询 / 单次值域计数 / 单次频数查询 / 单次前驱 / 后继 / 树上路径第 $k$ 小：
$$O(\log m)$$
从父节点版本建立一个新的树上节点版本：
$$O(\log m)$$
如果增加权值和后查询前 $k$ 小 / 大元素和，单次复杂度仍为：
$$O(\log m)$$
