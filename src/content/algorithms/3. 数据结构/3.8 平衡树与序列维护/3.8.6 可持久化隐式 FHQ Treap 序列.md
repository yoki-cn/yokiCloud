### 可持久化隐式 FHQ Treap 序列
---
```text
可持久化 FHQ Treap 有序多重集
    ↓
按 value / Compare 维护顺序
排名、第 k 小、前驱、后继

可持久化隐式 FHQ Treap 序列
    ↓
按中序位置维护顺序
插入、删除、翻转、区间修改、区间查询
```

```cpp
template<class T = i64>
struct PersistentImplicitTreap {
    struct Node {
        int ch[2]{}, size{}, stamp{};
        unsigned pri{};
        T value{}, sum{}, add{};
        bool rev{};
    };
    vector<Node> tr{Node{}};
    vector<int> root{0};
    u64 rng_state;
    int epoch{1};
    explicit PersistentImplicitTreap(
        u64 seed = chrono::steady_clock::now().time_since_epoch().count()
    ) : rng_state(seed) {}
    explicit PersistentImplicitTreap(
        const vector<T> &a,
        u64 seed = chrono::steady_clock::now().time_since_epoch().count()
    ) : rng_state(seed) { root[0] = build_tree(a); }
    void reserve(int nodes, int versions = 0) {
        tr.reserve(nodes + 1);
        if (versions) root.reserve(versions + 1);
    }
    // ===== 节点与 COW：所有修改操作依赖，不建议删除 =====
    int node_size(int p) const { return p ? tr[p].size : 0; }
    T node_sum(int p) const { return p ? tr[p].sum : T{}; }
    unsigned rng() {
        u64 z = (rng_state += 0x9e3779b97f4a7c15ULL);
        z = (z ^ (z >> 30)) * 0xbf58476d1ce4e5b9ULL;
        z = (z ^ (z >> 27)) * 0x94d049bb133111ebULL;
        return unsigned(z ^ (z >> 31));
    }
    int new_node(const T &value) {
        tr.push_back({{}, 1, epoch, rng(), value, value, T{}, false});
        return int(tr.size()) - 1;
    }
    // 当前操作内已经复制过则直接复用，否则复制后再修改
    int touch(int p) {
        if (!p || tr[p].stamp == epoch) return p;
        tr.push_back(tr[p]);
        tr.back().stamp = epoch;
        return int(tr.size()) - 1;
    }
    void pull(int p) {
        tr[p].size = 1 + node_size(tr[p].ch[0]) + node_size(tr[p].ch[1]);
        tr[p].sum = node_sum(tr[p].ch[0]) + tr[p].value + node_sum(tr[p].ch[1]);
    }
    void begin_modify() { ++ epoch; }
    // ===== 懒标记：range_add / reverse 及带标记的 split / merge 依赖 =====
    void apply_add(int p, const T &value) {
        if (!p) return;
        tr[p].value += value;
        tr[p].sum += value * tr[p].size;
        tr[p].add += value;
    }
    void apply_rev(int p) {
        if (!p) return;
        swap(tr[p].ch[0], tr[p].ch[1]);
        tr[p].rev ^= 1;
    }
    // 子节点仍可能属于旧版本，因此下传前必须 touch()
    void push(int p) {
        if (tr[p].rev) {
            for (int d = 0; d < 2; ++ d) if (tr[p].ch[d]) {
                tr[p].ch[d] = touch(tr[p].ch[d]);
                apply_rev(tr[p].ch[d]);
            }
            tr[p].rev = false;
        }
        if (tr[p].add != T{}) {
            T add = tr[p].add;
            for (int d = 0; d < 2; ++ d) if (tr[p].ch[d]) {
                tr[p].ch[d] = touch(tr[p].ch[d]);
                apply_add(tr[p].ch[d], add);
            }
            tr[p].add = T{};
        }
    }
    // ===== split / merge：split(p, k) 将前 k 个元素分到左侧 =====
    pair<int, int> split(int p, int k) {
        if (!p) return {0, 0};
        p = touch(p);
        push(p);
        int left = node_size(tr[p].ch[0]);
        if (left >= k) {
            auto [a, b] = split(tr[p].ch[0], k);
            tr[p].ch[0] = b;
            pull(p);
            return {a, p};
        }
        auto [a, b] = split(tr[p].ch[1], k - left - 1);
        tr[p].ch[1] = a;
        pull(p);
        return {p, b};
    }
    // 要求 a 的所有元素在序列中位于 b 之前
    int merge(int a, int b) {
        if (!a || !b) return a | b;
        if (tr[a].pri < tr[b].pri) {
            a = touch(a);
            push(a);
            tr[a].ch[1] = merge(tr[a].ch[1], b);
            pull(a);
            return a;
        }
        b = touch(b);
        push(b);
        tr[b].ch[0] = merge(a, tr[b].ch[0]);
        pull(b);
        return b;
    }
    // ===== 单点核心：比统一 split + merge 的节点常数更小 =====
    int insert_node(int p, int k, int x) {
        if (!p) return x;
        if (tr[x].pri < tr[p].pri) {
            auto [a, b] = split(p, k);
            tr[x].ch[0] = a;
            tr[x].ch[1] = b;
            pull(x);
            return x;
        }
        p = touch(p);
        push(p);
        int left = node_size(tr[p].ch[0]);
        if (k <= left) tr[p].ch[0] = insert_node(tr[p].ch[0], k, x);
        else tr[p].ch[1] = insert_node(tr[p].ch[1], k - left - 1, x);
        pull(p);
        return p;
    }
    int erase_node(int p, int k) {
        p = touch(p);
        push(p);
        int left = node_size(tr[p].ch[0]);
        if (k < left) tr[p].ch[0] = erase_node(tr[p].ch[0], k);
        else if (k > left) tr[p].ch[1] = erase_node(tr[p].ch[1], k - left - 1);
        else return merge(tr[p].ch[0], tr[p].ch[1]);
        pull(p);
        return p;
    }
    int set_node(int p, int k, const T &value) {
        p = touch(p);
        push(p);
        int left = node_size(tr[p].ch[0]);
        if (k < left) tr[p].ch[0] = set_node(tr[p].ch[0], k, value);
        else if (k > left) tr[p].ch[1] = set_node(tr[p].ch[1], k - left - 1, value);
        else tr[p].value = value;
        pull(p);
        return p;
    }
    // ===== O(n) 笛卡尔树建树：不需要批量建树时可删除 =====
    int build_tree(const vector<T> &a) {
        if (a.empty()) return 0;
        vector<int> st;
        st.reserve(a.size());
        for (const T &value : a) {
            int x = new_node(value), last = 0;
            while (!st.empty() && tr[st.back()].pri >= tr[x].pri) {
                last = st.back();
                st.pop_back();
            }
            tr[x].ch[0] = last;
            if (!st.empty()) tr[st.back()].ch[1] = x;
            st.push_back(x);
        }
        int rt = st.front();
        vector<int> order{rt};
        order.reserve(a.size());
        for (int i = 0; i < order.size(); ++ i) {
            int p{ order[i] };
            if (tr[p].ch[0]) order.push_back(tr[p].ch[0]);
            if (tr[p].ch[1]) order.push_back(tr[p].ch[1]);
        }
        for (int p : views::reverse(order)) pull(p);
        return rt;
    }
    // ===== 版本与基础修改：每次调用均追加一个版本 =====
    int copy_version(int ver) {
        root.push_back(root[ver]);
        return int(root.size()) - 1;
    }
    int size(int ver) const { return node_size(root[ver]); }
    bool empty(int ver) const { return root[ver] == 0; }
    int insert(int ver, int pos, const T &value) {
        assert(0 <= pos && pos <= size(ver));
        begin_modify();
        int x = new_node(value);
        root.push_back(insert_node(root[ver], pos, x));
        return int(root.size()) - 1;
    }
    int insert(int ver, int pos, const vector<T> &values) {
        assert(0 <= pos && pos <= size(ver));
        if (values.empty()) return copy_version(ver);
        begin_modify();
        auto [a, b] = split(root[ver], pos);
        int mid = build_tree(values);
        root.push_back(merge(merge(a, mid), b));
        return int(root.size()) - 1;
    }
    int erase(int ver, int pos) {
        assert(0 <= pos && pos < size(ver));
        begin_modify();
        root.push_back(erase_node(root[ver], pos));
        return int(root.size()) - 1;
    }
    int erase(int ver, int l, int r) {
        assert(0 <= l && l <= r && r <= size(ver));
        if (l == r) return copy_version(ver);
        begin_modify();
        auto [a, c] = split(root[ver], r);
        auto [x, b] = split(a, l);
        root.push_back(merge(x, c));
        return int(root.size()) - 1;
    }
    int set(int ver, int pos, const T &value) {
        assert(0 <= pos && pos < size(ver));
        begin_modify();
        root.push_back(set_node(root[ver], pos, value));
        return int(root.size()) - 1;
    }
    // ===== 区间修改：[l, r)；若不需要对应操作可删除公开接口 =====
    int range_add(int ver, int l, int r, const T &value) {
        assert(0 <= l && l <= r && r <= size(ver));
        if (l == r || value == T{}) return copy_version(ver);
        begin_modify();
        auto [a, c] = split(root[ver], r);
        auto [x, b] = split(a, l);
        b = touch(b);
        apply_add(b, value);
        root.push_back(merge(merge(x, b), c));
        return int(root.size()) - 1;
    }
    int reverse(int ver, int l, int r) {
        assert(0 <= l && l <= r && r <= size(ver));
        if (r - l <= 1) return copy_version(ver);
        begin_modify();
        auto [a, c] = split(root[ver], r);
        auto [x, b] = split(a, l);
        b = touch(b);
        apply_rev(b);
        root.push_back(merge(merge(x, b), c));
        return int(root.size()) - 1;
    }
    // ===== 只读查询 =====
    T get_node(int p, int k, T extra, bool flip) const {
        int lc = tr[p].ch[flip ? 1 : 0];
        int rc = tr[p].ch[flip ? 0 : 1];
        int left = node_size(lc);
        T next_extra = extra + tr[p].add;
        bool next_flip = flip ^ tr[p].rev;
        if (k < left) return get_node(lc, k, next_extra, next_flip);
        if (k == left) return tr[p].value + extra;
        return get_node(rc, k - left - 1, next_extra, next_flip);
    }
    T get(int ver, int pos) const {
        assert(0 <= pos && pos < size(ver));
        return get_node(root[ver], pos, T{}, false);
    }
    T range_sum_node(int p, int l, int r, T extra, bool flip) const {
        if (!p || l >= r) return T{};
        if (l == 0 && r == node_size(p)) return tr[p].sum + extra * tr[p].size;
        int lc = tr[p].ch[flip ? 1 : 0];
        int rc = tr[p].ch[flip ? 0 : 1];
        int left = node_size(lc);
        T next_extra = extra + tr[p].add;
        bool next_flip = flip ^ tr[p].rev;
        T ans{};
        if (l < left) {
            ans += range_sum_node(lc, l, min(r, left), next_extra, next_flip);
        }
        if (l <= left && left < r) ans += tr[p].value + extra;
        if (r > left + 1) {
            ans += range_sum_node(rc, max(0, l - left - 1),
                r - left - 1, next_extra, next_flip);
        }
        return ans;
    }
    T range_sum(int ver, int l, int r) const {
        assert(0 <= l && l <= r && r <= size(ver));
        return range_sum_node(root[ver], l, r, T{}, false);
    }
    T all_sum(int ver) const { return node_sum(root[ver]); }
};
```
#### 核心语义
“键”不是 `value`，而是**隐式下标**
```cpp
split(p, k)
```
始终表示：
```text
原序列 = A + B
|A| = k
```
即：
```text
A = 前 k 个元素
B = 剩余元素
```
所有公开区间统一使用 **0base 左闭右开 `[l, r)`**
#### 初始化与版本
- `PersistentImplicitTreap<T> tree`：版本 `0` 为空序列
- `PersistentImplicitTreap<T> tree(a)`：版本 `0` 为初始序列 `a`，建树 $O(n)$
- `root[ver]`：版本 `ver` 的根
- `size(ver)`：版本长度，$O(1)$
- `empty(ver)`：是否为空，$O(1)$
- `copy_version(ver)`：直接复制根，生成内容完全相同的新版本，$O(1)$
- `reserve(nodes, versions)`：预留节点 / 版本容量
#### 单点修改
```cpp
tree.insert(ver, pos, value);
```
在原下标 `pos` 元素之前插入：
```text
0 ... pos-1 | value | pos ...
```
其中 `0 <= pos <= size(ver)`
```cpp
tree.erase(ver, pos);
```
删除下标 `pos`
```cpp
tree.set(ver, pos, value);
```
修改一个元素
这三个操作都返回新版本编号
#### 区间修改
```cpp
tree.erase(ver, l, r);
tree.range_add(ver, l, r, delta);
tree.reverse(ver, l, r);
```
均作用于 `[l, r)`，并返回新版本编号
例如：
```text
[0, 1, 2, 3, 4]
reverse(ver, 1, 4)

=> [0, 3, 2, 1, 4]
```
#### 查询
```cpp
tree.get(ver, pos);
tree.range_sum(ver, l, r);
tree.all_sum(ver);
```
分别查询：
- 下标 `pos` 的值
- `[l, r)` 的区间和
- 整个版本的元素和
**这些查询不会 `split`，也不会 `push` 或复制节点**，因此不会消耗持久化空间
#### 批量插入
```cpp
tree.insert(ver, pos, values);
```
将整个 `values` 插入到 `pos` 前
其中新插入部分使用 $O(m)$ 的笛卡尔树建树，而不是逐元素执行 `insert()`
#### 注意事项
##### 1. `touch()` 是持久化正确性的核心
普通 FHQ Treap 可以直接：
```cpp
tr[p].ch[0] = ...
```
可持久化版本不行
一个节点可能仍被许多历史版本引用，因此任何写操作之前都必须保证它属于**本次修改**
```cpp
p = touch(p);
```
如果：
```cpp
tr[p].stamp != epoch
```
就复制一个节点；反之说明这个节点已经是当前操作创建的副本，可以继续原地修改
所以：
> `stamp / epoch` 并不是版本号本身，而是一次修改过程的 COW 标记
##### 2. 懒标记下传时，儿子也必须先 `touch`
这是最容易写错的地方：
```cpp
tr[p].ch[d] = touch(tr[p].ch[d]);
apply_add(tr[p].ch[d], add);
```
不能直接：
```cpp
apply_add(tr[p].ch[d], add); // 错
```
否则会修改旧版本共享的儿子节点
##### 3. 为什么查询不能直接 `push`
普通隐式 Treap 的查询经常写：
```cpp
push(p);
```
这里不能这么做，因为 `push()` 本身会修改节点和儿子
所以 `get()` / `range_sum()` 使用：
```cpp
extra
flip
```
把尚未真正下传的 `add / rev` 沿递归传下去
因此纯查询：
```cpp
tree.get(...)
tree.range_sum(...)
```
**零新增节点**
##### 4. `rev` 的实现已经立即交换儿子
```cpp
swap(tr[p].ch[0], tr[p].ch[1]);
tr[p].rev ^= 1;
```
所以查询时：
```cpp
flip
```
表示的是“来自祖先、尚未实际作用到当前节点的翻转”
不要再把 `tr[p].rev` 当作“当前节点的左右儿子还没交换”，否则下标方向会算反
##### 5. 持久化结构不要回收历史节点
即使执行：
```cpp
erase(ver, l, r)
```
被删除区间的旧节点仍可能被历史版本引用，不能放回普通 freelist 重用
##### 6. 类型要求
当前版本的 `T` 同时承担：
```text
元素值
区间和
区间加标记
```
因此至少需要支持：
```cpp
T{}
a + b
a += b
a * int
a == b
a != b
```
整数序列通常直接使用：
```cpp
PersistentImplicitTreap<i64>
```
若值和区间和可能需要不同类型，例如元素为 `int`、和必须用 `i128`，则建议再把 `value` 与 `sum` 拆成两个模板参数，而不是强行让这里的 `T` 承担两种含义
#### 常见用法
##### 从历史版本分叉
```cpp
PersistentImplicitTreap<i64> tree({1, 2, 3, 4});
int v1 = tree.insert(0, 2, 10);
// v1 = [1, 2, 10, 3, 4]
int v2 = tree.reverse(v1, 1, 4);
// v2 = [1, 3, 10, 2, 4]
int v3 = tree.range_add(0, 0, 2, 100);
// 从版本 0 分叉
// v3 = [101, 102, 3, 4]
```
历史版本完全不受影响：
```cpp
cout << tree.get(0, 0) << '\n';  // 1
cout << tree.get(v3, 0) << '\n'; // 101
```
##### 插入和删除
```cpp
int ver = 0;
ver = tree.insert(ver, 0, 5);
ver = tree.insert(ver, 1, 7);
ver = tree.insert(ver, 1, 6);
// [5, 6, 7]
ver = tree.erase(ver, 1);
// [5, 7]
```
##### 区间删除
```cpp
int ver = tree.erase(old_ver, l, r); // 删除 [l, r)
```
##### 区间加与区间和
```cpp
int ver = tree.range_add(old_ver, l, r, delta);
i64 ans = tree.range_sum(ver, l, r);
```
##### 区间翻转
```cpp
int ver = tree.reverse(old_ver, l, r);
```
##### 批量插入
```cpp
vector<i64> b{3, 1, 4, 1, 5};
int ver = tree.insert(old_ver, pos, b);
```
比连续执行：
```cpp
for (i64 x : b) {
    old_ver = tree.insert(old_ver, pos++, x);
}
```
更合适
##### 每次输入操作都要求产生版本
```cpp
int ver;
if (op == 1) {
    ver = tree.insert(old_ver, pos, x);
} else if (op == 2) {
    ver = tree.erase(old_ver, pos);
} else if (op == 3) {
    cout << tree.range_sum(old_ver, l, r) << '\n';
    ver = tree.copy_version(old_ver);
}
```
#### 功能裁剪
如果**不需要批量建树 / 批量插入**，可以删除：
```cpp
build_tree()
PersistentImplicitTreap(const vector<T> &a, ...)
insert(ver, pos, const vector<T> &values)
```
如果**不需要区间翻转**，可以删除：
```cpp
Node::rev
apply_rev()
reverse()
```
同时删除 `push()` 中 `rev` 部分；`get_node()` / `range_sum_node()` 中的 `flip` 逻辑也可同步简化为固定访问 `ch[0] / ch[1]`
如果**不需要区间加**，可以删除：
```cpp
Node::add
apply_add()
range_add()
```
同时删除 `push()` 中 `add` 部分，以及只读查询中的 `extra` 传递
如果**区间加和区间和都不要**，还能继续删除：
```cpp
Node::sum
node_sum()
range_sum_node()
range_sum()
all_sum()
```
并把：
```cpp
void pull(int p) {
    tr[p].size = 1 + node_size(tr[p].ch[0]) + node_size(tr[p].ch[1]);
}
```
作为唯一维护信息
最终就会退化成一个相当短的：
> **可持久化动态数组 / Rope 式序列**

只保留：
```text
split / merge
insert / erase / set / get
版本分叉
```
这部分裁剪关系正是这类中大型板子最值得明确记录的内容
#### 复杂度
设当前序列长度为 $n$，批量插入长度为 $m$

| 操作                                                                      |             期望时间 |             新增节点 |
| ----------------------------------------------------------------------- | ---------------: | ---------------: |
| 初始 `build`                                                              |           $O(n)$ |           $O(n)$ |
| `copy_version` / `all_sum`                                              |           $O(1)$ |              $0$ |
| 单点 `insert` / 单点 `erase` / `set` / 区间 `erase` / `range_add` / `reverse` |      $O(\log n)$ |      $O(\log n)$ |
| `get` / `range_sum`                                                     |      $O(\log n)$ |              $0$ |
| 批量插入 $m$ 个元素                                                            | $O(m+\log(n+m))$ | $O(m+\log(n+m))$ |
Treap 随机平衡，因此这些均为期望复杂度；理论最坏树高仍为 $O(n)$
