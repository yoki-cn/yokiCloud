### FHQ Treap 有序多重集
---
```cpp
template<class T, class Compare = less<T>>
struct FHQTreap {
    struct Node {
        int ch[2]{}, size{}, cnt{};
        unsigned pri{};
        T value{};
    };
    Compare comp;
    mt19937 rng;
    vector<Node> tr{Node{}};
    vector<int> bin;
    int root{};
    explicit FHQTreap(
        unsigned seed = unsigned(chrono::steady_clock::now().time_since_epoch().count()),
        Compare comp_ = Compare()
    ) : comp(move(comp_)), rng(seed) {}
    explicit FHQTreap(
        vector<T> a,
        unsigned seed = unsigned(chrono::steady_clock::now().time_since_epoch().count()),
        Compare comp_ = Compare()
    ) : comp(move(comp_)), rng(seed) {
        tr.reserve(a.size() + 1);
        build(move(a));
    }
private:
    int node_size(int p) const { return tr[p].size; }
    bool equal(const T &a, const T &b) const { return !comp(a, b) && !comp(b, a); }
    // ===== 节点创建与维护：所有修改操作依赖，不建议删除 =====
    int new_node(const T &value, int cnt = 1) {
        int p;
        if (bin.empty()) {
            p = tr.size();
            tr.push_back({});
        } else {
            p = bin.back();
            bin.pop_back();
        }
        tr[p] = Node{{}, cnt, cnt, unsigned(rng()), value};
        return p;
    }
    void recycle(int p) {
        tr[p] = Node{};
        bin.push_back(p);
    }
    void pull(int p) {
        tr[p].size = node_size(tr[p].ch[0]) + tr[p].cnt + node_size(tr[p].ch[1]);
    }
    // ===== 批量建树：若只从空树逐个 insert，可删除该函数与 vector 构造函数 =====
    void build(vector<T> a) {
        if (a.empty()) return;
        ranges::sort(a, comp);
        vector<int> st;
        st.reserve(a.size());
        for (int l = 0, r; l < a.size(); l = r) {
            for (r = l + 1; r < a.size() && equal(a[l], a[r]); ++ r) {}
            int x = new_node(a[l], r - l), last = 0;
            while (!st.empty() && tr[st.back()].pri > tr[x].pri) {
                last = st.back();
                st.pop_back();
            }
            tr[x].ch[0] = last;
            if (!st.empty()) tr[st.back()].ch[1] = x;
            st.push_back(x);
        }
        root = st.front();
        vector<int> order{root};
        order.reserve(tr.size() - 1);
        for (int i = 0; i < order.size(); ++ i) {
            int p = order[i];
            if (tr[p].ch[0]) order.push_back(tr[p].ch[0]);
            if (tr[p].ch[1]) order.push_back(tr[p].ch[1]);
        }
        for (int p : order | views::reverse) pull(p);
    }
    // ===== FHQ 核心：insert / erase / unite 依赖，不建议删除 =====
    // < value | >= value
    pair<int, int> split_less(int p, const T &value) {
        if (!p) return {0, 0};
        if (comp(tr[p].value, value)) {
            auto [a, b] = split_less(tr[p].ch[1], value);
            tr[p].ch[1] = a;
            pull(p);
            return {p, b};
        }
        auto [a, b] = split_less(tr[p].ch[0], value);
        tr[p].ch[0] = b;
        pull(p);
        return {a, p};
    }
    // <= value | > value
    pair<int, int> split_leq(int p, const T &value) {
        if (!p) return {0, 0};
        if (!comp(value, tr[p].value)) {
            auto [a, b] = split_leq(tr[p].ch[1], value);
            tr[p].ch[1] = a;
            pull(p);
            return {p, b};
        }
        auto [a, b] = split_leq(tr[p].ch[0], value);
        tr[p].ch[0] = b;
        pull(p);
        return {a, p};
    }
    // 要求 a 中所有键严格小于 b 中所有键
    int merge(int a, int b) {
        if (!a || !b) return a | b;
        if (tr[a].pri < tr[b].pri) {
            tr[a].ch[1] = merge(tr[a].ch[1], b);
            pull(a);
            return a;
        }
        tr[b].ch[0] = merge(a, tr[b].ch[0]);
        pull(b);
        return b;
    }
    bool erase_node(int &p, const T &value) {
        if (!p) return false;
        if (comp(value, tr[p].value)) {
            bool ok = erase_node(tr[p].ch[0], value);
            if (ok) pull(p);
            return ok;
        }
        if (comp(tr[p].value, value)) {
            bool ok = erase_node(tr[p].ch[1], value);
            if (ok) pull(p);
            return ok;
        }
        if (tr[p].cnt > 1) {
            -- tr[p].cnt;
            pull(p);
        } else {
            int q = p;
            p = merge(tr[p].ch[0], tr[p].ch[1]);
            recycle(q);
        }
        return true;
    }
public:
    // ===== 基础接口：默认操作 root；传入 int &rt 可在同一节点池维护多棵树 =====
    int size(int p) const { return node_size(p); }
    int size() const { return size(root); }
    bool empty(int p) const { return p == 0; }
    bool empty() const { return root == 0; }
    void reserve(int nodes) {
        tr.reserve(nodes + 1);
        bin.reserve(nodes);
    }
    void clear() {
        tr.resize(1);
        tr[0] = Node{};
        bin.clear();
        root = 0;
    }
    void insert(int &p, const T &value) {
        auto [a, c] = split_less(p, value);
        auto [b, d] = split_leq(c, value);
        if (b) {
            ++ tr[b].cnt;
            pull(b);
        } else {
            b = new_node(value);
        }
        p = merge(merge(a, b), d);
    }
    void insert(const T &value) { insert(root, value); }
    bool erase(int &p, const T &value) { return erase_node(p, value); }
    bool erase(const T &value) { return erase(root, value); }
    // ===== 只读查询：不 split，不改变树形 =====
    int count(int p, const T &value) const {
        while (p) {
            if (comp(value, tr[p].value)) p = tr[p].ch[0];
            else if (comp(tr[p].value, value)) p = tr[p].ch[1];
            else return tr[p].cnt;
        }
        return 0;
    }
    int count(const T &value) const { return count(root, value); }
    bool contains(int p, const T &value) const { return count(p, value) != 0; }
    bool contains(const T &value) const { return contains(root, value); }
    // 严格小于 value 的元素个数
    int order_of_key(int p, const T &value) const {
        int ans = 0;
        while (p) {
            if (comp(tr[p].value, value)) {
                ans += node_size(tr[p].ch[0]) + tr[p].cnt;
                p = tr[p].ch[1];
            } else {
                p = tr[p].ch[0];
            }
        }
        return ans;
    }
    int order_of_key(const T &value) const { return order_of_key(root, value); }
    // k 为 0base
    optional<T> kth(int p, int k) const {
        if (k < 0 || k >= node_size(p)) return nullopt;
        while (p) {
            int left = node_size(tr[p].ch[0]);
            if (k < left) {
                p = tr[p].ch[0];
            } else if (k < left + tr[p].cnt) {
                return tr[p].value;
            } else {
                k -= left + tr[p].cnt;
                p = tr[p].ch[1];
            }
        }
        return nullopt;
    }
    optional<T> kth(int k) const { return kth(root, k); }
    // 严格前驱：最大的 < value
    optional<T> predecessor(int p, const T &value) const {
        int ans = 0;
        while (p) {
            if (comp(tr[p].value, value)) {
                ans = p;
                p = tr[p].ch[1];
            } else {
                p = tr[p].ch[0];
            }
        }
        if (!ans) return nullopt;
        return tr[ans].value;
    }
    optional<T> predecessor(const T &value) const { return predecessor(root, value); }
    // 严格后继：最小的 > value
    optional<T> successor(int p, const T &value) const {
        int ans = 0;
        while (p) {
            if (comp(value, tr[p].value)) {
                ans = p;
                p = tr[p].ch[0];
            } else {
                p = tr[p].ch[1];
            }
        }
        if (!ans) return nullopt;
        return tr[ans].value;
    }
    optional<T> successor(const T &value) const { return successor(root, value); }
    // ===== Treap Union：若不需要任意两个有序多重集合并，可直接删除 =====
    // a、b 必须属于同一节点池且互不共享节点；调用后旧 a、b 不再独立使用
    int unite(int a, int b) {
        if (!a || !b) return a | b;
        if (tr[a].pri > tr[b].pri) swap(a, b);
        auto [l, ge] = split_less(b, tr[a].value);
        auto [eq, r] = split_leq(ge, tr[a].value);
        if (eq) {
            tr[a].cnt += tr[eq].cnt;
            recycle(eq);
        }
        tr[a].ch[0] = unite(tr[a].ch[0], l);
        tr[a].ch[1] = unite(tr[a].ch[1], r);
        pull(a);
        return a;
    }
};
```
#### 接口说明
##### 默认单棵树
* `FHQTreap<T> tr`：构造空有序多重集
* `FHQTreap<T> tr(a)`：由初始序列构造；先排序并压缩相同键
* `insert(value)`：插入一个值，期望 $O(\log d)$
* `erase(value)`：删除一个值；不存在返回 `false`，期望 $O(\log d)$
* `count(value)`：该值出现次数，期望 $O(\log d)$
* `contains(value)`：是否存在，期望 $O(\log d)$
* `order_of_key(value)`：严格小于 `value` 的元素个数，期望 $O(\log d)$
* `kth(k)`：`0base` 第 `k` 小；越界返回 `nullopt`，期望 $O(\log d)$
* `predecessor(value)`：严格前驱，期望 $O(\log d)$
* `successor(value)`：严格后继，期望 $O(\log d)$
* `size()` / `empty()`：元素总数 / 是否为空
* `reserve(nodes)`：预留节点池空间
* `clear()`：清空当前对象并保留 `vector` 容量
这里 $d$ 表示**不同键的数量**，而不是元素总数
##### 重复值
与旧版相比，这版不再让每个重复值占一个 Treap 节点，而是：
```cpp
Node::cnt
```
记录该键的出现次数
因此：
```text
{5, 5, 5, 5}
```
只占一个节点：
```text
value = 5
cnt   = 4
size  = 4
```
这在重复值很多时明显更省空间，也让 `Treap Union` 合并重复键更加自然
##### 同一节点池维护多棵树
所有常用接口都有接受根编号的版本：
```cpp
int rt = 0;
tr.insert(rt, x);
tr.erase(rt, x);
tr.order_of_key(rt, x);
tr.kth(rt, k);
tr.predecessor(rt, x);
tr.successor(rt, x);
```
因此一个 `FHQTreap` 对象可以作为**共享节点池**：
```cpp
FHQTreap<int> tr;
vector<int> root(n);
```
每个 `root[i]` 都可以表示一棵独立 Treap
这正是 `unite()` 最实用的使用方式
#### Treap Union / `unite`
##### 接口
```cpp
int c = tr.unite(a, b);
```
与普通：
```cpp
merge(a, b)
```
完全不同
`merge(a,b)` 要求：
```text
a 中所有键 < b 中所有键
```
而：
```cpp
unite(a, b)
```
**没有键域顺序限制**
例如：
```text
A = {1, 3, 3, 8}
B = {2, 3, 7}

unite(A, B)

=> {1, 2, 3, 3, 3, 7, 8}
```
相同键直接累加 `cnt`
##### 使用方式
```cpp
FHQTreap<int> tr;
vector<int> rt(n);
for (int i = 0; i < n; ++ i) {
    tr.insert(rt[i], a[i]);
}
rt[u] = tr.unite(rt[u], rt[v]);
rt[v] = 0;
```
调用后：
```text
rt[u]
```
拥有两棵树中的全部元素
原来的：
```text
rt[u] / rt[v]
```
不能再作为两棵独立树同时继续使用
尤其应当写：
```cpp
rt[u] = tr.unite(rt[u], rt[v]);
rt[v] = 0;
```
而不要保留 `rt[v]`
##### 适用场景
比较典型的是：
```text
树上子树信息合并
DSU / 连通块合并
每个节点维护一个有序多重集
分治过程中合并若干集合
```
例如：
```cpp
for (int v : g[u]) {
    dfs(v, u);
    rt[u] = tr.unite(rt[u], rt[v]);
    rt[v] = 0;
}
```
合并后可以直接查询：
```cpp
auto x = tr.kth(rt[u], k);
int cnt = tr.order_of_key(rt[u], value);
```
#### 注意事项
* 当前版本使用**每个不同键一个节点 + `cnt` 记录重数**。`size` 统计元素总数，而节点数统计不同键数量
* `Compare` 的等价关系是：
```cpp
!comp(a, b) && !comp(b, a)
```
两个在 `Compare` 意义下等价的对象会被压缩到同一个节点
* `kth()` 使用 `0base`
* 常用题目中的 `1base` 排名为：
```cpp
tr.order_of_key(x) + 1
```
* `predecessor()` / `successor()` 均为**严格**前驱 / 后继
* 查询均为纯遍历，不会调用 `split()`，所以查询不会改变 Treap 树形
* `split_size()` 已经删除。权值 Treap 不需要按中序前 `k` 个节点破坏性分裂；而且当前节点带有 `cnt`，按元素数量分裂可能需要把一个节点的 `cnt` 拆成两份。按排名分裂更适合放在隐式 Treap 中
* `merge(a,b)` 只用于键域严格有序的两棵树；任意集合合并必须使用 `unite(a,b)`
* `unite(a,b)` 要求两棵树：
  * 来自**同一个 `FHQTreap` 对象**
  * 节点集合互不共享
  * 调用后视为消费掉旧的 `a,b`
* `unite()` 不是持久化操作，会直接修改两棵输入树
* 节点被真正删除或 `unite()` 中出现重复键时会进入 `bin`，后续插入复用
* 随机优先级碰撞不影响正确性；树高期望为 $O(\log d)$，理论最坏仍可能达到 $O(d)$
#### 常见用法
##### 六种经典平衡树操作
```cpp
FHQTreap<int> tr;
tr.insert(x);
tr.erase(x);
int rank = tr.order_of_key(x) + 1;
if (auto v = tr.kth(k - 1)) {
    cout << *v << '\n';
}
if (auto v = tr.predecessor(x)) {
    cout << *v << '\n';
}
if (auto v = tr.successor(x)) {
    cout << *v << '\n';
}
```
##### 查询重复次数
```cpp
cout << tr.count(x) << '\n';
if (tr.contains(x)) {
    // x 存在
}
```
##### 初始序列批量建树
如果一开始已经有全部元素：
```cpp
vector<int> a(n);
for (int &x : a) cin >> x;
FHQTreap<int> tr(move(a));
```
内部会：
```text
ranges::sort
→ 合并重复值
→ O(d) 构造随机笛卡尔树
```
整体瓶颈为排序
如果只从空树在线插入，可以把整个 `build()` 和对应构造函数删掉
##### 自定义键
```cpp
struct Key {
    i64 value{};
    int id{};
};
struct KeyLess {
    bool operator()(const Key &a, const Key &b) const {
        return tie(a.value, a.id) < tie(b.value, b.id);
    }
};
FHQTreap<Key, KeyLess> tr;
tr.insert({value, id});
int rank = tr.order_of_key({value, INT_MIN}) + 1;
if (auto x = tr.kth(k - 1)) {
    cout << x->value << ' ' << x->id << '\n';
}
```
由于 `id` 参与比较：
```text
(value, id1)
(value, id2)
```
是两个不同键，因此不会合并 `cnt`
##### 树上合并
```cpp
FHQTreap<int> tr;
vector<int> rt(n);
for (int u = 0; u < n; ++ u) {
    tr.insert(rt[u], a[u]);
}
auto dfs = Y([&](auto self, int u, int fa) -> void {
    for (int v : g[u]) if (v != fa) {
        self(v, u);
        rt[u] = tr.unite(rt[u], rt[v]);
        rt[v] = 0;
    }
});
dfs(0, -1);
```
例如查询整个子树的第 `k` 小：
```cpp
auto ans = tr.kth(rt[u], k);
```
#### 复杂度
设：
* $n$：元素总数
* $d$：不同键数量
* `unite()` 的两棵树含有 $n_d,m_d$ 个不同键节点，且 $n_d\ge m_d$

| 操作                        |                                                      期望时间 |
| ------------------------- | --------------------------------------------------------: |
| `insert`                  |                                               $O(\log d)$ |
| `erase`                   |                                               $O(\log d)$ |
| `count / contains`        |                                               $O(\log d)$ |
| `order_of_key`            |                                               $O(\log d)$ |
| `kth`                     |                                               $O(\log d)$ |
| `predecessor / successor` |                                               $O(\log d)$ |
| 初始批量构造                    |                                              $O(n\log n)$ |
| `unite`                   | $O!\left(m_d\log\left(\frac{n_d}{m_d}+1\right)\right)$ 期望 |
空间复杂度：
$$O(d)$$
节点回收后，长期在线插删时实际节点池规模接近“历史上同时存在的最大不同键数”，而不是总操作次数
