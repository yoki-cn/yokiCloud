### Splay 序列与区间暴露
---
```cpp
template<class T>
struct SplaySeq {
    struct Node {
        int ch[2]{}, fa{}, size{};
        bool rev{};
        T value{};
    };
    vector<Node> tr{Node{}};
    vector<int> bin, stk;
    int root{};
    int guard[2]{};
    SplaySeq() { build({}); }
    explicit SplaySeq(const vector<T> &a) { build(a); }
private:
    int node_size(int p) const { return tr[p].size; }
    // ===== 节点维护：所有操作依赖，不建议删除 =====
    void pull(int p) {
        tr[p].size = node_size(tr[p].ch[0]) + 1 + node_size(tr[p].ch[1]);
    }
    int new_node(const T &value = T{}) {
        int p;
        if (bin.empty()) {
            p = tr.size();
            tr.push_back({});
        } else {
            p = bin.back();
            bin.pop_back();
            tr[p] = {};
        }
        tr[p].size = 1;
        tr[p].value = value;
        return p;
    }
    // 删除大区间时避免递归回收造成栈溢出
    void recycle(int p) {
        if (!p) return;
        stk.clear();
        stk.push_back(p);
        for (int i = 0; i < stk.size(); ++ i) {
            int x = stk[i];
            for (int c : tr[x].ch) if (c) stk.push_back(c);
        }
        for (int x : stk | views::reverse) {
            tr[x] = {};
            bin.push_back(x);
        }
        stk.clear();
    }
    // ===== 区间翻转：不需要 reverse() 时可删除 rev / apply_rev()
    // 并把 push() 改为空函数 =====
    void apply_rev(int p) {
        if (!p) return;
        swap(tr[p].ch[0], tr[p].ch[1]);
        tr[p].rev ^= 1;
    }
    void push(int p) {
        if (!p || !tr[p].rev) return;
        apply_rev(tr[p].ch[0]);
        apply_rev(tr[p].ch[1]);
        tr[p].rev = false;
    }
    // ===== O(n) 平衡建树：构造和批量插入依赖 =====
    int build_rec(const vector<int> &id, int l, int r, int fa) {
        if (l >= r) return 0;
        int m = (l + r) / 2, p = id[m];
        tr[p].fa = fa;
        tr[p].ch[0] = build_rec(id, l, m, p);
        tr[p].ch[1] = build_rec(id, m + 1, r, p);
        pull(p);
        return p;
    }
    int build_values(const vector<T> &a) {
        vector<int> id;
        id.reserve(a.size());
        for (const T &value : a) id.push_back(new_node(value));
        return build_rec(id, 0, id.size(), 0);
    }
    // ===== Splay 核心：rotate / splay，不建议删除 =====
    void push_path(int p, int goal = 0) {
        stk.clear();
        for (int x = p; x != goal; x = tr[x].fa) stk.push_back(x);
        for (int x : stk | views::reverse) push(x);
    }
    void rotate(int p) {
        int fa = tr[p].fa, g = tr[fa].fa;
        int d = tr[fa].ch[1] == p;
        int x = tr[p].ch[d ^ 1];
        if (g) tr[g].ch[tr[g].ch[1] == fa] = p;
        else root = p;
        tr[p].fa = g;
        tr[p].ch[d ^ 1] = fa;
        tr[fa].fa = p;
        tr[fa].ch[d] = x;
        if (x) tr[x].fa = fa;
        pull(fa);
        pull(p);
    }
    void splay(int p, int goal = 0) {
        push_path(p, goal);
        while (tr[p].fa != goal) {
            int fa = tr[p].fa, g = tr[fa].fa;
            if (g != goal) {
                if ((tr[fa].ch[1] == p) == (tr[g].ch[1] == fa)) rotate(fa);
                else rotate(p);
            }
            rotate(p);
        }
        if (!goal) root = p;
    }
    // ===== 按排名定位：expose / 单点接口依赖 =====
    int kth_node(int k) {
        assert(0 <= k && k < node_size(root));
        int p = root;
        while (true) {
            push(p);
            int left = node_size(tr[p].ch[0]);
            if (k < left)  p = tr[p].ch[0];
            else if (k == left) return p;
            else {
                k -= left + 1;
                p = tr[p].ch[1];
            }
        }
    }
    // ===== 区间暴露：将真实序列 [l,r) 暴露为右边界的左子树 =====
    int expose(int l, int r) {
        assert(0 <= l && l <= r && r <= size());
        int left = kth_node(l);
        splay(left);
        int right = kth_node(r + 1);
        splay(right, left);
        return tr[right].ch[0];
    }
    // ===== 区间拆接：erase / move / insert 依赖 =====
    int detach(int l, int r) {
        int mid = expose(l, r);
        int right = tr[root].ch[1];
        tr[right].ch[0] = 0;
        if (mid) tr[mid].fa = 0;
        pull(right);
        pull(root);
        return mid;
    }
    void attach(int pos, int p) {
        if (!p) return;
        assert(0 <= pos && pos <= size());
        int left = kth_node(pos);
        splay(left);
        int right = kth_node(pos + 1);
        splay(right, left);
        tr[right].ch[0] = p;
        tr[p].fa = right;
        pull(right);
        pull(left);
    }
public:
    // ===== 初始化与状态 =====
    void build(const vector<T> &a) {
        tr.clear();
        tr.push_back({});
        bin.clear();
        stk.clear();
        root = 0;
        vector<int> id;
        id.reserve(a.size() + 2);
        guard[0] = new_node();
        id.push_back(guard[0]);
        for (const T &value : a) id.push_back(new_node(value));
        guard[1] = new_node();
        id.push_back(guard[1]);
        root = build_rec(id, 0, id.size(), 0);
    }
    void clear() { build({}); }
    void reserve(int nodes) {
        tr.reserve(nodes + 3);
        bin.reserve(nodes);
        stk.reserve(nodes + 2);
    }
    int size() const { return node_size(root) - 2; }
    bool empty() const { return size() == 0; }
    // ===== 基础区间操作 =====
    void reverse(int l, int r) {
        assert(0 <= l && l <= r && r <= size());
        if (l < r) apply_rev(expose(l, r));
    }
    // ===== 插入：若只需要文艺平衡树，可整段删除 =====
    void insert(int pos, const T &value) {
        assert(0 <= pos && pos <= size());
        attach(pos, new_node(value));
    }
    void insert(int pos, const vector<T> &a) {
        assert(0 <= pos && pos <= size());
        if (!a.empty()) attach(pos, build_values(a));
    }
    // ===== 删除：若不需要删除，可删除 recycle / bin 以及本段 =====
    T erase(int pos) {
        assert(0 <= pos && pos < size());
        int p = detach(pos, pos + 1);
        T value = std::move(tr[p].value);
        recycle(p);
        return value;
    }
    void erase(int l, int r) {
        assert(0 <= l && l <= r && r <= size());
        if (l < r) recycle(detach(l, r));
    }
    // ===== 区间搬移：不需要时可直接删除 =====
    // pos 是移动前原序列的位置，表示把 [l,r) 移动到 pos 前
    void move(int l, int r, int pos) {
        assert(0 <= l && l <= r && r <= size());
        assert(0 <= pos && pos <= size());
        if (l >= r || (l <= pos && pos <= r)) return;
        int mid = detach(l, r);
        if (pos > r) pos -= r - l;
        attach(pos, mid);
    }
    // ===== 单点操作：不需要时可直接删除 =====
    T get(int pos) {
        assert(0 <= pos && pos < size());
        int p = kth_node(pos + 1);
        splay(p);
        return tr[p].value;
    }
    void set(int pos, const T &value) {
        assert(0 <= pos && pos < size());
        int p = kth_node(pos + 1);
        splay(p);
        tr[p].value = value;
    }
    // ===== 导出序列：输出 / 调试需要时保留 =====
    vector<T> to_vector() {
        vector<T> a;
        a.reserve(size());
        vector<int> st;
        int p = root;
        while (p || !st.empty()) {
            while (p) {
                push(p);
                st.push_back(p);
                p = tr[p].ch[0];
            }
            p = st.back();
            st.pop_back();
            if (p != guard[0] && p != guard[1]) a.push_back(tr[p].value);
            p = tr[p].ch[1];
        }
        return a;
    }
};
```
#### 接口说明
##### 两个哨兵
内部中序序列实际上是：
```text
左哨兵 + 真实序列 + 右哨兵
```
所以真实长度直接是：
```cpp
node_size(root) - 2
```
不再额外维护一个 `n`，这样 `detach / attach` 时也不会出现“树大小和 `n` 忘记同步”的问题
真实下标 `pos` 对应内部排名：
```text
pos + 1
```
##### 区间暴露 `expose(l,r)`
对于真实区间：
```text
[l,r)
```
先找到：
```cpp
left  = kth_node(l);
right = kth_node(r + 1);
```
然后：
```cpp
splay(left);
splay(right, left);
```
此时结构固定为：
```text
          left
             \
            right
            /
        [l,r)
```
因此：
```cpp
tr[right].ch[0]
```
恰好就是整个目标区间
所以以后任何区间操作，本质都是：
```cpp
int p = expose(l, r);
// 对 p 整棵子树操作
```
#### 注意事项
- 外部统一使用 `0base` 和 `[l,r)`
- 两个哨兵永远不属于真实序列，也不会被公开接口删除或搬移
- `kth_node()` 查询的是**内部中序排名**，所以真实元素 `pos` 对应 `pos + 1`
- 在存在 `rev` 时，下降到儿子前必须 `push()`；`splay()` 旋转前必须把从祖先到目标节点的整条路径下传
- `reverse(l,r)` 只修改目标子树的翻转标记，不需要重新 `pull()`，因为翻转不会改变子树大小
- `move(l,r,pos)` 中 `pos` 是**移动前原序列的位置**。若 `pos` 位于 `[l,r]`，移动后序列不变
- `erase(l,r)` 为了回收节点，需要遍历被删除的全部元素，因此为 $O(\log n+r-l)$；如果不要求回收节点，可以直接丢弃子树，把删除降到均摊 $O(\log n)$
- Splay 单次操作可能是 $O(n)$，一串操作均摊为 $O(\log n)$
- 这是**序列 Splay**，不按 `value` 排序；排名、第 $k$ 小、前驱、后继请使用前面的 FHQ Treap 有序多重集
- 与隐式 FHQ Treap 相比，Splay 最有特色的是 `splay(x, goal)` 和稳定的区间 `expose`，也是理解 LCT 的重要前置结构
#### 常见用法
##### 文艺平衡树
```cpp
vector<int> a(n);
iota(a.begin(), a.end(), 1);
SplaySeq<int> sp(a);
while (m --) {
    int l, r;
    cin >> l >> r;
    sp.reverse(l - 1, r);
}
for (int x : sp.to_vector()) {
    cout << x << ' ';
}
```
##### 动态插入和删除
```cpp
SplaySeq<int> sp(a);
sp.insert(pos, x);
sp.insert(pos, vector<int>{1, 2, 3});
int old = sp.erase(pos);
sp.erase(l, r);
```
##### 区间搬移
```cpp
sp.move(l, r, pos);
```
表示：
> 把原序列 `[l,r)` 剪下来，插到原位置 `pos` 之前

例如：
```text
0 1 2 3 4 5
move(1, 3, 5)
0 3 4 1 2 5
```
##### 交换两个相邻区间
若：
```text
A [l,m) [m,r) D
```
想变成：
```text
A [m,r) [l,m) D
```
直接：
```cpp
sp.move(l, m, r);
```
##### 区间循环右移
把 `[l,r)` 循环右移 `k` 位：
```cpp
int len = r - l;
if (len) {
    k %= len;
    if (k) sp.move(r - k, r, l);
}
```
循环左移同理：
```cpp
if (len) {
    k %= len;
    if (k) sp.move(l, l + k, r);
}
```
##### 单点访问
```cpp
T x = sp.get(pos);
sp.set(pos, value);
```
#### 常见扩展：加入区间聚合
例如需要维护**区间和**，只需要增加几处
`Node` 增加：
```cpp
T sum{};
```
`new_node()` 增加：
```cpp
tr[p].sum = value;
```
`pull()` 改成：
```cpp
void pull(int p) {
    tr[p].size = node_size(tr[p].ch[0]) + 1 + node_size(tr[p].ch[1]);
    tr[p].sum =
        (tr[p].ch[0] ? tr[tr[p].ch[0]].sum : T{}) +
        tr[p].value +
        (tr[p].ch[1] ? tr[tr[p].ch[1]].sum : T{});
}
```
然后区间和就是：
```cpp
T range_sum(int l, int r) {
    if (l == r) return T{};
    return tr[expose(l, r)].sum;
}
```
`set()` 中修改完：
```cpp
tr[p].value = value;
pull(p);
```
即可
同样的方式可以维护：
```text
sum
min / max
xor
最大子段和
区间哈希
```
只要信息能由左右子树和当前节点在 `pull()` 中合并即可
##### 加入区间加
若还需要：
```text
range_add(l,r,x)
range_sum(l,r)
```
再增加：
```cpp
T add{};
```
以及：
```cpp
void apply_add(int p, const T &x) {
    if (!p) return;
    tr[p].value += x;
    tr[p].sum += x * tr[p].size;
    tr[p].add += x;
}
```
并在 `push()` 中下传
修改整段以后，因为区间聚合值发生了变化，还应更新两个边界：
```cpp
void range_add(int l, int r, const T &x) {
    if (l == r) return;
    int p = expose(l, r);
    apply_add(p, x);
    int right = tr[root].ch[1];
    pull(right);
    pull(root);
}
```
这就是 Splay 的一个优势：
```text
expose 区间
↓
整棵目标子树打标记
↓
恢复两个边界信息
```
结构非常固定
#### 功能裁剪
如果题目只是经典文艺平衡树，只需要保留：
```text
Node::size / rev
new_node / pull
apply_rev / push
build
push_path / rotate / splay
kth_node / expose
reverse
to_vector
```
可以整个删除：
```text
bin / recycle
build_values
detach / attach
insert
erase
move
get / set
```
如果连区间翻转都不需要，则再删除：
```text
Node::rev
apply_rev
reverse
```
并让：
```cpp
void push(int) {}
```
或者进一步删掉所有 `push()`
#### 复杂度

| 操作                                                                                    |           时间复杂度 |
| ------------------------------------------------------------------------------------- | --------------: |
| `build`                                                                               |          $O(n)$ |
| `kth_node` / `expose` / `reverse` / 单点 `insert` / `erase(pos)` / `move` / `get / set` |  均摊 $O(\log n)$ |
| 批量插入 $m$ 个元素                                                                          |   $O(m+\log n)$ |
| `erase(l,r)`                                                                          | $O(\log n+r-l)$ |
| `to_vector`                                                                           |          $O(n)$ |
空间复杂度为 $O(n)$；启用节点回收后，被删除节点可以继续复用
