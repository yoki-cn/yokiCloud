### 可持久化 FHQ Treap 有序多重集
---
**解决问题：** 基于任意历史版本插入 / 删除一个键，并查询各版本的排名、第 $k$ 小、前驱、后继等有序多重集信息
相同值使用多个独立节点保存；每次修改产生一个新版本，未修改的子树与旧版本共享
```cpp
template<class T, class Compare = less<T>>
struct PersistentFHQTreap {
    struct Node {
        int ch[2]{}, size{};
        unsigned pri{};
        T value{};
    };
    vector<Node> tr{{}};
    vector<int> root{ 0 };
    u64 rng_state;
    Compare comp;
    explicit PersistentFHQTreap(
        u64 seed = chrono::steady_clock::now().time_since_epoch().count(),
        Compare comp_ = Compare()
    ) : rng_state(seed), comp(move(comp_)) {}
    void reserve(int nodes, int versions = 0) {
        tr.reserve(nodes + 1);
        if (versions) root.reserve(versions + 1);
    }
    // ===== 节点与随机优先级：所有修改操作依赖，不建议删除 =====
    int node_size(int p) const { return tr[p].size; }
    unsigned rng() {
        u64 z = (rng_state += 0x9e3779b97f4a7c15ULL);
        z = (z ^ (z >> 30)) * 0xbf58476d1ce4e5b9ULL;
        z = (z ^ (z >> 27)) * 0x94d049bb133111ebULL;
        return unsigned(z ^ (z >> 31));
    }
    int new_node(const T &value) {
        tr.push_back({{}, 1, rng(), value});
        return int(tr.size()) - 1;
    }
    int clone(int p) {
        tr.push_back(tr[p]);
        return int(tr.size()) - 1;
    }
    void pull(int p) {
        tr[p].size = 1 + node_size(tr[p].ch[0]) + node_size(tr[p].ch[1]);
    }
    // ===== 持久化 split / merge：插入和删除依赖 =====
    // < value | >= value
    pair<int, int> split_less(int p, const T &value) {
        if (!p) return {0, 0};
        int q = clone(p);
        if (comp(tr[q].value, value)) {
            auto [a, b] = split_less(tr[q].ch[1], value);
            tr[q].ch[1] = a;
            pull(q);
            return {q, b};
        }
        auto [a, b] = split_less(tr[q].ch[0], value);
        tr[q].ch[0] = b;
        pull(q);
        return {a, q};
    }
    // 要求 a 中所有元素排在 b 中所有元素之前
    int merge(int a, int b) {
        if (!a || !b) return a | b;
        if (tr[a].pri < tr[b].pri) {
            int p = clone(a);
            tr[p].ch[1] = merge(tr[p].ch[1], b);
            pull(p);
            return p;
        }
        int p = clone(b);
        tr[p].ch[0] = merge(a, tr[p].ch[0]);
        pull(p);
        return p;
    }
    // ===== 修改核心：沿修改路径复制 =====
    int insert_node(int p, int x) {
        if (!p) return x;
        if (tr[x].pri < tr[p].pri) {
            auto [a, b] = split_less(p, tr[x].value);
            tr[x].ch[0] = a;
            tr[x].ch[1] = b;
            pull(x);
            return x;
        }
        int q = clone(p);
        if (comp(tr[x].value, tr[q].value)) tr[q].ch[0] = insert_node(tr[q].ch[0], x);
        else tr[q].ch[1] = insert_node(tr[q].ch[1], x);
        pull(q);
        return q;
    }
    int erase_node(int p, const T &value, bool &erased) {
        if (!p) return 0;
        if (comp(value, tr[p].value)) {
            int c = erase_node(tr[p].ch[0], value, erased);
            if (!erased) return p;
            int q = clone(p);
            tr[q].ch[0] = c;
            pull(q);
            return q;
        }
        if (comp(tr[p].value, value)) {
            int c = erase_node(tr[p].ch[1], value, erased);
            if (!erased) return p;
            int q = clone(p);
            tr[q].ch[1] = c;
            pull(q);
            return q;
        }
        erased = true;
        return merge(tr[p].ch[0], tr[p].ch[1]);
    }
    // ===== 版本修改：insert / erase 均追加一个新版本 =====
    int insert(int ver, const T &value) {
        int x = new_node(value);
        root.push_back(insert_node(root[ver], x));
        return int(root.size()) - 1;
    }
    int erase(int ver, const T &value) {
        bool erased = false;
        int r = erase_node(root[ver], value, erased);
        root.push_back(r);
        return int(root.size()) - 1;
    }
    // 若题目要求“每次操作都产生版本”，查询操作后可复制原版本
    int copy_version(int ver) {
        root.push_back(root[ver]);
        return int(root.size()) - 1;
    }
    // ===== 基础查询：均不产生新节点 =====
    int size(int ver) const { return node_size(root[ver]); }
    bool empty(int ver) const { return root[ver] == 0; }
    // Compare 意义下严格排在 value 前面的元素个数
    int order_of_key(int ver, const T &value) const {
        int p = root[ver], ans = 0;
        while (p) {
            if (comp(tr[p].value, value)) {
                ans += node_size(tr[p].ch[0]) + 1;
                p = tr[p].ch[1];
            } else {
                p = tr[p].ch[0];
            }
        }
        return ans;
    }
    // 0base 第 k 小；Compare 改变时对应第 k 个元素
    optional<T> kth(int ver, int k) const {
        if (k < 0 || k >= size(ver)) return nullopt;
        int p = root[ver];
        while (p) {
            int left = node_size(tr[p].ch[0]);
            if (k < left) p = tr[p].ch[0];
            else if (k == left) return tr[p].value;
            else {
                k -= left + 1;
                p = tr[p].ch[1];
            }
        }
        return nullopt;
    }
    // 最大的严格 < value
    optional<T> predecessor(int ver, const T &value) const {
        int p = root[ver], ans = 0;
        while (p) {
            if (comp(tr[p].value, value)) {
                ans = p;
                p = tr[p].ch[1];
            } else {
                p = tr[p].ch[0];
            }
        }
        if (!ans) return nullopt;
        return tr[ans].value;
    }
    // 最小的严格 > value
    optional<T> successor(int ver, const T &value) const {
        int p = root[ver], ans = 0;
        while (p) {
            if (comp(value, tr[p].value)) {
                ans = p;
                p = tr[p].ch[0];
            } else {
                p = tr[p].ch[1];
            }
        }
        if (!ans) return nullopt;
        return tr[ans].value;
    }
    // ===== 等值查询：若不需要 count / contains，以下代码可直接删除 =====
    int count_leq(int ver, const T &value) const {
        int p = root[ver], ans = 0;
        while (p) {
            if (!comp(value, tr[p].value)) {
                ans += node_size(tr[p].ch[0]) + 1;
                p = tr[p].ch[1];
            } else {
                p = tr[p].ch[0];
            }
        }
        return ans;
    }
    int count(int ver, const T &value) const {
        return count_leq(ver, value) - order_of_key(ver, value);
    }
    bool contains(int ver, const T &value) const {
        int p = root[ver];
        while (p) {
            if (comp(value, tr[p].value)) p = tr[p].ch[0];
            else if (comp(tr[p].value, value)) p = tr[p].ch[1];
            else return true;
        }
        return false;
    }
};
```
#### 版本与修改
- `root[0]`：初始空版本
- `insert(ver, value)`：在版本 `ver` 中插入一个独立的 `value`，追加并返回新版本编号；期望 $O(\log n)$
- `erase(ver, value)`：删除版本 `ver` 中的一个等价值；若不存在则新版本直接共享原根，不创建 Treap 节点；期望 $O(\log n)$
- `copy_version(ver)`：直接复制版本根并产生新版本，$O(1)$
- `reserve(nodes, versions)`：预留节点池 / 版本数组容量，最好在第一次修改前调用
#### 查询
- `size(ver)` / `empty(ver)`：版本大小 / 是否为空，$O(1)$
- `order_of_key(ver, value)`：返回满足 `comp(x, value)` 的元素数；默认 `less<T>` 时即严格小于 `value` 的元素数，期望 $O(\log n)$
- `kth(ver, k)`：返回 `0base` 的第 `k` 个元素；越界返回 `nullopt`，期望 $O(\log n)$
- `predecessor(ver, value)`：严格前驱；不存在返回 `nullopt`，期望 $O(\log n)$
- `successor(ver, value)`：严格后继；不存在返回 `nullopt`，期望 $O(\log n)$
- `count(ver, value)`：与 `value` 在 `Compare` 意义下等价的元素数，期望 $O(\log n)$
- `contains(ver, value)`：是否至少存在一个等价值，期望 $O(\log n)$
其中等价关系为：
```cpp
!comp(a, b) && !comp(b, a)
```
所以自定义 `Compare` 时，并不要求 `T` 提供 `==`
#### 注意事项
- **版本编号与节点编号不是一回事。** `ver` 必须是 `root` 中已有的版本编号；内部节点编号不要拿来当版本使用
- 所有旧版本可达节点都不会被原地修改；`split_less()`、`merge()` 以及修改路径上的节点都会按需复制
- 相同值使用多个独立节点。`erase()` 只删除其中一个，但不保证对应哪一个物理节点，多重集语义不受影响
- 重复值统一向右插入，因此内部满足“左侧严格在当前值之前，右侧不在当前值之前”；不要自行修改 `insert_node()` 中相等值的分支而不同步修改 `split_less()`
- `predecessor()` / `successor()` 是**严格**前驱 / 后继；若希望“$\le x$ 最大值”或“$\ge x$ 最小值”，不能直接代替使用
- `kth()` 使用 `0base`
- 随机优先级只决定树形，不影响有序性正确性；树高及操作复杂度为期望 $O(\log n)$，理论最坏仍为 $O(n)$
- 默认随机种子来自时间；调试时若希望树形完全可复现，可以显式传固定种子
- 持久化数据结构通常会产生大量节点。如果能估计上界，建议提前 `reserve()`，避免 `vector` 多次扩容搬迁
#### 常见用法
##### 从旧版本分叉
```cpp
PersistentFHQTreap<int> tree;
int v1 = tree.insert(0, 3);   // {3}
int v2 = tree.insert(v1, 5);  // {3, 5}
int v3 = tree.insert(v1, 1);  // {1, 3}
cout << *tree.kth(v2, 1) << '\n'; // 5
cout << *tree.kth(v3, 0) << '\n'; // 1
```
##### 重复值与删除
```cpp
int v1 = tree.insert(0, 5);
int v2 = tree.insert(v1, 5);
int v3 = tree.insert(v2, 5);
cout << tree.count(v3, 5) << '\n'; // 3
int v4 = tree.erase(v3, 5);
cout << tree.count(v4, 5) << '\n'; // 2
cout << tree.count(v3, 5) << '\n'; // 3，旧版本不变
```
##### 排名、第 k 小、前驱与后继
```cpp
int rank = tree.order_of_key(ver, x); // 严格小于 x 的数量
auto kth = tree.kth(ver, k);
auto pre = tree.predecessor(ver, x);
auto nxt = tree.successor(ver, x);

if (kth) cout << *kth << '\n';
if (pre) cout << *pre << '\n';
if (nxt) cout << *nxt << '\n';
```
若题目中的“排名”使用 `1base`：
```cpp
int rank = tree.order_of_key(ver, x) + 1;
```
##### 每次操作都必须生成版本
例如输入的第 `i` 次操作要求产生版本 `i`，即使这次只是查询：
```cpp
int ver = old_ver;

if (op == 1) {
    ver = tree.insert(old_ver, x);
} else if (op == 2) {
    ver = tree.erase(old_ver, x);
} else {
    cout << tree.order_of_key(old_ver, x) + 1 << '\n';
    ver = tree.copy_version(old_ver);
}
```
##### 自定义键
如果需要按 `(value, id)` 区分元素：
```cpp
struct Key {
    i64 value{};
    int id{};
};
struct KeyLess {
    bool operator()(const Key &a, const Key &b) const {
        return tie(a.value, a.id) < tie(b.value, b.id);
    }
};
PersistentFHQTreap<Key, KeyLess> tree;
int ver = tree.insert(0, {value, id});
int rank = tree.order_of_key(ver, {value, INT_MIN});
auto kth = tree.kth(ver, k);
```
这里需要注意：一旦把 `id` 加入比较键，两个 `{value, id}` 只有完整键等价时才被视为同一个元素；如果只是想统计某个 `value` 的全部元素，应根据键域分别求边界排名
#### 复杂度

| 操作                                                                |          时间复杂度 |    新增 Treap 节点 |
| ----------------------------------------------------------------- | -------------: | -------------: |
| `insert`                                                          | 期望 $O(\log n)$ | 期望 $O(\log n)$ |
| `erase`，值存在                                                       | 期望 $O(\log n)$ | 期望 $O(\log n)$ |
| `erase`，值不存在                                                      | 期望 $O(\log n)$ |         $O(1)$ |
| `copy_version / size / empty`                                     |         $O(1)$ |            $0$ |
| `order_of_key / kth / predecessor / successor / count / contains` | 期望 $O(\log n)$ |            $0$ |
若共有 $V$ 个版本、$M$ 次实际改变集合内容的修改，总空间期望为
$$O(V + M\log n)$$
