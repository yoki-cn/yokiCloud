### 隐式 FHQ Treap 序列
---
**解决问题：** 将序列下标隐式维护在 FHQ Treap 的中序遍历中，支持动态插入 / 删除、单点访问 / 修改，以及区间加、区间和、区间翻转
```cpp
template<class T = i64>
struct ImplicitFHQTreap {
    struct Node {
        int ch[2]{}, size{};
        unsigned pri{};
        T value{}, sum{}, add{};
        bool rev{};
    };
    mt19937 rng;
    vector<Node> tr{Node{}};
    int root{};
    explicit ImplicitFHQTreap(
        unsigned seed = unsigned(chrono::steady_clock::now().time_since_epoch().count())
    ) : rng(seed) {}
    explicit ImplicitFHQTreap(
        const vector<T> &a,
        unsigned seed = unsigned(chrono::steady_clock::now().time_since_epoch().count())
    ) : rng(seed) {
        tr.reserve(a.size() + 1);
        root = build_tree(a);
    }
private:
    int node_size(int p) const { return tr[p].size; }
    T node_sum(int p) const { return tr[p].sum; }
    // ===== 节点创建与维护：build / split / merge / 修改操作依赖，不建议删除 =====
    int new_node(const T &value) {
        int p = tr.size();
        tr.push_back({{}, 1, unsigned(rng()), value, value, T{}, false});
        return p;
    }
    void pull(int p) {
        tr[p].size = 1 + node_size(tr[p].ch[0]) + node_size(tr[p].ch[1]);
        tr[p].sum = node_sum(tr[p].ch[0]) + tr[p].value + node_sum(tr[p].ch[1]);
    }
    // ===== 区间加：若不需要，删除 Node::add / apply_add() / range_add()
    // 同时删除 push() 中对应逻辑 =====
    void apply_add(int p, const T &value) {
        if (!p) return;
        tr[p].value += value;
        tr[p].sum += value * tr[p].size;
        tr[p].add += value;
    }
    // ===== 区间翻转：若不需要，删除 Node::rev / apply_rev() / reverse()
    // 同时删除 push() 中对应逻辑 =====
    void apply_rev(int p) {
        if (!p) return;
        swap(tr[p].ch[0], tr[p].ch[1]);
        tr[p].rev ^= 1;
    }
    // ===== 懒标记下传：只要保留 add / rev 中任意一种，就必须保留 =====
    void push(int p) {
        if (tr[p].rev) {
            apply_rev(tr[p].ch[0]);
            apply_rev(tr[p].ch[1]);
            tr[p].rev = false;
        }
        if (tr[p].add != T{}) {
            apply_add(tr[p].ch[0], tr[p].add);
            apply_add(tr[p].ch[1], tr[p].add);
            tr[p].add = T{};
        }
    }
    // ===== O(n) 笛卡尔树建树：若只从空序列单点插入，可删除
    // 同时删除 vector 构造函数和批量 insert(pos, values) =====
    int build_tree(const vector<T> &a) {
        if (a.empty()) return 0;
        vector<int> st;
        st.reserve(a.size());
        for (const T &value : a) {
            int x = new_node(value), last = 0;
            while (!st.empty() && tr[st.back()].pri > tr[x].pri) {
                last = st.back();
                st.pop_back();
            }
            tr[x].ch[0] = last;
            if (!st.empty()) tr[st.back()].ch[1] = x;
            st.push_back(x);
        }
        int rt = st.front();
        vector<int> order{rt};
        order.reserve(a.size());
        for (int i = 0; i < iorder.size(); ++ i) {
            int p = order[i];
            for (int c : tr[p].ch) if (c) order.push_back(c);
        }
        for (int p : order | views::reverse) pull(p);
        return rt;
    }
    // ===== 单点辅助：get / set 直接沿树下降，不需要 split =====
    T get_node(int p, int k) {
        while (p) {
            push(p);
            int left = node_size(tr[p].ch[0]);
            if (k < left) p = tr[p].ch[0];
            else if (k == left) return tr[p].value;
            else {
                k -= left + 1;
                p = tr[p].ch[1];
            }
        }
        return T{};
    }
    void set_node(int p, int k, const T &value) {
        push(p);
        int left = node_size(tr[p].ch[0]);
        if (k < left) set_node(tr[p].ch[0], k, value);
        else if (k > left) set_node(tr[p].ch[1], k - left - 1, value);
        else tr[p].value = value;
        pull(p);
    }
    // ===== 区间辅助：将 root 分成 [0,l) / [l,r) / [r,n) =====
    array<int, 3> split_range(int l, int r) {
        auto [a, c] = split(root, r);
        auto [x, b] = split(a, l);
        return {x, b, c};
    }
    void join_range(int a, int b, int c) { root = merge(merge(a, b), c); }
public:
    int size(int p) const { return node_size(p); }
    int size() const { return node_size(root); }
    bool empty() const { return root == 0; }
    void reserve(int nodes) { tr.reserve(nodes + 1); }
    void clear() {
        tr.resize(1);
        tr[0] = Node{};
        root = 0;
    }
    // ===== FHQ 核心：split 按元素数量分裂，merge 按原序列顺序拼接 =====
    // 前 k 个元素 | 剩余元素
    pair<int, int> split(int p, int k) {
        if (!p) return {0, 0};
        push(p);
        int left = node_size(tr[p].ch[0]);
        if (k <= left) {
            auto [a, b] = split(tr[p].ch[0], k);
            tr[p].ch[0] = b;
            pull(p);
            return {a, p};
        }
        auto [a, b] = split(tr[p].ch[1], k - left - 1);
        tr[p].ch[1] = a;
        pull(p);
        return {p, b};
    }
    // 返回序列 a + b
    int merge(int a, int b) {
        if (!a || !b) return a | b;
        if (tr[a].pri < tr[b].pri) {
            push(a);
            tr[a].ch[1] = merge(tr[a].ch[1], b);
            pull(a);
            return a;
        }
        push(b);
        tr[b].ch[0] = merge(a, tr[b].ch[0]);
        pull(b);
        return b;
    }
    // ===== 基础序列接口：插入 / 删除 / 单点访问修改 =====
    // 在原下标 pos 之前插入
    void insert(int pos, const T &value) {
        assert(0 <= pos && pos <= size());
        auto [a, b] = split(root, pos);
        root = merge(merge(a, new_node(value)), b);
    }
    // 在 pos 前插入整个 values
    void insert(int pos, const vector<T> &values) {
        assert(0 <= pos && pos <= size());
        if (values.empty()) return;
        auto [a, b] = split(root, pos);
        root = merge(merge(a, build_tree(values)), b);
    }
    T erase(int pos) {
        assert(0 <= pos && pos < size());
        auto [a, b, c] = split_range(pos, pos + 1);
        T value = move(tr[b].value);
        join_range(a, 0, c);
        return value;
    }
    // 删除 [l, r)
    void erase(int l, int r) {
        assert(0 <= l && l <= r && r <= size());
        if (l == r) return;
        auto part = split_range(l, r);
        root = merge(part[0], part[2]);
    }
    T get(int pos) {
        assert(0 <= pos && pos < size());
        return get_node(root, pos);
    }
    void set(int pos, const T &value) {
        assert(0 <= pos && pos < size());
        set_node(root, pos, value);
    }
    // ===== 区间修改：[l, r) =====
    void range_add(int l, int r, const T &value) {
        assert(0 <= l && l <= r && r <= size());
        if (l == r) return;
        auto [a, b, c] = split_range(l, r);
        apply_add(b, value);
        join_range(a, b, c);
    }
    void reverse(int l, int r) {
        assert(0 <= l && l <= r && r <= size());
        if (r - l <= 1) return;
        auto [a, b, c] = split_range(l, r);
        apply_rev(b);
        join_range(a, b, c);
    }
    // ===== 区间和：若不需要，删除 Node::sum / node_sum() / range_sum() / all_sum()
    // 并将 pull() 改为只维护 size =====
    T range_sum(int l, int r) {
        assert(0 <= l && l <= r && r <= size());
        if (l == r) return T{};
        auto [a, b, c] = split_range(l, r);
        T ans = node_sum(b);
        join_range(a, b, c);
        return ans;
    }
    T all_sum() const { return node_sum(root); }
    // ===== 导出序列：只用于输出 / 调试时可删除 =====
    vector<T> to_vector() {
        vector<T> a;
        a.reserve(size());
        vector<int> st;
        int p = root;
        while (p || !st.empty()) {
            while (p) {
                push(p);
                st.push_back(p);
                p = tr[p].ch[0];
            }
            p = st.back();
            st.pop_back();
            a.push_back(tr[p].value);
            p = tr[p].ch[1];
        }

        return a;
    }
};
```
#### 接口说明
##### 核心语义
隐式 Treap **不按 `value` 排序**
节点的逻辑下标由：
```cpp
size(left)
```
隐式决定
因此：
```cpp
auto [a, b] = split(root, k);
```
表示：
```text
a = 原序列前 k 个元素
b = 原序列剩余元素
```
也就是：
```text
[0, k) | [k, n)
```
这也是它和上一份 FHQ Treap 有序多重集最本质的区别
##### 基础接口

| 接口                 | 含义              |       期望复杂度 |
| ------------------ | --------------- | ----------: |
| `size()`           | 序列长度            |      $O(1)$ |
| `empty()`          | 是否为空            |      $O(1)$ |
| `insert(pos,x)`    | 在 `pos` 前插入 `x` | $O(\log n)$ |
| `insert(pos,a)`    | 批量插入序列 `a`      |  $O(\|a\|)$ |
| `erase(pos)`       | 删除并返回 `pos`     | $O(\log n)$ |
| `erase(l,r)`       | 删除 `[l,r)`      | $O(\log n)$ |
| `get(pos)`         | 读取单点            | $O(\log n)$ |
| `set(pos,x)`       | 单点赋值            | $O(\log n)$ |
| `range_add(l,r,x)` | `[l,r)` 全部加 `x` | $O(\log n)$ |
| `reverse(l,r)`     | 翻转 `[l,r)`      | $O(\log n)$ |
| `range_sum(l,r)`   | `[l,r)` 区间和     | $O(\log n)$ |
| `all_sum()`        | 整体和             |      $O(1)$ |
| `to_vector()`      | 导出整个序列          |      $O(n)$ |
所有公开下标统一使用 **`0base`**，所有区间统一使用 **`[l,r)`**

---
#### `split` 与 `merge`
##### `split(p, k)`
```cpp
auto [a, b] = tr.split(p, k);
```
切成：
```text
a = 前 k 个元素
b = 剩余元素
```
例如：
```text
p = [1, 2, 3, 4, 5]
k = 2
a = [1, 2]
b = [3, 4, 5]
```
##### `merge(a, b)`
隐式 Treap 的：
```cpp
merge(a, b)
```
只表示：
```text
序列 a + 序列 b
```
#### 注意事项
##### 懒标记必须在下降前 `push`
例如 `split()`：
```cpp
push(p);
int left = size(tr[p].ch[0]);
```
顺序不能反
特别是存在 `rev` 时，如果没有先下传，逻辑左儿子和当前物理 `ch[0]` 可能不一致
`merge()` 同理，在修改某个节点的儿子之前必须先 `push()`
##### `apply_add()` 必须同时修改 `value` 和 `sum`
这里使用的是标准的：
```cpp
tr[p].value += value;
tr[p].sum += value * tr[p].size;
tr[p].add += value;
```
也就是说 lazy 标记表示：
> 整棵子树的逻辑值已经更新，只是对子节点的修改尚未真正下传

所以 `sum` 在不 `push()` 的情况下依然始终正确。参考模板中的 Treap-序列同样采用区间 `add` 与 `sum` 的懒标记维护
##### `reverse` 立即交换左右儿子
```cpp
swap(tr[p].ch[0], tr[p].ch[1]);
tr[p].rev ^= 1;
```
因此当前节点的物理结构已经翻转，只需要记录：
> 两个儿子的内部序列还需要继续翻转

不要只写：
```cpp
tr[p].rev ^= 1;
```
却不交换两个儿子
##### `range_sum()` 会改变 Treap 内部树形
它会执行：
```cpp
split
split
读取 sum
merge
merge
```
因此：
```cpp
range_sum()
```
不是 `const`
不过它不会改变**逻辑序列**，只可能改变随机 Treap 的内部指针组织
##### 删除区间不会回收节点
当前：
```cpp
erase(l, r)
```
直接丢弃中间子树，因此时间仍然是期望：
$$O(\log n)$$
但被删除节点仍然留在：
```cpp
tr
```
中，不会重新利用
因此空间复杂度更准确地说是：
> `O(初始元素数 + 历史插入元素总数)`

而不是严格的当前 $O(n)$
这是故意的：如果为了回收 `[l,r)` 中的全部节点去遍历整棵被删除子树，则一次区间删除会变成：
$$O(\log n+r-l)$$
若题目存在极大量的“插入一大段 → 删除一大段”并且内存敏感，可以采用后面“常见用法”中的 `rebuild()` 周期性回收
##### `T` 的要求
完整版本要求 `T` 支持：
```cpp
T{}
a + b
a += b
a * int
a != b
```
通常直接：
```cpp
ImplicitFHQTreap<i64>
```
即可
如果只需要维护一般对象序列，可以裁掉 `sum / add` 部分
#### 功能裁剪
##### 只需要动态序列 + 翻转
删除：
```cpp
Node::sum
Node::add
node_sum()
apply_add()
range_add()
range_sum()
all_sum()
```
然后：
```cpp
void pull(int p) {
    tr[p].size = 1 + node_size(tr[p].ch[0]) + node_size(tr[p].ch[1]);
}
```
`push()` 只保留：
```cpp
void push(int p) {
    if (tr[p].rev) {
        apply_rev(tr[p].ch[0]);
        apply_rev(tr[p].ch[1]);
        tr[p].rev = false;
    }
}
```
此时 `T` 不再需要任何数值运算
##### 不需要区间翻转
删除：
```cpp
Node::rev
apply_rev()
reverse()
```
并删除 `push()` 中对应部分
##### 完全没有懒标记
如果既不要：
```text
range_add
reverse
```
那么：
```cpp
push()
```
以及所有：
```cpp
push(p);
```
都可以一起删除
这会得到一份非常短的：
> 动态数组型隐式 FHQ Treap

##### 不需要批量建树
删除：
```cpp
build_tree()
ImplicitFHQTreap(const vector<T> &a, ...)
insert(pos, const vector<T> &values)
```
只保留单点：
```cpp
insert(pos, value)
```

即可
#### 常见用法
##### 动态数组
```cpp
ImplicitFHQTreap<i64> tr({1, 2, 3, 4});
tr.insert(2, 10);
// [1, 2, 10, 3, 4]
tr.erase(1);
// [1, 10, 3, 4]
tr.set(2, 100);
// [1, 10, 100, 4]
cout << tr.get(2) << '\n';
```
##### 区间加、区间和、翻转
```cpp
tr.range_add(l, r, delta);
i64 sum = tr.range_sum(l, r);
tr.reverse(l, r);
```
均使用：
```text
[l, r)
```
##### 批量插入
```cpp
vector<i64> b{3, 1, 4, 1, 5};
tr.insert(pos, b);
```
内部不是逐个执行：
```cpp
insert()
```
而是先对 `b` 用随机优先级构造 Cartesian Tree，再一次 `merge()`
因此复杂度是：
$$O(|b|+\log n)$$
##### 剪切一段并删除
如果只是剪掉：
```text
[l,r)
```
直接：
```cpp
tr.erase(l, r);
```
即可
如果还需要保留被剪下来的 Treap 根，可以直接利用核心接口：
```cpp
auto [ab, c] = tr.split(tr.root, r);
auto [a, b] = tr.split(ab, l);
tr.root = tr.merge(a, c);
// b 就是被剪下来的 [l, r)
```
需要注意：
```cpp
b
```
仍然属于同一个 `tr` 节点池
##### 交换两个相邻区间
将：
```text
A B C D
```
中的：
```text
B = [l,m)
C = [m,r)
```
交换成：
```text
A C B D
```
可以：
```cpp
auto [abc, d] = tr.split(tr.root, r);
auto [ab, c] = tr.split(abc, m);
auto [a, b] = tr.split(ab, l);

tr.root = tr.merge(tr.merge(tr.merge(a, c), b), d);
```
这本质就是：
```text
[l,m) 与 [m,r) 交换
```
也是隐式 Treap 很经典的应用
##### 区间循环移动
例如将：
```text
[l,r)
```
循环右移 `k` 位
```cpp
int len = r - l;
k %= len;
auto [abc, d] = tr.split(tr.root, r);
auto [a, bc] = tr.split(abc, l);
auto [b, c] = tr.split(bc, len - k);
tr.root = tr.merge(tr.merge(tr.merge(a, c), b), d);
```
例如：
```text
[1, 2, 3, 4, 5]
[l,r) = [1,5)
k = 2

=> [1, 4, 5, 2, 3]
```
如果 `k == 0` 可以直接跳过
##### 导出整个序列
```cpp
vector<i64> a = tr.to_vector();
```
调试 / 对拍尤其方便：
```cpp
for (i64 x : tr.to_vector()) {
    cerr << x << ' ';
}
```
##### 大量删除后的重构
如果题目长期执行：
```text
大量插入
大量区间删除
大量插入
大量区间删除
```
当前节点池可能累计得很大
可以偶尔执行：
```cpp
auto a = tr.to_vector();
tr.clear();
tr.reserve(a.size());
tr.root = tr.build_tree(a);
```
不过 `build_tree()` 当前是 `private`
如果题目确实需要这种操作，可以在结构体中补一个：
```cpp
void rebuild() {
    auto a = to_vector();
    clear();
    tr.reserve(a.size() + 1);
    root = build_tree(a);
}
```
#### 复杂度
Treap 树高期望为 $O(\log n)$

| 操作                                                                                |          期望时间 |
| --------------------------------------------------------------------------------- | ------------: |
| 初始 `build`                                                                        |        $O(n)$ |
| `split / merge` / `insert(pos,x)`                                                 |   $O(\log n)$ |
| 批量插入 $m$ 个元素                                                                      | $O(m+\log n)$ |
| `erase(pos)` / `erase(l,r)` / `get / set` / `range_add` / `reverse` / `range_sum` |   $O(\log n)$ |
| `all_sum`                                                                         |        $O(1)$ |
| `to_vector`                                                                       |        $O(n)$ |
理论最坏树高仍可能为 $O(n)$【
节点池空间为：$$O(n_0+I)$$a
