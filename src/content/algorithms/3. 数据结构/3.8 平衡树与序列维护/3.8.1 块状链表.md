### 块状链表
---
**解决问题：** 维护动态序列，支持按位置访问、单点 / 批量插入、单点 / 区间删除。相比隐式平衡树实现更短，操作复杂度约为 $O(\sqrt n)$
```text
块状链表
    动态长度
    ↓
按“位置”分块
插入 / 删除 / 访问 / 批量编辑

ODT
    固定长度
    ↓
按“连续相等值”分段
区间赋值 / 整段变换 / 枚举常值段
```

```cpp
template<class T, int B = 700>
struct BlockList {
    static_assert(B >= 2);
    using Block = vector<T>;
    using Iter = typename list<Block>::iterator;
    using CIter = typename list<Block>::const_iterator;
    list<Block> blocks;
    int n{};
    explicit BlockList(const vector<T> &a = {}) { build(a); }
    int size() const { return n; }
    bool empty() const { return n == 0; }
private:
    // ===== 建块：构造和 rebuild() 依赖，不建议删除 =====
    void build(const vector<T> &a) {
        blocks.clear();
        n = a.size();
        for (int l = 0; l < n; l += B) {
            blocks.emplace_back(a.begin() + l, a.begin() + min(n, l + B));
        }
        if (blocks.empty()) {
            blocks.emplace_back();
        } else if (blocks.size() > 1 && int(blocks.back().size()) < B / 2) {
            auto last = prev(blocks.end());
            auto pre = prev(last);
            pre->insert(pre->end(),
                make_move_iterator(last->begin()),
                make_move_iterator(last->end()));
            blocks.erase(last);
        }
    }
    // ===== 定位：所有按位置操作依赖，不建议删除 =====
    pair<Iter, int> locate(int pos) {
        auto it = blocks.begin();
        while (next(it) != blocks.end() && pos >= int(it->size())) {
            pos -= it->size();
            ++ it;
        }
        return {it, pos};
    }
    pair<CIter, int> locate(int pos) const {
        auto it = blocks.begin();
        while (next(it) != blocks.end() && pos >= int(it->size())) {
            pos -= it->size();
            ++ it;
        }
        return {it, pos};
    }
    // ===== 块分裂与重平衡：插入 / 删除依赖，不建议删除 =====
    // 保证全局位置 pos 成为块边界，返回 pos 开始的块
    Iter split_at(int pos) {
        if (pos == n) return blocks.end();
        auto [it, k] = locate(pos);
        if (k == 0) return it;
        Block right(
            make_move_iterator(it->begin() + k),
            make_move_iterator(it->end())
        );
        it->erase(it->begin() + k, it->end());
        return blocks.insert(next(it), move(right));
    }
    Iter split_big(Iter it) {
        if (int(it->size()) <= 2 * B) return next(it);
        int k = it->size() / 2;
        Block right(
            make_move_iterator(it->begin() + k),
            make_move_iterator(it->end())
        );
        it->erase(it->begin() + k, it->end());
        return blocks.insert(next(it), move(right));
    }
    Iter rebalance(Iter it) {
        if (it == blocks.end()) return it;
        while (blocks.size() > 1) {
            if (it->empty()) {
                auto nxt = next(it);
                if (nxt != blocks.end()) {
                    blocks.erase(it);
                    it = nxt;
                } else {
                    auto pre = prev(it);
                    blocks.erase(it);
                    it = pre;
                }
                continue;
            }
            if (int(it->size()) > 2 * B) {
                split_big(it);
                return it;
            }
            if (int(it->size()) >= B / 2) return it;
            auto nxt = next(it);
            if (nxt != blocks.end()) {
                it->insert(it->end(),
                    make_move_iterator(nxt->begin()),
                    make_move_iterator(nxt->end()));
                blocks.erase(nxt);
            } else {
                auto pre = prev(it);
                pre->insert(pre->end(),
                    make_move_iterator(it->begin()),
                    make_move_iterator(it->end()));
                blocks.erase(it);
                it = pre;
            }
        }
        return it;
    }
    void fix_pos(int pos) {
        if (n == 0) return;
        pos = clamp(pos, 0, n - 1);
        auto [it, k] = locate(pos);
        rebalance(it);
    }
public:
    // ===== 基础接口：按位置访问 / 单点插入删除 =====
    T &operator[](int pos) {
        assert(0 <= pos && pos < n);
        auto [it, k] = locate(pos);
        return (*it)[k];
    }
    const T &operator[](int pos) const {
        assert(0 <= pos && pos < n);
        auto [it, k] = locate(pos);
        return (*it)[k];
    }
    void insert(int pos, T value) {
        assert(0 <= pos && pos <= n);
        auto [it, k] = locate(pos);
        it->insert(it->begin() + k, move(value));
        ++ n;
        split_big(it);
    }
    T erase(int pos) {
        assert(0 <= pos && pos < n);
        auto [it, k] = locate(pos);
        T value = move((*it)[k]);
        it->erase(it->begin() + k);
        -- n;
        if (n == 0) {
            blocks.clear();
            blocks.emplace_back();
        } else {
            rebalance(it);
        }
        return value;
    }
    // ===== 批量操作：若只需要单点插删，可以删除以下两个接口与 split_at() =====
    void insert(int pos, vector<T> values) {
        assert(0 <= pos && pos <= n);
        if (values.empty()) return;
        int m = values.size();
        auto at = split_at(pos);
        if (n == 0 && blocks.size() == 1 && blocks.front().empty()) {
            blocks.clear();
            at = blocks.end();
        }
        for (int l = 0; l < m; l += B) {
            int r = min(m, l + B);
            Block block(
                make_move_iterator(values.begin() + l),
                make_move_iterator(values.begin() + r)
            );
            blocks.insert(at, move(block));
        }
        n += m;
        if (pos > 0) fix_pos(pos - 1);
        fix_pos(pos);
        fix_pos(pos + m - 1);
        if (pos + m < n) fix_pos(pos + m);
    }
    void erase(int l, int r) {
        assert(0 <= l && l <= r && r <= n);
        if (l == r) return;
        auto last = split_at(r);
        auto first = split_at(l);
        blocks.erase(first, last);
        n -= r - l;
        if (n == 0) {
            blocks.clear();
            blocks.emplace_back();
            return;
        }
        if (l > 0) fix_pos(l - 1);
        if (l < n) fix_pos(l);
    }
    // ===== 遍历与重构：调试、输出或题目需要顺序扫描时保留 =====
    template<class F>
    void for_each(int l, int r, F f) const {
        assert(0 <= l && l <= r && r <= n);
        if (l == r) return;
        auto [it, k] = locate(l);
        int len = r - l;
        while (len) {
            int take = min(len, int(it->size()) - k);
            for (int i = 0; i < take; ++ i) f((*it)[k + i]);
            len -= take;
            ++ it;
            k = 0;
        }
    }
    vector<T> to_vector() const {
        vector<T> a;
        a.reserve(n);
        for (const auto &block : blocks) {
            a.insert(a.end(), block.begin(), block.end());
        }
        return a;
    }
    void rebuild() { build(to_vector()); }
};
```
#### 接口说明
- `BlockList(a)`：由序列 `a` 构造
- `size()` / `empty()`：当前元素个数 / 是否为空，$O(1)$
- `operator[](pos)`：访问下标 `pos`，$O(n/B)$
- `insert(pos, value)`：在原下标 `pos` 前插入一个元素
- `erase(pos)`：删除并返回下标 `pos` 的元素
- `insert(pos, values)`：在 `pos` 前批量插入序列 `values`
- `erase(l, r)`：删除 `[l,r)`
- `for_each(l,r,f)`：顺序访问 `[l,r)` 中每个元素，不修改结构
- `to_vector()`：恢复完整连续序列
- `rebuild()`：重新按目标块长分块
#### 块大小维护
结构大致维持：
```text
B / 2 <= 块大小 <= 2B
```
边界和只有一个块时允许例外
删除后如果某块过小，就与相邻块合并；合并后过大再重新分裂。这一点很重要——参考资料中的块状链表同样专门强调需要合并，否则结构可能退化。
#### 注意事项
- 所有下标均为 `0base`，区间统一为 `[l,r)`
- `insert(pos, ...)` 表示插在原 `pos` 之前；`pos == size()` 即尾插
- `B` 应取约 $\Theta(\sqrt n)$。常见数据规模下 `500~1000` 通常都可以，再根据常数调整
- `operator[]` 只需要扫描块，不扫描块内元素，因此为 $O(n/B)$
- 单点插入 / 删除还需要移动块内 `vector`，因此有额外 $O(B)$
- 批量操作比逐个调用 `insert/erase` 更重要：完整块可以整体加入 / 删除
- 如果题目大量需要区间翻转、区间加、区间聚合等操作，通常更适合隐式 FHQ Treap，而不是继续给这个结构堆懒标记
- `rebuild()` 是 $O(n)$，不应每次操作后调用；它适合调试或非常特殊的周期性重构策略
#### 常见用法
##### 动态数组
```cpp
BlockList<int> a({1, 2, 3, 4});
a.insert(2, 10);
// [1, 2, 10, 3, 4]
a.erase(1);
// [1, 10, 3, 4]
a[2] = 100;
// [1, 10, 100, 4]
```
##### 批量插入和区间删除
这正是块状链表相对普通 `vector` 更值得使用的情况：
```cpp
vector<int> b{5, 6, 7, 8};
a.insert(pos, b);
a.erase(l, r);
```
如果 `b` 后面不再使用，可以避免一次复制：
```cpp
a.insert(pos, move(b));
```
##### 文本编辑器
经典应用就是动态文本
```cpp
BlockList<char> text;
vector<char> s(input.begin(), input.end());
text.insert(cursor, move(s));
text.erase(cursor, cursor + len);
text.for_each(cursor, cursor + len, [&](char c) {
    cout << c;
});
```
##### 整体输出 / 对拍
```cpp
vector<int> a = blocks.to_vector();
```
如果经过大量自定义操作，怀疑块分布不好，也可以：
```cpp
blocks.rebuild();
```
##### 特化：区间统计
块状链表本身没有维护区间信息。如果只是偶尔查询，可以直接遍历：
```cpp
i64 sum = 0;
a.for_each(l, r, [&](int x) {
    sum += x;
});
```
如果这种区间查询非常频繁，就不应该继续用这个基础版本，应给块维护统计信息，或者直接换其他数据结构
#### 复杂度
设当前长度为 $n$，目标块长为 $B$

|操作|时间复杂度|
|---|--:|
|构造|$O(n)$|
|`size / empty`|$O(1)$|
|`operator[]`|$O(n/B)$|
|单点 `insert / erase`|$O(n/B+B)$|
|批量插入 $m$ 个元素|$O(n/B+B+m)$|
|删除 `[l,r)`|$O(n/B+B+k)$|
|`for_each(l,r)`|$O(n/B+r-l)$|
|`to_vector / rebuild`|$O(n)$|
其中 $k$ 为删除涉及的块数；对于有非平凡析构函数的 `T`，删除元素本身还要计入对应析构开销
取 $B=\Theta(\sqrt n)$ 时，单点插删约为 $O(\sqrt n)$
