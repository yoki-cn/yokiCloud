### 可持久化 DSU
---
使用可持久化隐式线段树维护每个点的并查集状态：
- `val[x] < 0`：`x` 是根，集合大小为 `-val[x]`
- `val[x] >= 0`：`val[x]` 是 `x` 的父节点
- 未出现在线段树中的点默认 `val[x] = -1`
因此初始版本不需要真正建立整棵线段树
```cpp
struct PersistentDSU {
private:
    struct Node {
        int ch[2]{};
        int val{ -1 };
    };
    int n;
    vector<Node> tr{{}};
    vector<int> root{ 0 }, cc;
    // 查询版本 rt 中点 x 的 parent / -size；未修改过的点默认为 -1
    int get(int p, int x) const {
        int l{ 0 }, r{ n };
        while (p && r - l > 1) {
            int m{ (l + r) / 2 };
            if (x < m) {
                p = tr[p].ch[0];
                r = m;
            } else {
                p = tr[p].ch[1];
                l = m;
            }
        }
        return p ? tr[p].val : -1;
    }
    // 在版本 p 上单点修改并返回新的线段树根
    int set_node(int p, int l, int r, int x, int val) {
        int q{ int(tr.size()) };
        tr.push_back(p ? tr[p] : Node{});
        if (r - l == 1) {
            tr[q].val = val;
            return q;
        }
        int m{ (l + r) / 2 };
        if (x < m) tr[q].ch[0] = set_node(tr[q].ch[0], l, m, x, val);
        else tr[q].ch[1] = set_node(tr[q].ch[1], m, r, x, val);
        return q;
    }
    // 不做路径压缩；按大小合并保证父链长度 O(log n)
    int find_root(int rt, int x) const {
        while (true) {
            int p{ get(rt, x) };
            if (p < 0) return x;
            x = p;
        }
    }
    int new_version(int rt, int cnt) {
        root.push_back(rt);
        cc.push_back(cnt);
        return int(root.size()) - 1;
    }
public:
    explicit PersistentDSU(int n_, int q = 0) : n(n_), cc{ n } {
        if (q > 0) {
            int lg{};
            while ((1LL << lg) < n) ++ lg;
            tr.reserve(1 + 2LL * q * (lg + 1));
            root.reserve(q + 1);
            cc.reserve(q + 1);
        }
    }
    int versions() const { return int(root.size()); }
    int find(int ver, int x) const { return find_root(root[ver], x); }
    bool same(int ver, int x, int y) const { return find_root(root[ver], x) == find_root(root[ver], y); }
    int size(int ver, int x) const {
        int rt{ root[ver] };
        x = find_root(rt, x);
        return -get(rt, x);
    }
    int count(int ver) const { return cc[ver]; }
    // 从版本 ver 合并 x、y，并始终追加一个新版本
    int unite(int ver, int x, int y) {
        int rt{ root[ver] };
        x = find_root(rt, x);
        y = find_root(rt, y);
        if (x == y) return new_version(rt, cc[ver]);
        int sx{ -get(rt, x) }, sy{ -get(rt, y) };
        if (sx < sy) {
            swap(x, y);
            swap(sx, sy);
        }
        rt = set_node(rt, 0, n, y, x);
        rt = set_node(rt, 0, n, x, -(sx + sy));
        return new_version(rt, cc[ver] - 1);
    }
    // ===== 若题目每次操作一定执行 unite()，以下接口可直接删除 =====
    int copy_version(int ver) { return new_version(root[ver], cc[ver]); }
};
```
#### 接口说明
##### 状态表示
版本 `0` 为初始版本，此时所有点互不连通
底层使用一个可持久化数组表示 DSU：
- 值 `< 0`：当前点为根，绝对值为集合大小
- 值 `>= 0`：当前点的父节点编号
- 从未修改的位置隐式取值 `-1`
因此不需要构造初始线段树
##### 接口
- `PersistentDSU(n, q)`：建立 `n` 个点；`q` 可选，用于预留最多约 `q` 次版本操作的空间，构造本身 $O(1)$
- `versions()`：返回当前版本数量，$O(1)$
- `find(ver, x)`：返回版本 `ver` 中 `x` 的代表元，$O(\log^2 n)$
- `same(ver, x, y)`：判断指定版本中两点是否连通，$O(\log^2 n)$
- `size(ver, x)`：返回指定版本中 `x` 所在集合大小，$O(\log^2 n)$
- `count(ver)`：返回指定版本的连通块数量，$O(1)$
- `unite(ver, x, y)`：从版本 `ver` 合并两点，追加并返回新版本编号，$O(\log^2 n)$
- `copy_version(ver)`：追加一个与 `ver` 完全相同的版本并返回新版本编号，$O(1)$
#### 注意事项
- 构造后初始版本编号为 `0`
- `unite(ver, x, y)` **始终产生一个新版本**；即使两点原本已经连通，也会追加一个内容相同的版本
- 可以从任意历史版本继续调用 `unite()`，因此自然支持版本分叉
- `copy_version(ver)` 只是新增一个根引用，不复制底层节点
- 不使用路径压缩；按大小合并保证 DSU 父链长度为 $O(\log n)$
- 查询一次父节点需要在线段树中进行一次 $O(\log n)$ 单点查询，因此 `find()` 为 $O(\log^2 n)$
- 一次成功 `unite()` 只持久化修改两个位置：新根的大小与旧根的父亲，共新增 $O(\log n)$ 个节点
- `find(ver, x)` 返回的代表元只对该版本有意义；不同版本中同一集合的代表元不必相同
- 默认点编号为 `0base`
- 要求 `n > 0`、`0 <= x < n`、`0 <= ver < versions()`
- 构造参数 `q` 只用于 `reserve()`，给小或不给都不影响正确性
#### 常见用法
##### 从历史版本继续分叉
```cpp
PersistentDSU dsu(4, 4);

int v1 = dsu.unite(0, 0, 1);   // {0,1}, {2}, {3}
int v2 = dsu.unite(0, 2, 3);   // 从 v0 分叉：{0}, {1}, {2,3}
int v3 = dsu.unite(v1, 1, 2);  // {0,1,2}, {3}

cout << dsu.same(0, 0, 1) << '\n';   // 0
cout << dsu.same(v1, 0, 1) << '\n';  // 1
cout << dsu.same(v2, 0, 1) << '\n';  // 0
cout << dsu.same(v2, 2, 3) << '\n';  // 1
cout << dsu.same(v3, 0, 2) << '\n';  // 1
```
##### 每次操作对应一个版本
```cpp
vector<int> ver(q + 1);
ver[0] = 0;

for (int i = 1; i <= q; ++ i) {
    if (op[i] == 1) {
        ver[i] = dsu.unite(ver[k], x, y);
    } else {
        ver[i] = dsu.copy_version(ver[k]);
    }
}
```
题目操作编号和内部版本编号只有在“每次操作恰好创建一个版本”时才能直接一一对应；更稳妥的方式是像上面一样显式保存 `ver[i]`
