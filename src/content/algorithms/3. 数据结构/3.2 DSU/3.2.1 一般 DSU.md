### 一般 DSU
---
```cpp
struct DSU {
    int cc;
    vector<int> pa, siz;
    explicit DSU(int n) : cc(n), pa(n), siz(n, 1) { ranges::iota(pa, 0); }
    int find(int x) {
        while (x != pa[x]) x = pa[x] = pa[pa[x]];
        return x;
    }
    bool same(int x, int y) { return find(x) == find(y); }
    bool unite(int x, int y) {
        x = find(x), y = find(y);
        if (x == y) return false;
        if (siz[x] < siz[y]) swap(x, y);
        pa[y] = x;
        siz[x] += siz[y];
        -- cc;
        return true;
    }
    int size(int x) { return siz[find(x)]; }
    int count() const { return cc; }
    bool is_root(int x) const { return pa[x] == x; }
};
```
#### 注意事项
- 默认点编号为 `0base`
- `siz[x]` 只有在 `x` 为根时有效；外部查询使用 `size(x)`
- 使用路径压缩，不能直接用于可撤销并查集
- 本板子采用按大小合并，不需要再维护秩
