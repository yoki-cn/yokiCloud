### Poly（多项式与形式幂级数）
---
```cpp
// ===== 模平方根：仅 Poly<Z>::sqrt(n) 自动开方依赖 =====
// 若不需要自动 FPS 平方根，可删除本函数与 sqrt(n)
optional<Z> mod_sqrt(Z a) {
    constexpr int P{ Z::mod() };
    if (a == Z(0)) return Z(0);
    if (a.pow((P - 1) / 2) != Z(1)) return nullopt;
    if (P % 4 == 3) return a.pow((P + 1) / 4);
    int s{ countr_zero(unsigned(P - 1)) };
    i64 q{ (P - 1) >> s };
    Z z{ 2 };
    while (z.pow((P - 1) / 2) == Z(1)) z += Z(1);
    Z c{ z.pow(q) }, x{ a.pow((q + 1) / 2) }, t{ a.pow(q) };
    int m{ s };
    while (t != Z(1)) {
        int i{ 1 };
        Z tt{ t * t };
        for (; i < m && tt != Z(1); ++ i) tt *= tt;
        assert(i < m);
        Z b{ c.pow(1LL << (m - i - 1)) };
        x *= b;
        c = b * b;
        t *= c;
        m = i;
    }
    if ((-x).val() < x.val()) x = -x;
    return x;
}
template<class T = Z>
struct Poly {
    vector<T> a;
    Poly() = default;
    explicit Poly(int n) : a(n) {}
    explicit Poly(vector<T> a_) : a(move(a_)) {}
    Poly(initializer_list<T> a_) : a(a_) {}
    // ===== 基础容器：几乎所有功能依赖，不建议删除 =====
    int size() const { return int(a.size()); }
    bool empty() const { return a.empty(); }
    T operator[](int i) const { return 0 <= i && i < size() ? a[i] : T{}; }
    T &operator[](int i) { return a[i]; }
    void resize(int n) { a.resize(n); }
    // 删除最高次连续零项；divmod() 依赖
    void norm() {
        while (!a.empty() && a.back() == T{}) a.pop_back();
    }
    // 截断为恰好 n 项；不足补零，只复制实际需要的前缀
    Poly trunc(int n) const {
        assert(n >= 0);
        Poly f(n);
        copy_n(a.begin(), min(n, size()), f.a.begin());
        return f;
    }
    // 乘 x^k；k < 0 时删除前 -k 项
    // sqrt() / pow() / mulT() 依赖
    Poly shift(int k) const {
        if (k >= 0) {
            Poly f(size() + k);
            copy(a.begin(), a.end(), f.a.begin() + k);
            return f;
        }
        int d{ -k };
        if (d >= size()) return Poly();
        return Poly(vector<T>(a.begin() + d, a.end()));
    }
    // 反转系数；指定 n 时先截断 / 补零到 n 项再反转
    // divmod() / mulT() 依赖
    Poly rev(int n = -1) const {
        Poly f{ n < 0 ? *this : trunc(n) };
        ranges::reverse(f.a);
        return f;
    }
    // 单点求值；完全独立
    T eval(T x) const {
        T res{};
        for (int i = size() - 1; i >= 0; -- i) res = res * x + a[i];
        return res;
    }
    // ===== 加减：多数 FPS 与多项式除法依赖 =====
    Poly &operator+=(const Poly &g) {
        if (g.size() > size()) a.resize(g.size());
        for (int i = 0; i < g.size(); ++ i) a[i] += g.a[i];
        return *this;
    }
    Poly &operator-=(const Poly &g) {
        if (g.size() > size()) a.resize(g.size());
        for (int i = 0; i < g.size(); ++ i) a[i] -= g.a[i];
        return *this;
    }
    friend Poly operator+(Poly f, const Poly &g) { return f += g; }
    friend Poly operator-(Poly f, const Poly &g) { return f -= g; }
    friend Poly operator-(Poly f) {
        for (T &x : f.a) x = -x;
        return f;
    }
    // ===== 标量乘除：sqrt() / pow() 等依赖 =====
    Poly &operator*=(const T &k) {
        for (T &x : a) x *= k;
        return *this;
    }
    Poly &operator/=(const T &k) {
        for (T &x : a) x /= k;
        return *this;
    }
    friend Poly operator*(Poly f, const T &k) { return f *= k; }
    friend Poly operator*(const T &k, Poly f) { return f *= k; }
    friend Poly operator/(Poly f, const T &k) { return f /= k; }
    // ===== 多项式乘法：所有快速 FPS / divmod / mulT 依赖 =====
    friend Poly operator*(const Poly &f, const Poly &g) {
        return Poly(::conv(f.a, g.a));
    }
    Poly &operator*=(const Poly &g) { return *this = *this * g; }
    // ===== FPS 逆：log / sqrt / divmod 依赖 =====
    Poly inv(int n) const {
        assert(n >= 0);
        if (n == 0) return {};
        assert(!empty() && a[0] != T{});
        Poly x{ T(1) / a[0] };
        for (int m = 1; m < n; m <<= 1) {
            int k{ min(m << 1, n) };
            x = (x * (Poly{ T(2) } - trunc(k) * x)).trunc(k);
        }
        return x.trunc(n);
    }
    // ===== FPS 平方根：若不需要 sqrt，可删除这两个成员与上方 mod_sqrt =====
    // 已知常数项平方根 s0；要求 a[0] != 0
    Poly sqrt(int n, T s0) const {
        assert(n >= 0);
        if (n == 0) return {};
        assert(!empty());
        assert(a[0] != T{} && s0 != T{} && s0 * s0 == a[0]);
        Poly x{ s0 };
        T inv2{ T(1) / T(2) };
        for (int m = 1; m < n; m <<= 1) {
            int k{ min(m << 1, n) };
            x = ((x + (trunc(k) * x.inv(k)).trunc(k)) * inv2).trunc(k);
        }
        return x.trunc(n);
    }
    // Z 专用：支持前导零，自动求最低非零项的模平方根
    Poly sqrt(int n) const {
        static_assert(is_same_v<T, Z>);
        assert(n >= 0);
        if (n == 0) return {};
        int t{ 0 };
        while (t < min(size(), n) && a[t] == T{}) ++ t;
        if (t == min(size(), n)) return Poly(n);
        assert(t % 2 == 0);
        int s{ t >> 1 }, m{ n - 2 * s };
        Poly h{ shift(-2 * s).trunc(m) };
        auto s0{ mod_sqrt(h.a[0]) };
        assert(s0.has_value());
        return h.sqrt(m, *s0).shift(s).trunc(n);
    }
    // ===== 多项式除法：不需要 / % 时可整体删除 divmod 与两个运算符 =====
    // 依赖 norm / rev / inv / 多项式乘法 / 加减
    friend pair<Poly, Poly> divmod(Poly f, Poly g) {
        f.norm();
        g.norm();
        assert(!g.empty());
        if (f.size() < g.size()) return { Poly(), f };
        int n{ f.size() }, m{ g.size() }, k{ n - m + 1 };
        Poly rf{ f.rev() };
        rf.resize(k);
        Poly q{ (rf * g.rev().inv(k)).trunc(k).rev() };
        Poly r{ f - q * g };
        r.resize(m - 1);
        q.norm();
        r.norm();
        return { q, r };
    }
    friend Poly operator/(Poly f, const Poly &g) {
        return divmod(move(f), g).first;
    }
    friend Poly operator%(Poly f, const Poly &g) {
        return divmod(move(f), g).second;
    }
    // ===== 微积分与 log / exp / pow =====
    // deriv / integr：log 依赖
    // log：inv 依赖
    // exp：log 依赖
    // pow：log + exp + shift + 标量运算依赖
    Poly deriv() const {
        if (size() <= 1) return Poly();
        Poly f(size() - 1);
        for (int i = 1; i < size(); ++ i) f.a[i - 1] = a[i] * T(i);
        return f;
    }
    Poly integr() const {
        Poly f(size() + 1);
        for (int i = 0; i < size(); ++ i) f.a[i + 1] = a[i] / T(i + 1);
        return f;
    }
    Poly log(int n) const {
        assert(n >= 0);
        if (n == 0) return {};
        assert(!empty() && a[0] == T(1));
        return (deriv() * inv(n)).integr().trunc(n);
    }
    Poly exp(int n) const {
        assert(n >= 0);
        if (n == 0) return {};
        assert(empty() || a[0] == T{});
        Poly x{ T(1) };
        for (int m = 1; m < n; m <<= 1) {
            int k{ min(m << 1, n) };
            x = (x * (Poly{ T(1) } - x.log(k) + trunc(k))).trunc(k);
        }
        return x.trunc(n);
    }
    Poly pow(i64 k, int n) const {
        assert(k >= 0 && n >= 0);
        if (n == 0) return {};
        if (k == 0) {
            Poly f(n);
            f.a[0] = T(1);
            return f;
        }
        int t{ 0 };
        while (t < min(size(), n) && a[t] == T{}) ++ t;
        if (t == min(size(), n) || (t && k > (n - 1LL) / t)) return Poly(n);
        T c{ a[t] };
        int m{ int(n - 1LL * t * k) };
        Poly f{ shift(-t).trunc(m) / c };
        f = (f.log(m) * T(k)).exp(m);
        T ck{ 1 }, base{ c };
        for (i64 e{ k }; e; e >>= 1, base *= base) {
            if (e & 1) ck *= base;
        }
        return (f * ck).shift(int(1LL * t * k)).trunc(n);
    }
    // ===== 转置乘法：主要供多点求值 / 插值等高级模块使用 =====
    // 仅依赖 rev / shift / 多项式乘法
    Poly mulT(const Poly &g) const {
        if (g.empty()) return Poly();
        return ((*this) * g.rev()).shift(1 - g.size());
    }
};
```

#### 接口说明

- `Poly<T>`：系数数组 `a`，低次到高次，`size()/empty()/resize()`；`f[i]` 常量版本越界补零，非 const 版本必须在范围内。
- `trunc(n)`：前 `n` 项，不足补零；`shift(k)`：乘 $x^k$（负数表示砍掉低次项）；`rev(n)`：截断/补零到 `n` 项再翻转（不传参数则整段翻转）。
- `norm()`：移除最高次零项；`eval(x)`：Horner 单点求值。
- `+ - *`：多项式运算；`* /` 标量；`divmod(f,g)` 得商余，`f/g`、`f%g` 为普通多项式除法。
- `inv(n)`：模 $x^n$ 乘法逆；`sqrt(n,s0)`：常数项已知平方根；`sqrt(n)`：仅 `Z` 自动 Tonelli–Shanks 并处理前导零。
- `deriv()` / `integr()`：形式求导 / 积分；`log(n)`、`exp(n)`、`pow(k,n)`：均返回恰好 `n` 项。
- `mulT(g)`：转置乘法 `f * reverse(g)` 后删除前 `|g|-1` 项，主要用于多点求值。

#### 注意事项

- `inv(n)` 要求 `f[0]!=0`；`log(n)` 要求 `f[0]=1`；`exp(n)` 要求 `f[0]=0`；`pow(k,n)` 只接受非负整数 `k`。
- `sqrt(n,s0)` 要求 `f[0]=s0²!=0` 且 `2` 可逆；自动 `sqrt(n)` 要求平方根存在，前导零个数为偶数，选择模意义下较小的常数项根。
- 形式积分除以 `i+1`，使用模数 `998244353` 时必须确保需要的所有分母可逆，常见条件 `n < 998244353`。
- `Poly<i64>` 的基础加减乘可用，但**FPS 求逆、除法、积分、对数等必须在合适的域上使用**；整数截断除法不满足要求。
- 普通多项式的 `/` 和 `%` 与 FPS `inv()` 是不同操作；普通除法要求 `g` 非零且最高项可逆。
- 多项式采用两种零的表示：空数组与固定长度全零。`norm()` 会去掉所有最高零；FPS 截断通常保留固定长度。
- 只需基础运算可以删除 `mod_sqrt`、两种 `sqrt`、`divmod`、`deriv/integr/log/exp/pow`、`mulT`；保留所需接口依赖链。

#### 常见用法

```cpp
Poly<Z> f{1, 2, 3}, g{1, 1};
auto sum = f + g;
auto mul = f * g;
auto val = f.eval(Z(2));
auto pre = f.trunc(5);             // {1,2,3,0,0}
auto inv = g.inv(5);              // (1+x)^(-1) mod x^5
auto lg = g.log(5);               // log(1+x)
auto ex = Poly<Z>{0, 1}.exp(5);  // exp(x)
auto power = g.pow(1000000007LL, 5);
auto [q, r] = divmod(f, g);
```

求平方根时有两个版本：

```cpp
Poly<Z> sq{1, 2, 1};
auto a = sq.sqrt(3, Z(1)); // 手动给常数项根，少一次模 sqrt
// auto b = sq.sqrt(3);    // 自动选择一个根（仅 Z）
```

应用中最容易遗漏的前导零处理（求 $f^k$）：

```cpp
Poly<Z> f{0, 0, 2, 3};
auto g = f.pow(3, 10); // 前 6 项为 0，常数系数移位由 pow 自动处理
```

#### 复杂度

- `trunc/shift/rev/norm/deriv/integr/eval`：$O(n)$；`+/-`：$O(n)$；多项式乘法：$O(M(n))$。
- `inv/sqrt/log/exp/pow/divmod`：$O(M(n))$ 量级（常数不同）；`mulT`：$O(M(n))$；$M(n)=O(n \log n)$ 为 NTT 卷积。
