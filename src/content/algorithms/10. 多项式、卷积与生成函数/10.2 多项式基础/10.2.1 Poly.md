### Poly
---
依赖已有：
```cpp
using Z = MInt<998244353>;
template<class T>
vector<T> conv(const vector<T> &a, const vector<T> &b);
```
其中 `conv()` 使用前面的统一卷积后端：`vector<Z>` 自动使用 NTT，其余类型使用对应的通用卷积
```cpp
// ===== 模平方根：仅 Poly<Z>::sqrt(n) 自动开方依赖 =====
// 若不需要自动 FPS 平方根，可删除本函数与 sqrt(n)
optional<Z> mod_sqrt(Z a) {
    constexpr int P{ Z::mod() };
    if (a == Z(0)) return Z(0);
    if (a.pow((P - 1) / 2) != Z(1)) return nullopt;
    if (P % 4 == 3) return a.pow((P + 1) / 4);
    int s{ countr_zero(unsigned(P - 1)) };
    i64 q{ (P - 1) >> s };
    Z z{ 2 };
    while (z.pow((P - 1) / 2) == Z(1)) z += Z(1);
    Z c{ z.pow(q) }, x{ a.pow((q + 1) / 2) }, t{ a.pow(q) };
    int m{ s };
    while (t != Z(1)) {
        int i{ 1 };
        Z tt{ t * t };
        for (; i < m && tt != Z(1); ++ i) tt *= tt;
        assert(i < m);
        Z b{ c.pow(1LL << (m - i - 1)) };
        x *= b;
        c = b * b;
        t *= c;
        m = i;
    }
    if ((-x).val() < x.val()) x = -x;
    return x;
}
template<class T = Z>
struct Poly {
    vector<T> a;
    Poly() = default;
    explicit Poly(int n) : a(n) {}
    explicit Poly(vector<T> a_) : a(move(a_)) {}
    Poly(initializer_list<T> a_) : a(a_) {}
    // ===== 基础容器：几乎所有功能依赖，不建议删除 =====
    int size() const { return int(a.size()); }
    bool empty() const { return a.empty(); }
    T operator[](int i) const { return 0 <= i && i < size() ? a[i] : T{}; }
    T &operator[](int i) { return a[i]; }
    void resize(int n) { a.resize(n); }
    // 删除最高次连续零项；divmod() 依赖
    void norm() {
        while (!a.empty() && a.back() == T{}) a.pop_back();
    }
    // 截断为恰好 n 项；不足补零，只复制实际需要的前缀
    Poly trunc(int n) const {
        assert(n >= 0);
        Poly f(n);
        copy_n(a.begin(), min(n, size()), f.a.begin());
        return f;
    }
    // 乘 x^k；k < 0 时删除前 -k 项
    // sqrt() / pow() / mulT() 依赖
    Poly shift(int k) const {
        if (k >= 0) {
            Poly f(size() + k);
            copy(a.begin(), a.end(), f.a.begin() + k);
            return f;
        }
        int d{ -k };
        if (d >= size()) return Poly();
        return Poly(vector<T>(a.begin() + d, a.end()));
    }
    // 反转系数；指定 n 时先截断 / 补零到 n 项再反转
    // divmod() / mulT() 依赖
    Poly rev(int n = -1) const {
        Poly f{ n < 0 ? *this : trunc(n) };
        ranges::reverse(f.a);
        return f;
    }
    // 单点求值；完全独立
    T eval(T x) const {
        T res{};
        for (int i = size() - 1; i >= 0; -- i) res = res * x + a[i];
        return res;
    }
    // ===== 加减：多数 FPS 与多项式除法依赖 =====
    Poly &operator+=(const Poly &g) {
        if (g.size() > size()) a.resize(g.size());
        for (int i = 0; i < g.size(); ++ i) a[i] += g.a[i];
        return *this;
    }
    Poly &operator-=(const Poly &g) {
        if (g.size() > size()) a.resize(g.size());
        for (int i = 0; i < g.size(); ++ i) a[i] -= g.a[i];
        return *this;
    }
    friend Poly operator+(Poly f, const Poly &g) { return f += g; }
    friend Poly operator-(Poly f, const Poly &g) { return f -= g; }
    friend Poly operator-(Poly f) {
        for (T &x : f.a) x = -x;
        return f;
    }
    // ===== 标量乘除：sqrt() / pow() 等依赖 =====
    Poly &operator*=(const T &k) {
        for (T &x : a) x *= k;
        return *this;
    }
    Poly &operator/=(const T &k) {
        for (T &x : a) x /= k;
        return *this;
    }
    friend Poly operator*(Poly f, const T &k) { return f *= k; }
    friend Poly operator*(const T &k, Poly f) { return f *= k; }
    friend Poly operator/(Poly f, const T &k) { return f /= k; }
    // ===== 多项式乘法：所有快速 FPS / divmod / mulT 依赖 =====
    friend Poly operator*(const Poly &f, const Poly &g) {
        return Poly(::conv(f.a, g.a));
    }
    Poly &operator*=(const Poly &g) { return *this = *this * g; }
    // ===== FPS 逆：log / sqrt / divmod 依赖 =====
    Poly inv(int n) const {
        assert(n >= 1 && !empty() && a[0] != T{});
        Poly x{ T(1) / a[0] };
        for (int m = 1; m < n; m <<= 1) {
            int k{ min(m << 1, n) };
            x = (x * (Poly{ T(2) } - trunc(k) * x)).trunc(k);
        }
        return x.trunc(n);
    }
    // ===== FPS 平方根 =====
    // 已知常数项平方根 s0；要求 a[0] != 0
    Poly sqrt(int n, T s0) const {
        assert(n >= 1 && !empty());
        assert(a[0] != T{} && s0 != T{} && s0 * s0 == a[0]);
        Poly x{ s0 };
        T inv2{ T(1) / T(2) };
        for (int m = 1; m < n; m <<= 1) {
            int k{ min(m << 1, n) };
            x = ((x + (trunc(k) * x.inv(k)).trunc(k)) * inv2).trunc(k);
        }
        return x.trunc(n);
    }
    // Z 专用：支持前导零，自动求最低非零项的模平方根
    Poly sqrt(int n) const {
        static_assert(is_same_v<T, Z>);
        assert(n >= 1);
        int t{ 0 };
        while (t < min(size(), n) && a[t] == T{}) ++ t;
        if (t == min(size(), n)) return Poly(n);
        assert(t % 2 == 0);
        int s{ t >> 1 }, m{ n - 2 * s };
        Poly h{ shift(-2 * s).trunc(m) };
        auto s0{ mod_sqrt(h.a[0]) };
        assert(s0.has_value());
        return h.sqrt(m, *s0).shift(s).trunc(n);
    }
    // ===== 多项式除法 =====
    // 依赖 norm / rev / inv / 多项式乘法 / 加减
    friend pair<Poly, Poly> divmod(Poly f, Poly g) {
        f.norm();
        g.norm();
        assert(!g.empty());
        if (f.size() < g.size()) return { Poly(), f };
        int n{ f.size() }, m{ g.size() }, k{ n - m + 1 };
        Poly rf{ f.rev() };
        rf.resize(k);
        Poly q{ (rf * g.rev().inv(k)).trunc(k).rev() };
        Poly r{ f - q * g };
        r.resize(m - 1);
        q.norm();
        r.norm();
        return { q, r };
    }
    friend Poly operator/(Poly f, const Poly &g) {
        return divmod(move(f), g).first;
    }
    friend Poly operator%(Poly f, const Poly &g) {
        return divmod(move(f), g).second;
    }
    // ===== 微积分与 log / exp / pow =====
    // deriv / integr：log 依赖
    // log：inv 依赖
    // exp：log 依赖
    // pow：log + exp + shift + 标量运算依赖
    Poly deriv() const {
        if (size() <= 1) return Poly();
        Poly f(size() - 1);
        for (int i = 1; i < size(); ++ i) f.a[i - 1] = a[i] * T(i);
        return f;
    }
    Poly integr() const {
        Poly f(size() + 1);
        for (int i = 0; i < size(); ++ i) f.a[i + 1] = a[i] / T(i + 1);
        return f;
    }
    Poly log(int n) const {
        assert(n >= 1 && !empty() && a[0] == T(1));
        return (deriv() * inv(n)).integr().trunc(n);
    }
    Poly exp(int n) const {
        assert(n >= 1 && (empty() || a[0] == T{}));
        Poly x{ T(1) };
        for (int m = 1; m < n; m <<= 1) {
            int k{ min(m << 1, n) };
            x = (x * (Poly{ T(1) } - x.log(k) + trunc(k))).trunc(k);
        }
        return x.trunc(n);
    }
    Poly pow(i64 k, int n) const {
        assert(k >= 0 && n >= 1);
        if (k == 0) {
            Poly f(n);
            f.a[0] = T(1);
            return f;
        }
        int t{ 0 };
        while (t < min(size(), n) && a[t] == T{}) ++ t;
        if (t == min(size(), n) || 1LL * t * k >= n) return Poly(n);
        T c{ a[t] };
        int m{ n - 1LL * t * k };
        Poly f{ shift(-t) / c };
        f = (f.log(m) * T(k)).exp(m);
        T ck{ 1 }, base{ c };
        for (i64 e{ k }; e; e >>= 1, base *= base) {
            if (e & 1) ck *= base;
        }
        return (f * ck).shift(int(1LL * t * k)).trunc(n);
    }
    // ===== 转置乘法：主要供多点求值 / 插值等高级模块使用 =====
    // 仅依赖 rev / shift / 多项式乘法
    Poly mulT(const Poly &g) const {
        if (g.empty()) return Poly();
        return ((*this) * g.rev()).shift(1 - g.size());
    }
};
```
#### 依赖与卷积后端
`Poly` 不再维护自己的 NTT
所有多项式乘法统一调用：
```cpp
conv(f.a, g.a);
```
因此：
```cpp
Poly<Z>
```
自动使用前面已经实现的 `998244353` NTT，而：
```cpp
Poly<i64>
Poly<long double>
```
等类型使用 `conv<T>()` 对应的通用后端
这样 FFT、NTT、任意模卷积等底层实现都只维护一份
#### 基础容器
系数始终按低次到高次保存：
```cpp
Poly f{ a0, a1, a2 };
```
表示：
$$
f(x)=a_0+a_1x+a_2x^2
$$
主要接口：
* `Poly(n)`：构造恰好 `n` 个零系数
* `Poly(vector<T>)`：从系数数组构造
* `Poly{a0, a1, ...}`：初始化列表构造
* `size()`：当前**存储长度**
* `empty()`：是否没有存储系数
* `operator[](i) const`：读取系数，越界返回 `0`
* `operator[](i)`：返回系数引用，要求 $0\le i<size()$
* `resize(n)`：修改存储长度
* `norm()`：删除最高端连续零系数
`size()` 不一定等于次数加一；只有 `norm()` 后才保证末尾不存在无意义的零项
##### `trunc(n)`
```cpp
Poly g = f.trunc(n);
```
返回：
$$
f(x)\bmod x^n
$$
并保证结果**恰好存储 `n` 项**
如果原长度不足则补零，如果原长度超过 `n` 则只复制前 `n` 项
这一点对所有 FPS 接口很重要：返回长度统一等于请求的截断长度
##### `shift(k)`
当 $k\ge0$：
$$
f(x)\longmapsto x^k f(x)
$$
当 $k<0$ 时删除最低的 $-k$ 项：
$$
\sum_i a_ix^i
\longmapsto
\sum_{i\ge-k}a_ix^{i+k}
$$
例如：
```cpp
Poly f{ 1, 2, 3 };
f.shift(2);   // {0, 0, 1, 2, 3}
f.shift(-1);  // {2, 3}
```
##### `rev(n)`
```cpp
Poly g = f.rev();
```
反转当前全部系数
指定长度：
```cpp
Poly g = f.rev(n);
```
则先执行：
```cpp
f.trunc(n)
```
再反转
主要供多项式除法、转置乘法等高级操作使用
##### `eval(x)`
使用 Horner 法计算：
$$
f(x)
$$
```cpp
T y = f.eval(x);
```
时间复杂度为 $O(|f|)$
##### 基本运算
支持：
```cpp
f + g
f - g
-f

f * g
f *= g

f * k
k * f
f / k
f *= k
f /= k
```
其中：
```cpp
f * g
```
自动调用统一的：
```cpp
conv()
```
后端
##### FPS
所有 FPS 接口中的 `n` 都表示结果模：
$$
x^n
$$
并返回恰好 `n` 项
##### `inv(n)`
```cpp
Poly g = f.inv(n);
```
满足：
$$
f(x)g(x)\equiv1\pmod{x^n}
$$
要求：
* `n >= 1`
* `f[0]` 存在且可逆
使用 Newton 迭代：
$$
g\leftarrow g(2-fg)
$$
##### `sqrt(n, s0)`
已知：
$$
s_0^2=f_0
$$
时：
```cpp
Poly g = f.sqrt(n, s0);
```
求：
$$
g(x)^2\equiv f(x)\pmod{x^n}
$$
要求：
* `n >= 1`
* $f_0\ne0$
* $s_0\ne0$
* $s_0^2=f_0$
* `2` 可逆
使用 Newton 迭代：
$$
g\leftarrow\frac12\left(g+\frac fg\right)
$$
##### `sqrt(n)`
`Z` 专用自动版本：
```cpp
Poly g = f.sqrt(n);
```
允许常数项为 `0`
设最低非零项为：
$$
a_tx^t
$$
存在平方根必须满足：
$$
t\equiv0\pmod2
$$
且 $a_t$ 在模 `Z::mod()` 下存在平方根
本接口自动使用 `mod_sqrt()` 求最低非零项平方根；条件不满足时触发断言。
如果：
$$
f(x)\equiv0\pmod{x^n}
$$
则返回长度 `n` 的零多项式
##### `deriv()`
形式导数：
$$
\left(\sum_{i\ge0}a_ix^i\right)'=\sum_{i\ge1}ia_ix^{i-1}
$$
##### `integr()`
常数项取 `0` 的形式积分：
$$
\int f(x)\,dx=\sum_{i\ge0}\frac{a_i}{i+1}x^{i+1}
$$
要求所有出现的 $i+1$ 在系数体系中可逆
对于默认 `Z`，通常要求需要处理的次数小于模数
##### `log(n)`
```cpp
Poly g = f.log(n);
```
要求：
$$
f_0=1
$$
使用：
$$
\log f=\int\frac{f'}f
$$
##### `exp(n)`
```cpp
Poly g = f.exp(n);
```
要求：
$$
f_0=0
$$
返回：
$$
\exp f(x)\pmod{x^n}
$$
使用 Newton 形式：
$$
g\leftarrow
g\left(1-\log g+f\right)
$$
 ##### `pow(k, n)`
```cpp
Poly g = f.pow(k, n);
```
计算：
$$
f(x)^k\bmod x^n
$$
要求：
$$
k\ge0
$$
设最低非零项为：
$$
f(x)=c x^t h(x), \quad h(0)=1
$$
则：
$$
f(x)^k=c^k x^{tk}\exp(k\log h(x))
$$
如果：
$$
tk\ge n
$$
直接返回长度 `n` 的零多项式
特别地：
```cpp
f.pow(0, n)
```
始终返回：
$$
1\pmod{x^n}
$$
包括 `f` 为零多项式的情况
##### 多项式除法
```cpp
auto [q, r] = divmod(f, g);
```
满足：
$$
f=qg+r,
\qquad
\deg r<\deg g.
$$
也支持：
```cpp
Poly q = f / g;
Poly r = f % g;
```
要求：
* `g` 不是零多项式
* `g` 的最高次项可逆
实现通过反转多项式，把高次项除法转化为 FPS 求逆
当：
$$
n=|f|,\qquad m=|g|
$$
且 $n\ge m$ 时，只需要求商的：
$$
n-m+1
$$
项，因此实现会先截掉反转后的 `f` 中不影响商的部分，避免无意义的大卷积
##### `mulT(g)`
```cpp
Poly h = f.mulT(g);
```
计算转置卷积形式：
$$
h=\left(f\cdot\operatorname{rev}(g)\right)\operatorname{shift}(1-|g|)
$$
等价于：
```cpp
return (f * g.rev()).shift(1 - g.size());
```
若 `g` 非空，结果长度与 `f.size()` 一致
主要作为多点求值、插值等高级多项式算法的底层接口
#### 注意事项
`Poly<T>` 的基础容器、加减乘和求值只要求 `T` 支持对应运算；但 FPS 操作通常工作在域或至少需要相关元素可逆的环中。因此：
```cpp
Poly<i64>
```
适合：
```text
加减
普通乘法
shift / trunc / rev
eval
```
但通常不能直接使用：
```text
inv / integr / log / exp / sqrt
```
因为整数除法不是形式幂级数需要的域除法
对于默认 `Poly<Z>`，快速乘法继承已有 `conv()` 的 NTT 限制。如果某次乘法补齐后的 NTT 长度超过 `NTT::MAX_N`，普通 `conv()` 同样无法直接处理；需要另行选择适合超长卷积的后端
自动 `sqrt(n)` 使用 `mod_sqrt()`，因此默认假设 `Z::mod()` 为奇素数
`norm()` 会改变存储长度，而 `trunc(n)` 始终保证固定长度。FPS 计算通常应使用 `trunc()`，多项式次数判断和除法再使用 `norm()`
#### 常见用法
##### 多项式乘法
```cpp
Poly f{ Z(1), Z(2), Z(3) };
Poly g{ Z(4), Z(5) };
Poly h = f * g;
```
对应：
$$
(1+2x+3x^2)(4+5x)
$$
默认使用已有 NTT 后端
普通整数类型：
```cpp
Poly<i64> f{ 1, 2, 3 };
Poly<i64> g{ 4, 5 };
auto h = f * g;
// {4, 13, 22, 15}
```
使用 `conv<i64>()` 对应的通用后端
##### 求逆
```cpp
Poly f{ Z(2), Z(3), Z(1) };
Poly g = f.inv(n);
```
满足：
$$
fg\equiv1\pmod{x^n}
$$
##### 对数与指数
```cpp
Poly f{ Z(1), Z(2), Z(3) };
Poly lf = f.log(n);
```
以及：
```cpp
Poly f{ Z(0), Z(2), Z(3) };
Poly ef = f.exp(n);
```
满足：
$$
\exp(\log f)\equiv f\pmod{x^n}
$$
或：
$$
\log(\exp f)\equiv f\pmod{x^n}
$$
前提是对应常数项条件成立
##### 整数幂
```cpp
Poly f{ Z(0), Z(2), Z(3) };
Poly g = f.pow(k, n);
```
得到：
$$
f^k\bmod x^n
$$
##### FPS 平方根
自动选择最低非零项的一个模平方根：
```cpp
Poly g = f.sqrt(n);
```
如果已经知道常数项平方根：
```cpp
Z s0 = ...;
Poly g = f.sqrt(n, s0);
```
后者不依赖 `mod_sqrt()`
##### 多项式除法
```cpp
auto [q, r] = divmod(f, g);
```
或者：
```cpp
Poly q = f / g;
Poly r = f % g;
```
满足：
$$
f=qg+r \quad \deg r<\deg g
$$
#### 复杂度
记长度为 $n$ 的多项式乘法复杂度为：
$$
M(n)
$$
使用当前 `Z + NTT` 后端时：
$$
M(n)=O(n\log n)
$$
若某个其他系数类型使用朴素卷积：
$$
M(n)=O(n^2)
$$
Newton 迭代每次将正确长度翻倍，因此在通常的卷积复杂度条件下：
$$
M(1)+M(2)+M(4)+\cdots+M(n)=O(M(n))
$$
所以各 FPS 操作保持一次乘法量级的渐进复杂度

| 操作             |                 时间复杂度 |
| -------------- | --------------------: |
| `trunc(n)`     |                $O(n)$ |
| `shift(k)`     |          $O(\|f\|+k)$ |
| `rev()`        |            $O(\|f\|)$ |
| `rev(n)`       |                $O(n)$ |
| `eval(x)`      |            $O(\|f\|)$ |
| `+ / -`        | $O(\max(\|f\|,\|g\|)$ |
| 标量乘除           |            $O(\|f\|)$ |
| `f * g`        |   $O(M(\|f\|+\|g\|))$ |
| `deriv()`      |            $O(\|f\|)$ |
| `integr()`     |            $O(\|f\|)$ |
| `inv(n)`       |             $O(M(n))$ |
| `log(n)`       |             $O(M(n))$ |
| `exp(n)`       |             $O(M(n))$ |
| `sqrt(n)`      |             $O(M(n))$ |
| `pow(k, n)`    |      $O(M(n)+\log k)$ |
| `divmod(f, g)` |         $O(M(\|f\|))$ |
| `mulT(g)`      |   $O(M(\|f\|+\|g\|))$ |
| `mod_sqrt()`   |      $O(\log^2 P)$ 最坏 |
除底层 NTT 的共享缓存外，FPS Newton 迭代的主要额外空间为：
$$
O(n)
$$
