### FFT
---
```cpp
template<class Real = double>
struct FFT {
    static_assert(is_floating_point_v<Real>);
    using C = complex<Real>;
    inline static int base = 1;
    inline static vector<int> rev{0, 1};
    inline static vector<C> roots{C(0), C(1)};

    static void ensure(int n) {
        int k = countr_zero(unsigned(n));
        if (k <= base) return;
        rev.resize(n);
        for (int i = 0; i < n; ++ i) {
            rev[i] = (rev[i >> 1] >> 1) | ((i & 1) << (k - 1));
        }
        roots.resize(n);
        while (base < k) {
            int m = 1 << base;
            long double angle = acosl(-1.L) / m;
            complex<long double> w(cosl(angle), sinl(angle));
            for (int i = m / 2; i < m; ++ i) {
                roots[2 * i] = roots[i];
                auto z = complex<long double>(roots[i].real(), roots[i].imag()) * w;
                roots[2 * i + 1] = C(Real(z.real()), Real(z.imag()));
            }
            ++ base;
        }
    }
    static void fft(vector<C> &a, bool inverse = false) {
        int n = a.size();
        assert(n > 0 && has_single_bit(unsigned(n)));
        ensure(n);
        int shift = base - countr_zero(unsigned(n));
        for (int i = 0; i < n; ++ i) {
            int j = rev[i] >> shift;
            if (i < j) swap(a[i], a[j]);
        }
        for (int k = 1; k < n; k <<= 1) {
            for (int i = 0; i < n; i += 2 * k) {
                for (int j = 0; j < k; ++ j) {
                    C x = a[i + j], y = a[i + j + k] * roots[k + j];
                    a[i + j] = x + y;
                    a[i + j + k] = x - y;
                }
            }
        }
        if (inverse) {
            reverse(a.begin() + 1, a.end());
            Real inv = Real(1) / n;
            for (C &x : a) x *= inv;
        }
    }
    template<class A, class B>
    static vector<C> conv_complex(const vector<A> &a, const vector<B> &b) {
        if (a.empty() || b.empty()) return {};
        int need = a.size() + b.size() - 1;
        if (1LL * a.size() * b.size() <= 4096) {
            vector<C> c(need);
            for (int i = 0; i < int(a.size()); ++ i) {
                for (int j = 0; j < int(b.size()); ++ j) {
                    c[i + j] += C(a[i]) * C(b[j]);
                }
            }
            return c;
        }
        int n = int(bit_ceil(unsigned(need)));
        vector<C> f(n), g(n);
        for (int i = 0; i < int(a.size()); ++ i) f[i] = C(a[i]);
        for (int i = 0; i < int(b.size()); ++ i) g[i] = C(b[i]);
        fft(f);
        fft(g);
        for (int i = 0; i < n; ++ i) f[i] *= g[i];
        fft(f, true);
        f.resize(need);
        return f;
    }
    // 不打包：使用两次独立正向 FFT，降低额外消去误差。
    template<class A, class B>
    static vector<Real> conv_real(const vector<A> &a, const vector<B> &b) {
        if (a.empty() || b.empty()) return {};
        int need = a.size() + b.size() - 1;
        if (1LL * a.size() * b.size() <= 4096) {
            vector<Real> c(need);
            for (int i = 0; i < int(a.size()); ++ i) {
                for (int j = 0; j < int(b.size()); ++ j) {
                    c[i + j] += Real(a[i]) * Real(b[j]);
                }
            }
            return c;
        }
        auto f = conv_complex(a, b);
        vector<Real> c(f.size());
        for (int i = 0; i < int(f.size()); ++ i) c[i] = f[i].real();
        return c;
    }
    template<class A, class B>
    static vector<i64> conv_round(const vector<A> &a, const vector<B> &b) {
        auto c = conv_real(a, b);
        vector<i64> res(c.size());
        for (int i = 0; i < int(c.size()); ++ i) res[i] = llround(c[i]);
        return res;
    }
};
```

#### 接口说明

- `FFT<Real>::fft(a, inverse = false)`：原地复数 FFT / IFFT，长度须为非零的 2 的幂；$O(n\log n)$。
- `FFT<Real>::conv_complex(a, b)`：复系数卷积，支持实数或复数输入；$O(N\log N)$。
- `FFT<Real>::conv_real(a, b)`：实系数卷积，采用两次独立正向 FFT，不使用复数打包；$O(N\log N)$。
- `FFT<Real>::conv_round(a, b)`：对实系数卷积结果执行 `llround`，返回 `vector<i64>`；$O(N\log N)$。

#### 注意事项

- 所有卷积均为**浮点近似计算**。即使不打包，`conv_round()` 也不能保证对任意大整数输入都精确；若需模数卷积的严格正确性，请使用 NTT / 合法范围内的 CRT。
- `conv_round()` 的逐项正确性要求计算误差不越过半整数取整边界，真实答案和取整输入必须落在 `i64` 可表示范围内。
- 旋转因子预处理在 `long double` 精度下计算，再存为 `Real`；这只降低误差，**不是精度保证**。
- `fft()` 会修改输入；`conv_*()` 返回完整长度 $|a|+|b|-1$ 的卷积，空数组返回空数组。
- 小规模 $|a||b|\le4096$ 直接朴素卷积；复用缓存的 `rev`、`roots` 按本模板实例使用过的最大变换长度保留内存。
- 默认 `double` 偏快；大整数近似卷积可考虑 `long double`，但若题目要求严格整数结果，优先选择有数学精确性保证的方法。

#### 常见用法

```cpp
vector<int> a{1, 2, 3}, b{4, 5};
auto c = FFT<>::conv_round(a, b);         // {4, 13, 22, 15}
auto r = FFT<>::conv_real(a, b);          // vector<double>
auto hi = FFT<long double>::conv_round(a, b);

using C = complex<double>;
auto z = FFT<>::conv_complex(vector<C>{{1, 1}}, vector<C>{{2, 3}});
// z[0] = -1 + 5i

vector<C> f{1, 2, 3, 0};
FFT<>::fft(f);                             // 正向 FFT
FFT<>::fft(f, true);                       // 原地还原原数组
```

#### 复杂度

设 $N=\operatorname{bit\_ceil}(|a|+|b|-1)$。

| 操作 | 时间复杂度 | 额外空间复杂度 |
|---|---|---|
| `fft()` | $O(n\log n)$ | $O(n)$ 缓存（复用） |
| `conv_complex()` | $O(N\log N)$ | $O(N)$ |
| `conv_real()` | $O(N\log N)$ | $O(N)$ |
| `conv_round()` | $O(N\log N)$ | $O(N)$ |

当 $|a||b|\le4096$ 时，卷积使用 $O(|a||b|)$ 朴素算法。
