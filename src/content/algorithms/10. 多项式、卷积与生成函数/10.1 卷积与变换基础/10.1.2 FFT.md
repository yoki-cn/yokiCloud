### FFT
---
```cpp
template<class Real = double>
struct FFT {
    static_assert(is_floating_point_v<Real>);
    using C = complex<Real>;
    using HC = complex<long double>;
    static constexpr i64 BRUTE{ 4096 };
    inline static int base{ 1 };
    inline static vector<int> rev{ 0, 1 };
    inline static vector<C> roots{ C(0), C(1) };
    inline static vector<HC> hroots{ HC(0), HC(1) };
    inline static constexpr long double PI{ numbers::pi_v<long double> };
    vector<C> fa, fb;
    static C mul(C a, C b) {
        return C(a.real() * b.real() - a.imag() * b.imag(), a.real() * b.imag() + a.imag() * b.real());
    }
    static int fft_size(int need) { return bit_ceil(unsigned(need)); }
    static void ensure(int n) {
        int nbase{ countr_zero(unsigned(n)) }; // __builtin_ctz(n)
        if (nbase <= base) return;
        rev.resize(n);
        for (int i = 0; i < n; ++ i) rev[i] = (rev[i >> 1] >> 1) | ((i & 1) << (nbase - 1));
        roots.resize(n);
        hroots.resize(n);
        while (base < nbase) {
            long double angle{ PI / (1LL << base) };
            HC z(cosl(angle), sinl(angle));
            for (int i = 1 << (base - 1); i < (1 << base); ++ i) {
                hroots[i << 1] = hroots[i];
                hroots[i << 1 | 1] = hroots[i] * z;
                roots[i << 1] = C(hroots[i << 1]);
                roots[i << 1 | 1] = C(hroots[i << 1 | 1]);
            }
            ++ base;
        }
    }
    static void fft(vector<C> &a, bool inverse = false) {
        int n = a.size();
        assert(n && (n & (n - 1)) == 0);
        ensure(n);
        if (inverse) reverse(a.begin() + 1, a.end());
        int shift{ base - countr_zero(unsigned(n)) };
        for (int i = 0; i < n; ++ i) {
            int j{ rev[i] >> shift };
            if (i < j) swap(a[i], a[j]);
        }
        for (int k = 1; k < n; k <<= 1) {
            for (int i = 0; i < n; i += k << 1) {
                for (int j = 0; j < k; ++ j) {
                    C u{ a[i + j] }, v{ mul(a[i + j + k], roots[j + k]) };
                    a[i + j] = u + v;
                    a[i + j + k] = u - v;
                }
            }
        }
        if (inverse) {
            Real inv{ Real(1) / n };
            for (C &x : a) x *= inv;
        }
    }
    template<class A, class B>
    vector<Real> conv_real(const vector<A> &a, const vector<B> &b) {
        static_assert(is_arithmetic_v<A> && is_arithmetic_v<B>);
        if (a.empty() || b.empty()) return {};
        int na = a.size(), nb = b.size(), need{ na + nb - 1 };
        if (i64(na) * nb <= BRUTE) {
            vector<Real> c(need);
            for (int i = 0; i < na; ++ i) {
                for (int j = 0; j < nb; ++ j) {
                    c[i + j] += Real(a[i]) * Real(b[j]);
                }
            }
            return c;
        }
        int n{ fft_size(need) };
        fa.assign(n, C{});
        fb.assign(n, C{});
        for (int i = 0; i < na; ++ i) fa[i] = Real(a[i]);
        for (int i = 0; i < nb; ++ i) fb[i] = Real(b[i]);
        fft(fa);
        fft(fb);
        for (int i = 0; i < n; ++ i) fa[i] = mul(fa[i], fb[i]);
        fft(fa, true);
        vector<Real> c(need);
        for (int i = 0; i < need; ++ i) c[i] = fa[i].real();
        return c;
    }
    template<class A, class B>
    vector<C> conv_complex(const vector<A> &a, const vector<B> &b) {
        if (a.empty() || b.empty()) return {};
        int na = a.size(), nb = b.size(), need{ na + nb - 1 };
        if (i64(na) * nb <= BRUTE) {
            vector<C> c(need);
            for (int i = 0; i < na; ++ i) {
                for (int j = 0; j < nb; ++ j) c[i + j] += mul(C(a[i]), C(b[j]));
            }
            return c;
        }
        int n{ fft_size(need) };
        fa.assign(n, C{});
        fb.assign(n, C{});
        for (int i = 0; i < na; ++ i) fa[i] = C(a[i]);
        for (int i = 0; i < nb; ++ i) fb[i] = C(b[i]);
        fft(fa);
        fft(fb);
        for (int i = 0; i < n; ++ i) fa[i] = mul(fa[i], fb[i]);
        fft(fa, true);
        return vector<C>(fa.begin(), fa.begin() + need);
    }
    template<class A, class B>
    vector<i64> conv_round(const vector<A> &a, const vector<B> &b) {
        static_assert(is_integral_v<A> && is_integral_v<B>);
        vector<Real> c{ conv_real(a, b) };
        vector<i64> res(c.size());
        for (int i = 0; i < c.size(); ++ i) res[i] = llround(c[i]);
        return res;
    }
    // ===== 任意模整数卷积 =====
    // 高低位拆分 MTT；相比 conv_round() 可处理更大系数，但仍属于浮点算法
    // 若要求严格无浮点误差，优先使用多模 NTT + CRT
    template<class T>
    vector<int> conv_mod(const vector<T> &a, const vector<T> &b, int mod) {
        static_assert(is_integral_v<T>);
        assert(mod > 0);
        if (a.empty() || b.empty()) return {};
        int na = a.size(), nb = b.size(), need{ na + nb - 1 };
        if (mod == 1) return vector<int>(need);
        auto norm = [&](T x) -> i64 {
            i128 v{ x };
            v %= mod;
            if (v < 0) v += mod;
            return i64(v);
        };
        if (i64(na) * nb <= BRUTE) {
            vector<int> c(need);
            for (int i = 0; i < na; ++ i) {
                i64 x{ norm(a[i]) };
                for (int j = 0; j < nb; ++ j) {
                    i64 y{ norm(b[j]) };
                    c[i + j] = (i128(c[i + j]) + i128(x) * y) % mod;
                }
            }
            return c;
        }
        long double root_mod{ sqrt((long double)mod) };
        i64 cut{ i64(root_mod) };
        while (i128(cut + 1) * (cut + 1) <= mod) ++ cut;
        while (i128(cut) * cut > mod) -- cut;
        int n{ fft_size(need) };
        fa.assign(n, C{});
        fb.assign(n, C{});
        for (int i = 0; i < na; ++ i) {
            i64 x{ norm(a[i]) };
            fa[i] = C(Real(x / cut), Real(x % cut));
        }
        for (int i = 0; i < nb; ++ i) {
            i64 x{ norm(b[i]) };
            fb[i] = C(Real(x / cut), Real(x % cut));
        }
        fft(fa);
        fft(fb);
        Real scale{ Real(0.5) / n };
        for (int i = 0; i <= n / 2; ++ i) {
            int j{ (-i) & (n - 1) };
            C ai{ fa[i] }, aj{ fa[j] }, bi{ fb[i] }, bj{ fb[j] };
            C ahi{ (ai + conj(aj)) * scale }, dli{ ai - conj(aj) };
            C ali{ dli.imag() * scale, -dli.real() * scale };
            C ahj{ (aj + conj(ai)) * scale }, dlj{ aj - conj(ai) };
            C alj{ dlj.imag() * scale, -dlj.real() * scale };
            fa[j] = mul(ahi, bi);
            fb[j] = mul(ali, bi);
            if (i != j) {
                fa[i] = mul(ahj, bj);
                fb[i] = mul(alj, bj);
            }
        }
        fft(fa);
        fft(fb);
        vector<int> res(need);
        for (int i = 0; i < need; ++ i) {
            i64 hi{ llround(fa[i].real()) };
            i64 mid{ llround(fa[i].imag()) + llround(fb[i].real()) };
            i64 lo{ llround(fb[i].imag()) };
            i128 value{ (i128(hi) * cut + mid) % mod };
            value = (value * cut + lo) % mod;
            if (value < 0) value += mod;
            res[i] = int(value);
        }
        return res;
    }
};
```
#### 接口说明
##### 成员变量
* `Real`：FFT 使用的浮点类型，默认为 `double`，也可使用 `long double`
* `C`：`complex<Real>` 的别名，FFT 实际使用的复数类型
* `HC`：`complex<long double>` 的别名，仅用于更高精度地生成单位根
* `BRUTE`：朴素卷积阈值；当 $|a||b|\leq \text{BRUTE}$ 时直接使用 $O(|a||b|)$ 卷积，当前为 `4096`
* `base`：当前已经预处理到的最大 FFT 层数，可支持长度不超过 $2^{base}$ 的变换
* `rev`：位逆序数组，同一 `FFT<Real>` 类型的所有对象共享
* `roots`：实际 FFT 使用的单位根，同一 `FFT<Real>` 类型的所有对象共享
* `hroots`：使用 `long double` 保存的高精度单位根，用于生成 `roots`
* `PI`：高精度圆周率
* `fa`、`fb`：卷积使用的两个工作数组，对象内部复用
`base`、`rev`、`roots`、`hroots` 为 `inline static`，只会随着遇到的最大 FFT 长度增长；同一 `FFT<Real>` 的多个对象共享这套缓存
不同 `Real` 实例化之间缓存独立，例如 `FFT<double>` 与 `FFT<long double>` 分别维护自己的预处理结果
##### 接口
* `FFT<Real>`：FFT 封装，`Real` 默认为 `double`
* `mul(a, b)`：复数乘法，$O(1)$
* `fft_size(need)`：返回不小于 `need` 的最小二次幂，$O(1)$；通常要求 `need >= 1`
* `ensure(n)`：保证长度 `n` 对应的位逆序和单位根已经预处理；首次扩展到 `n` 时为 $O(n)$
* `fft(a, inverse)`：原地执行 FFT / IFFT，$O(n\log n)$；要求 `a.size() > 0` 且为 $2$ 的幂
* `conv_real(a, b)`：普通实数卷积，返回 `vector<Real>`，$O(N\log N)$
* `conv_complex(a, b)`：复数卷积，返回 `vector<C>`，$O(N\log N)$
* `conv_round(a, b)`：整数卷积，FFT 后使用 `llround()` 恢复整数，返回 `vector<i64>`，$O(N\log N)$
* `conv_mod(a, b, mod)`：任意正模数整数卷积，使用高低位拆分 MTT，返回 `vector<int>`，$O(N\log N)$；要求 `mod > 0`
其中卷积长度
$$
L=|a|+|b|-1,
$$
FFT 长度
$$
N=2^{\lceil\log_2 L\rceil}.
$$
当 $|a||b|\leq\text{BRUTE}$ 时，各卷积接口自动改用朴素卷积
##### `fft(a, inverse)`
原地进行离散傅里叶变换：
```cpp
FFT<>::fft(a);        // FFT
FFT<>::fft(a, true);  // IFFT，并自动除以 a.size()
```
要求：
* `a` 非空
* `a.size()` 为 $2$ 的幂
若需要手动计算卷积：
```cpp
FFT<>::fft(a);
FFT<>::fft(b);
for (int i = 0; i < n; ++ i) a[i] = FFT<>::mul(a[i], b[i]);
FFT<>::fft(a, true);
```
普通卷积通常直接调用下面的封装接口即可
##### `conv_real(a, b)`
计算普通卷积
$$
c_k=\sum_{i+j=k}a_i b_j
$$
```cpp
FFT<> F;
vector<double> c = F.conv_real(a, b);
```
* 输入元素必须为算术类型
* 返回长度为 `a.size() + b.size() - 1`
* 任一输入为空时返回空数组
* 返回值为浮点近似值
##### `conv_complex(a, b)`
计算复数序列卷积：
```cpp
FFT<> F;
auto c = F.conv_complex(a, b);
```
输入元素需要能够转换为 `C`，返回 `vector<FFT<>::C>`
FFT 分支分别变换两个序列，逐点相乘后进行逆变换
##### `conv_round(a, b)`
计算整数卷积：
```cpp
FFT<> F;
vector<i64> c = F.conv_round(a, b);
```
输入元素必须为整数类型
内部先调用 `conv_real()`，再使用 `llround()` 将每个系数恢复到最近整数，因此支持正数、负数和 `0`，但正确性受到浮点误差限制
##### `conv_mod(a, b, mod)`
计算
$$
c_k=\sum_{i+j=k}a_i b_j\pmod {mod}.
$$
```cpp
FFT<> F;
vector<int> c = F.conv_mod(a, b, mod);
```
要求：
* `a`、`b` 的元素为整数类型
* `mod > 0`
输入会先规范到 $[0,mod)$，因此可以直接传入负数；返回值也均位于 $[0,mod)$
`mod == 1` 时直接返回全 `0` 数组
该接口按约 $\sqrt{mod}$ 为基数将每个系数拆为高低两部分，再通过 FFT 同时计算各部分卷积。相比直接使用 `conv_round()`，它能够显著降低大系数整数卷积的浮点误差
`mod` 不要求为质数，也不要求存在适用于 NTT 的原根
#### 注意事项
* 本实现使用 `bit_ceil`、`countr_zero`、`numbers::pi_v`，需要 **C++20**
* 若使用 C++17，可分别将它们替换为手动求二次幂、`__builtin_ctz` 和 `acosl(-1)`
* `conv_real()` 与 `conv_complex()` 返回浮点近似值，不应直接使用 `==` 判断理论上的实数结果
* `conv_round()` 只有在真实系数能够存入 `i64`，并且最终浮点误差小于 $0.5$ 时，`llround()` 才能恢复正确答案
* “结果不溢出 `i64`”并不足以保证 `conv_round()` 正确；序列较长或系数较大时仍可能积累明显浮点误差
* `conv_mod()` 通过高低位拆分显著提高整数卷积的可用范围，但本质仍属于浮点算法，不提供数学意义上的绝对精确保证
* 对任意模卷积要求严格无误差时，应使用多模 NTT + CRT
* `FFT<double>` 一般速度更快；精度不足时可改为 `FFT<long double>`，但运行时间和常数通常都会明显增加
* `hroots` 使用 `long double` 生成单位根，再转换为 `C`，用于减小单位根预处理阶段的精度损失
* `base`、`rev`、`roots`、`hroots` 为静态共享缓存，会保留到该 `FFT<Real>` 曾经处理过的最大规模
* `fa`、`fb` 属于当前对象；连续进行卷积时其容量通常可以复用
* 因此长期空间占用主要由历史上出现过的最大 FFT 长度决定
* 当 $|a||b|\leq4096$ 时，各卷积接口自动使用朴素卷积，避免小规模 FFT 的常数开销
#### 常见用法
##### 整数卷积
```cpp
FFT<> F;
vector<i64> a{ 1, 2, 3 };
vector<i64> b{ 4, 5 };
auto c = F.conv_round(a, b);
```
得到：
```text
4 13 22 15
```
对应
$$
(1+2x+3x^2)(4+5x)
=4+13x+22x^2+15x^3
$$
##### 实数卷积
```cpp
FFT<> F;
vector<double> a{ 1.5, 2.0 };
vector<double> b{ 3.0, 4.0 };
auto c = F.conv_real(a, b);
```
结果为浮点近似值
##### 复数卷积
```cpp
using C = FFT<>::C;
FFT<> F;
vector<C> a{ C(1, 2), C(3, 4) };
vector<C> b{ C(5, 6), C(7, 8) };
auto c = F.conv_complex(a, b);
```
##### 任意模卷积
```cpp
FFT<> F;
vector<int> a{ 1, 2, 3 };
vector<int> b{ 4, 5 };
auto c = F.conv_mod(a, b, 1000000007);
```
模数可以是任意正整数，无需具有 NTT 原根，也不要求 `mod - 1` 含有足够大的 $2$ 次幂因子
##### 输入含负数的任意模卷积
```cpp
FFT<> F;
vector<i64> a{ -1, 2, -3 };
vector<i64> b{ 4, -5 };
auto c = F.conv_mod(a, b, 1000000007);
```
无需手动进行取模规范化
##### 精度不足时
```cpp
FFT<long double> F;
auto c = F.conv_round(a, b);
```
只需修改模板参数，其余调用方式不变
##### 手动 FFT
```cpp
using C = FFT<>::C;
int n = FFT<>::fft_size(need);
vector<C> a(n);
for (int i = 0; i < m; ++ i) a[i] = value[i];
FFT<>::fft(a);
// 频域处理
FFT<>::fft(a, true);
```
#### 复杂度
设非空输入长度分别为
$$
n=|a|,\quad m=|b|
$$
卷积结果长度为
$$
L=n+m-1
$$
实际 FFT 长度为
$$
N=2^{\lceil\log_2 L\rceil}
$$
当
$$
nm\leq\text{BRUTE}
$$
时，各卷积接口直接使用朴素卷积，时间复杂度为 $O(nm)$，不进行 FFT
否则：

| 接口                    |        时间复杂度 |                  额外 / 工作空间 |
| --------------------- | -----------: | -------------------------: |
| `mul(a, b)`           |       $O(1)$ |                     $O(1)$ |
| `fft_size(need)`      |       $O(1)$ |                     $O(1)$ |
| `ensure(N)`           |  首次扩展 $O(N)$ |                $O(N)$ 静态缓存 |
| `fft(a, inverse)`     | $O(N\log N)$ | $O(1)$ 临时空间，另有 $O(N)$ 静态缓存 |
| `conv_real(a, b)`     | $O(N\log N)$ |                     $O(N)$ |
| `conv_complex(a, b)`  | $O(N\log N)$ |                     $O(N)$ |
| `conv_round(a, b)`    | $O(N\log N)$ |                     $O(N)$ |
| `conv_mod(a, b, mod)` | $O(N\log N)$ |                     $O(N)$ |
`conv_real()` 和 `conv_complex()` 的 FFT 分支各进行两次正变换和一次逆变换
`conv_round()` 的主要复杂度来自 `conv_real()`，额外的取整过程为 $O(L)$
`conv_mod()` 的 FFT 分支进行两次初始 FFT、$O(N)$ 的频域拆分与组合，再进行两次 FFT 恢复结果，因此渐进复杂度仍为 $O(N\log N)$，但常数高于普通卷积
`rev`、`roots`、`hroots` 会保留到同一 `FFT<Real>` 曾经使用过的最大规模。若历史最大变换长度为 $N_{\max}$，则静态持久缓存空间为 $O(N_{\max})$
