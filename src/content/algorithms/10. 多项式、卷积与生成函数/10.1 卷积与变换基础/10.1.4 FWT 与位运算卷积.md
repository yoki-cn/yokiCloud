### FWT（OR / AND / XOR 卷积）
---
```cpp
namespace FWT {
    template<int type, class T>
    void transform(vector<T> &a, bool inverse = false) {
        int n = a.size();
        assert(n > 0 && has_single_bit(unsigned(n)));
        for (int k = 1; k < n; k <<= 1) {
            for (int i = 0; i < n; i += 2 * k) {
                for (int j = 0; j < k; ++ j) {
                    T &x = a[i + j], &y = a[i + j + k];
                    if constexpr (type == 0) { // OR: 子集 zeta / Möbius
                        if (inverse) y -= x;
                        else y += x;
                    } else if constexpr (type == 1) { // AND: 超集 zeta / Möbius
                        if (inverse) x -= y;
                        else x += y;
                    } else { // XOR: Walsh-Hadamard
                        T u = x, v = y;
                        x = u + v;
                        y = u - v;
                    }
                }
            }
        }
        if constexpr (type == 2) {
            if (inverse) for (T &x : a) x /= T(n);
        }
    }
    template<int type, class T>
    vector<T> conv(const vector<T> &a, const vector<T> &b) {
        if (a.empty() || b.empty()) return {};
        int n = int(bit_ceil(unsigned(max(a.size(), b.size()))));
        vector<T> f(n), g;
        copy(a.begin(), a.end(), f.begin());
        transform<type>(f);
        if (&a == &b) {
            for (T &x : f) x *= x;
        } else {
            g.resize(n);
            copy(b.begin(), b.end(), g.begin());
            transform<type>(g);
            for (int i = 0; i < n; ++ i) f[i] *= g[i];
        }
        transform<type>(f, true);
        return f;
    }
    template<class T>
    vector<T> conv(const vector<T> &a, const vector<T> &b, int type) {
        if (type == 0) return conv<0>(a, b);
        if (type == 1) return conv<1>(a, b);
        assert(type == 2);
        return conv<2>(a, b);
    }
}
template<class T> vector<T> conv_or(const vector<T> &a, const vector<T> &b) {
    return FWT::conv<0>(a, b);
}
template<class T> vector<T> conv_and(const vector<T> &a, const vector<T> &b) {
    return FWT::conv<1>(a, b);
}
template<class T> vector<T> conv_xor(const vector<T> &a, const vector<T> &b) {
    return FWT::conv<2>(a, b);
}
```

#### 接口说明

- `conv_or / conv_and / conv_xor(a,b)`：按位或、按位与、按位异或卷积，结果长度为 `bit_ceil(max(|a|,|b|))`。
- `FWT::transform<type>(a,inverse)`：`type=0/1/2` 对应 OR/AND/XOR，编译期常量让蝶形内不出现动态分支；`inverse=true` 为逆变换。
- `FWT::conv<type>(a,b)`：快速编译期分发；也保留 `FWT::conv(a,b,type)` 运行时入口（三路只在函数入口分支）。

#### 注意事项

- 输入表示 mask `0..2^k-1`；不足长度自动补零。多于 $2^k$ 不会折回。
- XOR 逆变换需要 $2^k$ 在系数环中可逆。使用 `Z` 没问题；**整数 `vector<i64>` 不适合逐步做 `/n` 后再用于非整数代数扩展**，且中间值可能溢出。
- OR 卷积 `a[A]b[B]` 汇总到 `A|B`，不是要求 `A∩B=∅` 的子集卷积。

#### 常见用法

```cpp
vector<Z> a(1 << k), b(1 << k);
auto c0 = conv_or(a, b);
auto c1 = conv_and(a, b);
auto c2 = conv_xor(a, b);
// 只做一次变换、多次频域乘法：
FWT::transform<2>(a);
FWT::transform<2>(a, true);
```

#### 复杂度
- 三种变换及卷积均为 $O(k\cdot 2^k)$ 时间、$O(2^k)$ 额外空间。
