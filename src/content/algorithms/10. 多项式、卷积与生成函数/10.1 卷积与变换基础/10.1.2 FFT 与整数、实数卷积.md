#### FFT 与整数 / 实数卷积
```cpp
template<class Real = double>
struct FFT {
    static_assert(is_floating_point_v<Real>);
    using C = complex<Real>;
    static constexpr i64 BRUTE{ 4096 };
    inline static int base{ 1 };
    inline static vector<int> rev{ 0, 1 };
    inline static vector<C> roots{ C(0), C(1) };
    inline static const Real PI{ acos(Real(-1)) };
    vector<C> fa, fb;
    static C mul(C a, C b) {
        return C(a.real() * b.real() - a.imag() * b.imag(), a.real() * b.imag() + a.imag() * b.real());
    }
    static int fft_size(int need) {
        int n{ 1 };
        while (n < need) n <<= 1;
        return n;
    }
    static void ensure(int n) {
        int nbase{ __builtin_ctz(n) };
        if (nbase <= base) return;
        rev.resize(n);
        for (int i = 0; i < n; ++ i) rev[i] = (rev[i >> 1] >> 1) | ((i & 1) << (nbase - 1));
        roots.resize(n);
        while (base < nbase) {
            C z(cos(PI / (1 << base)), sin(PI / (1 << base)));
            for (int i = 1 << (base - 1); i < (1 << base); ++ i) {
                roots[2 * i] = roots[i];
                roots[2 * i + 1] = mul(roots[i], z);
            }
            ++ base;
        }
    }
    // ===== FFT 核心：a.size() 必须为 2 的幂 =====
    static void fft(vector<C> &a, bool inverse = false) {
        int n{ int(a.size()) };
        assert(n && (n & (n - 1)) == 0);
        ensure(n);
        // 逆变换等价于反转频率下标后做正变换，单位根只需维护一份
        if (inverse) reverse(a.begin() + 1, a.end());
        int shift{ base - __builtin_ctz(n) };
        for (int i = 0; i < n; ++ i) {
            int j{ rev[i] >> shift };
            if (i < j) swap(a[i], a[j]);
        }
        for (int k = 1; k < n; k <<= 1) {
            for (int i = 0; i < n; i += 2 * k) {
                for (int j = 0; j < k; ++ j) {
                    C u{ a[i + j] }, v{ mul(a[i + j + k], roots[k + j]) };
                    a[i + j] = u + v;
                    a[i + j + k] = u - v;
                }
            }
        }
        if (inverse) {
            Real inv{ Real(1) / n };
            for (C &x : a) x *= inv;
        }
    }
    // ===== 实数卷积：整数卷积依赖；打包两个实序列 =====
    template<class A, class B>
    vector<Real> convolution_real(const vector<A> &a, const vector<B> &b) {
        static_assert(is_arithmetic_v<A> && is_arithmetic_v<B>);
        if (a.empty() || b.empty()) return {};
        int na{ int(a.size()) }, nb{ int(b.size()) }, need{ na + nb - 1 };
        if (i64(na) * nb <= BRUTE) {
            vector<Real> c(need);
            for (int i = 0; i < na; ++ i) {
                for (int j = 0; j < nb; ++ j) c[i + j] += Real(a[i]) * Real(b[j]);
            }
            return c;
        }
        int n{ fft_size(need) };
        fa.resize(n);
        ranges::fill(fa, C{});
        for (int i = 0; i < na; ++ i) fa[i].real(Real(a[i]));
        for (int i = 0; i < nb; ++ i) fa[i].imag(Real(b[i]));
        // F = DFT(a) + i DFT(b)，利用共轭对称拆出两个频谱
        fft(fa);
        for (int i = 0; i <= n / 2; ++ i) {
            int j{ (-i) & (n - 1) };
            C x{ fa[i] }, y{ conj(fa[j]) };
            C va{ (x + y) * Real(0.5) }, vb{ (x - y) * C(0, Real(-0.5)) };
            C z{ mul(va, vb) };
            fa[i] = z;
            if (i != j) fa[j] = conj(z);
        }
        fft(fa, true);
        vector<Real> c(need);
        for (int i = 0; i < need; ++ i) c[i] = fa[i].real();
        return c;
    }
    // ===== 复数卷积：若系数始终为实数 / 整数，可整段删除 =====
    template<class A, class B>
    vector<C> convolution_complex(const vector<A> &a, const vector<B> &b) {
        if (a.empty() || b.empty()) return {};
        int na{ int(a.size()) }, nb{ int(b.size()) }, need{ na + nb - 1 };
        if (i64(na) * nb <= BRUTE) {
            vector<C> c(need);
            for (int i = 0; i < na; ++ i) {
                for (int j = 0; j < nb; ++ j) c[i + j] += mul(C(a[i]), C(b[j]));
            }
            return c;
        }
        int n{ fft_size(need) };
        fa.resize(n);
        fb.resize(n);
        ranges::fill(fa, C{});
        ranges::fill(fb, C{});
        for (int i = 0; i < na; ++ i) fa[i] = C(a[i]);
        for (int i = 0; i < nb; ++ i) fb[i] = C(b[i]);
        fft(fa);
        fft(fb);
        for (int i = 0; i < n; ++ i) fa[i] = mul(fa[i], fb[i]);
        fft(fa, true);
        return vector<C>(fa.begin(), fa.begin() + need);
    }
    // ===== 整数恢复与自动分派：convolution() 依赖以上所有卷积接口；裁剪任一分支时同步删除 =====
    template<class A, class B>
    vector<i64> convolution_ll(const vector<A> &a, const vector<B> &b) {
        static_assert(is_integral_v<A> && is_integral_v<B>);
        vector<Real> c{ convolution_real(a, b) };
        vector<i64> res(c.size());
        for (int i = 0; const auto &x : c) res[i ++] = llround(x);
        return res;
    }
    template<class A, class B>
    auto convolution(const vector<A> &a, const vector<B> &b) {
        if constexpr (is_integral_v<A> && is_integral_v<B>) return convolution_ll(a, b);
        else if constexpr (is_arithmetic_v<A> && is_arithmetic_v<B>) return convolution_real(a, b);
        else return convolution_complex(a, b);
    }
};
```
##### 接口说明
###### `fft(a, inverse)`
原地执行 FFT
```cpp
fft(a, false); // 正变换
fft(a, true);  // 逆变换
```
要求：
- `a.size() >= 1`
- `a.size()` 是 $2$ 的幂
逆变换已经自动除以 $n$
因此标准卷积流程就是：
```cpp
fft(a, false);
fft(b, false);
for (int i = 0; i < n; ++ i) a[i] *= b[i];
fft(a, true);
```
###### `convolution_real(a, b)`
计算普通卷积：$$c_k=\sum_{i+j=k}a_i b_j$$
返回：
```cpp
vector<ld>
```
长度严格为：
$$|a|+|b|-1$$
任一输入为空时返回空数组
接口做成模板：
```cpp
template<class T>
vector<ld> convolution_real(...);
```
所以可以直接：
```cpp
vector<double>
vector<ld>
vector<int>
vector<i64>
```
作为输入
但**内部统一转换为 `ld`**，输出也是浮点近似值
###### `convolution_ll(a, b)`
计算整数卷积：
```cpp
vector<i64> c = convolution_ll(a, b);
```
理论答案仍然是：
$$c_k=\sum_{i+j=k}a_i b_j$$
FFT 逆变换后使用：
```cpp
llround(...)
```
恢复最近整数
支持：
- 正数
- `0`
- 负数
它不是模卷积
###### `FFTPlan`
```cpp
FFTPlan plan(need);
```
预处理：
- 位逆序数组 `rev`
- 单位根 `roots`
其中：
```cpp
plan.n
```
是不小于 `need` 的最小二次幂
`init(need)` 如果对应长度和当前 `n` 相同，会直接返回，因此：
```cpp
FFTPlan plan;
for (...) {
    auto c = plan.convolution_ll(a, b);
}
```
当这些卷积需要相同 FFT 长度时，不会反复计算 `rev` 与单位根
###### `FFTPlan::transform(a, inverse)`
要求：
```cpp
a.size() == plan.n
```
例如：
```cpp
FFTPlan plan(1000); // plan.n = 1024
vector<Complex> a(plan.n);
plan.transform(a, false);
plan.transform(a, true);
```
##### 思想
设：
$$A(x)=\sum_i a_ix^i,\quad B(x)=\sum_i b_ix^i$$
它们的乘积：
$$C(x)=A(x)B(x)$$
满足：
$$c_k=\sum_{i+j=k}a_i b_j$$
直接计算需要：
$$O(nm)$$
FFT 利用卷积定理：
$$\operatorname{DFT}(A*B)=\operatorname{DFT}(A)\cdot\operatorname{DFT}(B)$$
因此：
1. 把两个系数数组补到足够长的 $2$ 的幂
2. 分别 FFT
3. 逐点相乘
4. 逆 FFT
5. 得到卷积
将复杂度降至：
$$O(N\log N)$$
其中：
$$N=2^{\lceil\log_2(n+m-1)\rceil}$$
##### 注意事项
###### 精度是 FFT 整数卷积最重要的限制
`convolution_ll()` **不是任意 `i64` 输入都一定精确**
它正确需要同时满足：
1. 真实卷积系数可以存入 `i64`
2. 浮点 FFT 的最终误差严格小于 $0.5$
设：
$$A=\max_i |a_i|,\quad B=\max_j |b_j|$$
则一个简单的整数大小上界为：
$$|c_k| \le \min(|a|,|b|)AB$$
这个上界可以检查**整数溢出风险**，但不能单独证明 FFT 浮点误差一定安全

> [!warning]  
> 如果数据长度和系数都很大，或者题目明确要求严格精确，不要仅因为理论答案能放进 `i64` 就认为 `convolution_ll()` 一定安全
> 
> 后续的 **任意模卷积 / MTT / 多模 NTT + CRT** 才是处理大系数精确卷积的方案
###### `FFTPlan` 不一定比普通 `fft()` 更快
普通版本每一层只调用一次 `sin/cos`，因此一次 FFT 只有： $O(\log N)$ 次三角函数计算
而 `FFTPlan::init()` 一次会预计算 $N$ 个单位根
所以：
- **单次卷积**：直接用普通版通常更合适
- **大量相同 FFT 长度的变换**：`FFTPlan` 才更有意义
###### 不要和后续 MTT 混淆
这一页的：
```cpp
convolution_ll()
```
是**直接把整数转成浮点数，FFT 后四舍五入**
之后的 MTT 则会进行**系数拆分，再使用 FFT 控制数值范围**
二者适用精度范围不同
##### 裁剪指南
###### 只需要手动 FFT
保留：
```cpp
ld
Complex
PI
fft()
```
###### 只需要整数卷积
保留：
```cpp
ld / Complex / PI
fft()
convolution_ll()
```
可以删除：
```cpp
convolution_real()
FFTPlan
```
###### 只需要实数卷积
保留：
```cpp
ld / Complex / PI
fft()
convolution_real()
```
删除：
```cpp
convolution_ll()
FFTPlan
```
###### 需要大量重复卷积
可以只保留：
```cpp
ld / Complex / PI
FFTPlan
```
因为 `FFTPlan` 自己有：
```cpp
transform()
convolution_real()
convolution_ll()
```
此时甚至可以把前面的普通：
```cpp
fft()
convolution_real()
convolution_ll()
```
**全部删除**
##### 常见用法
###### 整数卷积
```cpp
vector<i64> a{ 1, 2, 3 };
vector<i64> b{ 4, 5 };
auto c = convolution_ll(a, b);
```
结果：
```cpp
4 13 22 15
```
对应：
$$(1+2x+3x^2)(4+5x)=4+13x+22x^2+15x^3$$
###### 含负数
```cpp
vector<i64> a{ 1, -2, 3 };
vector<i64> b{ -4, 5 };
auto c = convolution_ll(a, b);
```
无需额外处理符号
###### 实数卷积
```cpp
vector<double> a{ 1.5, 2.0 };
vector<double> b{ 3.0, 4.0 };
auto c = convolution_real(a, b);
```
返回：
```cpp
vector<ld>
```
不要对结果使用裸 `==` 判断
###### 重复相同规模卷积
```cpp
FFTPlan plan;
for (int t = 0; t < q; ++ t) {
    vector<i64> a, b;
    // ...
    auto c = plan.convolution_ll(a, b);
}
```
只要每次：
$$|a|+|b|-1$$
落到同一个二次幂区间，`init()` 就不会重新预处理
###### 手动使用变换
```cpp
int n{ 1 };
while (n < need) n <<= 1;
vector<Complex> a(n);
for (int i = 0; i < m; ++ i) a[i] = value[i];
fft(a, false);
// 频域处理
fft(a, true);
```
如果只是普通卷积，不需要手动调用，应直接用 `convolution_real()` 或 `convolution_ll()`
#### 复杂度
设：
$$N=2^{\lceil\log_2(|a|+|b|-1)\rceil}$$

|接口|时间复杂度|额外空间|
|---|--:|--:|
|`fft(a, inverse)`|$O(N\log N)$|$O(1)$|
|`convolution_real(a,b)`|$O(N\log N)$|$O(N)$|
|`convolution_ll(a,b)`|$O(N\log N)$|$O(N)$|
|`FFTPlan::init(need)`|$O(N)$|$O(N)$|
|`FFTPlan::transform()`|$O(N\log N)$|$O(1)$|
|`FFTPlan::convolution_real()`|$O(N\log N)$|$O(N)$|
|`FFTPlan::convolution_ll()`|$O(N\log N)$|$O(N)$|
其中 `FFTPlan` 的 `rev + roots` 会额外长期占用 $O(N)$ 空间
