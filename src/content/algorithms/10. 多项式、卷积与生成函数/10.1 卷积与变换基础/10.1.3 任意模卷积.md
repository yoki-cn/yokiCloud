#### 任意模卷积
##### 三模 NTT + CRT
```cpp
namespace AnyModConv {
    constexpr int P1{ 167772161 }; // 5 * 2^25 + 1
    constexpr int P2{ 469762049 }; // 7 * 2^26 + 1
    constexpr int P3{ 998244353 }; // 119 * 2^23 + 1
    constexpr int G{ 3 }, MAX_N{ 1 << 23 };
    constexpr i64 P12{ 1LL * P1 * P2 };
    // ===== 单模快速幂：NTT 与 CRT 依赖，不建议删除 =====
    template<int MOD>
    int qpow(int a, i64 b) {
        i64 res{ 1 };
        for (; b; b >>= 1, a = i64(a) * a % MOD) {
            if (b & 1) res = res * a % MOD;
        }
        return res;
    }
    // ===== 单模 NTT：三个辅助卷积依赖，不建议删除 =====
    template<int MOD>
    void ntt(vector<int> &a, bool inverse) {
        int n = a.size();
        assert(n && (n & (n - 1)) == 0 && (MOD - 1) % n == 0);
        // 位逆序置换
        for (int i = 1, j = 0; i < n; ++ i) {
            int bit{ n >> 1 };
            for (; j & bit; bit >>= 1) j ^= bit;
            j ^= bit;
            if (i < j) swap(a[i], a[j]);
        }
        for (int len = 2; len <= n; len <<= 1) {
            int wlen{ qpow<MOD>(G, (MOD - 1) / len) };
            if (inverse) wlen = qpow<MOD>(wlen, MOD - 2);
            for (int i = 0; i < n; i += len) {
                i64 w{ 1 };
                for (int j = 0; j < len / 2; ++ j) {
                    int u{ a[i + j] }, v{ int(w * a[i + j + len / 2] % MOD) };
                    int x{ u + v }, y{ u - v };
                    if (x >= MOD) x -= MOD;
                    if (y < 0) y += MOD;
                    a[i + j] = x;
                    a[i + j + len / 2] = y;
                    w = w * wlen % MOD;
                }
            }
        }
        if (inverse) {
            int inv_n{ qpow<MOD>(n, MOD - 2) };
            for (int &x : a) x = i64(x) * inv_n % MOD;
        }
    }
    // ===== 单个 NTT 质数下的卷积：主接口内部依赖，不建议删除 =====
    template<int MOD>
    vector<int> convolution_ntt(const vector<i64> &a, const vector<i64> &b) {
        if (a.empty() || b.empty()) return {};
        int need = a.size() + b.size() - 1, n{ 1 };
        while (n < need) n <<= 1;
        vector<int> x(n), y(n);
        for (int i = 0; i < a.size(); ++ i) x[i] = a[i] % MOD;
        for (int i = 0; i < b.size(); ++ i) y[i] = b[i] % MOD;
        ntt<MOD>(x, false);
        ntt<MOD>(y, false);
        for (int i = 0; i < n; ++ i) x[i] = i64(x[i]) * y[i] % MOD;
        ntt<MOD>(x, true);
        x.resize(need);
        return x;
    }
    // ===== 任意模卷积主接口：三模 NTT + CRT =====
    // mod 为任意正 int；输入按模 mod 的意义处理，可含负数
    template<class T>
    vector<int> convolution_mod(const vector<T> &a, const vector<T> &b, int mod) {
        static_assert(is_integral_v<T>);
        assert(mod > 0);
        if (a.empty() || b.empty()) return {};
        // 先规范到 [0, mod)，确保 CRT 恢复的是确定的非负整数卷积
        vector<i64> x(a.size()), y(b.size());
        for (int i = 0; i < a.size(); ++ i) {
            i128 v{ a[i] };
            v %= mod;
            if (v < 0) v += mod;
            x[i] = i64(v);
        }
        for (int i = 0; i < b.size(); ++ i) {
            i128 v{ b[i] };
            v %= mod;
            if (v < 0) v += mod;
            y[i] = i64(v);
        }
        int need = x.size() + y.size() - 1;
        // 小规模直接在目标模数下朴素计算
        if (1ULL * x.size() * y.size() <= 4096) {
            vector<int> c(need);
            for (int i = 0; i < x.size(); ++ i) {
                for (int j = 0; j < y.size(); ++ j) {
                    c[i + j] = (c[i + j] + i128(x[i]) * y[j]) % mod;
                }
            }
            return c;
        }
        int n{ 1 };
        while (n < need) n <<= 1;
        assert(n <= MAX_N);
        vector<int> c1{ convolution_ntt<P1>(x, y) };
        vector<int> c2{ convolution_ntt<P2>(x, y) };
        vector<int> c3{ convolution_ntt<P3>(x, y) };
        // x = r1 + P1*t1 + P1*P2*t2
        static const i64 inv_p1_p2{ qpow<P2>(P1 % P2, P2 - 2) };
        static const i64 inv_p12_p3{ qpow<P3>(P12 % P3, P3 - 2) };
        vector<int> res(need);
        for (int i = 0; i < need; ++ i) {
            i64 t1{ (c2[i] - c1[i] + P2) % P2 * inv_p1_p2 % P2 };
            i64 x12{ c1[i] + i64(P1) * t1 };
            i64 t2{ (c3[i] - x12 % P3 + P3) % P3 * inv_p12_p3 % P3 };
            i128 value{ i128(x12) + i128(P12) * t2 };
            res[i] = value % mod;
        }
        return res;
    }
}
```
##### MTT：FFT 拆系数任意模卷积
这一部分依赖上一份 [[10.1.2 FFT 与整数、实数卷积]] 中的：
```cpp
using ld = long double;
using Complex = complex<ld>;
void fft(vector<Complex> &a, bool inverse);
```
`cpplibforCP` 的任意模 FFT 多项式同样采用将系数拆成高低位、再把两部分打包进复数的方法，而且明确提醒必须拆位控制精度
```cpp
// ===== MTT：若三模 NTT + CRT 已满足需求，以下代码可直接删除 =====
// 依赖 1.2 的 Complex / fft()；存在浮点精度风险
template<class T>
vector<int> convolution_mtt(const vector<T> &a, const vector<T> &b, int mod) {
    static_assert(is_integral_v<T>);
    assert(mod > 0);
    if (a.empty() || b.empty()) return {};
    constexpr i64 BASE{ 1 << 15 }, MASK{ BASE - 1 };
    int need = a.size() + b.size() - 1, n{ 1 };
    while (n < need) n <<= 1;
    // 小规模直接计算，避免 FFT 常数与精度问题
    if (1ULL * a.size() * b.size() <= 4096) {
        vector<int> c(need);
        for (int i = 0; i < a.size(); ++ i) {
            i128 x{ a[i] };
            x %= mod;
            if (x < 0) x += mod;
            for (int j = 0; j < b.size(); ++ j) {
                i128 y{ b[j] };
                y %= mod;
                if (y < 0) y += mod;
                c[i + j] = (c[i + j] + x * y) % mod;
            }
        }
        return c;
    }
    // A = a0 + i*a1，B = b0 + i*b1
    vector<Complex> A(n), B(n), C(n);
    for (int i = 0; i < a.size(); ++ i) {
        i128 v{ a[i] };
        v %= mod;
        if (v < 0) v += mod;
        i64 x{ i64(v) };
        A[i] = Complex(x & MASK, x >> 15);
    }
    for (int i = 0; i < b.size(); ++ i) {
        i128 v{ b[i] };
        v %= mod;
        if (v < 0) v += mod;
        i64 x{ i64(v) };
        B[i] = Complex(x & MASK, x >> 15);
    }
    fft(A, false);
    fft(B, false);
    // conj(B[n-i]) 是 b0 - i*b1 的 DFT
    C[0] = conj(B[0]);
    for (int i = 1; i < n; ++ i) C[i] = conj(B[n - i]);
    for (int i = 0; i < n; ++ i) {
        B[i] *= A[i];
        C[i] *= A[i];
    }
    fft(B, true);
    fft(C, true);
    vector<int> res(need);
    for (int i = 0; i < need; ++ i) {
        // B = (a0b0-a1b1) + i(a0b1+a1b0)
        // C = (a0b0+a1b1) + i(a1b0-a0b1)
        i64 lo{ llround((B[i].real() + C[i].real()) / 2) };
        i64 mid{ llround(B[i].imag()) };
        i64 hi{ llround((C[i].real() - B[i].real()) / 2) };
        i128 value{ i128(lo) + i128(mid) * BASE + i128(hi) * BASE * BASE };
        value %= mod;
        if (value < 0) value += mod;
        res[i] = value;
    }
    return res;
}
```
##### 三模 NTT + CRT
分别计算：
$$c_i\bmod P_1,\quad c_i\bmod P_2,\quad c_i\bmod P_3$$
因为三个模数两两互质，由 CRT：
$$c_i \pmod{P_1P_2P_3}$$
唯一确定
这里选：
$$P_1=167772161=5\cdot2^{25}+1$$
$$P_2=469762049=7\cdot2^{26}+1$$
$$P_3=998244353=119\cdot2^{23}+1$$
三者原根都可以取：
$$3$$
所以公共最大 NTT 长度为：
$$2^{23}$$
三模乘积：
$$P_1P_2P_3=78674626319836206717730817\approx7.87\times10^{25}$$
###### 为什么这三个模数足够？
接口要求：
```cpp
int mod
```
因此：
$$1\le mod\le2^{31}-1$$
输入首先规范为：
$$0\le a_i,b_i<mod$$
又因为 NTT 长度不超过：
$$2^{23}$$
所以：
$$|a|+|b|-1\le2^{23}$$
单个卷积系数最多包含：$\min(|a|,|b|)\le2^{22}$ 个乘积
因此：
$$c_k \le 2^{22}(2^{31}-2)^2$$
右边约为：
$$1.93\times10^{25}$$
严格小于：
$$P_1P_2P_3 \approx7.87\times10^{25}$$

所以在这份接口的全部合法范围内：
$$0\le c_k<P_1P_2P_3$$
因此 CRT 得到的不只是“一个同余类”，而是**原始整数卷积系数本身**
然后再计算：
$$c_k\bmod mod$$
就是严格正确的答案
##### CRT 合并
设三个 NTT 结果为：
$$r_1,\ r_2,\ r_3$$
先写：
$$x=r_1+P_1t_1$$
要求：
$$x\equiv r_2\pmod{P_2}$$
所以：
$$t_1=(r_2-r_1)P_1^{-1}\pmod{P_2}$$
得到：
$$x_{12}=r_1+P_1t_1$$
再写：
$$x=x_{12}+P_1P_2t_2$$
要求：
$$x\equiv r_3\pmod{P_3}$$
因此：
$$t_2=(r_3-x_{12})(P_1P_2)^{-1}\pmod{P_3}$$
最终：
$$x=x_{12}+P_1P_2t_2$$
其中最后一步超过 `i64`，因此必须：
```cpp
i128 value
```
不能等乘完以后再强转
##### MTT
设目标模数下一个系数：
$$x=x_0+Bx_1$$
这里：
$$B=2^{15}$$
于是：
$$a=a_0+Ba_1,\quad b=b_0+Bb_1$$
卷积为：
$$ab=a_0b_0+B(a_0b_1+a_1b_0)+B^2a_1b_1$$
因此只需得到三类卷积：
$$a_0*b_0$$
$$a_0*b_1+a_1*b_0$$
$$a_1*b_1$$
将：
$$a_0+ia_1,\quad b_0+ib_1$$
打包进复数，可以减少 FFT 次数
`cpplibforCP` 当前的 FFT 多项式后端同样使用 `bit = 15` 拆高低位，并使用复数打包恢复三部分卷积
##### 接口说明
###### `AnyModConv::convolution_mod(a,b,mod)`
推荐主接口：
```cpp
auto c = AnyModConv::convolution_mod(a, b, mod);
```
计算：
$$c_k=\sum_{i+j=k}a_ib_j\pmod{mod}$$
要求：
- `mod > 0` 
- `mod` 类型为 `int`
- 输入系数类型 `T` 为整数类型
- NTT 补齐后的长度不超过 $2^{23}$
任一输入为空时返回 `{}`
返回 `vector<int>`
每项满足：
$$0\le c_i<mod$$
输入无需提前规范，可以：
```cpp
vector<i64> a{ 1, -2, 3 };
```
内部会统一 `% mod`
###### `convolution_mtt(a,b,mod)`
可选 FFT 版本：
```cpp
auto c = convolution_mtt(a, b, mod);
```
接口语义与三模版相同
依赖：
```text
10.1.2 FFT 与整数 / 实数卷积
├── ld
├── Complex
└── fft()
```
不依赖三模 NTT
##### 注意事项
###### 首选三模 NTT + CRT
在支持范围内，它是**整数意义上严格精确**的
不会受到：
- `double`
- `long double`
- FFT 舍入
影响
如果：
$$|a|+|b|-1\le2^{23}$$
而且目标模数能放进正 `int`，默认优先：
```cpp
AnyModConv::convolution_mod()
```
###### MTT 不是严格无误差算法
MTT 已通过拆系数显著降低数值规模，但底层仍然使用浮点 FFT
因此仍可能发生：
```text
真实值 = ... .0000
FFT 结果 = ... .5001
llround 得到错误整数
```
尤其在：
- 长度很大
- 平台 `ld` 实际精度较低
- 数据构造极端
时应谨慎
`cpplibforCP` 的对应实现也明确标记其拆位是为了保证精度，并提醒恢复阶段存在精度损失

> [!tip]  
> **严格正确优先：三模 NTT + CRT**
> 
> **长度超过 $2^{23}$ 或特别在意常数，并且能接受 FFT 精度条件：再考虑 MTT**
###### 为什么 MTT 选 `15` 位？
使用：
$$B=2^{15}$$
如果：
$$mod<2^{31}$$
那么高位约只剩 $16$ 位
拆位后 FFT 中参与卷积的基本数字远小于直接放入 $10^9$ 量级系数，可以显著减小舍入误差
不要随意把 `1 << 15` 改成`1 << 20`
位数越大，单次 FFT 中的数值越大，精度风险越高
###### 主版实际上不依赖 10.1.1 的固定 `Z` NTT
虽然算法思想依赖 NTT，但 10.1.1 的实现是：
```text
Z + 998244353
```
而这里必须同时在三个不同模数下计算
因此 10.1.3 内部保留了一份很短的：
```cpp
template<int MOD>
ntt()
```
这是有意的，不是无意义重复
这样实际赛场使用：
```text
只需要 998244353
→ 抄 1.1

只需要 1e9+7 任意模卷积
→ 直接抄 1.3

只需要 FFT MTT
→ 抄 1.2 + 1.3 的 MTT 部分
```
不需要为了一个任意模卷积再把整个 `Poly` 体系带上
###### `mod == 1` 合法
此时所有答案都是 `0`
接口仍然可以正常使用
不过如果题面明确 `mod = 1` 当然直接输出零最快
###### 三模版不是任意长度
三个质数各自支持：
$$2^{25},\quad2^{26},\quad2^{23}$$
三模同时使用，因此取最小值：
$$\boxed{2^{23}}$$
不要看到前两个支持更长，就把 `MAX_N` 单独改大

##### 常见用法
###### 模 `1e9+7` 卷积
```cpp
vector<int> a{ 1, 2, 3 }, b{ 4, 5, 6 };
auto c = AnyModConv::convolution_mod(a, b, 1000000007);
```
得到：
```text
4 13 28 27 18
```
###### 任意合数模数
不要求目标模数为质数：
```cpp
int mod{ 1000000000 };
auto c = AnyModConv::convolution_mod(a, b, mod);
```
因为目标模数只在最后取模，CRT 的三个内部模数才要求是适合 NTT 的质数
所以：
```text
质数模
合数模
```
都可以
这是一个非常重要的使用范围
###### 输入含负数
```cpp
vector<i64> a{ 1, -2, 3 }, b{ -4, 5 };
auto c = AnyModConv::convolution_mod(a, b, 1000000007);
```
无需提前手动：
```cpp
(x % mod + mod) % mod
```
###### 使用 MTT
已有 10.1.2 的 FFT 后：
```cpp
vector<int> a(n), b(m);
auto c = convolution_mtt(a, b, mod);
```
和三模版的输出语义完全相同
因此算法切换时基本只需要修改函数名
##### 复杂度
设：
$$N=2^{\lceil\log_2(n+m-1)\rceil}$$
###### 三模 NTT + CRT
每个模数需要：
- 2 次正 NTT
- 1 次逆 NTT
共 9 次 NTT，因此：
$$O(N\log N)$$
CRT 合并：
$$O(n+m)$$
总复杂度：
$$\boxed{O(N\log N)}$$
空间复杂度：
$$\boxed{O(N)}$$
常数大约是单模 NTT 的 3 倍
###### MTT
代码进行：
- `A` 正 FFT
- `B` 正 FFT
- 两个结果逆 FFT
共 4 次 FFT
因此：
$$\boxed{O(N\log N)}$$
时间
$$\boxed{O(N)}$$
空间

|接口|时间复杂度|是否严格精确|最大数论长度|
|---|--:|---|--:|
|三模 NTT + CRT|$O(N\log N)$|✅|$2^{23}$|
|MTT|$O(N\log N)$|⚠️ 浮点条件下|无固定 NTT 上限|
|朴素任意模卷积|$O(nm)$|✅|无|
它和前两份的关系现在很清楚：
```text
10.1.1 NTT 与普通卷积
    └── 998244353，精确，最快最常用

10.1.2 FFT 与整数 / 实数卷积
    └── 实数、精度安全的普通整数

10.1.3 任意模卷积
    ├── 三模 NTT + CRT：任意 int 模，精确主版
    └── MTT：FFT 拆位，可选高速版
```
