### 子集卷积（按 popcount 分层）
---
> 依赖：`10.1.4 FWT` 中的 `FWT::transform`（OR 变换）；不依赖 NTT。

```cpp
namespace SubsetConv {
    // c[S] = sum_{A subseteq S} a[A] * b[S \ A]
    template<class T = Z>
    vector<T> conv_subset(const vector<T> &a, const vector<T> &b) {
        if (a.empty() || b.empty()) return {};
        int need{ int(max(a.size(), b.size())) };
        int n{ int(bit_ceil(unsigned(need))) };
        int k{ countr_zero(unsigned(n)) };
        // f[r][S] / g[r][S]：仅保留 popcount(S) = r 的系数，再做子集 Zeta
        vector<vector<T>> f(k + 1, vector<T>(n));
        bool square{ &a == &b };
        vector<vector<T>> g;
        if (!square) g.assign(k + 1, vector<T>(n));
        for (unsigned s = 0; s < a.size(); ++ s) f[popcount(s)][s] = a[s];
        if (!square) {
            for (unsigned s = 0; s < b.size(); ++ s) g[popcount(s)][s] = b[s];
        }
        for (auto &v : f) FWT::transform<0>(v);
        if (!square) for (auto &v : g) FWT::transform<0>(v);
        // 每个 mask 上按集合大小做普通卷积；从高 rank 向低 rank 可原地覆盖 f
        for (int s = 0; s < n; ++ s) {
            for (int r = k; r >= 0; -- r) {
                T x{};
                for (int i = 0; i <= r; ++ i) {
                    x += f[i][s] * (square ? f[r - i][s] : g[r - i][s]);
                }
                f[r][s] = x;
            }
        }
        for (auto &v : f) FWT::transform<0>(v, true);
        vector<T> res(n);
        for (unsigned s = 0; s < n; ++ s) res[s] = f[popcount(s)][s];
        return res;
    }
}
template<class T = Z>
vector<T> conv_subset(const vector<T> &a, const vector<T> &b) {
    return SubsetConv::conv_subset(a, b);
}
```

#### 接口说明

- `conv_subset(a,b)`：返回 `c[S] = Σ_{A⊆S}a[A]b[S\A]`，其中 `S\A` 是集合差，要求两侧**不相交**。
- 输入至少有一项时输出长度为 `bit_ceil(max(|a|,|b|))`，小于长度的 mask 缺省系数按零处理。

#### 注意事项

- 与 OR 卷积不同：OR 卷积允许 `A∩B ≠ ∅`，本卷积不允许。
- 要求状态数为 $2^k$，且所存数组索引等于集合的二进制 mask。
- 存储 $O(k2^k)$：当 `k` 比较大时内存压力很高；若只需要 OR/AND/XOR 卷积，不要引入此实现。
- 此实现复用 `FWT::transform<0>(v,...)`；只使用子集卷积时可将 OR 变换函数单独复制过来。

#### 常见用法

```cpp
vector<Z> a(1 << k), b(1 << k);
auto c = conv_subset(a, b);
// 集合划分：a[A] * b[S\\A]，枚举有序分组 (A, S\\A)
// 平方：直接传同一容器引用，避免第二套分层数组
c = conv_subset(a, a);
```

#### 复杂度

- $O(k^2\cdot 2^k)$ 时间、$O(k\cdot 2^k)$ 空间。
