### NTT（998244353）
---
```cpp
template<class T>
vector<T> conv(const vector<T> &a, const vector<T> &b) {
    if (a.empty() || b.empty()) return {};
    vector<T> c(a.size() + b.size() - 1);
    for (int i = 0; i < int(a.size()); ++ i) {
        for (int j = 0; j < int(b.size()); ++ j) c[i + j] += a[i] * b[j];
    }
    return c;
}
namespace NTT {
    constexpr int MOD = 998244353, G = 3, MAX_N = 1 << 23;
    static_assert(Z::mod() == MOD);
    inline int base = 1;
    inline vector<int> rev{0, 1};
    inline vector<Z> roots{0, 1};

    int ntt_size(int need) { return int(bit_ceil(unsigned(need))); }
    void ensure(int n) {
        int k = countr_zero(unsigned(n));
        if (k <= base) return;
        rev.resize(n);
        for (int i = 0; i < n; ++ i) {
            rev[i] = (rev[i >> 1] >> 1) | ((i & 1) << (k - 1));
        }
        roots.resize(n);
        while (base < k) {
            Z e = Z(G).pow((MOD - 1) >> (base + 1));
            for (int i = 1 << (base - 1); i < (1 << base); ++ i) {
                roots[2 * i] = roots[i];
                roots[2 * i + 1] = roots[i] * e;
            }
            ++ base;
        }
    }
    void dft(vector<Z> &a) {
        int n = a.size();
        assert(n > 0 && has_single_bit(unsigned(n)) && n <= MAX_N);
        ensure(n);
        int shift = base - countr_zero(unsigned(n));
        for (int i = 0; i < n; ++ i) {
            int j = rev[i] >> shift;
            if (i < j) swap(a[i], a[j]);
        }
        for (int k = 1; k < n; k <<= 1) {
            for (int i = 0; i < n; i += 2 * k) {
                for (int j = 0; j < k; ++ j) {
                    Z x = a[i + j], y = a[i + j + k] * roots[k + j];
                    a[i + j] = x + y;
                    a[i + j + k] = x - y;
                }
            }
        }
    }
    void idft(vector<Z> &a) {
        assert(!a.empty());
        reverse(a.begin() + 1, a.end());
        dft(a);
        Z inv = (1 - MOD) / int(a.size());
        for (Z &x : a) x *= inv;
    }
    vector<Z> conv(const vector<Z> &a, const vector<Z> &b) {
        if (a.empty() || b.empty()) return {};
        if (min(a.size(), b.size()) <= 40) return ::conv<Z>(a, b);
        int need = a.size() + b.size() - 1, n = ntt_size(need);
        assert(n <= MAX_N);
        vector<Z> f(n);
        copy(a.begin(), a.end(), f.begin());
        dft(f);
        if (&a == &b) {
            for (Z &x : f) x *= x;
        } else {
            vector<Z> g(n);
            copy(b.begin(), b.end(), g.begin());
            dft(g);
            for (int i = 0; i < n; ++ i) f[i] *= g[i];
        }
        idft(f);
        f.resize(need);
        return f;
    }
}
vector<Z> conv(const vector<Z> &a, const vector<Z> &b) {
    return NTT::conv(a, b);
}
```

#### 接口说明

- `conv(a, b)`：普通卷积，自动选朴素乘法或 NTT；相同容器引用传入 `conv(a,a)` 会复用变换。
- `NTT::dft(a)` / `NTT::idft(a)`：原地正变换 / 逆变换，`a.size()` 必须是合法的 2 的幂。
- `NTT::ntt_size(need)`：不小于 `need` 的最小 2 的幂（`need > 0`）。
- `NTT::MAX_N`：$2^23$；`ensure(n)` 为内部缓存，无需主动调用。
- 泛型 `conv<T>(a,b)`：朴素乘法，适用于非 `Z` 系数；不会使用 NTT。

#### 注意事项

- 依赖已有静态模整数 `Z = MInt<998244353>`，并要求提供 `Z::pow()`、`val()` 与四则运算；需 C++20（`bit_ceil`、`countr_zero`、`has_single_bit`）。
- 当 `a.size()+b.size()-1 > 2^23` 时本版本**不支持**；不默认带大内存的超长卷积。若题目需要可分块。
- `dft` 会修改输入；做逆变换前频域必须满足预期语义。非 `Z` 的 `conv` 为 $O(nm)$。
- `conv(a,a)` 与 `conv(a, vector<Z>(a))` 结果相同，但只有前者触发平方优化。

#### 常见用法

```cpp
vector<Z> a{1, 2, 3}, b{4, 5};
auto c = conv(a, b);       // [4,13,22,15]
auto sq = conv(a, a);      // 平方只做一次正向 NTT
int k = 3;
c.resize(min<int>(k, c.size())); // 取前 k 项
```

频域逐项乘法（需自行补零到 `NTT::ntt_size(a.size()+b.size()-1)`）：

```cpp
int n = NTT::ntt_size(a.size() + b.size() - 1);
a.resize(n); b.resize(n);
NTT::dft(a); NTT::dft(b);
for (int i = 0; i < n; ++ i) a[i] *= b[i];
NTT::idft(a);
```

#### 复杂度

- 朴素分支：$O(|a||b|)$，其余 $O(N \log N)$ 时间、$O(N)$ 额外空间；`N=bit_ceil(|a|+|b|-1)`。
