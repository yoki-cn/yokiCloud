### NTT
---
```cpp
template<class T>
vector<T> conv(const vector<T> &a, const vector<T> &b) {
    if (a.empty() || b.empty()) return {};
    vector<T> c(a.size() + b.size() - 1);
    for (int i = 0; i < a.size(); ++ i) {
        for (int j = 0; j < b.size(); ++ j) c[i + j] += a[i] * b[j];
    }
    return c;
}
namespace NTT {
    static_assert(Z::mod() == 998244353);
    constexpr int MOD{ 998244353 }, G{ 3 };
    constexpr int MAX_LOG{ 23 }, MAX_N{ 1 << MAX_LOG };
    constexpr int BRUTE{ 60 };
    inline int base{ 1 };
    inline vector<int> rev{ 0, 1 };
    inline vector<Z> roots{ 0, 1 };
    static int ntt_size(unsigned need) { return bit_ceil(need); }
    void ensure(unsigned n) {
        int nbase{ countr_zero(n) };
        if (nbase <= base) return;
        rev.resize(n);
        for (int i = 0; i < n; ++ i) rev[i] = (rev[i >> 1] >> 1) | ((i & 1) << (nbase - 1));
        roots.resize(n);
        while (base < nbase) {
            Z e{ Z(G).pow((MOD - 1) >> (base + 1)) };
            for (int i = 1 << (base - 1); i < (1 << base); ++ i) {
                roots[i << 1] = roots[i];
                roots[i << 1 | 1] = roots[i] * e;
            }
            ++ base;
        }
    }
    void dft(vector<Z> &a) {
        unsigned n = a.size();
        assert(n && (n & (n - 1)) == 0 && n <= MAX_N);
        ensure(n);
        int shift{ base - countr_zero(n) };
        for (int i = 0; i < n; ++ i) {
            int j{ rev[i] >> shift };
            if (i < j) swap(a[i], a[j]);
        }
        for (int k = 1; k < n; k <<= 1) {
            for (int i = 0; i < n; i += k << 1) {
                for (int j = 0; j < k; ++ j) {
                    Z u{ a[i + j] }, v{ a[i + j + k] * roots[k + j] };
                    a[i + j] = u + v;
                    a[i + j + k] = u - v;
                }
            }
        }
    }
    void idft(vector<Z> &a) {
        assert(!a.empty());
        reverse(a.begin() + 1, a.end());
        dft(a);
        // n | MOD - 1，因此 -(MOD - 1) / n 即 n 在模 MOD 下的逆元
        Z inv_n{ (1 - MOD) / int(a.size()) };
        for (Z &x : a) x *= inv_n;
    }
    vector<Z> conv(const vector<Z> &a, const vector<Z> &b) {
        if (a.empty() || b.empty()) return {};
        if (min(a.size(), b.size()) <= BRUTE) return ::conv<Z>(a, b);
        int need{ int(a.size() + b.size() - 1) }, n{ ntt_size(need) };
        assert(n <= MAX_N);
        vector<Z> f(n);
        ranges::copy(a, f.begin());
        dft(f);
        // conv(a, a) 时少做一次 DFT；不需要该优化可删掉本分支
        if (&a == &b) {
            for (Z &x : f) x *= x;
        } else {
            vector<Z> g(n);
            ranges::copy(b, g.begin());
            dft(g);
            for (int i = 0; i < n; ++ i) f[i] *= g[i];
        }
        idft(f);
        f.resize(need);
        return f;
    }
    // ===== 超长卷积：若保证结果长度不超过 MAX_N，以下函数可直接删除 =====
    // 将多项式按 MAX_N / 2 分块，使每对块的卷积均可用长度 MAX_N 的 NTT
    vector<Z> conv_large(const vector<Z> &a, const vector<Z> &b) {
        if (a.empty() || b.empty()) return {};
        int need{ a.size() + b.size() - 1 };
        if (need <= MAX_N) return NTT::conv(a, b);
        constexpr int B{ MAX_N >> 1 };
        int na{ (a.size() + B - 1) / B }, nb{ (b.size() + B - 1) / B };
        vector<vector<Z>> fa(na, vector<Z>(MAX_N));
        vector<vector<Z>> fb(nb, vector<Z>(MAX_N));
        for (int i = 0; i < a.size(); ++ i) fa[i / B][i % B] = a[i];
        for (int i = 0; i < b.size(); ++ i) fb[i / B][i % B] = b[i];
        for (auto &f : fa) dft(f);
        for (auto &f : fb) dft(f);
        vector<Z> res(need), f(MAX_N);
        // 第 s 个输出块由所有 i + j = s 的输入块卷积之和组成
        for (int s = 0; s < na + nb - 1; ++ s) {
            ranges::fill(f, Z());
            int l{ max(0, s - nb + 1) }, r{ min(na - 1, s) };
            for (int i = l; i <= r; ++ i) {
                for (int j = s - i, k = 0; k < MAX_N; ++ k) f[k] += fa[i][k] * fb[j][k];
            }
            idft(f);
            int offset{ s * B }, len{ min(MAX_N, need - offset) };
            for (int i = 0; i < len; ++ i) res[offset + i] += f[i];
        }
        return res;
    }
}
vector<Z> conv(const vector<Z> &a, const vector<Z> &b) { return NTT::conv(a, b); }
vector<Z> conv_large(const vector<Z> &a, const vector<Z> &b) { return NTT::conv_large(a, b); }
```
#### 依赖
```cpp
using Z = MInt<998244353>;
```
要求 `Z` 至少支持：
* 从整数构造
* `+`、`-`、`*`
* `/ int`
* `pow(i64)`
* `static mod()`
NTT 部分固定工作于模数
$$
998244353=119\cdot2^{23}+1
$$
原根取 `3`，因此单次 NTT 支持的最大长度为
$$
2^{23}
$$
#### NTT 常量与缓存
* `MOD = 998244353`：模数
* `G = 3`：`MOD` 的原根
* `MAX_LOG = 23`：支持的最大 NTT 层数
* `MAX_N = 1 << MAX_LOG`：单次 NTT 最大长度，即 $2^{23}$
* `BRUTE = 60`：普通卷积阈值
* `base`：当前已经预处理到的最大层数
* `rev`：位逆序置换数组
* `roots`：单位根数组
`base`、`rev`、`roots` 为命名空间内共享缓存
调用规模逐渐增大时，`ensure()` 只继续补充尚未预处理的部分；已经计算过的单位根不会重复计算
#### `NTT::ntt_size(need)`
```cpp
int n = NTT::ntt_size(need);
```
返回不小于 `need` 的最小二次幂：
$$
n=2^{\lceil\log_2 need\rceil}
$$
例如：
```cpp
NTT::ntt_size(1);  // 1
NTT::ntt_size(5);  // 8
NTT::ntt_size(8);  // 8
```
通常用于计算卷积需要的 NTT 长度
要求 `need > 0`
#### `NTT::ensure(n)`
```cpp
NTT::ensure(n);
```
保证长度 `n` 所需的：
* 位逆序数组 `rev`
* 单位根数组 `roots`
已经完成预处理
通常无需手动调用，`dft()` 会自动调用
要求 `n` 为 $2$ 的幂
#### `NTT::dft(a)`
```cpp
NTT::dft(a);
```
对 `a` 原地进行正向 NTT
要求：
* `a` 非空
* `a.size()` 为 $2$ 的幂
* `a.size() <= MAX_N`
变换后 `a` 表示对应多项式在各个单位根上的点值
#### `NTT::idft(a)`
```cpp
NTT::idft(a);
```
对 `a` 原地进行逆 NTT，并自动乘以
$$
n^{-1}\pmod {998244353}
$$
要求与 `dft()` 相同
代码中的
```cpp
Z inv_n{ (1 - MOD) / int(a.size()) };
```
利用了有效 NTT 长度 $n$ 一定满足
$$
n\mid MOD-1
$$
若
$$
MOD-1=kn
$$
则
$$
n(-k)=1-MOD\equiv1\pmod {MOD}
$$
因此
$$
-\frac{MOD-1}{n}
$$
恰好是 $n$ 在模 `MOD` 下的逆元
#### 平方卷积优化
调用：
```cpp
auto c = NTT::conv(a, a);
```
时，两参数引用同一个 `vector` 对象，因此代码会进入：
```cpp
if (&a == &b)
```
只进行一次 DFT，然后逐点平方：
```cpp
for (Z &x : f) x *= x;
```
相比普通卷积少进行一次正向 NTT
注意该优化判断的是**对象地址相同**，而不是内容相同：
```cpp
vector<Z> a, b = a;
NTT::conv(a, a); // 触发平方优化
NTT::conv(a, b); // 不触发，即使 a == b
```
这样不会额外产生 $O(n)$ 的内容比较开销
#### `NTT::conv_large(a, b)`
```cpp
auto c = NTT::conv_large(a, b);
```
用于结果长度超过 `MAX_N`，无法直接进行一次 NTT 的卷积
若
$$
|a|+|b|-1\leq MAX_N
$$
会直接调用：
```cpp
NTT::conv(a, b);
```
否则令
$$
B=\frac{MAX_N}{2}=2^{22}
$$
并将两个多项式分别拆成长度至多 $B$ 的块：
$$
A(x)=A_0(x)+x^BA_1(x)+x^{2B}A_2(x)+\cdots
$$
$$
B(x)=B_0(x)+x^BB_1(x)+x^{2B}B_2(x)+\cdots
$$
于是
$$
A(x)B(x)=\sum_s x^{sB}\left(\sum_{i+j=s}A_i(x)B_j(x)\right)
$$
由于每个块长度至多为 $B$，两个块卷积后的长度至多为
$$
2B-1<MAX_N
$$
所以所有块卷积都可以使用长度 `MAX_N` 的 NTT
代码首先对每个输入块进行一次 DFT，然后对于每个块编号 `s`，直接在频域累加所有满足
$$
i+j=s
$$
的块乘积，最后只进行一次 IDFT
因此不会对每一对块分别执行完整的 DFT / IDFT
#### 全局 `conv()`
```cpp
vector<Z> conv(const vector<Z> &a, const vector<Z> &b);
```
是：
```cpp
NTT::conv(a, b)
```
的全局快捷接口
因此：
```cpp
vector<Z> a, b;
auto c = conv(a, b);
```
会自动使用 NTT
而对于其他类型：
```cpp
vector<i64> a, b;
auto c = conv(a, b);
```
会匹配前面的函数模板，执行普通 $O(nm)$ 卷积
如果显式写：
```cpp
::conv<Z>(a, b);
```
则会强制调用模板版本，即普通 $O(nm)$ 卷积；`NTT::conv()` 的小规模分支正是通过这种方式调用朴素后端
#### 全局 `conv_large()`
```cpp
vector<Z> conv_large(const vector<Z> &a, const vector<Z> &b);
```
等价于：
```cpp
NTT::conv_large(a, b);
```
用于无需显式写命名空间的超长卷积调用
#### 注意事项
* `998244353 - 1` 含有因子 $2^{23}$，因此单次 NTT 的最大合法长度为 `MAX_N = 1 << 23`
* 普通 `NTT::conv()` 限制的是**实际 NTT 长度**，而不只是答案长度；内部已经通过 `assert(n <= MAX_N)` 检查
* 若结果长度超过 `MAX_N`，应使用 `conv_large()`
* `conv_large()` 可以突破单次 NTT 的长度限制，但并不意味着可以无代价地处理任意巨大数组；块数较多时，时间与内存都会快速增长
* `conv_large()` 会为每个输入块保存一个长度为 `MAX_N` 的频域数组，因此其内存常数很大，更适合作为超过 $2^{23}$ 限制时的补充方案
* `conv_large()` 中选择 `B = MAX_N / 2`，是为了保证任意两个块的线性卷积长度严格不超过 `MAX_N`
* `dft()` 与 `idft()` 都会直接修改传入数组
* `dft()` 的输入长度必须为 $2$ 的幂，通常使用 `ntt_size()` 补到合法长度
* 不要直接对任意长度的原数组调用 `dft()`
* `ensure()` 的缓存只增不减；运行过程中曾经进行过的大规模 NTT 会使 `rev` 和 `roots` 保留对应空间
* 小规模卷积自动使用 $O(nm)$ 算法，这是有意的常数优化，而不是 NTT 失效
* `conv(a, a)` 的平方优化只在两个参数确实引用同一个对象时触发
#### 常见用法
##### 普通模卷积
```cpp
vector<Z> a{ 1, 2, 3 };
vector<Z> b{ 4, 5 };
auto c = conv(a, b);
```
得到：
```text
4 13 22 15
```
因为
$$
(1+2x+3x^2)(4+5x)=4+13x+22x^2+15x^3.
$$
##### 多项式平方
```cpp
vector<Z> a{ 1, 2, 3 };
auto c = conv(a, a);
```
由于两个参数是同一个对象，大规模情况下只进行一次正向 NTT
##### 截断卷积结果
若只需要前 `k` 项：
```cpp
auto c = conv(a, b);
if (c.size() > k) c.resize(k);
```
如果已经知道高次输入项不可能影响前 `k` 项，也可以预先截断：
```cpp
if (a.size() > k) a.resize(k);
if (b.size() > k) b.resize(k);
auto c = conv(a, b);
if (c.size() > k) c.resize(k);
```
这在形式幂级数中很常见
##### 手动使用 DFT
```cpp
int need{ int(a.size() + b.size() - 1) };
int n{ NTT::ntt_size(need) };
vector<Z> f(n), g(n);
ranges::copy(a, f.begin());
ranges::copy(b, g.begin());
NTT::dft(f);
NTT::dft(g);
for (int i = 0; i < n; ++ i) f[i] *= g[i];
NTT::idft(f);
f.resize(need);
```
这与普通 NTT 卷积的核心过程一致
只有需要在频域进行额外处理时，通常才需要手动调用 `dft()` / `idft()`
##### 超长卷积
```cpp
vector<Z> a, b;
// |a| + |b| - 1 可能超过 2^23
auto c = conv_large(a, b);
```
当结果长度不超过 `MAX_N` 时，它会自动退化为普通 `NTT::conv()`
因此如果无法方便地提前判断规模，也可以直接使用 `conv_large()`；但通常仍推荐在确定需要突破 `MAX_N` 时才调用它，以明确表达用途
##### 普通类型卷积
全局模板还可以直接用于其他支持乘加的类型：
```cpp
vector<i64> a{ 1, 2, 3 };
vector<i64> b{ 4, 5 };
auto c = conv(a, b);
```
此时不会使用 NTT，而是执行普通双重循环卷积
#### 复杂度
设
$$
n=|a|,\quad m=|b|
$$
结果长度为
$$
L=n+m-1
$$
##### `dft()` / `idft()`
设变换长度为 $N$
时间复杂度：
$$
O(N\log N)
$$
`ensure()` 首次扩展到长度 $N$ 时还需要：
$$
O(N)
$$
预处理
`rev` 与 `roots` 共占：
$$
O(N)
$$
静态缓存
除缓存外，`dft()` 本身为原地操作，只需要 $O(1)$ 额外临时空间
##### `NTT::conv()`
当
$$
\min(n,m)\leq BRUTE
$$
时，使用普通卷积：
$$
O(nm)
$$
否则令

$$
N=2^{\lceil\log_2(n+m-1)\rceil}
$$
一般情况需要：
* 两次 DFT
* $O(N)$ 次逐点乘法
* 一次 IDFT
因此时间复杂度为：
$$
O(N\log N)
$$
空间复杂度为：
$$
O(N)
$$
另有最多 $O(N)$ 的共享 NTT 预处理缓存
当调用：
```cpp
conv(a, a)
```
并成功触发平方优化时，只需要：
* 一次 DFT
* $O(N)$ 次逐点平方
* 一次 IDFT
渐进复杂度仍为：
$$
O(N\log N)
$$
但常数更小
##### `conv_large()`
令
$$
M=MAX_N=2^{23}
$$
块长为
$$
B=M/2
$$
设两个输入分别被拆成
$$
p=\left\lceil\frac nB\right\rceil
\quad
q=\left\lceil\frac mB\right\rceil
$$
个块
首先需要对全部输入块执行 DFT：
$$
O((p+q)M\log M)
$$
对于所有块对 $(i,j)$，其频域乘积恰好在 `s = i + j` 时被累加一次。共有 $pq$ 对，每一对需要遍历 $M$ 个频点，因此为：
$$
O(pqM)
$$
共有
$$
p+q-1
$$
个输出块，每个输出块执行一次 IDFT，因此为：
$$
O((p+q)M\log M)
$$
总时间复杂度：
$$
O((p+q)M\log M+pqM)
$$
需要保存全部输入块的频域结果，因此主要空间复杂度为：
$$
O((p+q)M)
$$
另外还需要结果数组与一个长度 $M$ 的临时频域数组
因此 `conv_large()` 更适合**输入仅略微超过单次 NTT 上限，或者只有一侧块数较少**的情况；当 $p,q$ 都很大时，$pqM$ 的频域乘加和巨大的块缓存都会成为主要开销
