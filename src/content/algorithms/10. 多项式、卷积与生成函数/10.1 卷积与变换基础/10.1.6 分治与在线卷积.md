### CDQ 分治卷积与在线卷积
---
> 依赖：全局 `conv(a,b)`；若要快速卷积，系数用 `Z` 并引入 `10.1.1 NTT`。

```cpp
// ===== CDQ 分治卷积：固定核 + 前向递推；不需要时可整段删除 =====
// solve(i, s) 根据 sum_{j<i} f[j] * g[i-j] 确定 f[i]；g[0] 不参与 s
template<class T, class F>
vector<T> cdq_conv(int n, const vector<T> &g, F solve) {
    vector<T> f(n), sum(n);
    auto dfs = [&](auto &&self, int l, int r) -> void {
        if (r - l == 1) {
            f[l] = solve(l, sum[l]);
            return;
        }
        int m{ (l + r) >> 1 };
        self(self, l, m);
        // 左半已确定，将 f[l..m) 对 sum[m..r) 的贡献批量加入
        int len{ min(r - l, int(g.size())) };
        if (len > 1) {
            vector<T> a(f.begin() + l, f.begin() + m);
            vector<T> b(g.begin() + 1, g.begin() + len);
            vector<T> c{ ::conv(a, b) };
            for (int i = m; i < r; ++ i) {
                int k{ i - l - 1 };
                if (k < c.size()) sum[i] += c[k];
            }
        }
        self(self, m, r);
    };
    if (n) dfs(dfs, 0, n);
    return f;
}
// ===== Semi-Relaxed：f 预先已知，g 逐项加入；不需要时可整段删除 =====
template<class T = Z>
struct SemiRelaxedConv {
    vector<T> f, g, h;
    explicit SemiRelaxedConv(vector<T> f_) : f(move(f_)) {}
private:
    void ensure(int n) {
        if (h.size() < n) h.resize(n);
    }
    // 已知前 q 项 g；d = lowbit(q)
    // 新完成块 g[q-d..q) × f[0..2d) 中，只把次数 [d,2d) 加到未来 h[q..q+d)
    void add_future(int q, int d) {
        int m{ min(2 * d, int(f.size())) };
        if (!m) return;
        vector<T> a(g.end() - d, g.end());
        vector<T> b(f.begin(), f.begin() + m);
        vector<T> c{ ::conv(a, b) };
        int r{ min(2 * d, int(c.size())) };
        ensure(q + r - d);
        for (int i = d; i < r; ++ i) h[q + i - d] += c[i];
    }
public:
    // 加入 g[i]，返回完整的 h[i]
    T append(const T &x) {
        int i{ int(g.size()) };
        g.push_back(x);
        ensure(i + 1);
        // pending() 已提前维护其它项；这里只补当前未知的 f[0] * g[i]
        if (!f.empty()) h[i] += f[0] * x;
        int q{ i + 1 }, d{ q & -q };
        add_future(q, d);
        return h[i];
    }
    // ===== 查询辅助：只需要 append() 时以下函数可直接删除 =====
    int size() const { return int(g.size()); }
    // 要求 0 <= i < size()；返回已经完整确定的 h[i]
    const T &operator[](int i) const { return h[i]; }
    // 加入 g[n] 前，h[n] 中已经由 g[0..n) 确定的部分
    T pending() const {
        int n{ size() };
        return n < h.size() ? h[n] : T{};
    }
};
// ===== Relaxed：f、g 均逐项加入；不需要时可整段删除 =====
template<class T = Z>
struct RelaxedConv {
    vector<T> f, g, h;
private:
    void ensure(int n) {
        if (int(h.size()) < n) h.resize(n);
    }
    // 将局部卷积次数 [d,2d) 加到全局 h[q..q+d)
    void add_slice(int q, int d, const vector<T> &c) {
        int r{ min(2 * d, int(c.size())) };
        ensure(q + r - d);
        for (int i = d; i < r; ++ i) h[q + i - d] += c[i];
    }
    // 已知前 q 项；d = lowbit(q)，提前维护下一段尚未完成的卷积系数
    void add_future(int q, int d) {
        if (q == d) {
            // q 为 2 的幂：整个已知前缀第一次形成一个完整块
            vector<T> a(f.begin(), f.begin() + d);
            vector<T> b(g.begin(), g.begin() + d);
            add_slice(q, d, ::conv(a, b));
            return;
        }
        // 新块 [q-d,q) 分别与另一侧最前面的 2d 项交叉
        vector<T> a(f.begin() + q - d, f.begin() + q);
        vector<T> b(g.begin(), g.begin() + 2 * d);
        vector<T> c{ ::conv(a, b) };
        a.assign(g.begin() + q - d, g.begin() + q);
        b.assign(f.begin(), f.begin() + 2 * d);
        vector<T> t{ ::conv(a, b) };
        for (int i = 0; i < t.size(); ++ i) c[i] += t[i];
        add_slice(q, d, c);
    }
public:
    // 加入 f[i], g[i]，返回完整的 h[i]
    T append(const T &x, const T &y) {
        int i = f.size();
        f.push_back(x);
        g.push_back(y);
        ensure(i + 1);
        // h[i] 的内部项已经由之前的块提前加入，只补两个新出现的端点
        if (i == 0) h[0] += x * y;
        else h[i] += x * g[0] + y * f[0];
        int q{ i + 1 }, d{ q & -q };
        add_future(q, d);
        return h[i];
    }
    // ===== 查询辅助：只需要 append() 时以下函数可直接删除 =====
    int size() const { return int(f.size()); }
    // 要求 0 <= i < size()；返回已经完整确定的 h[i]
    const T &operator[](int i) const { return h[i]; }
    // 加入 f[n], g[n] 前，h[n] 中由旧系数产生的内部贡献
    T pending() const {
        int n{ size() };
        return n < int(h.size()) ? h[n] : T{};
    }
};
```

#### 接口说明

- `cdq_conv(n,g,solve)`：求 `f[i]=solve(i, Σ_{j<i}f[j]g[i-j])`，只计算旧项贡献，`g[0]` 不使用。
- `SemiRelaxedConv<T>(f)`：核 `f` 固定；`append(g_i)` 返回 `Σ_{j=0}^i f[j]g[i-j]`。
- `RelaxedConv<T>()`：两个数组同步在线加入；`append(f_i,g_i)` 返回卷积第 `i` 项。
- `pending()`：尚未加入下一项前已确定的 `h[n]` 的旧项部分。`size()`、`operator[]` 查询历史状态。

#### 注意事项

- 两种在线结构只允许**末尾追加**，不允许修改已有项；若两边已全部给出，直接 `conv` 更快。
- `SemiRelaxedConv::pending()` 不含新项 `f[0]*g[n]`；`RelaxedConv::pending()` 不含两项新端点 `f[0]*g[n] + f[n]*g[0]`（`n>0`）。
- CDQ 中 `solve(i,sum)` 应只依赖已经计算的 `f[0..i)`，不能读取未来项。`g[0]` 不参与 `sum`，即使它非零也是如此。
- 所有数据结构复用全局 `conv`；对于非 `Z` 的数组，只会执行朴素乘法，不保证快速复杂度。

#### 常见用法

```cpp
// f[0]=1, f[i]=sum_{j<i} f[j]*g[i-j]
vector<Z> g(n);
auto f = cdq_conv<Z>(n, g, [&](int i, Z sum) -> Z {
    return i == 0 ? Z(1) : sum;
});

// 固定核的在线卷积：加入序列 x
SemiRelaxedConv<Z> s(kernel);
Z old = s.pending();
Z value = s.append(x);

// 双侧在线卷积
RelaxedConv<Z> r;
Z prev = r.pending();
Z cur = r.append(x, y);
```

#### 复杂度

- CDQ：$O(N \log^2 N)$ 时间，$O(N)$ 存储加上临时卷积（使用 NTT）。
- 半在线 / 全在线：总计典型 $O(N \log^2 N)$ 时间，$O(N)$ 持久存储，临时卷积需 $O(N)$。
