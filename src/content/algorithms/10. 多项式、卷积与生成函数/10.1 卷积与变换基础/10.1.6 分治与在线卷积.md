### 分治与在线卷积
---
**解决问题：** 当卷积的一侧或两侧不能一次性全部得到，或者当前系数依赖之前已经求出的卷积贡献时，使用 CDQ、Semi-Relaxed 或 Relaxed Convolution 批量处理历史贡献
依赖前面的统一卷积接口：
```cpp
conv(a, b);
```
默认 `T = Z` 时使用 NTT
```cpp
// ===== CDQ 分治卷积：固定核 + 前向递推；不需要时可整段删除 =====
// solve(i, s) 根据 sum_{j<i} f[j] * g[i-j] 确定 f[i]；g[0] 不参与 s
template<class T, class F>
vector<T> cdq_conv(int n, const vector<T> &g, F solve) {
    vector<T> f(n), sum(n);
    auto dfs = [&](auto &&self, int l, int r) -> void {
        if (r - l == 1) {
            f[l] = solve(l, sum[l]);
            return;
        }
        int m{ (l + r) >> 1 };
        self(self, l, m);
        // 左半已确定，将 f[l..m) 对 sum[m..r) 的贡献批量加入
        int len{ min(r - l, int(g.size())) };
        if (len > 1) {
            vector<T> a(f.begin() + l, f.begin() + m);
            vector<T> b(g.begin() + 1, g.begin() + len);
            vector<T> c{ ::conv(a, b) };
            for (int i = m; i < r; ++ i) {
                int k{ i - l - 1 };
                if (k < c.size()) sum[i] += c[k];
            }
        }
        self(self, m, r);
    };
    if (n) dfs(dfs, 0, n);
    return f;
}
// ===== Semi-Relaxed：f 预先已知，g 逐项加入；不需要时可整段删除 =====
template<class T = Z>
struct SemiRelaxedConv {
    vector<T> f, g, h;
    explicit SemiRelaxedConv(vector<T> f_) : f(move(f_)) {}
private:
    void ensure(int n) {
        if (h.size() < n) h.resize(n);
    }
    // 已知前 q 项 g；d = lowbit(q)
    // 新完成块 g[q-d..q) × f[0..2d) 中，只把次数 [d,2d) 加到未来 h[q..q+d)
    void add_future(int q, int d) {
        int m{ min(2 * d, int(f.size())) };
        if (!m) return;
        vector<T> a(g.end() - d, g.end());
        vector<T> b(f.begin(), f.begin() + m);
        vector<T> c{ ::conv(a, b) };
        int r{ min(2 * d, int(c.size())) };
        ensure(q + r - d);
        for (int i = d; i < r; ++ i) h[q + i - d] += c[i];
    }
public:
    // 加入 g[i]，返回完整的 h[i]
    T append(const T &x) {
        int i{ int(g.size()) };
        g.push_back(x);
        ensure(i + 1);
        // pending() 已提前维护其它项；这里只补当前未知的 f[0] * g[i]
        if (!f.empty()) h[i] += f[0] * x;
        int q{ i + 1 }, d{ q & -q };
        add_future(q, d);
        return h[i];
    }
    // ===== 查询辅助：只需要 append() 时以下函数可直接删除 =====
    int size() const { return int(g.size()); }
    // 要求 0 <= i < size()；返回已经完整确定的 h[i]
    const T &operator[](int i) const { return h[i]; }
    // 加入 g[n] 前，h[n] 中已经由 g[0..n) 确定的部分
    T pending() const {
        int n{ size() };
        return n < h.size() ? h[n] : T{};
    }
};
// ===== Relaxed：f、g 均逐项加入；不需要时可整段删除 =====
template<class T = Z>
struct RelaxedConv {
    vector<T> f, g, h;
private:
    void ensure(int n) {
        if (int(h.size()) < n) h.resize(n);
    }
    // 将局部卷积次数 [d,2d) 加到全局 h[q..q+d)
    void add_slice(int q, int d, const vector<T> &c) {
        int r{ min(2 * d, int(c.size())) };
        ensure(q + r - d);
        for (int i = d; i < r; ++ i) h[q + i - d] += c[i];
    }
    // 已知前 q 项；d = lowbit(q)，提前维护下一段尚未完成的卷积系数
    void add_future(int q, int d) {
        if (q == d) {
            // q 为 2 的幂：整个已知前缀第一次形成一个完整块
            vector<T> a(f.begin(), f.begin() + d);
            vector<T> b(g.begin(), g.begin() + d);
            add_slice(q, d, ::conv(a, b));
            return;
        }
        // 新块 [q-d,q) 分别与另一侧最前面的 2d 项交叉
        vector<T> a(f.begin() + q - d, f.begin() + q);
        vector<T> b(g.begin(), g.begin() + 2 * d);
        vector<T> c{ ::conv(a, b) };
        a.assign(g.begin() + q - d, g.begin() + q);
        b.assign(f.begin(), f.begin() + 2 * d);
        vector<T> t{ ::conv(a, b) };
        for (int i = 0; i < t.size(); ++ i) c[i] += t[i];
        add_slice(q, d, c);
    }
public:
    // 加入 f[i], g[i]，返回完整的 h[i]
    T append(const T &x, const T &y) {
        int i = f.size();
        f.push_back(x);
        g.push_back(y);
        ensure(i + 1);
        // h[i] 的内部项已经由之前的块提前加入，只补两个新出现的端点
        if (i == 0) h[0] += x * y;
        else h[i] += x * g[0] + y * f[0];
        int q{ i + 1 }, d{ q & -q };
        add_future(q, d);
        return h[i];
    }
    // ===== 查询辅助：只需要 append() 时以下函数可直接删除 =====
    int size() const { return int(f.size()); }
    // 要求 0 <= i < size()；返回已经完整确定的 h[i]
    const T &operator[](int i) const { return h[i]; }
    // 加入 f[n], g[n] 前，h[n] 中由旧系数产生的内部贡献
    T pending() const {
        int n{ size() };
        return n < int(h.size()) ? h[n] : T{};
    }
};
```
#### 接口说明
三种版本的使用条件不同：

| 接口                | 第一侧  | 第二侧  | 当前答案何时得到               | 典型用途    |
| ----------------- | ---- | ---- | ---------------------- | ------- |
| `cdq_conv()`      | 递推产生 | 固定核  | 递归到当前位置                | 前向递推 DP |
| `SemiRelaxedConv` | 全部已知 | 在线产生 | `append(g[i])` 后       | 固定核在线卷积 |
| `RelaxedConv`     | 在线产生 | 在线产生 | `append(f[i], g[i])` 后 | 真正在线卷积  |
| `conv()`          | 全部已知 | 全部已知 | 一次性                    | 普通卷积    |
##### `cdq_conv(n, g, solve)`
计算一个满足前向依赖的序列
当处理位置 `i` 时，传给：
```cpp
solve(i, sum)
```
的 `sum` 为：
$$
\sum_{j<i}f_jg_{i-j}
$$
也就是说：
```cpp
auto f = cdq_conv<Z>(n, g, [&](int i, Z sum) {
    return ...;
});
```
必须能够仅根据：
$$
f_0,f_1,\ldots,f_{i-1}
$$
以及其它已知信息确定 `f[i]`
`g` 可以短于 `n`，不存在的系数视为 `0`
`n == 0` 时返回空数组
##### `SemiRelaxedConv(f)`
固定：
$$
F(x)=\sum_{i\ge0}f_ix^i
$$
预先全部已知，另一侧：
$$
G(x)=g_0+g_1x+\cdots
$$
逐项加入
```cpp
SemiRelaxedConv<Z> online(f);
Z h0 = online.append(g0);
Z h1 = online.append(g1);
```
第 `i` 次 `append(g[i])` 返回：
$$
h_i=[x^i]F(x)G(x)=\sum_{j=0}^{i}f_jg_{i-j}
$$
超出 `f.size()` 的系数视为 `0`
##### `SemiRelaxedConv::pending()`
假设已经加入：
$$
g_0,\ldots,g_{n-1}
$$
则：
```cpp
online.pending()
```
返回：
$$
\sum_{j=1}^{n}f_jg_{n-j}
$$
其中超出 `f` 范围的项自动忽略
因此下一项：
$$
h_n=f_0g_n+\operatorname{pending()}
$$
这使得它可以直接用于固定核递推
如果：
$$
f_0=0,
$$
则在 `g[n]` 尚未知时：
$$
h_n=\operatorname{pending()}
$$
已经完整确定
##### `RelaxedConv::append(x, y)`
两侧均逐项产生：
```cpp
RelaxedConv<Z> online;
Z hi = online.append(fi, gi);
```
加入第 `i` 对系数后返回：
$$
h_i=\sum_{j=0}^{i}f_jg_{i-j}
$$
调用之前不需要知道未来的 `f`、`g`
##### `RelaxedConv::pending()`
已经加入：
$$
f_0,\ldots,f_{n-1},\quad g_0,\ldots,g_{n-1}
$$
后：
```cpp
online.pending()
```
返回下一项中的内部贡献：
$$
\sum_{j=1}^{n-1}f_jg_{n-j}
$$
因此：
$$
h_n=f_0g_n+f_ng_0+\operatorname{pending()}.
$$
未来两个端点系数尚未产生，所以不包含在 `pending()` 中
##### `size()` 与 `operator[]`
```cpp
int n = online.size();
Z x = online[i];
```
其中 `operator[](i)` 要求：
$$
0\le i<\operatorname{size()}
$$
这些位置已经是最终卷积结果
内部 `h` 可能还保存了若干**未来位置的部分贡献**，但这些位置不能通过 `operator[]` 当作完整答案使用；需要下一项的已知部分时使用 `pending()`
#### 注意事项
##### CDQ 中 `g[0]` 不参与 `sum`
`cdq_conv()` 维护的是：
$$
\sum_{j<i}f_jg_{i-j}
$$
因为：
$$
i-j\ge1
$$
所以不会出现 `g[0]`
如果原递推含有当前项：
$$
f_i g_0
$$
需要在 `solve()` 中自行代数处理
例如：
$$
f_i=b_i+g_0f_i+\operatorname{sum}
$$
应写成：
$$
f_i=\frac{b_i+\operatorname{sum}}{1-g_0}
$$
##### `pending()` 依赖提前维护未来贡献
在线结构不是只在需要 `h[i]` 时才临时计算它。
处理完前 `q` 项后，令：
$$
d=\operatorname{lowbit}(q)
$$
刚刚完成的长度 `d` 块会通过一次或两次普通卷积，把当前已经能够确定的贡献提前加入：
$$
h_q,h_{q+1},\ldots
$$
中的对应区间
因此下一次 `append()` 时，当前系数的内部贡献已经准备好，只需要补入新系数产生的端点项
这也是 `pending()` 能保持 $O(1)$ 查询的原因
##### Semi-Relaxed 固定侧不能修改
构造：
```cpp
SemiRelaxedConv<Z> online(kernel);
```
后，内部保存了一份 `kernel`
之后只能按顺序：
```cpp
online.append(x);
```
不能再修改过去已经参与计算的固定侧或在线侧系数
##### Relaxed 也不支持修改历史
Relaxed Convolution 的“在线”是：
```text
不断在末尾加入新系数
```
而不是：
```text
任意修改过去的系数
```
一旦第 `i` 项已经 append，就视为永久确定
##### 两边已经全部已知时不要使用在线卷积
如果一开始已经有完整：
```cpp
vector<Z> a, b;
```
直接：
```cpp
auto c = conv(a, b);
```
即可
NTT 下普通卷积为：
$$
O(n\log n)
$$
而这些分治 / 在线结构通常为：
$$
O(n\log^2 n)
$$
在线结构的价值是**未来系数尚未产生时必须先得到当前答案**，而不是比普通卷积更快
##### 非 `Z` 时不会自动获得 NTT 复杂度
三套结构最终都依赖：
```cpp
conv(a, b);
```
因此复杂度由底层卷积决定
若 `T = Z`，统一 `conv()` 使用 NTT
若 `T = i64` 等类型仍走朴素卷积，则整体通常退化为 $O(n^2)$ 量级
#### 常见用法
##### CDQ 求三角递推
若：
$$
f_i=b_i+\sum_{j<i}f_jg_{i-j}
$$
可以直接：
```cpp
vector<Z> g(n), b(n);
auto f = cdq_conv<Z>(n, g, [&](int i, Z sum) {
    return b[i] + sum;
});
```
如果：
$$
f_0=1
$$
则：
```cpp
auto f = cdq_conv<Z>(n, g, [&](int i, Z sum) {
    if (i == 0) return Z(1);
    return b[i] + sum;
});
```
##### 半在线普通卷积
```cpp
vector<Z> f(n); // 预先全部已知
SemiRelaxedConv<Z> online(f);
for (int i = 0; i < n; ++ i) {
    Z gi = ...;
    Z hi = online.append(gi);
    // hi = sum_{j=0}^i f[j] * g[i-j]
}
```
##### 用 `pending()` 做固定核递推
考虑：
$$
x_n=b_n+\sum_{j=1}^{n}k_jx_{n-j}
$$
即：
$$
k_0=0
$$
可以写成：
```cpp
vector<Z> kernel(n);
kernel[0] = 0;
SemiRelaxedConv<Z> online(kernel);
vector<Z> x(n);
for (int i = 0; i < n; ++ i) {
    x[i] = b[i] + online.pending();
    online.append(x[i]);
}
```
这里 `pending()` 在 `x[i]` 尚未加入时已经等于：
$$
\sum_{j=1}^{i}k_jx_{i-j}
$$
如果 $k_0\ne0$ 且递推为：
$$
x_n=b_n+\sum_{j=0}^{n}k_jx_{n-j}
$$
则：
$$
x_n=\frac{b_n+\operatorname{pending()}}{1-k_0}
$$
##### 全在线普通卷积
```cpp
RelaxedConv<Z> online;

for (int i = 0; i < n; ++ i) {
    Z fi = ...;
    Z gi = ...;
    Z hi = online.append(fi, gi);
}
```
每次立即得到：
$$
h_i=\sum_{j=0}^{i}f_jg_{i-j}
$$
##### 使用全在线的下一项内部贡献
```cpp
RelaxedConv<Z> online;
for (int i = 0; i < n; ++ i) {
    Z known = online.pending();
    Z fi = ...; // 可以使用 known 推导
    Z gi = ...;
    Z hi = online.append(fi, gi);
}
```
在加入第 `i` 项前：
$$
\operatorname{known}=\sum_{j=1}^{i-1}f_jg_{i-j}
$$
#### 复杂度
设要得到前 `n` 个卷积系数，长度为 $k$ 的普通卷积复杂度记为：
$$
M(k).
$$
这些算法都按照二次幂块工作
对于大小：
$$
d=1,2,4,\ldots
$$
的块，大约会出现：
$$
O\left(\frac nd\right)
$$
次，因此更精确的总复杂度形式为：
$$
O\left(\sum_{d=1,2,4,\ldots\le n}\frac nd M(d)\right)
$$
##### 使用 NTT
当：
$$
M(d)=O(d\log d),
$$
有：
$$
\frac ndM(d)=O(n\log d)
$$
对所有二次幂规模求和得到：
$$
O(n\log^2n)
$$
因此：

| 接口                |     NTT 时间复杂度 | 主要额外空间 |   |      |
| ----------------- | ------------: | -----: | - | ---- |
| `cdq_conv()`      | $O(n\log^2n)$ | $O(n)$ |   |      |
| `SemiRelaxedConv` | $O(n\log^2n)$ |    $O( | f | +n)$ |
| `RelaxedConv`     | $O(n\log^2n)$ | $O(n)$ |   |      |
| 普通 `conv()`       |  $O(n\log n)$ | $O(n)$ |   |      |
其中 `SemiRelaxedConv` 的复杂度指处理前 `n` 次 `append()`；构造时保存固定序列还需要 $O(|f|)$ 空间
##### 使用朴素卷积
若底层：
$$
M(d)=O(d^2)
$$
则：
$$
\sum_d\frac nd\cdot d^2=O\left(n\sum_d d\right)=O(n^2)
$$
因此泛型并不意味着任意 `T` 都能达到 $O(n\log^2n)$；真正的大规模版本通常使用 `Z + NTT`
