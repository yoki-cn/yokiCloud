### 循环与负循环卷积
---
依赖前面的统一卷积接口：
```cpp
conv(a, b);
```
其中 `vector<Z>` 自动使用 NTT，其余类型使用对应的普通卷积后端
```cpp
namespace CyclicConv {
    // A(x) mod (x^n - 1)
    template<class T = Z>
    vector<T> cyclic_reduce(const vector<T> &a, int n) {
        assert(n > 0);
        vector<T> res(n);
        for (int i = 0; i < a.size(); ++ i) res[i % n] += a[i];
        return res;
    }
    // A(x) mod (x^n + 1)
    template<class T = Z>
    vector<T> negacyclic_reduce(const vector<T> &a, int n) {
        assert(n > 0);
        vector<T> res(n);
        for (int i = 0; i < a.size(); ++ i) {
            if ((i / n) & 1) res[i % n] -= a[i];
            else res[i % n] += a[i];
        }
        return res;
    }
    // ===== 通用循环卷积：先折回输入，再调用统一 conv() 后端 =====
    template<class T = Z>
    vector<T> conv_cyclic(const vector<T> &a, const vector<T> &b, int n) {
        assert(n > 0);
        if (a.empty() || b.empty()) return vector<T>(n);
        vector<T> f{ cyclic_reduce(a, n) };
        if (&a == &b) return cyclic_reduce(::conv(f, f), n);
        vector<T> g{ cyclic_reduce(b, n) };
        return cyclic_reduce(::conv(f, g), n);
    }
    // ===== 通用负循环卷积：先折回输入，再调用统一 conv() 后端 =====
    template<class T = Z>
    vector<T> conv_negacyclic(const vector<T> &a, const vector<T> &b, int n) {
        assert(n > 0);
        if (a.empty() || b.empty()) return vector<T>(n);
        vector<T> f{ negacyclic_reduce(a, n) };
        if (&a == &b) return negacyclic_reduce(::conv(f, f), n);
        vector<T> g{ negacyclic_reduce(b, n) };
        return negacyclic_reduce(::conv(f, g), n);
    }
    // ===== Z + NTT 直接循环卷积：长度 n 直接变换，避免普通卷积使用约 2n 的 NTT =====
    vector<Z> conv_cyclic_ntt(const vector<Z> &a, const vector<Z> &b, int n) {
        assert(n > 0 && (n & (n - 1)) == 0 && n <= NTT::MAX_N);
        vector<Z> f{ cyclic_reduce(a, n) };
        NTT::dft(f);
        if (&a == &b) {
            for (Z &x : f) x *= x;
        } else {
            vector<Z> g{ cyclic_reduce(b, n) };
            NTT::dft(g);
            for (int i = 0; i < n; ++ i) f[i] *= g[i];
        }
        NTT::idft(f);
        return f;
    }
    // ===== Z + NTT 直接负循环卷积：需要存在 2n 次单位根 =====
    vector<Z> conv_negacyclic_ntt(const vector<Z> &a, const vector<Z> &b, int n) {
        assert(n > 0 && (n & (n - 1)) == 0 && 2 * n <= NTT::MAX_N);
        vector<Z> f{ negacyclic_reduce(a, n) }, g;
        bool square{ &a == &b };
        if (!square) g = negacyclic_reduce(b, n);
        // w^n = -1；乘 w^i 后转化为普通长度 n 循环卷积
        Z w{ Z(NTT::G).pow((NTT::MOD - 1) / (2 * n)) }, pw{ 1 };
        for (int i = 0; i < n; ++ i) {
            f[i] *= pw;
            if (!square) g[i] *= pw;
            pw *= w;
        }
        NTT::dft(f);
        if (square) {
            for (Z &x : f) x *= x;
        } else {
            NTT::dft(g);
            for (int i = 0; i < n; ++ i) f[i] *= g[i];
        }
        NTT::idft(f);
        Z iw{ w.inv() };
        pw = 1;
        for (int i = 0; i < n; ++ i) {
            f[i] *= pw;
            pw *= iw;
        }
        return f;
    }
}
template<class T = Z>
vector<T> conv_cyclic(const vector<T> &a, const vector<T> &b, int n) {
    return CyclicConv::conv_cyclic(a, b, n);
}
template<class T = Z>
vector<T> conv_negacyclic(const vector<T> &a, const vector<T> &b, int n) {
    return CyclicConv::conv_negacyclic(a, b, n);
}
vector<Z> conv_cyclic_ntt(const vector<Z> &a, const vector<Z> &b, int n) {
    return CyclicConv::conv_cyclic_ntt(a, b, n);
}
vector<Z> conv_negacyclic_ntt(const vector<Z> &a, const vector<Z> &b, int n) {
    return CyclicConv::conv_negacyclic_ntt(a, b, n);
}
```
#### 接口说明
##### `cyclic_reduce(a, n)`
计算：
$$
A(x)\bmod(x^n-1)
$$
因为：
$$
x^n=1
$$
所以：
$$
x^i=x^{i\bmod n}
$$
因此所有下标模 `n` 相同的系数直接累加
返回长度始终为 `n`
输入长
###### `conv_cyclic(a, b, n)`
计算：
$$
A(x)B(x)\pmod{x^n-1}
$$
```cpp
auto c = conv_cyclic(a, b, n);
```
返回长度始终为 `n`
若两个输入已经是长度 `n` 的状态，则
$$
c_k= \sum_{\substack{0\le i,j<n\\i+j\equiv k\pmod n}} a_i b_j
$$
实现会先分别计算：
$$
A\bmod(x^n-1),\qquad
B\bmod(x^n-1),
$$
再调用统一的：
```cpp
conv()
```
进行普通卷积，最后再次模 $x^n-1$
这利用了：
$$
(AB)\bmod f=\big((A\bmod f)(B\bmod f)\big)\bmod f.
$$
因此即使原输入长度远大于 `n`，真正进入普通卷积的数组长度也始终只有 `n`
##### `conv_negacyclic(a, b, n)`
计算：
$$
A(x)B(x)\pmod{x^n+1}
$$
```cpp
auto c = conv_negacyclic(a, b, n);
```
当两个输入次数均小于 `n` 时：
$$
c_k=\sum_{i+j=k}a_i b_j-\sum_{i+j=k+n}a_i b_j
$$
与循环卷积的核心区别为：
$$
x^n=
\begin{cases}
1,&\text{循环卷积},\\
-1,&\text{负循环卷积}
\end{cases}
$$
###### `conv_cyclic_ntt(a, b, n)`
针对 `Z = MInt<998244353>` 的直接 NTT 循环卷积：
```cpp
auto c = conv_cyclic_ntt(a, b, n);
```
要求：
* `n > 0`
* `n` 为 $2$ 的幂
* $n\le2^{23}$
通用 `conv_cyclic()` 在两个长度为 `n` 的多项式之间先做普通卷积，需要表示最多 $2n-1$ 个系数，因此当 `n` 为二次幂时通常进行长度 `2n` 的 NTT
直接循环卷积工作于：
$$
\mathbb F[x]/(x^n-1)
$$
因此只需要长度 `n` 的 NTT
两个参数引用同一个对象时：
```cpp
conv_cyclic_ntt(a, a, n);
```
只进行一次正向 NTT，再逐点平方
##### `conv_negacyclic_ntt(a, b, n)`
针对 `Z` 的直接 NTT 负循环卷积：
```cpp
auto c = conv_negacyclic_ntt(a, b, n);
```
要求：
* `n > 0`
* `n` 为 $2$ 的幂
* $2n\le2^{23}$
即：
$$
n\le2^{22}
$$
取：
$$
w=g^{(MOD-1)/(2n)}
$$
则：
$$
w^n=-1.
$$
先进行：
$$
a_i\leftarrow a_iw^i
\quad
b_i\leftarrow b_iw^i
$$
再做普通长度 `n` 循环卷积
当某项发生一次环绕，即：
$$
i+j=k+n
$$
有：
$$
w^{i+j}=w^{k+n}=-w^k,
$$
因此环绕项会自动获得负号
最后再乘：
$$
w^{-i}
$$
即可恢复负循环卷积结果
##### 全局接口
推荐业务代码直接使用：
```cpp
conv_cyclic(a, b, n);
conv_negacyclic(a, b, n);

conv_cyclic_ntt(a, b, n);
conv_negacyclic_ntt(a, b, n);
```
与现有卷积接口统一：
```cpp
conv(a, b);                  // 普通卷积
conv_large(a, b);            // 超长 998244353 卷积
conv_mod(a, b, mod);         // 任意模卷 
conv_or(a, b);               // OR 卷积
conv_and(a, b);              // AND 卷积
conv_xor(a, b);              // XOR 卷积
conv_cyclic(a, b, n);        // 模 x^n - 1
conv_negacyclic(a, b, n);    // 模 x^n + 1
```
#### 注意事项
##### `n` 表示周期，而不是输入长度
接口中的：
```cpp
conv_cyclic(a, b, n);
```

表示在：
$$
\mathbb F[x]/(x^n-1)
$$
中计算
并不要求：
```cpp
a.size() == n;
b.size() == n;
```
输入可以短于或长于 `n`
长度不足的部分视为 `0`；长度超过 `n` 的部分会先按照对应模多项式折回
###### 空输入返回长度 `n` 的零数组
普通卷积通常满足：
```cpp
conv({}, b) == {};
```
但循环与负循环卷积工作在固定的 `n` 维商环中，因此：
```cpp
conv_cyclic({}, b, n);
conv_negacyclic({}, b, n);
```
均返回：
```cpp
vector<T>(n);
```
从而保证返回长度始终为 `n`
##### 通用版会先折回输入
不要先对任意长的 `a`、`b` 做完整普通卷积，再折回答案
因为：
$$
AB\bmod f=((A\bmod f)(B\bmod f))\bmod f,
$$
可以先将两个输入压缩到长度 `n`
对于：
```text
a.size() >> n
b.size() >> n
```
这种情况可以显著减少实际卷积规模
##### 直接 NTT 版与通用版的区别
若：
```cpp
a.size() <= n;
b.size() <= n;
```
且 `n` 是二次幂，则通用版中的普通卷积需要最多：
$$
2n-1
$$
个系数，因此 NTT 长度通常为：
$$
2n
$$
直接循环 / 负循环 NTT 只使用长度：
$$
n
$$
因此渐进复杂度相同，但直接版常数明显更小，内存也更少
##### 直接循环与负循环的长度上限不同
对于 `998244353`：
$$
MOD-1=119\cdot2^{23}
$$
直接循环卷积只需要 $n$ 次单位根，因此：
$$
n\le2^{23}
$$
直接负循环卷积需要 $2n$ 次单位根，因此：
$$
2n\le2^{23}
$$
即：
$$
n\le2^{22}
$$
##### 平方卷积优化
下面这些调用：
```cpp
conv_cyclic(a, a, n);
conv_negacyclic(a, a, n);
conv_cyclic_ntt(a, a, n);
conv_negacyclic_ntt(a, a, n);
```
都会利用两个参数引用同一对象的信息
通用版可以将平方优化继续传给底层 `conv(f, f)`；直接 NTT 版则只进行一次正向 NTT
两个内容相等但对象不同的数组不会触发：
```cpp
vector<Z> b = a;
conv_cyclic_ntt(a, b, n); // 不触发平方优化
```
#### 常见用法
##### 普通循环卷积
```cpp
vector<Z> a, b;
auto c = conv_cyclic(a, b, n);
```
得到：
$$
A(x)B(x)\bmod(x^n-1)
$$
##### 普通负循环卷积
```cpp
auto c = conv_negacyclic(a, b, n);
```
得到：
$$
A(x)B(x)\bmod(x^n+1)
$$
##### 直接 NTT 循环卷积
当 `n` 是合法 NTT 长度，并且性能敏感时：
```cpp
vector<Z> a, b;
auto c = conv_cyclic_ntt(a, b, n);
```
要求：
$$
n\le2^{23}
$$
##### 直接 NTT 负循环卷积
```cpp
auto c = conv_negacyclic_ntt(a, b, n);
```
要求：
$$
n\le2^{22}
$$
如果 `n` 不满足直接 NTT 的条件，仍然可以使用通用：
```cpp
conv_negacyclic(a, b, n);
```
只要其底层普通卷积能够处理对应规模
#### 复杂度
设：
$$
A=|a| \quad B=|b|
$$
周期为 `n`
记两个长度至多为 `n` 的数组进行普通卷积的复杂度为：
$$
M(n)
$$
##### 折回
```cpp
cyclic_reduce(a, n);
negacyclic_reduce(a, n);
```
需要初始化长度 `n` 的结果并扫描输入，因此时间复杂度为：
$$
O(|a|+n)
$$
空间复杂度为：
$$
O(n)
$$
##### 通用循环 / 负循环卷积
首先分别折回两个输入：
$$
O(A+B+n)
$$
随后只对两个长度 `n` 的数组调用普通卷积，再线性折回答案
总时间复杂度为：
$$
O(A+B+n+M(n))
$$
通常简写为：
$$
O(A+B+M(n))
$$
当 `T = Z` 且使用 NTT 时：
$$
M(n)=O(n\log n)
$$
因此：
$$
O(A+B+n\log n)
$$
若原输入本身长度均为 $O(n)$，则为：
$$
O(n\log n)
$$
对于二次幂 `n`，通用 NTT 版本通常实际执行长度约 `2n` 的变换
##### 直接 NTT 版本
先折回输入，然后执行长度恰为 `n` 的 NTT
时间复杂度：
$$
O(A+B+n\log n)
$$
若输入长度均为 $O(n)$：
$$
O(n\log n)
$$
空间复杂度：
$$
O(n)
$$
平方卷积会少进行一次正向 NTT，但渐进复杂度不变

| 接口                             |            时间复杂度 | 返回长度 |
| ------------------------------ | ---------------: | ---: |
| `cyclic_reduce(a, n)`          |     $O(\|a\|+n)$ |  $n$ |
| `negacyclic_reduce(a, n)`      |         $O(a+n)$ |  $n$ |
| `conv_cyclic(a, b, n)`         |    $O(A+B+M(n))$ |  $n$ |
| `conv_negacyclic(a, b, n)`     |    $O(A+B+M(n))$ |  $n$ |
| `conv_cyclic_ntt(a, b, n)`     | $O(A+B+n\log n)$ |  $n$ |
| `conv_negacyclic_ntt(a, b, n)` | $O(A+B+n\log n)$ |  $n$ |
