### 任意模数卷积（三模 NTT + CRT）
---
```cpp
namespace AnyModConv {
    constexpr int P1 = 167772161, P2 = 469762049, P3 = 998244353;
    constexpr int G = 3, MAX_N = 1 << 23;
    constexpr i64 P12 = 1LL * P1 * P2;

    template<int P>
    int power(int a, int b) {
        i64 r = 1;
        for (; b; b >>= 1, a = 1LL * a * a % P) {
            if (b & 1) r = r * a % P;
        }
        return r;
    }
    template<int P>
    void ntt(vector<int> &a, bool inverse = false) {
        int n = a.size();
        assert(n > 0 && has_single_bit(unsigned(n)) && (P - 1) % n == 0);
        for (int i = 1, j = 0; i < n; ++ i) {
            int b = n >> 1;
            while (j & b) j ^= b, b >>= 1;
            j ^= b;
            if (i < j) swap(a[i], a[j]);
        }
        for (int len = 2; len <= n; len <<= 1) {
            int w = power<P>(G, (P - 1) / len);
            if (inverse) w = power<P>(w, P - 2);
            for (int i = 0; i < n; i += len) {
                i64 t = 1;
                for (int j = 0; j < len / 2; ++ j) {
                    int x = a[i + j], y = t * a[i + j + len / 2] % P;
                    int u = x + y, v = x - y;
                    if (u >= P) u -= P;
                    if (v < 0) v += P;
                    a[i + j] = u;
                    a[i + j + len / 2] = v;
                    t = t * w % P;
                }
            }
        }
        if (inverse) {
            int inv = power<P>(n, P - 2);
            for (int &x : a) x = 1LL * x * inv % P;
        }
    }
    template<int P>
    vector<int> multiply(const vector<int> &a, const vector<int> &b) {
        int need = a.size() + b.size() - 1;
        int n = int(bit_ceil(unsigned(need)));
        assert(n <= MAX_N);
        vector<int> f(n);
        for (int i = 0; i < int(a.size()); ++ i) f[i] = a[i] % P;
        ntt<P>(f);
        if (&a == &b) {
            for (int &x : f) x = 1LL * x * x % P;
        } else {
            vector<int> g(n);
            for (int i = 0; i < int(b.size()); ++ i) g[i] = b[i] % P;
            ntt<P>(g);
            for (int i = 0; i < n; ++ i) f[i] = 1LL * f[i] * g[i] % P;
        }
        ntt<P>(f, true);
        f.resize(need);
        return f;
    }
    template<class T>
    vector<int> conv_mod(const vector<T> &a, const vector<T> &b, int mod) {
        static_assert(is_integral_v<T>);
        assert(mod > 0);
        if (a.empty() || b.empty()) return {};
        int need = a.size() + b.size() - 1;
        if (mod == 1) return vector<int>(need);
        vector<int> x(a.size()), y;
        auto norm = [&](T v) -> int {
            i64 r = i64(v % mod);
            if (r < 0) r += mod;
            return int(r);
        };
        for (int i = 0; i < int(a.size()); ++ i) x[i] = norm(a[i]);
        bool square = &a == &b;
        if (!square) {
            y.resize(b.size());
            for (int i = 0; i < int(b.size()); ++ i) y[i] = norm(b[i]);
        }
        const vector<int> &z = square ? x : y;
        if (min(x.size(), z.size()) <= 40) {
            vector<int> c(need);
            for (int i = 0; i < int(x.size()); ++ i) {
                for (int j = 0; j < int(z.size()); ++ j) {
                    c[i + j] = (c[i + j] + 1LL * x[i] * z[j]) % mod;
                }
            }
            return c;
        }
        assert(bit_ceil(unsigned(need)) <= MAX_N);
        auto c1 = multiply<P1>(x, z);
        auto c2 = multiply<P2>(x, z);
        auto c3 = multiply<P3>(x, z);
        static const i64 i12 = power<P2>(P1, P2 - 2);
        static const i64 i123 = power<P3>(P12 % P3, P3 - 2);
        vector<int> c(need);
        for (int i = 0; i < need; ++ i) {
            i64 t = (c2[i] - c1[i] + P2) * i12 % P2;
            i64 v = c1[i] + P1 * t;
            i64 u = ((c3[i] - v % P3 + P3) % P3) * i123 % P3;
            c[i] = (v + (__int128)P12 * u) % mod;
        }
        return c;
    }
}
template<class T>
vector<int> conv_mod(const vector<T> &a, const vector<T> &b, int mod) {
    return AnyModConv::conv_mod(a, b, mod);
}
```

#### 接口说明

- `conv_mod(a,b,mod)`：返回 `vector<int>`，系数在 `[0,mod)`；支持负数输入与任意 `1 <= mod <= INT_MAX`。
- `AnyModConv::ntt<P>(a,inverse)`：单素数原地 NTT，仅供该实现内部使用。
- `AnyModConv::multiply<P>(a,b)`：内部单模卷积，可单独按需复制。

#### 注意事项

- **严格精确的前提**：`bit_ceil(|a|+|b|-1) <= 2^23`、`mod <= INT_MAX`。此时卷积每个非负系数不超过 `min(|a|,|b|)(mod-1)^2`，在长度限制下小于 `P1*P2*P3`，CRT 重建唯一。
- 通过先按目标模数归一化，避免输入负值；`i64` 乘法安全，最终 CRT 汇总必须使用 `__int128`。
- 若 `mod == 998244353`，应优先直接用单模 `NTT::conv`，速度约为三模的 3 倍。
- 不支持系数绝对值超出整型 `T`、结果长度超限、目标模数超出 `int` 的情况。仅当有明确需要时再加大长度/任意大模数方案。

#### 常见用法

```cpp
vector<i64> a{1, -2, 3}, b{4, 5};
auto c = conv_mod(a, b, 1000000007); // {4, 1000000004, 2, 15}
auto d = conv_mod(a, a, 1000000000); // 平方优化
```

#### 复杂度

- $O(N \log N)$ 时间（3 次独立卷积），$O(N)$ 额外空间；小数组回退 $O(|a||b|)$。
