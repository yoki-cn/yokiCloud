### 三模 NTT + CRT
---
```cpp
namespace AnyModConv {
    constexpr int P1{ 167772161 }; // 5 * 2^25 + 1
    constexpr int P2{ 469762049 }; // 7 * 2^26 + 1
    constexpr int P3{ 998244353 }; // 119 * 2^23 + 1
    constexpr int G{ 3 }, MAX_N{ 1 << 23 }, BRUTE{ 60 };
    constexpr i64 P12{ 1LL * P1 * P2 };
    static int ntt_size(int need) { return bit_ceil(unsigned(need)); }
    template<int MOD>
    int qpow(int a, i64 b) {
        i64 res{ 1 };
        for (; b; b >>= 1, a = i64(a) * a % MOD) {
            if (b & 1) res = res * a % MOD;
        }
        return res;
    }
    template<int MOD>
    void ntt(vector<int> &a, bool inverse = false) {
        int n{ int(a.size()) };
        assert(n && (n & (n - 1)) == 0 && (MOD - 1) % n == 0);
        for (int i = 1, j = 0; i < n; ++ i) {
            int bit{ n >> 1 };
            for (; j & bit; bit >>= 1) j ^= bit;
            j ^= bit;
            if (i < j) swap(a[i], a[j]);
        }
        for (int len = 2; len <= n; len <<= 1) {
            int wlen{ qpow<MOD>(G, (MOD - 1) / len) };
            if (inverse) wlen = qpow<MOD>(wlen, MOD - 2);
            for (int i = 0; i < n; i += len) {
                i64 w{ 1 };
                for (int j = 0; j < len / 2; ++ j) {
                    int u{ a[i + j] }, v{ int(w * a[i + j + len / 2] % MOD) };
                    int x{ u + v }, y{ u - v };
                    if (x >= MOD) x -= MOD;
                    if (y < 0) y += MOD;
                    a[i + j] = x;
                    a[i + j + len / 2] = y;
                    w = w * wlen % MOD;
                }
            }
        }
        if (inverse) {
            int inv_n{ qpow<MOD>(n, MOD - 2) };
            for (int &x : a) x = i64(x) * inv_n % MOD;
        }
    }
    // ===== 单个 NTT 质数下的卷积：conv_mod() 内部依赖，不建议单独使用 =====
    template<int MOD>
    vector<int> conv_ntt(const vector<i64> &a, const vector<i64> &b) {
        if (a.empty() || b.empty()) return {};
        int need{ int(a.size() + b.size() - 1) }, n{ ntt_size(need) };
        assert(n <= MAX_N);
        vector<int> x(n);
        for (int i = 0; i < a.size(); ++ i) x[i] = a[i] % MOD;
        ntt<MOD>(x);
        // conv_ntt(a, a) 时少做一次正向 NTT
        if (&a == &b) {
            for (int &v : x) v = i64(v) * v % MOD;
        } else {
            vector<int> y(n);
            for (int i = 0; i < b.size(); ++ i) y[i] = b[i] % MOD;
            ntt<MOD>(y);
            for (int i = 0; i < n; ++ i) x[i] = i64(x[i]) * y[i] % MOD;
        }
        ntt<MOD>(x, true);
        x.resize(need);
        return x;
    }
    // ===== 任意正 int 模卷积：三模 NTT + CRT，严格精确 =====
    template<class T>
    vector<int> conv_mod(const vector<T> &a, const vector<T> &b, int mod) {
        static_assert(is_integral_v<T>);
        assert(mod > 0);
        if (a.empty() || b.empty()) return {};
        int na = a.size(), nb = b.size(), need{ na + nb - 1 };
        if (mod == 1) return vector<int>(need);
        auto norm = [&](T value) -> i64 {
            i128 x{ value };
            x %= mod;
            if (x < 0) x += mod;
            return i64(x);
        };
        bool square{ &a == &b };
        vector<i64> x(na), y;
        for (int i = 0; i < na; ++ i) x[i] = norm(a[i]);
        if (!square) {
            y.resize(nb);
            for (int i = 0; i < nb; ++ i) y[i] = norm(b[i]);
        }
        const vector<i64> &z{ square ? x : y };
        if (min(na, nb) <= BRUTE) {
            vector<int> res(need);
            for (int i = 0; i < na; ++ i) {
                for (int j = 0; j < nb; ++ j) {
                    res[i + j] = (res[i + j] + x[i] * z[j]) % mod;
                }
            }
            return res;
        }
        int n{ ntt_size(need) };
        assert(n <= MAX_N);
        vector<int> c1{ conv_ntt<P1>(x, z) };
        vector<int> c2{ conv_ntt<P2>(x, z) };
        vector<int> c3{ conv_ntt<P3>(x, z) };
        static const i64 inv_p1_p2{ qpow<P2>(P1 % P2, P2 - 2) };
        static const i64 inv_p12_p3{ qpow<P3>(P12 % P3, P3 - 2) };
        vector<int> res(need);
        for (int i = 0; i < need; ++ i) {
            i64 t1{ (c2[i] - c1[i] + P2) % P2 * inv_p1_p2 % P2 };
            i64 x12{ c1[i] + i64(P1) * t1 };
            i64 t2{ (c3[i] - x12 % P3 + P3) % P3 * inv_p12_p3 % P3 };
            i128 value{ i128(x12) + i128(P12) * t2 };
            res[i] = value % mod;
        }
        return res;
    }
}
template<class T>
vector<int> conv_mod(const vector<T> &a, const vector<T> &b, int mod) {
    return AnyModConv::conv_mod(a, b, mod);
}
```
#### 接口说明
##### 常量
* `P1 = 167772161`：$5\cdot2^{25}+1$
* `P2 = 469762049`：$7\cdot2^{26}+1$
* `P3 = 998244353`：$119\cdot2^{23}+1$
* `G = 3`：三个 NTT 质数均可使用的原根
* `MAX_N = 2^23`：三模共同支持的最大 NTT 长度
* `BRUTE = 60`：小规模卷积阈值
* `P12 = P1P2`：CRT 合并使用
三个模数的乘积为
$$
P_1P_2P_3=78'674'626'319'836'206'717'730'817
$$
##### `AnyModConv::ntt_size(need)`
返回不小于 `need` 的最小二次幂：
```cpp
int n = AnyModConv::ntt_size(need);
```
即
$$
n=2^{\lceil\log_2 need\rceil}
$$
要求 `need > 0`
##### `AnyModConv::ntt<MOD>(a, inverse)`
内部单模 NTT
```cpp
AnyModConv::ntt<998244353>(a);
AnyModConv::ntt<998244353>(a, true);
```
要求：
* `a` 非空
* `a.size()` 为 $2$ 的幂
* `a.size()` 整除 `MOD - 1`
* `MOD` 为本板子使用的 NTT 质数之一
这是 `conv_ntt()` 的底层实现，通常无需直接调用
##### `AnyModConv::conv_ntt<MOD>(a, b)`
在一个内部 NTT 质数下计算卷积：
```cpp
auto c = AnyModConv::conv_ntt<998244353>(a, b);
```
主要供 `conv_mod()` 内部使用，不建议作为普通单模卷积接口
若两个参数引用同一个对象：
```cpp
AnyModConv::conv_ntt<P1>(a, a);
```
只进行一次正向 NTT，再逐点平方
普通 `998244353` 卷积应直接使用前面的：
```cpp
conv(a, b);
```
无需经过任意模卷积板子
##### `AnyModConv::conv_mod(a, b, mod)`
主接口：
```cpp
auto c = AnyModConv::conv_mod(a, b, mod);
```
计算
$$
c_k=\sum_{i+j=k}a_ib_j\pmod {mod}
$$
要求：
* `T` 为整数类型
* `mod > 0`
* `mod` 能存入正 `int`
* NTT 分支补齐后的长度不超过 $2^{23}$
返回：
```cpp
vector<int>
```
且每项均满足
$$
0\le c_i<mod
$$
输入无需提前取模，可以包含负数
当：
```cpp
min(a.size(), b.size()) <= BRUTE
```
时直接在目标模数下使用朴素卷积
否则分别在三个 NTT 质数下计算卷积，并通过 CRT 恢复原始整数卷积系数，最后对目标 `mod` 取模
`mod == 1` 时直接返回全 `0` 数组
##### 全局 `conv_mod(a, b, mod)`
```cpp
auto c = conv_mod(a, b, mod);
```
等价于：
```cpp
auto c = AnyModConv::conv_mod(a, b, mod);
```
推荐业务代码直接使用全局接口，与：
```cpp
conv(a, b);
conv_large(a, b);
```
保持一致
#### 注意事项
##### 使用范围
该接口支持任意正 `int` 模数：
```cpp
conv_mod(a, b, 1000000007); // 质数
conv_mod(a, b, 1000000000); // 合数
```
目标模数不需要：
* 是质数
* 存在适合 NTT 的原根
* 满足 `mod - 1` 含有较大的二次幂因子
只有内部三个辅助模数需要支持 NTT
##### 为什么三模 CRT 在这里严格正确
输入首先规范到：
$$
0\le a_i,b_i<mod<2^{31}
$$
若 NTT 补齐长度不超过 $2^{23}$，则
$$
|a|+|b|-1\le2^{23}
$$
因此单个卷积系数最多包含
$$
\min(|a|,|b|)\le2^{22}
$$
个乘积
于是原始非负整数卷积系数满足
$$
c_k<2^{22}\cdot2^{31}\cdot2^{31}=2^{84}
$$
而
$$
P_1P_2P_3>2^{86}
$$
所以一定有
$$
0\le c_k<P_1P_2P_3.
$$
因此三个余数唯一确定的不是某个不确定的同余类代表元，而就是原始整数卷积系数本身
最后再计算：
$$
c_k\bmod mod
$$
即可得到严格正确的任意模卷积结果
##### CRT 合并
设三个 NTT 结果分别为：
$$
r_1,\quad r_2,\quad r_3
$$
先恢复前两个模数：
$$
t_1=(r_2-r_1)P_1^{-1}\pmod {P_2},
$$
$$
x_{12}=r_1+P_1t_1.
$$
再加入第三个模数：
$$
t_2=(r_3-x_{12})(P_1P_2)^{-1}\pmod {P_3}
$$
$$
x=x_{12}+P_1P_2t_2
$$
最后一步可能超过 `i64`，因此必须使用：
```cpp
i128 value;
```
不能先在 `i64` 中乘法再强制转换
##### 长度限制
三个质数分别支持最大 NTT 长度：
$$
2^{25},\quad2^{26},\quad2^{23}
$$
三模必须同时使用，因此公共上限为：
$$
MAX_N=2^{23}
$$
不要因为前两个模数支持更长 NTT 就单独增大 `MAX_N`
这里限制的是**补齐后的 NTT 长度**：
```cpp
int n = AnyModConv::ntt_size(a.size() + b.size() - 1);
```
必须满足：
```cpp
n <= AnyModConv::MAX_N;
```
##### 与普通 NTT / FFT 任意模卷积的选择
目标模数就是 `998244353` 时，优先使用：
```cpp
auto c = conv(a, b);
```
只需要一套 NTT，常数明显小于三模 CRT
要求任意模且需要整数意义上的严格正确时，使用：
```cpp
auto c = conv_mod(a, b, mod);
```
如果能够接受浮点误差条件，也可以使用 FFT 板子的：
```cpp
FFT<> F;
auto c = F.conv_mod(a, b, mod);
```
其高低位拆分 MTT 常数通常更小，但本质仍是浮点算法
因此三者的定位为：
```text
conv(a, b)
    998244353，单模 NTT，严格精确

FFT<>::conv_mod(a, b, mod)
    任意模，MTT，浮点算法

conv_mod(a, b, mod)
    任意正 int 模，三模 NTT + CRT，严格精确
```
#### 常见用法
##### 模 `1e9+7` 卷积
```cpp
vector<int> a{ 1, 2, 3 };
vector<int> b{ 4, 5, 6 };
auto c = conv_mod(a, b, 1000000007);
```
得到：
```text
4 13 28 27 18
```
对应：
$$
(1+2x+3x^2)(4+5x+6x^2)
$$
##### 合数模数
目标模数不要求为质数：
```cpp
int mod{ 1000000000 };
auto c = conv_mod(a, b, mod);
```
##### 输入含负数
```cpp
vector<i64> a{ 1, -2, 3 };
vector<i64> b{ -4, 5 };
auto c = conv_mod(a, b, 1000000007);
```
内部会自动将所有输入规范到：
$$
[0,mod)
$$
无需提前手动写：
```cpp
(x % mod + mod) % mod;
```
#### 复杂度
设：
$$
n=|a|,\quad m=|b|
$$
结果长度为：
$$
L=n+m-1
$$
NTT 分支实际变换长度为：
$$
N=2^{\lceil\log_2L\rceil}
$$
##### 三模 NTT + CRT
普通情况下，每个模数需要：
* 两次正向 NTT
* 一次逆 NTT
三个模数共进行 `9` 次 NTT
NTT 部分：
$$
O(N\log N)
$$
CRT 合并：
$$
O(n+m)
$$
因此总时间复杂度为：
$$
O(N\log N)
$$
空间复杂度为：
$$
O(N)
$$
当调用：
```cpp
conv_mod(a, a, mod);
```
触发平方优化后，每个模数只需要：
* 一次正向 NTT
* 一次逆 NTT
总计 `6` 次 NTT
渐进复杂度仍为：
$$
O(N\log N)
$$
但常数更小
