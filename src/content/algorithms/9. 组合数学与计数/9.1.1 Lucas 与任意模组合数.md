### Lucas 与任意模组合数
---
固定质数模直接使用已有 `Comb::Lucas()`。本节只补充**运行时质数 Lucas**与**任意合数模组合数**
```cpp
// ===== 运行时质数 Lucas：p 很大但每个数位较小的一侧不大时使用 =====
i64 comb_p(i64 n, i64 k, i64 p) {
    if (n < 0 || k < 0 || k > n || n >= p) return 0;
    k = min(k, n - k);
    i64 a{ 1 }, b{ 1 };
    for (i64 i = 1; i <= k; ++ i) {
        a = i64(i128(a) * (n - k + i) % p);
        b = i64(i128(b) * i % p);
    }
    return i64(i128(a) * qpow_mod(b, p - 2, p) % p);
}
i64 lucas(i64 n, i64 k, i64 p) {
    if (n < 0 || k < 0 || k > n) return 0;
    i64 ans{ 1 };
    while (k) {
        i64 a{ n % p }, b{ k % p };
        if (b > a) return 0;
        ans = i64(i128(ans) * comb_p(a, b, p) % p);
        n /= p;
        k /= p;
    }
    return ans;
}
// ===== 任意合数模：素数幂组合数 + CRT =====
struct CombMod {
    struct PP {
        int p, e, m;
        vector<i64> pre;
        PP(int p_, int e_, int m_) : p(p_), e(e_), m(m_), pre(m + 1, 1) {
            for (int i = 1; i <= m; ++ i) {
                pre[i] = pre[i - 1];
                if (i % p) pre[i] = i64(i128(pre[i]) * i % m);
            }
        }
        i64 vp(i64 n) const {
            i64 res{};
            while (n) n /= p, res += n;
            return res;
        }
        i64 fac0(i64 n) const {
            i64 res{ 1 };
            while (n) {
                if ((n / m) & 1) res = i64(i128(res) * pre[m] % m);
                res = i64(i128(res) * pre[n % m] % m);
                n /= p;
            }
            return res;
        }
        i64 C(i64 n, i64 k) const {
            if (n < 0 || k < 0 || k > n) return 0;
            i64 c{ vp(n) - vp(k) - vp(n - k) };
            if (c >= e) return 0;
            i64 a{ fac0(n) }, b{ fac0(k) }, d{ fac0(n - k) };
            i64 iv{ *inverse_mod(i64(i128(b) * d % m), m) };
            return i64(i128(a) * iv % m * qpow_mod(p, c, m) % m);
        }
    };
    int m;
    vector<PP> a;
    explicit CombMod(int m_) : m(m_) {
        int x{ m };
        for (int p = 2; i64(p) * p <= x; ++ p) if (x % p == 0) {
            int e{}, q{ 1 };
            while (x % p == 0) {
                x /= p;
                ++ e;
                q *= p;
            }
            a.emplace_back(p, e, q);
        }
        if (x > 1) a.emplace_back(x, 1, x);
    }
    i64 C(i64 n, i64 k) const {
        if (n < 0 || k < 0 || k > n || m == 1) return 0;
        i64 ans{}, mod{ 1 };
        for (const auto &f : a) {
            i64 r{ f.C(n, k) }, d{ (r - ans) % f.m };
            if (d < 0) d += f.m;
            i64 t{ i64(i128(d) * *inverse_mod(mod % f.m, f.m) % f.m) };
            ans += mod * t;
            mod *= f.m;
            ans %= mod;
        }
        return ans;
    }
};
```
#### 选择

| 条件                          | 使用                    |
| --------------------------- | --------------------- |
| 固定质数模，$n<mod$               | `comb.C(n,k)`         |
| 固定质数模，$n\ge mod$            | `comb.Lucas(n,k)`     |
| 运行时质数 `p`，且 Lucas 每位较小的一侧不大 | `lucas(n,k,p)`        |
| 任意合数模，各素数幂因子可以预处理           | `CombMod(mod).C(n,k)` |
#### 接口
- `comb_p(n,k,p)`：计算 $0\le n<p$ 时的 $\binom nk\bmod p$；要求 `p` 为质数
- `lucas(n,k,p)`：运行时质数 Lucas，不做 $O(p)$ 预处理
- `CombMod(m)`：分解合数模并预处理各素数幂；要求 `m > 0`
- `CombMod::C(n,k)`：计算 $\binom nk\bmod m$
`CombMod` 内部的 `PP` 仅用于模 $p^e$ 计算，通常不需要单独调用
依赖已有：
```cpp
qpow_mod(a, b, mod);
inverse_mod(a, mod);
```
以及 `i128`
#### 注意事项
- `comb.Lucas()` 与 `lucas()` 都要求模数为质数
- `comb.Lucas()` 内部调用 `Comb::C()`；若模数很大且 Lucas 数位很大，自动扩容可能不可接受，此时使用无预处理的 `lucas()`
- `lucas()` 每个数位需要 $O(\min(k_i,n_i-k_i))$ 次乘法和一次快速幂，只适合较小的一侧不大
- `CombMod` 不能直接对 `n! / k!(n-k)!` 求逆；需要先拆出各素因子的指数，再对剩余部分求逆
- `CombMod` 需要为每个素数幂 $p^e$ 开 $O(p^e)$ 空间，因此模数含很大的素数幂时不适用
- `CombMod` 使用试除分解 `m`；这里 `m` 为 `int` 时已经足够，没必要为此依赖 Pollard--Rho
#### 常见用法
##### 固定质数模
```cpp
Comb<> comb;
Z ans = comb.Lucas(n, k);
```
##### 运行时质数
```cpp
i64 ans = lucas(n, k, p);
```
##### 任意合数模
```cpp
CombMod comb(mod);
i64 ans = comb.C(n, k);
```
#### 复杂度

| 操作 | 时间复杂度 | 空间复杂度 |
|---|---:|---:|
| `lucas(n,k,p)` | $O\left(\sum_i\min(k_i,n_i-k_i)+\log_p n\log p\right)$ | $O(1)$ |
| `CombMod` 构造 | $O(\sqrt m+\sum p_i^{e_i})$ | $O(\sum p_i^{e_i})$ |
| `CombMod::C` | $O\left(\sum_i\log_{p_i}n+\log m\right)$ | $O(1)$ 额外空间 |
