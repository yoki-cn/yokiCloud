### 排列组合与基础计数
---
#### 杨辉三角（Pascal's Triangle）
```cpp
vector<vector<i64>> pascal(int n, i64 mod) {
    vector c(n + 1, vector<i64>(n + 1));
    c[0][0] = 1 % mod;
    for (int i = 1; i <= n; ++ i) {
        c[i][0] = c[i][i] = 1 % mod;
        for (int j = 1; j < i; ++ j) {
            c[i][j] = (i128(c[i - 1][j - 1]) + c[i - 1][j]) % mod;
        }
    }
    return c;
}
```
- `pascal(n, mod)`：返回所有 $0\le k\le i\le n$ 的 $\binom{i}{k}\bmod mod$；要求 `n >= 0`、`mod > 0`，$O(n^2)$
#### 精确组合数
```cpp
i128 comb_i128(i64 n, i64 k) {
    if (n < 0 || k < 0 || k > n) return 0;
    k = min(k, n - k);
    i128 ans{ 1 };
    for (i64 i = 1; i <= k; ++ i) {
        i64 a{ n - k + i }, b{ i };
        i64 g{ gcd(a, b) };
        a /= g;
        b /= g;
        g = gcd(i64(ans % b), b);
        ans /= g;
        b /= g;
        ans *= a;
    }
    return ans;
}
template<class Big>
Big comb_big(i64 n, i64 k) {
    if (n < 0 || k < 0 || k > n) return Big(0);
    k = min(k, n - k);
    Big ans{ 1 };
    for (i64 i = 1; i <= k; ++ i) {
        ans *= n - k + i;
        ans /= i;
    }
    return ans;
}
```
