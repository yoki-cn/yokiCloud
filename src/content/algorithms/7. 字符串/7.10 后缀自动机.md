### SAM
---
```cpp
template<int A = 26, int offset = 'a'>
struct SAM {
    struct Node {
        int len{ 0 }, link{ 0 };
        array<int, A> next{};
    };
    vector<Node> t;
    vector<int> occ;
    int last{ 1 };
    explicit SAM(int n = 0) {
        t.reserve(2 * n + 2);
        occ.reserve(2 * n + 2);
        t.assign(2, Node());
        occ.assign(2, 0);
        t[0].len = -1;
        t[0].next.fill(1);
    }
    template<class T>
    static int id(const T &c) { return int(c) - offset; }
    // ===== 核心扩展：单串与广义 SAM 均依赖，不建议删除 =====
    int new_node() {
        t.emplace_back();
        occ.push_back(0);
        return int(t.size()) - 1;
    }
    // c 为 0..A-1 的字符编号
    int extend(int p, int c) {
        if (t[p].next[c]) {
            int q{ t[p].next[c] };
            if (t[q].len == t[p].len + 1) return q;
            int r{ new_node() };
            t[r].len = t[p].len + 1;
            t[r].link = t[q].link;
            t[r].next = t[q].next;
            t[q].link = r;
            while (t[p].next[c] == q) {
                t[p].next[c] = r;
                p = t[p].link;
            }
            return r;
        }
        int cur{ new_node() };
        t[cur].len = t[p].len + 1;
        while (!t[p].next[c]) {
            t[p].next[c] = cur;
            p = t[p].link;
        }
        t[cur].link = extend(p, c);
        return cur;
    }
    // ===== 顺序建串：若只从 Trie / 外部状态调用 extend(p, c)，可删除 last 与以下接口 =====
    template<class T>
    int extend(const T &c) {
        last = extend(last, id(c));
        ++ occ[last];
        return last;
    }
    void new_string() { last = 1; }
    template<class Seg>
    void build(const Seg &s) { for (const auto &c : s) extend(c); }
    // ===== 基础访问：简单封装，可按需删除 =====
    int next(int p, int c) const { return t[p].next[c]; }
    template<class T>
    int next(int p, const T &c) const { return next(p, id(c)); }
    int link(int p) const { return t[p].link; }
    int len(int p) const { return t[p].len; }
    int size() const { return int(t.size()); }
    // ===== 出现次数：若不统计，可删除 occ / add_occ() / order() / build_occ() / get_occ() =====
    // 同时删除 occ.reserve / occ.assign / occ.push_back / ++ occ[last]
    void add_occ(int p, int v = 1) { occ[p] += v; }
    vector<int> order() const {
        int mx{ 0 };
        for (int u = 1; u < size(); ++ u) mx = max(mx, t[u].len);
        vector<int> cnt(mx + 1), ord(size() - 1);
        for (int u = 1; u < size(); ++ u) ++ cnt[t[u].len];
        for (int i = 1; i <= mx; ++ i) cnt[i] += cnt[i - 1];
        for (int u = size() - 1; u >= 1; -- u) ord[-- cnt[t[u].len]] = u;
        return ord;
    }
    // 原地求 endpos 大小，节省一份 O(S) 数组；调用后 occ 不再是原始终点计数
    void build_occ() {
        auto ord{ order() };
        for (int i = int(ord.size()) - 1; i >= 0; -- i) {
            int u{ ord[i] };
            if (u != 1) occ[t[u].link] += occ[u];
        }
    }
    // 非破坏性版本：不修改 occ，但额外复制 O(S) 空间
    vector<int> get_occ() const {
        vector<int> res{ occ };
        auto ord{ order() };
        for (int i = int(ord.size()) - 1; i >= 0; -- i) {
            int u{ ord[i] };
            if (u != 1) res[t[u].link] += res[u];
        }
        return res;
    }
    // ===== 常用查询：各功能可单独删除；contains() 依赖 find() =====
    i64 count_substrings() const {
        i64 ans{ 0 };
        for (int u = 2; u < size(); ++ u) ans += t[u].len - t[t[u].link].len;
        return ans;
    }
    template<class Seg>
    int find(const Seg &s) const {
        int p{ 1 };
        for (const auto &c : s) {
            p = t[p].next[id(c)];
            if (!p) return 0;
        }
        return p;
    }
    template<class Seg>
    bool contains(const Seg &s) const { return find(s) != 0; }
    template<class Seg>
    int lcs(const Seg &s) const {
        int p{ 1 }, l{ 0 }, ans{ 0 };
        for (const auto &x : s) {
            int c{ id(x) };
            while (!t[p].next[c]) {
                p = t[p].link;
                l = t[p].len;
            }
            p = t[p].next[c];
            ans = max(ans, ++ l);
        }
        return ans;
    }
};
```
#### 成员变量
- `t[u].len`：状态 `u` 中最长子串的长度
- `t[u].link`：状态 `u` 的后缀链接
- `t[u].next[c]`：状态 `u` 经过字符编号 `c` 的转移
- 状态 `u` 表示的所有子串长度为  `(t[t[u].link].len, t[u].len]`
- `occ[u]`：
    - 建串完成后、统计前：直接落在状态 `u` 的前缀终点数量
    - `build_occ()` 后：状态 `u` 的 `endpos` 大小，即该状态对应子串的出现次数
- `last`：当前顺序插入字符串对应状态
- `0`：虚点，`len = -1`，所有字符均转移到根
- `1`：根，表示空串
#### 接口
- `SAM<A, offset>(n)`：构造，并按预计总插入长度 `n` 预留约 `2n + 2` 个状态
- `id(c)`：将元素映射到字符编号 `0..A-1`
- `extend(p, c)`：从状态 `p` 扩展字符编号 `c`，返回所得状态；用于广义 SAM / Trie 建 SAM
- `extend(c)`：向当前字符串末尾加入元素 `c`，更新 `last` 并记录一次前缀终点
- `new_string()`：令 `last = 1`，开始构造新的独立字符串
- `build(s)`：从当前 `last` 开始顺序插入序列 `s`
- `next(p, c)`：查询状态转移
- `link(p)`：返回后缀链接
- `len(p)`：返回状态最大长度
- `size()`：返回状态数，包含虚点 `0` 与根 `1`
- `add_occ(p, v)`：给状态增加 `v` 个直接终点标记，常用于 Trie 建广义 SAM
- `order()`：按 `len` 非递减返回全部真实状态，包含根、不包含虚点
- `build_occ()`：原地沿后缀链接汇总出现次数
- `get_occ()`：非破坏性返回所有状态的出现次数
- `count_substrings()`：返回本质不同非空子串数量
- `find(s)`：若 `s` 出现，返回其对应状态，否则返回 `0`
- `contains(s)`：判断 `s` 是否出现
- `lcs(s)`：返回 `s` 与当前 SAM 所表示字符串集合的最长公共子串长度

以字符串 `abcbc` 为例：
- **左图黑边是字符转移 `next`**
    - 从状态 `u` 沿字符 `c` 到 `v`，表示在 `u` 所表示的字符串后追加 `c`
    - 从根出发沿字符依次走，就在 SAM 中匹配一个子串
- **右图红边是后缀链接 `link`**
    - `u -> link[u]` 指向 `u` 所表示字符串类的最长严格后缀类
    - 始终有 `len[link[u]] < len[u]`
    - 因而 `link` 构成一棵以根为根的树
图中节点标出的 `ab`、`abc`、`bc` 等只是该状态的**最长代表串**。状态 `u` 实际表示一整个 `endpos` 等价类，其子串长度范围为 `(len[link[u]], len[u]]`
即
```text
len[link[u]] + 1 ... len[u]
```
例如图中的状态 `bc`：
```text
len[bc] = 2
link[bc] = root
```
因此它表示长度 `1..2` 的两个子串：
```text
c
bc
```
它们在 `abcbc` 中的结束位置完全相同：
```text
endpos(c)  = {3, 5}
endpos(bc) = {3, 5}
```
所以被压缩进同一个 SAM 状态
同理，状态 `abcb` 满足：
```text
len[abcb] = 4
len[link[abcb]] = len[b] = 1
```
因此它表示长度 `2..4` 的：
```text
cb
bcb
abcb
```
这些字符串的 `endpos` 也完全相同
##### 三个最重要的关系
看到 SAM 时优先想起：
$$\boxed{ u\text{ 表示长度 } (len[link[u]],len[u]] \text{ 的一组子串} }$$

同一状态内所有子串：
$$\boxed{ endpos\text{ 相同} \Longrightarrow occ\text{ 相同} }$$
后缀链接满足：
$$\boxed{ endpos(u)\subseteq endpos(link[u]) }$$
因此出现次数可以按 `len` 从大到小传播：
```cpp
occ[link[u]] += occ[u];
```

```text
节点最大字符串长度      t[u].len

```
例如右图中：
```text

```
这也解释了为什么从 `last = abcbc` 沿 `link` 一直走，会依次得到原串的后缀类：

```text
abcbc
bc
空串
```
图中绿色节点正对应这条从 `last` 沿后缀链接到根的链，即原串各后缀所在的状态
##### 做统计题时的速记
一个状态 `u` 表示：
```text
长度：
len[link[u]] + 1 ... len[u]

数量：
len[u] - len[link[u]]

出现次数：
occ[u]
```
因此：
**本质不同子串数**
```cpp
ans += len[u] - len[link[u]];
```
**出现至少 `k` 次的最长子串**
```cpp
if (occ[u] >= k) ans = max(ans, len[u]);
```
**P3804：长度 × 出现次数最大值**
```cpp
if (occ[u] > 1) {
    ans = max(ans, i64(occ[u]) * len[u]);
}
```
因为同一状态中的子串 `occ` 相同，所以需要最大长度时总是直接取 `len[u]`
> **SAM 状态 = `endpos` 等价类；`len` 给最长串，`link` 给长度下界，`occ` 对整个状态相同**

#### 注意事项
- `SAM<A, offset>` 要求元素满足 `0 <= int(c) - offset < A`
- 小写字母使用默认 `SAM<>`
- 整数编号序列使用 `SAM<A, 0>`
- `extend(p, c)` 中的 `c` 是已经映射好的 `0..A-1` 编号；字符需要写 `extend(p, SAM<>::id(ch))`
- `build(s)` 不会自动重置 `last`；插入多个互相独立的字符串时，每个字符串前调用 `new_string()`
- 单串 `extend(c)` 会自动执行 `++ occ[last]`
- clone 节点的初始 `occ` 必须为 `0`，不能复制原节点的 `occ`
- `occ` 使用 `int` 即可：出现次数不超过所有插入字符串总长度；最终乘积需要在运算前转为 `i64`
- `build_occ()` 会直接修改 `occ`：
    - 只能在全部字符串构造完成后调用
    - 不要调用两次
    - 调用后不要继续 `extend()` / `build()` / `add_occ()`
    - `build_occ()` 与 `get_occ()` 通常二选一，不要先 `build_occ()` 再调用 `get_occ()`
- 若需要保留原始终点计数或多次重新计算，使用 `get_occ()`
- `0` 是虚点，不参与普通状态答案；`1` 是根
- 固定 `array<int, A>` 适合小字符集；字符集很大时空间为 $O(SA)$，应考虑稀疏转移
#### 常见用法
##### 本质不同子串数量
```cpp
i64 ans{ sam.count_substrings() };
```
##### 查询模式串出现次数
普通规模、需要保留原 `occ`：
```cpp
auto occ{ sam.get_occ() };
int p{ sam.find(t) };
cout << (p ? occ[p] : 0) << '\n';
```
##### 原地统计出现次数
内存敏感时：
```cpp
sam.build_occ();
int p{ sam.find(t) };
cout << (p ? sam.occ[p] : 0) << '\n';
```
##### P3804：长度乘出现次数最大值
```cpp
SAM<> sam(s.size());
sam.build(s);
sam.build_occ();
i64 ans{ 0 };
for (int u = 2; u < sam.size(); ++ u) {
    if (sam.occ[u] > 1) { // 题目要求次数大于 1
        ans = max(ans, i64(sam.occ[u]) * sam.len(u));
    }
}
```
这里对每个状态只需要取 `len[u]`，因为同一状态中的所有子串拥有相同 `endpos`，出现次数相同，而长度越长乘积越大
##### 最长出现至少 `k` 次的子串
```cpp
sam.build_occ();
int ans{ 0 };
for (int u = 2; u < sam.size(); ++ u) {
    if (sam.occ[u] >= k) {
        ans = max(ans, sam.len(u));
    }
}
```
##### 多个独立字符串
```cpp
SAM<> sam(total_len);
for (const string &s : a) {
    sam.new_string();
    sam.build(s);
}
```
这样不会产生跨两个字符串边界的子串
##### Trie 建广义 SAM
设 Trie 中 `v` 是 `u` 经过字符编号 `c` 到达的儿子：
```cpp
sam_pos[root] = 1;
queue<int> q;
q.push(root);
while (!q.empty()) {
    int u{ q.front() };
    q.pop();
    for (int c = 0; c < A; ++ c) {
        int v{ trie[u][c] };
        if (!v) continue;
        sam_pos[v] = sam.extend(sam_pos[u], c);
        sam.add_occ(sam_pos[v], cnt[v]);
        q.push(v);
    }
}
```
若只需要子串集合，不统计出现次数，可以删掉 `add_occ()`
#### 复杂度
记：
- `M`：所有插入字符串总长度
- `S`：SAM 状态数，`S = O(M)`
- `L`：最长插入字符串长度
- `m`：查询串长度
- `A`：字符集大小

| 操作                                                        |      时间复杂度 |
| --------------------------------------------------------- | ---------: |
| `extend(c)` / `extend(p, c)`                              |  均摊 $O(A)$ |
| `build(s)`                                                | $O(\|s\|)$ |
| `new_string()` / `next / link / len / size` / `add_occ()` |     $O(1)$ |
| `order()` / `build_occ()` / `get_occ()`                   | $O(S + L)$ |
| `count_substrings()`                                      |     $O(S)$ |
| `find(s)` / `contains(s)` / `lcs(s)`                      |     $O(m)$ |
固定小字符集下，`A` 视为常数，因此构造总时间为 $O(M)$
空间复杂度：
- SAM 本体：$O(SA)$
- `occ`：$O(S)$
- `build_occ()` 临时空间：$O(S + L)$
- `get_occ()` 临时空间：$O(S + L)$，并额外保留一份 $O(S)$ 的结果数组
固定小字符集下可视为 $O(S)$
