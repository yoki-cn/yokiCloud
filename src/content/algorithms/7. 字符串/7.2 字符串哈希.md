### 字符串哈希
---
#### `u64` 自然溢出哈希
```cpp
struct RollingHash {
    u64 base;
    vector<u64> pw, h;
    static u64 default_base() {
        static const u64 b = [] {
            u64 x{ u64(chrono::steady_clock::now().time_since_epoch().count()) };
            x += 0x9e3779b97f4a7c15ULL;
            x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9ULL;
            x = (x ^ (x >> 27)) * 0x94d049bb133111ebULL;
            x ^= x >> 31;
            return x | (1ULL << 32) | 1;
        }();
        return b;
    }
    template<class T>
    static u64 encode(const T &x) {
        if constexpr (is_same_v<decay_t<T>, char>) return u64((unsigned char)x) + 1;
        else return u64(x) + 1;
    }
    template<class Seg>
    explicit RollingHash(const Seg &s, u64 base_ = default_base())
        : base(base_), pw(s.size() + 1, 1), h(s.size() + 1) {
        for (int i = 0; i < s.size(); ++ i) {
            pw[i + 1] = pw[i] * base;
            h[i + 1] = h[i] * base + encode(s[i]);
        }
    }
    int size() const { return int(h.size()) - 1; }
    u64 get(int l, int r) const { return h[r] - h[l] * pw[r - l]; }
    bool equal(int l1, int r1, int l2, int r2) const {
        return r1 - l1 == r2 - l2 && get(l1, r1) == get(l2, r2);
    }
    int lcp(int l1, int r1, int l2, int r2) const {
        int lo{ 0 }, hi{ min(r1 - l1, r2 - l2) };
        while (lo < hi) {
            int mid{ (lo + hi + 1) / 2 };
            if (get(l1, l1 + mid) == get(l2, l2 + mid)) lo = mid;
            else hi = mid - 1;
        }
        return lo;
    }
    u64 concat(u64 a, u64 b, int len_b) const { return a * pw[len_b] + b; }
};
```
#### 双模哈希
```cpp
struct DoubleHash {
    using H = pair<i64, i64>;
    static constexpr i64 mod1{ 1000000007 };
    static constexpr i64 mod2{ 1000000009 };
    i64 base;
    vector<i64> pw1, pw2, h1, h2;
    static i64 default_base() {
        static const i64 b{
            256 + i64(RollingHash::default_base() % (mod1 - 257))
        };
        return b;
    }
    template<class T>
    static u64 encode(const T &x) {
        if constexpr (is_same_v<decay_t<T>, char>) return u64((unsigned char)x) + 1;
        else return u64(x) + 1;
    }
    template<class Seg>
    explicit DoubleHash(const Seg &s, i64 base_ = default_base())
        : base(base_), pw1(s.size() + 1, 1), pw2(s.size() + 1, 1),
          h1(s.size() + 1), h2(s.size() + 1) {
        for (int i = 0; i < int(s.size()); ++ i) {
            i64 x1{ i64(encode(s[i]) % mod1) }, x2{ i64(encode(s[i]) % mod2) };
            pw1[i + 1] = pw1[i] * base % mod1;
            pw2[i + 1] = pw2[i] * base % mod2;
            h1[i + 1] = (h1[i] * base + x1) % mod1;
            h2[i + 1] = (h2[i] * base + x2) % mod2;
        }
    }
    int size() const { return int(h1.size()) - 1; }
    H get(int l, int r) const {
        i64 x{ (h1[r] - h1[l] * pw1[r - l]) % mod1 };
        i64 y{ (h2[r] - h2[l] * pw2[r - l]) % mod2 };
        if (x < 0) x += mod1;
        if (y < 0) y += mod2;
        return { x, y };
    }
    bool equal(int l1, int r1, int l2, int r2) const {
        return r1 - l1 == r2 - l2 && get(l1, r1) == get(l2, r2);
    }
    int lcp(int l1, int r1, int l2, int r2) const {
        int lo{ 0 }, hi{ min(r1 - l1, r2 - l2) };
        while (lo < hi) {
            int mid{ (lo + hi + 1) / 2 };
            if (get(l1, l1 + mid) == get(l2, l2 + mid)) lo = mid;
            else hi = mid - 1;
        }
        return lo;
    }
    H concat(H a, H b, int len_b) const {
        return { (a.first * pw1[len_b] + b.first) % mod1,
            (a.second * pw2[len_b] + b.second) % mod2
        };
    }
};
```
#### 接口说明
- `Hash(s)`：预处理序列 `s`，$O(n)$
- `Hash(s,base)`：使用指定底数预处理
- `get(l,r)`：返回 `[l,r)` 的归一化子串哈希，$O(1)$
- `equal(l1,r1,l2,r2)`：判断同一序列中的两个子串是否相等，$O(1)$
- `lcp(l1,r1,l2,r2)`：求两个子串的 LCP 长度，$O(\log n)$
- `concat(a,b,len_b)`：已知两个字符串的哈希，求其拼接哈希，$O(1)$
- `size()`：原序列长度，$O(1)$
`Seg` 只要求支持 `size()`、`operator[]`，可以直接用于：
```cpp
string
vector<int>
vector<i64>
```
等广义字符串
#### 注意事项
- 所有区间均为 `[l,r)
- 默认底数在运行时随机，但同一程序中同类型哈希对象共享同一个默认底数
- 如果显式传入底数，两个不同对象之间比较哈希时必须保证底数相同
- `DoubleHash` 碰撞概率明显低于单个 `u64` 哈希，但理论上仍可能碰撞
- 对最终只有少量候选且正确性要求极高的题，可以哈希筛选后再逐元素确认
- **哈希值不携带长度**；不同长度字符串作为集合键时必须同时保存长度
- 如果题目本身已有 KMP / Z / SA 等确定性算法能够直接完成，不要为了方便无条件换成哈希
- `concat()` 中的 `len_b` 是右侧字符串 `b` 的长度
#### 常见用法
##### 子串相等
```cpp
RollingHash h(s);
if (h.equal(l1, r1, l2, r2)) {
    // s[l1..r1) == s[l2..r2)
}
```
更重视碰撞风险时：
```cpp
DoubleHash h(s);
```
接口不变
##### 两个不同字符串比较子串
默认构造的两个对象共享底数：
```cpp
RollingHash a(s);
RollingHash b(t);
bool same{ r1 - l1 == r2 - l2 && a.get(l1, r1) == b.get(l2, r2) };
```
若显式设置底数，应写：
```cpp
u64 base{ RollingHash::default_base() };
RollingHash a(s, base);
RollingHash b(t, base);
```
##### 求子串 LCP
同一个字符串：
```cpp
int len{ h.lcp(l1, r1, l2, r2) };
```
不同字符串：
```cpp
int lo{ 0 }, hi{ min(r1 - l1, r2 - l2) };
while (lo < hi) {
    int mid{ (lo + hi + 1) / 2 };
    if (hs.get(l1, l1 + mid) == ht.get(l2, l2 + mid)) lo = mid;
    else hi = mid - 1;
}
```
得到两个子串的最长公共前缀长度 `lo`
##### 字典序比较两个子串
先求：
```cpp
int k{ h.lcp(l1, r1, l2, r2) };
int n1{ r1 - l1 }, n2{ r2 - l2 };
```
若：
```cpp
k == min(n1, n2)
```
则较短者字典序更小；否则比较：
```cpp
s[l1 + k] < s[l2 + k]
```
因此哈希可以把一次任意子串字典序比较做到：
$$O(\log n)$$
若需要大量字符串排序，SA + LCP/RMQ 往往更合适
##### 判断回文
建立正串与反串哈希：
```cpp
string rs{ s.rbegin(), s.rend() };
RollingHash h(s);
RollingHash rh(rs);
```
原串 `[l,r)` 在反串中对应：
```text
[n-r, n-l)
```
因此：
```cpp
bool pal{ h.get(l, r) == rh.get(n - r, n - l) };
```
即可 $O(1)$ 判断回文
##### 判断周期
长度为：
```cpp
int len{ r - l };
```
若 `p` 满足：
```cpp
h.get(l, r - p) == h.get(l + p, r)
```
则 `p` 是 `s[l..r)` 的一个周期
即：
$$\boxed{ p\text{ 是周期} \iff s[l..r-p)=s[l+p..r) }$$
如果进一步要求该子串由长度 `p` 的完整块重复组成，还需要：
```cpp
len % p == 0
```
##### 判断 Border
长度为 `k` 的前缀同时也是后缀：
```cpp
if (h.get(0, k) == h.get(n - k, n)) {
    // k 是一个 border 长度
}
```
对子串 `[l,r)` 同理：
```cpp
h.get(l, l + k) == h.get(r - k, r)
```
##### 拼接哈希
若：
```cpp
auto ha = h.get(l1, r1);
auto hb = h.get(l2, r2);
```
则：
```cpp
auto hab = h.concat(ha, hb, r2 - l2);
```
满足：
$$H(A+B)=H(A)\cdot base^{|B|}+H(B)$$
这个结论常用于：
- 线段树维护字符串哈希
- 分治拼接
- 动态字符串
- 区间翻转哈希
##### 哈希去重
若所有子串长度相同：
```cpp
set<u64> st;
for (...) {
    st.insert(h.get(l, r));
}
```
即可概率去重
不同长度时必须把长度也加入键：
```cpp
set<pair<int, u64>> st;
st.insert({ r - l, h.get(l, r) });
```
双哈希则使用：
```text
(length, hash1, hash2)
```
作为键
#### 二级结论
##### LCP + 哈希 = 子串比较基础工具
预处理 $O(n)$ 后：
- 子串相等：$O(1)$  
- LCP：$O(\log n)$
- 字典序比较：$O(\log n)$
- 回文判断：$O(1)$
- 周期判断：$O(1)$
因此很多“任意子串比较”问题都可以统一转成：
```text
Hash + 二分 LCP
```
##### 正反哈希
同时维护：
```text
forward hash
reverse hash
```
即可把大量与：
- 回文
- 翻转
- 镜像
有关的问题转为普通哈希相等
##### 动态字符串
静态 `RollingHash` 本身不能修改字符
若题目有单点修改 / 区间拼接，可利用：
$$H(A+B)=H(A)B^{|B|}+H(B)$$
将：
```text
(hash, length)
```
作为线段树 `Info`，即可维护动态字符串哈希
#### 复杂度
设序列长度为 $n$

| 操作                                           |       时间复杂度 |
| -------------------------------------------- | ----------: |
| 构造                                           |      $O(n)$ |
| `get()` / `equal()` / `concat()` / 回文 / 周期判断 |      $O(1)$ |
| `lcp()`                                      | $O(\log n)$ |
空间复杂度：$O(n)$
